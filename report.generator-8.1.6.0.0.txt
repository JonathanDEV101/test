      return this.converter.convertFromTable(table);

      NullPointerException npe2 = new NullPointerException(msg);
      npe2.setStackTrace(npe.getStackTrace());
      throw npe2;
    } 
  }

  }

  private void removeEmptyInpageLinks() {
    PageCollection pages = this.report.getPdfDocument().getPages();
    for (Page page : (Iterable<Page>)pages) {
      ArrayList<Annotation> annotationDeleteList = new ArrayList<>();
      AnnotationCollection annotations = page.getAnnotations();
      for (l0t<Annotation> l0t = annotations.iterator(); l0t.hasNext(); ) {
        Annotation annotation = l0t.next();
        if (annotation.getAnnotationType() == 13) {
          LinkAnnotation link = (LinkAnnotation)annotation;
          if (link.getAction() instanceof GoToAction) {
            GoToAction action = (GoToAction)link.getAction();
            TextFragmentAbsorber ta = new TextFragmentAbsorber();
            Rectangle rect = link.getRect();
            ta.setTextSearchOptions(new TextSearchOptions(rect));
            ta.visit(page);
            if (ta.getTextFragments().size() == 0) {
              annotationDeleteList.add(annotation);
              continue;
            } 
            if (action.getDestination() instanceof XYZExplicitDestination) {
              XYZExplicitDestination destination = (XYZExplicitDestination)action.getDestination();
              if (destination.getTop() > 70.0D || destination.getTop() < 40.0D) {
                XYZExplicitDestination newDestination = new XYZExplicitDestination(destination.getPageNumber(), destination.getLeft(), 50.0D, destination.getZoom());
                action.setDestination((IAppointment)newDestination);
              } 
            } 
          } 
        } 
      } 
      for (Annotation annotation : annotationDeleteList)
        annotations.delete(annotation); 
    } 
  }

package com.gigatronik.beds.report.impl;

import com.gigatronik.beds.report.api.ReportGenerator;
import com.gigatronik.beds.report.api.ReportGeneratorData;
import com.gigatronik.beds.report.impl.part.PartGenerator;
import java.io.File;
import java.util.ArrayList;
import java.util.Collections;
import java.util.Iterator;
import java.util.List;
import java.util.concurrent.ExecutionException;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Future;
import java.util.function.Supplier;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

public abstract class BaseReportGeneratorStarter<T extends ReportGenerator, D extends ReportGeneratorData> implements ReportGenerator<T, D> {
  private static final Logger LOGGER = LoggerFactory.getLogger(ImageLoader.class);
  
  private Future future;
  
  protected Supplier<FontProvider> fontProvider;
  
  protected ImageLoader imageLoader;
  
  protected ExecutorService service;
  
  protected List<PartGenerator<D>> partGeneratorList = new ArrayList<>();
  
  public BaseReportGeneratorStarter(Supplier<FontProvider> fontProvider, ImageLoader imageLoader, ExecutorService service) {
    this.fontProvider = fontProvider;
    this.imageLoader = imageLoader;
    this.service = service;
  }
  
  public Future generateAndSave(D data, File file) {
    this.future = this.service.submit(() -> {
          try {
            List<PartGenerator<D>> partGenerators = Collections.unmodifiableList(this.partGeneratorList);
            create(partGenerators);
            fill(partGenerators, data);
            save(file);
          } catch (Exception e) {
            LOGGER.error("Failed to build report, file={}", file, e);
          } 
        });
    this.service.submit(() -> cleanUp(this.partGeneratorList));
    return this.future;
  }
  
  public void waitForComplete() {
    if (!this.future.isDone())
      try {
        this.future.get();
        LOGGER.trace("build: {}", this.future.isCancelled() ? "canceled" : "complete");
      } catch (InterruptedException e) {
        LOGGER.error("Failed to build report, interrupted", (Throwable)e);
        Thread.currentThread().interrupt();
      } catch (ExecutionException e) {
        LOGGER.error("Failed to build report", (Throwable)e);
      }  
  }
  
  private void create(List<PartGenerator<D>> taskList) {
    taskList.forEach(partGenerator -> {
          if (isNullOrNotCanceled(this.future)) {
            LOGGER.trace("build: create\t\t\t{}", partGenerator);
            partGenerator.create();
            LOGGER.trace("build: create complete\t{}", partGenerator);
          } 
        });
  }
  
  private boolean isNullOrNotCanceled(Future future) {
    return (future == null || !future.isCancelled());
  }
  
  private void fill(List<PartGenerator<D>> taskList, D data) {
    List<PartGenerator<D>> fillTaskList = new ArrayList<>(taskList);
    while (!fillTaskList.isEmpty() && isNullOrNotCanceled(this.future)) {
      boolean notReady = true;
      Iterator<PartGenerator<D>> i = fillTaskList.iterator();
      while (i.hasNext() && isNullOrNotCanceled(this.future)) {
        PartGenerator<D> partGenerator = i.next();
        if (partGenerator.isReadyToFill(data)) {
          i.remove();
          LOGGER.trace("build: fill\t\t\t{}", partGenerator);
          partGenerator.fill(data);
          LOGGER.trace("build: fill complete\t{}", partGenerator);
          notReady = false;
        } 
      } 
      if (notReady && isNullOrNotCanceled(this.future))
        sleep(10L); 
    } 
    if (isNullOrNotCanceled(this.future))
      LOGGER.trace("build: fill all parts complete"); 
  }
  
  private void save(File file) {
    if (isNullOrNotCanceled(this.future)) {
      LOGGER.trace("build: save");
      saveReport(file);
      LOGGER.trace("build: save complete");
    } 
  }
  
  private void cleanUp(List<PartGenerator<D>> taskList) {
    Collections.reverse(taskList);
    taskList.forEach(PartGenerator::cleanUp);
  }
  
  private void sleep(long millis) {
    try {
      Thread.sleep(millis);
    } catch (InterruptedException e) {
      LOGGER.error("Failed to sleep {} ms", Long.valueOf(millis), e);
      Thread.currentThread().interrupt();
    } 
  }
  
  protected abstract void saveReport(File paramFile);
}

package com.gigatronik.beds.report.impl;

import com.aspose.pdf.Font;
import com.aspose.pdf.FontRepository;
import java.io.IOException;
import java.io.InputStream;
import org.apache.commons.lang3.reflect.MethodUtils;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

public class FontProvider {
  private static final Logger LOGGER = LoggerFactory.getLogger(FontProvider.class);
  
  private static FontProvider instance;
  
  private volatile Font fontBoschSansGlobal;
  
  private volatile Font fontBoschSansGlobalBold;
  
  public static FontProvider getInstance() {
    if (instance == null)
      instantiate(); 
    return new FontProvider(cloneFont(instance.fontBoschSansGlobal), cloneFont(instance.fontBoschSansGlobalBold));
  }
  
  private FontProvider() {}
  
  private FontProvider(Font fontBoschSansGlobal, Font fontBoschSansGlobalBold) {
    this.fontBoschSansGlobal = fontBoschSansGlobal;
    this.fontBoschSansGlobalBold = fontBoschSansGlobalBold;
  }
  
  public Font getRegular() {
    return this.fontBoschSansGlobal;
  }
  
  public Font getBold() {
    return this.fontBoschSansGlobalBold;
  }
  
  public Font getFont(boolean bold) {
    return bold ? this.fontBoschSansGlobalBold : this.fontBoschSansGlobal;
  }
  
  private static void instantiate() {
    if (instance == null) {
      FontProvider fontProvider = new FontProvider();
      fontProvider.initFonts();
      instance = fontProvider;
    } 
  }
  
  private static Font cloneFont(Font font) {
    try {
      return (Font)MethodUtils.invokeMethod(font, true, "clone");
    } catch (NoSuchMethodException|IllegalAccessException|java.lang.reflect.InvocationTargetException e) {
      LOGGER.error("Failed to clone font", (Throwable)e);
      return null;
    } 
  }
  
  private void initFonts() {
    LOGGER.trace("Loading fonts ...");
    this.fontBoschSansGlobal = loadFont("/fonts/BoschSansGlobal-Regular.ttf");
    this.fontBoschSansGlobalBold = loadFont("/fonts/BoschSansGlobal-Bold.ttf");
  }
  
  private Font loadFont(String fontFile) {
    LOGGER.trace("Loading font {}...", fontFile);
    try (InputStream is = FontProvider.class.getResourceAsStream(fontFile)) {
      Font font = FontRepository.openFont(is, 0);
      font.setEmbedded(true);
      font.setSubset(true);
      return font;
    } catch (IOException e) {
      LOGGER.warn("Error loading font {}", fontFile, e);
      return null;
    } 
  }
}

package com.gigatronik.beds.report.impl;

import java.awt.Graphics2D;
import java.awt.RenderingHints;
import java.awt.geom.AffineTransform;
import java.awt.image.BufferedImage;
import java.io.File;
import java.io.FileInputStream;
import java.io.IOException;
import java.io.InputStream;
import java.util.HashMap;
import java.util.Map;
import java.util.function.Consumer;
import javax.imageio.ImageIO;
import org.apache.commons.lang3.StringUtils;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

public class ImageLoader {
  private static final Logger LOGGER = LoggerFactory.getLogger(ImageLoader.class);
  
  private static final String IMAGE_SUFFIX = "png";
  
  private Map<String, String> imagePathMap = new HashMap<>();
  
  public void loadImage(Class contextClass, String imageResourcePath, double scaling, Consumer<InputStream> action) {
    try (InputStream inputStream = loadImage(contextClass, imageResourcePath, scaling)) {
      action.accept(inputStream);
    } catch (IOException e) {
      LOGGER.error("Failed to load image from resource '{}', scaling {}", imageResourcePath, scaling, e);
    } 
  }
  
  public InputStream loadImage(Class contextClass, String imageResourcePath, double scaling) throws IOException {
    String imageKey = imageResourcePath + ";" + imageResourcePath;
    String imagePath = this.imagePathMap.get(imageKey);
    File image = null;
    if (imagePath != null) {
      image = new File(imagePath);
      if (!image.exists())
        image = null; 
    } 
    if (image == null) {
      image = File.createTempFile("resource_", ".image");
      image.deleteOnExit();
      if (StringUtils.endsWithIgnoreCase(imageResourcePath, "png")) {
        reduceImage(contextClass, imageResourcePath, scaling, image);
      } else {
        LOGGER.debug("Ignoring reduceImage {} for image candidate, because its no valid supported file!", imageResourcePath);
      } 
      this.imagePathMap.put(imageKey, image.getAbsolutePath());
    } 
    return new FileInputStream(image);
  }
  
  protected void reduceImage(Class contextClass, String fileName, double scaling, File file) throws IOException {
    try (InputStream imgStream = contextClass.getResourceAsStream(fileName)) {
      BufferedImage origin = ImageIO.read(imgStream);
      int width = (int)(scaling * (double)origin.getWidth());
      int height = (int)(scaling * (double)origin.getHeight());
      BufferedImage smallImage = new BufferedImage(width, height, origin.getType());
      Graphics2D graphic = smallImage.createGraphics();
      graphic.setRenderingHint(RenderingHints.KEY_INTERPOLATION, RenderingHints.VALUE_INTERPOLATION_BILINEAR);
      graphic.setRenderingHint(RenderingHints.KEY_RENDERING, RenderingHints.VALUE_RENDER_QUALITY);
      graphic.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
      AffineTransform affineTransform = AffineTransform.getScaleInstance(scaling, scaling);
      graphic.drawRenderedImage(origin, affineTransform);
      graphic.dispose();
      ImageIO.write(smallImage, "png", file);
    } 
  }
}

public class ReportDataImpl<T> implements ReportData<T> {
  private static final Logger LOGGER = LoggerFactory.getLogger(ReportDataImpl.class);
  
  private T value;
  
  private boolean isSet;
  
  public synchronized T getValue() {
    while (!this.isSet && !Thread.currentThread().isInterrupted()) {
      try {
        wait();
      } catch (InterruptedException e) {
        LOGGER.error("Failed to wait for data");
        Thread.currentThread().interrupt();
      } 
    } 
    return this.value;
  }
  
  public synchronized boolean hasValue() {
    return this.isSet;
  }
  
  public synchronized void setValue(T value) {
    this.value = value;
    this.isSet = true;
    notifyAll();
  }
}

import com.gigatronik.ebike.common.configuration.ComplexResponse;
import com.gigatronik.ebike.common.util.PnTableWithChart;
import java.util.ArrayList;
import java.util.List;
import javax.swing.table.TableModel;

public class JTableStringModelDtoImpl implements JTableStringModelDto {
  private List<List<Object>> tableModel = new ArrayList<>();
  
  public List<List<Object>> getTableModel() {
    return this.tableModel;
  }
  
  public void setTableModel(List<List<Object>> tableModel) {
    this.tableModel = tableModel;
  }
  
  public String getValueForKey(String key) {
    for (int i = 0; i < this.tableModel.get(0).size(); i++) {
      Object o = this.tableModel.get(0).get(i);
      if (o instanceof PnTableWithChart) {
        PnTableWithChart pieChart = (PnTableWithChart)o;
        TableModel tableModelPie = pieChart.getTableModel();
        for (int j = 0; j < tableModelPie.getRowCount(); j++) {
          String value = tableModelPie.getValueAt(j, 0).toString();
          if (value.trim().equalsIgnoreCase(key))
            return getValueForAtIndex(i, j);

  public JTableStringModelDto convertFromTable(JTable table) {
    List<List<Object>> rowEntries = new ArrayList<>();
    List<Object> row1 = new ArrayList();
    List<Object> row2 = new ArrayList();
    JTableStringModelDto model = new JTableStringModelDtoImpl();
    for (int i = 0; i < table.getRowCount(); i++) {
      Object key = null;
      if (table.getValueAt(i, 0) instanceof THead) {
        THead tHead = (THead)table.getValueAt(i, 0);
        key = tHead.getValue();
      } else {
        key = table.getValueAt(i, 0);
      } 
      Object value = table.getValueAt(i, 1);
      if (key != null && value != null) {
        row1.add(key);
        row2.add(value);
      } 
    } 
    rowEntries.add(row1);
    rowEntries.add(row2);
    model.setTableModel(rowEntries);
    return model;
  }
}

    float borderWidth = 0.2F;
    MarginInfo pages = document.getPageInfo().getMargin();
    double maxWidth = document.getPageInfo().getWidth() - (pages.getLeft() + pages.getRight() + 51.012D);
    int colWidth = (int)(maxWidth / (double)columnCount);
    StringBuilder columnWidth = new StringBuilder();
    for (int i = 0; i < columnCount; i++)
      columnWidth.append(String.format("%d ", colWidth)); 
    Table table = new Table();
    table.getMargin().setLeft(25.506D);
    table.getMargin().setRight(25.506D);
    table.setColumnWidths(columnWidth.toString().trim());

public class DiagnosticServiceIntervalTableGenerator implements PartGenerator<ServiceReportGeneratorData> {

  private final DiagnosticReportGeneratorImpl parent;

  public void create() {

  public void fill(ServiceReportGeneratorData data) {
    this.parent.addTitleAboveTable(this.page, this.parent.getBundleValueForKey("EBikeReport.serviceReportTable.serviceinterval"), true);
    this.parent.addParagraphArialBold(this.page, "\r\n", 12.0F);
    Table serviceReportTable = this.parent.createTableAndAddToPage(this.document, this.page, 3);
    Row row = serviceReportTable.getRows().add();
    this.parent.addDefaultTableCell(row, "", false);
    this.parent.addDefaultTableCell(row, this.parent.getBundleValueForKey("EBikeReport.serviceReportTable.dateText"), true);
    this.parent.addDefaultTableCell(row, this.parent.getBundleValueForKey("EBikeReport.serviceReportTable.totalDist"), true);
    row = serviceReportTable.getRows().add();
    this.parent.addDefaultTableCell(row, this.parent.getBundleValueForKey("EBikeReport.serviceReportTable.planned"), false);
    boolean duConnectedAndServiceConfVisible = (data.getBikeIdent().isDuConnected() && data.isServiceConfTableVisible());
    if (duConnectedAndServiceConfVisible) {
      writePlannedServiceInterval(data, serviceReportTable);
    } else {
      this.parent.addDefaultTableCell(row, "", false);
      this.parent.addDefaultTableCell(row, "", false);
    } 
    row = serviceReportTable.getRows().add();
    this.parent.addDefaultTableCell(row, this.parent.getBundleValueForKey("EBikeReport.serviceReportTable.performed"), false);
    if (duConnectedAndServiceConfVisible) {
      writeHistoryServiceInterval(data, serviceReportTable);
    } else {
      this.parent.addDefaultTableCell(row, "", false);
      this.parent.addDefaultTableCell(row, "", false);
    } 
  }
  
  private void writePlannedServiceInterval(ServiceReportGeneratorData data, Table serviceReportTable) {
    boolean convertOdometerToMiles = "1".equalsIgnoreCase(data.getUnitDistance());
    Row row = serviceReportTable.getRows().get_Item(serviceReportTable.getRows().getCount() - 1);
    String[] nextService = data.getServiceIntervalDueDate();
    if (nextService != null) {
      if (!nextService[0].isEmpty() && !nextService[0].equals("2000-00-00")) {
        nextService[0] = I18NDateTimesFormats.reFormatDateByLocale(nextService[0], I18NUtil.getLocale());
        this.parent.addDefaultTableCell(row, nextService[0], false);
      } else {
        this.parent.addDefaultTableCell(row, "", false);
      } 
      if (!nextService[1].isEmpty()) {
        String nextServiceOdometer = String.format("%.0f km", Double.parseDouble(nextService[1]) / 1000.0D);
        if (convertOdometerToMiles)
          nextServiceOdometer = String.format("%.0f mi", ConverterUtil.kilometerToMile(Double.parseDouble(nextService[1]) / 1000.0D)); 
        this.parent.addDefaultTableCell(row, nextServiceOdometer, false);
      } else {
        this.parent.addDefaultTableCell(row, "", false);
      } 
    } else {
      this.parent.addDefaultTableCell(row, "", false);
      this.parent.addDefaultTableCell(row, "", false);
    } 
  }
  
  private void writeHistoryServiceInterval(ServiceReportGeneratorData data, Table serviceReportTable) {
    boolean firstentry = true;
    Row row = serviceReportTable.getRows().get_Item(serviceReportTable.getRows().getCount() - 1);
    if (data.getServiceHistoryData() == null || data.getServiceHistoryData().isEmpty()) {
      this.parent.addDefaultTableCell(row, "", false);
      this.parent.addDefaultTableCell(row, "", false);
    } else {
      int index = 0;
      while (index + 1 < data.getServiceHistoryData().size()) {
        if (!firstentry) {
          row = serviceReportTable.getRows().add();
          this.parent.addDefaultTableCell(row, "", false);
        } 
        firstentry = false;
        String date = data.getServiceHistoryData().get(index++);
        date = I18NDateTimesFormats.reFormatDateByLocale(date, I18NUtil.getLocale());
        this.parent.addDefaultTableCell(row, date, false);
        this.parent.addDefaultTableCell(row, data.getServiceHistoryData().get(index++), false);
      } 
    } 
  }
}

  private void addErrorCodes(ServiceReportGeneratorData data, Table table, String componentName, JTableStringModelDto tblErrorCode) {
    if (null == tblErrorCode)
      return; 
    if (tblErrorCode.getTableModel().get(0).isEmpty())
      return; 
    if (tblErrorCode.getTableModel().get(0).size() == 1) {
      writeOneErrorCode(table, componentName, tblErrorCode.getTableModel().get(0).get(0).toString().trim(), tblErrorCode.getTableModel().get(1).get(0).toString().trim());
      return;
    } 
    if (tblErrorCode.getTableModel().get(0).size() > 1) {
      Map<String, List<Integer>> errorCodeDounbleIndexes = data.getErrorCodeFilter().filterSameErrorcodes(tblErrorCode);
      Set<String> ecDuplicates = errorCodeDounbleIndexes.keySet();
      for (String key : ecDuplicates) {
        List<Integer> indexes = errorCodeDounbleIndexes.get(key);
        writeOneErrorCode(table, componentName, key, tblErrorCode.getTableModel().get(1).get(indexes.get(0).intValue()).toString().trim());
      } 
      List<Integer> allIndexes = new ArrayList<>();
      for (String key : ecDuplicates) {
        List<Integer> indexes = errorCodeDounbleIndexes.get(key);
        allIndexes.addAll(indexes);
      } 
      for (int i = 0; i < tblErrorCode.getTableModel().get(0).size(); i++) {
        if (!allIndexes.contains(Integer.valueOf(i))) {
          String restEc = ((List<String>)tblErrorCode.getTableModel().get(0)).get(i);
          String restDescr = ((List<String>)tblErrorCode.getTableModel().get(1)).get(i);
          if (restEc.length() > 3) {
            String firstThreeDigits = restEc.substring(2, restEc.length() - 3);
            writeOneErrorCode(table, componentName, firstThreeDigits, restDescr);
          } 
        } 
      } 
    } 
  }
  
  private void writeOneErrorCode(Table table, String componentName, String ec, String desc) {
    Row row = table.getRows().add();
    this.parent.addCellText(0.3F, componentName, this.parent.fontProvider.getRegular(), 7, AbstractReportGeneratorImpl.TABLETEXTLISTINGCOLOR9, row);
    String errorCode = ec;
    if (errorCode.length() > 3)
      errorCode = errorCode.substring(2, errorCode.length() - 3); 
    this.parent.addCellText(0.3F, errorCode, this.parent.fontProvider.getRegular(), 7, AbstractReportGeneratorImpl.TABLEENTRYTEXTCOLOR10, row);
    this.parent.addCellText(0.3F, desc, this.parent.fontProvider.getRegular(), 7, AbstractReportGeneratorImpl.TABLEENTRYTEXTCOLOR10, row);
  }

    this.tableHeader.setText(StringUtils.trimToEmpty(this.parent.getBundleValueForKey("EBikeReport.serviceReportTable.serviceinterval")));
    this.dateTextColumnHeader.setText(StringUtils.trimToEmpty(this.parent.getBundleValueForKey("EBikeReport.serviceReportTable.dateText")));
    this.totalDistColumnHeader.setText(StringUtils.trimToEmpty(this.parent.getBundleValueForKey("EBikeReport.serviceReportTable.totalDist")));
    this.nextIntervalTitle.setText(StringUtils.trimToEmpty(
          this.parent.getBundleValueForKey("EBikeConfigurationTab.ServiceConfiguration.nextInterval.title")));
    boolean convertOdometerToMiles = "1".equalsIgnoreCase(data.getUnitDistance());
    if (data.getBikeIdent().isDuConnected() && data.isServiceConfTableVisible()) {
      String[] nextService = data.getServiceIntervalDueDate();
      writeServiceValues(nextService, convertOdometerToMiles, this.currentRow);
    } else {
      this.parent.addEmptyCell(2, 0.3F, this.currentRow);
    } 
    this.currentRow = this.table.getRows().add();
    this.parent.addCell(0.3F, "EBikeReport.serviceReportTable.performed", this.parent.fontProvider.getRegular(), 7, AbstractReportGeneratorImpl.TABLETEXTLISTINGCOLOR9, this.currentRow);
    addServiceIntervallData(data, this.table, this.currentRow);
  }
  
  private void writeServiceValues(String[] nextService, boolean convertOdometerToMiles, Row currentRow) {
    if (nextService != null) {
      if (!nextService[0].isEmpty() && !nextService[0].equals("2000-00-00")) {
        nextService[0] = I18NDateTimesFormats.reFormatDateByLocale(nextService[0], I18NUtil.getLocale());
        this.parent.addCellText(0.3F, nextService[0], this.parent.fontProvider.getRegular(), 7, AbstractReportGeneratorImpl.TABLEENTRYTEXTCOLOR10, currentRow);
      } else {
        this.parent.addEmptyCell(1, 0.3F, currentRow);
      } 
      if (!nextService[1].isEmpty()) {
        if (convertOdometerToMiles) {
          this.parent.addCellText(0.3F, String.format("%.0f mi", 
                ConverterUtil.kilometerToMile(Double.parseDouble(nextService[1]) / 1000.0D)), this.parent.fontProvider.getRegular(), 7, AbstractReportGeneratorImpl.TABLEENTRYTEXTCOLOR10, currentRow);
        } else {
          this.parent.addCellText(0.3F, String.format("%.0f km", Double.parseDouble(nextService[1]) / 1000.0D), 
              this.parent.fontProvider.getRegular(), 7, AbstractReportGeneratorImpl.TABLEENTRYTEXTCOLOR10, currentRow);
        } 
      } else {
        this.parent.addEmptyCell(1, 0.3F, currentRow);
      } 
    } else {
      this.parent.addEmptyCell(2, 0.3F, currentRow);
    } 
  }
  
  private void addServiceIntervallData(ServiceReportGeneratorData data, Table table, Row currentRow) {
    if (data.getBikeIdent().isDuConnected() && data.isServiceConfTableVisible()) {
      if (data.getServiceHistoryData() == null || data.getServiceHistoryData().isEmpty()) {
        this.parent.addEmptyCell(2, 0.3F, currentRow);
      } else {
        addServiceIntervallEntry(table, data.getServiceHistoryData(), currentRow);
      } 
    } else {
      this.parent.addEmptyCell(2, 0.3F, currentRow);
    } 
  }
  
  private void addServiceIntervallEntry(Table table, List<String> serviceData, Row currentRow) {
    int index = 0;
    boolean firstentry = true;
    while (index + 1 < serviceData.size()) {
      if (!firstentry) {
        currentRow = table.getRows().add();
        this.parent.addEmptyCell(1, 0.3F, currentRow);
      } 
      firstentry = false;
      String date = serviceData.get(index++);
      date = I18NDateTimesFormats.reFormatDateByLocale(date, I18NUtil.getLocale());
      this.parent.addCellText(0.3F, date, this.parent.fontProvider.getRegular(), 7, AbstractReportGeneratorImpl.TABLEENTRYTEXTCOLOR10, currentRow);
      this.parent.addCellText(0.3F, serviceData.get(index++), this.parent.fontProvider.getRegular(), 7, AbstractReportGeneratorImpl.TABLEENTRYTEXTCOLOR10, currentRow);
    } 
  }
}

package com.gigatronik.beds.service.report.impl;

import com.gigatronik.beds.service.report.api.ErrorCodeFilter;
import com.gigatronik.beds.service.report.api.dto.JTableStringModelDto;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class StandardErrorCodeFilterImpl implements ErrorCodeFilter {
  public boolean findErrorCodeDuplicated(JTableStringModelDto tblErrorCode, int i, String firstThreeDigits, Map<String, List<Integer>> dblECIndex, List<Integer> indexOfDuplicatesForOneEC) {
    boolean found = false;
    int rowCount = tblErrorCode.getTableModel().get(0).size();
    for (int j = i + 1; j < rowCount; j++) {
      String coimpareErrorCode = ((List<String>)tblErrorCode.getTableModel().get(0)).get(j);
      if (coimpareErrorCode.length() > 3) {
        String compareFirstThreeDigits = coimpareErrorCode.substring(2, coimpareErrorCode.length() - 3);
        if (compareFirstThreeDigits.equalsIgnoreCase(firstThreeDigits)) {
          if (dblECIndex.containsKey(firstThreeDigits))
            break; 
          indexOfDuplicatesForOneEC.add(Integer.valueOf(j));
          found = true;
        } 
      } 
    } 
    return found;
  }
  
  public Map<String, List<Integer>> filterSameErrorcodes(JTableStringModelDto tblErrorCode) {
    int rowCount = tblErrorCode.getTableModel().get(0).size();
    Map<String, List<Integer>> dblECIndex = new HashMap<>();
    for (int i = 0; i < rowCount; i++) {
      String errorCode = ((List<String>)tblErrorCode.getTableModel().get(0)).get(i);
      if (errorCode.length() > 3) {
        String firstThreeDigits = errorCode.substring(2, errorCode.length() - 3);
        List<Integer> indexOfDuplicatesForOneEC = new ArrayList<>();
        indexOfDuplicatesForOneEC.add(Integer.valueOf(i));
        if (findErrorCodeDuplicated(tblErrorCode, i, firstThreeDigits, dblECIndex, indexOfDuplicatesForOneEC))
          dblECIndex.put(firstThreeDigits, indexOfDuplicatesForOneEC); 
      } 
    } 
    return dblECIndex;
  }
}

  private void changeListsToOnlyBatteryData(List<Map.Entry<String, String>> data, List<Map.Entry<String, String>> column1) {
    column1.clear();
    while (column1.size() + 1 < data.size()) {
      int index = column1.size();
      column1.add(data.get(index));
      data.remove(index);
    } 
  }
  
  public void fill(ServiceReportGeneratorData data) {
    List<Map.Entry<String, String>> column1;
    List<Map.Entry<String, String>> column2 = generateStatisticsTableBatteryColumnData(data);
    if (this.parent.getReportType() == ReportType.CUSTOMER_BMS_ONLY) {
      column1 = new ArrayList<>();
      changeListsToOnlyBatteryData(column2, column1);
    } else {
      column1 = generateStatisticsTableFirstColumnData(data);
    } 
    int entrySum = column1.size() + column2.size();
    if (entrySum == 0)
      return; 
    this.cellWrapperForTable.getParagraphs().add((BaseParagraph)this.table);
    this.tableHeader.setText(this.parent.getBundleValueForKey("EBikeReport.statisticsTable.title"));
    Iterator<Map.Entry<String, String>> it1 = column1.iterator();
    Iterator<Map.Entry<String, String>> it2 = column2.iterator();
    while (entrySum > 0) {
      if (it1.hasNext()) {
        Map.Entry<String, String> entry = it1.next();
        addEntryToTable(this.table, entry.getKey(), entry.getValue());
        entrySum--;
      } else {
        addEntryToTable(this.table, "", "");
      } 
      if (it2.hasNext()) {
        Map.Entry<String, String> entry = it2.next();
        addEntryToTable(this.table, entry.getKey(), entry.getValue());
        entrySum--;
        continue;
      } 
      addEntryToTable(this.table, "", "");
    } 
    endDynamicTable(this.table);
  }

  private Map.Entry<String, String> addStatisticsTableEntryCheckContraints(ServiceReportGeneratorData data, String key, String value) {
    Map.Entry<String, String> entry = null;
    if (StringUtils.isEmpty(value)) {
      if (BikeType.getValueByXmlAttribute(data.getBikeIdent().getBikeType()) == BikeType.CLASSIC_PLUS)
        entry = createSimpleEntry(key, value); 
    } else {
      entry = createSimpleEntry(key, value);
    } 
    return entry;
  }
  
  private void addEntryToTable(Table table, String keyText, String valueText) {
    Row row;
    float borderThickness = 0.3F;
    if (this.tableEntryCount < 2)
      borderThickness = 0.0F; 
    if (this.tableEntryCount % 2 == 0) {
      row = table.getRows().add();
    } else {
      row = table.getRows().get_Item(table.getRows().getCount() - 1);
    } 
    this.parent.addCellText(borderThickness, keyText, this.parent.fontProvider.getRegular(), 7, AbstractReportGeneratorImpl.TABLETEXTLISTINGCOLOR9, row);
    this.parent.addCellText(borderThickness, valueText, this.parent.fontProvider.getRegular(), 7, AbstractReportGeneratorImpl.TABLEENTRYTEXTCOLOR10, row);
    this.tableEntryCount++;
  }
  
  private void endDynamicTable(Table table) {
    if (this.tableEntryCount % 2 != 0) {
      Row row = table.getRows().get_Item(table.getRows().getCount() - 1);
      this.parent.addEmptyCell(2, 0.3F, row);
    } 
  }
  
  private Map.Entry<String, String> createSimpleEntry(String key, String value) {
    return new AbstractMap.SimpleEntry<>(key, value);
  }
}
