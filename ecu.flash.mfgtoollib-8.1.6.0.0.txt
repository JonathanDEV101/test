package com.mfgtool;

import com.sun.jna.Callback;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

public class WinProgressUpdateCallback implements Callback {
  private static final Logger LOGGER = LoggerFactory.getLogger(WinProgressUpdateCallback.class);
  
  private final MfgToolLibCtrl ctrl;
  
  public WinProgressUpdateCallback(MfgToolLibCtrl ctrl) {
    this.ctrl = ctrl;
  }
  
  public void callback(WinOperateResult.ByReference operateResult) {
    LOGGER.info("<--- Operation ID: {}", Integer.valueOf(operateResult.OperationID));
    LOGGER.info("     Command Index: {}", Integer.valueOf(operateResult.cmdIndex));
    LOGGER.info("     Phase Index: {}", Integer.valueOf(operateResult.CurrentPhaseIndex));
    StringBuilder builder = new StringBuilder();
    char[] cmdInfo = operateResult.cmdInfo;
    for (char c : cmdInfo) {
      if (Character.isDefined(c) && c != '\000') {
        builder.append(c);
      } else {
        break;
      } 
    } 
    LOGGER.info("     Command Info : {}", builder.toString());
    LOGGER.info("     ProgressWithinCommand: {}", Boolean.valueOf(operateResult.bProgressWithinCommand));
    LOGGER.info("     Progress within command: {}", Integer.valueOf(operateResult.DoneWithinCommand));
    LOGGER.info("     Total within command: {}", Integer.valueOf(operateResult.TotalWithinCommand));
    LOGGER.info("     Command status {} --->", Integer.valueOf(operateResult.cmdStatus));
    if (operateResult.cmdStatus == 2) {
      this.ctrl.setErrorOccurred(builder.toString());
      LOGGER.error(" --> Error while executing command!!!");
    } else if (operateResult.cmdStatus == 1) {
      if (!operateResult.bProgressWithinCommand)
        this.ctrl.setProgress(operateResult.CurrentPhaseIndex, operateResult.cmdIndex); 
    } else if (operateResult.cmdStatus == 3) {
      this.ctrl.setCommandRunning(operateResult.bProgressWithinCommand, operateResult.DoneWithinCommand, operateResult.TotalWithinCommand, builder.toString());
    } 
  }
}
