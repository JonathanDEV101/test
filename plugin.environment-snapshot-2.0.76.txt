  public EnvironmentSnapshotDartPlugin.CheckResult execute() {
    EnvironmentSnapshotDartPlugin.CheckResult<String> result;

    }

  }

    OperatingSystem operatingSystem = this.systemInfo.getOperatingSystem();
    OSProcess[] processes = operatingSystem.getProcesses(0, OperatingSystem.ProcessSort.NAME);

    } 
    return new SingleCheckResultImpl<>(this, builder.toString());
  }
