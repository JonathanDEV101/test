package com.gigatronik.beds.service.hiddriverwrapper.impl;

import java.util.LinkedList;
import java.util.concurrent.Future;
import java.util.concurrent.ScheduledExecutorService;
import java.util.concurrent.TimeUnit;
import java.util.function.Consumer;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

public class EventRateLimiter<T> implements Runnable {
  private static final Logger LOGGER = LoggerFactory.getLogger(EventRateLimiter.class);
  
  private final ScheduledExecutorService executor;
  
  private final Consumer<T> callback;
  
  private final long eventTimeout;
  
  private final LinkedList<T> eventQueue = new LinkedList<>();
  
  private Future future;
  
  public EventRateLimiter(ScheduledExecutorService executor, Consumer<T> callback, long eventTimeout) {
    this.executor = executor;
    this.callback = callback;
    this.eventTimeout = eventTimeout;
  }
  
  public void run() {
    sendNextEvent();
  }
  
  public synchronized void fireEvent(T event) {
    LOGGER.debug("Queue an event {}", event);
    this.eventQueue.addLast(event);
    if (this.future == null) {
      LOGGER.debug("Start event sender (one event each {} ms)", Long.valueOf(this.eventTimeout));
      this.future = this.executor.scheduleAtFixedRate(this, 0L, this.eventTimeout, TimeUnit.MILLISECONDS);
    } 
  }
  
  private synchronized void sendNextEvent() {
    if (this.future == null)
      return; 
    T event = this.eventQueue.pollFirst();
    if (event == null) {
      LOGGER.debug("No more events in queue, stop event sender");
      this.future.cancel(false);
      this.future = null;
      return;
    } 
    LOGGER.debug("Send next event from queue {}", event);
    this.callback.accept(event);
  }
}

package com.gigatronik.beds.service.hiddriverwrapper.impl;

import java.util.LinkedList;
import java.util.Queue;

class ScheduledRx {
  private Queue<byte[]> inBuffer = new LinkedList<>();
  
  synchronized void waitForResponse(long readTimeout) throws InterruptedException {
    long scheduledTime = System.currentTimeMillis();
    long remainingWaitingTime = readTimeout;
    boolean timedOut = false;
    while (!isResponseAvailable() && !timedOut && remainingWaitingTime > 0L) {
      wait(remainingWaitingTime);
      long currentTime = System.currentTimeMillis();
      timedOut = (currentTime - readTimeout >= scheduledTime);
      remainingWaitingTime = scheduledTime - currentTime + readTimeout;
    } 
  }
  
  synchronized boolean isResponseAvailable() {
    return !this.inBuffer.isEmpty();
  }
  
  synchronized void setResponse(byte[] response) {
    if (response != null) {
      this.inBuffer.add(response);
      notifyAll();
    } 
  }
  
  synchronized void clearBuffer() {
    this.inBuffer.clear();
  }
  
  synchronized byte[] getResponse() {
    if (isResponseAvailable())
      return this.inBuffer.poll(); 
    return new byte[0];
  }
}
