  public String toString() {
    String str = "";
    for (int i = 0; i < this.buffer.length && this.buffer[i] != 0; i += 2)

    return str;
  }

package com.gigatronik.framework.plugin.hardware.usb.hid;

import com.gigatronik.dart2.framework.plugins.exceptions.HardwareInterfaceException;
import org.apache.commons.codec.binary.Hex;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

public class RxPoller implements Runnable {
  final Logger logger = LoggerFactory.getLogger(getClass());
  
  private boolean doStop;
  
  private final Thread THREAD;
  
  private final HidDeviceImpl DEVICE;
  
  public RxPoller(HidDeviceImpl device) {
    this.doStop = false;
    this.DEVICE = device;
    this.THREAD = new Thread(this);
    this.THREAD.setName(String.format("GtIoT-HidRxPoller_0x%02X_0x%02X", this.DEVICE.pid, this.DEVICE.vid));
    this.THREAD.start();
    this.logger.debug("RX-Polling Thread {} spawned", this.THREAD.getName());
  }
  
  public void run() {
    while (!this.doStop) {
      readSingleFrame();
      try {
        Thread.sleep(this.DEVICE.getTurnaroundTime());
      } catch (InterruptedException e) {
        this.logger.error("Thread sleep interrupted", (Throwable)e);
      } 
    } 
  }
  
  public void stop() {
    this.doStop = true;
    try {
      this.THREAD.join();
    } catch (InterruptedException e) {
      this.logger.error("Thread join interrupted", (Throwable)e);
    } 
  }
  
  private boolean readSingleFrame() {
    boolean dataAvailable = false;
    if (this.DEVICE.isConnected() && this.DEVICE.isOpen())
      try {
        byte[] rxbuffer = new byte[65];
        int result = this.DEVICE.read(rxbuffer, 0);
        dataAvailable = (result > 0);
        if (dataAvailable) {
          this.logger.trace("RX! Device: {} Data: {}", this.DEVICE, new String(Hex.encodeHex(rxbuffer, false)).replaceAll("(.{2})", "$1 "));
          this.DEVICE.onData(rxbuffer);
        } 
      } catch (HardwareInterfaceException e) {
        this.logger.error("Unable to read", (Throwable)e);
      }  
    return dataAvailable;
  }
}
