  SwMinRequiredVersionCheckResult compareEcuFirmwareVersionWithMinRequiredVersionFromContainer(BoschConstants.SwType swType, Version swVersionEcu) throws IOException, NumberFormatException, VersionCheckFailedException {
    String swMinRequiredVersionString = null;
    if (this.cffController.isCff3Loaded())
      swMinRequiredVersionString = this.cffController.getCff3Interface().getMinRequiredVersionUpdateConstraintCff3(swType, this.componentType, this.containerConfig.getConfiguredParId()); 
    if (this.cffController.isCff2Loaded())

    LOGGER.info("SwMinRequiredVersion for ECU '{}' for application-id '{}' for swType '{}' from container is '{}'", this.ecuName, this.containerConfig.getConfiguredParId(), swType, swMinRequiredVersionString);

      throw new VersionCheckFailedException("SwMinRequiredVersion='" + swMinRequiredVersion + "' or swVersionEcu='" + swVersionEcu + "' is invalid!");

  private FlashProcedure.FlashState checkContainerBikeCompatibility(boolean forceFlash) {
    FlashProcedure.CheckResult checkResult = check();

      LOGGER.info("Skip programming of '{}' because it is up to date", this.ecuName);
      fireProgressChanged(100);
      return FlashProcedure.FlashState.SKIPPED;
    } 
    if (checkResult.isNegative()) {
      LOGGER.info("Cancel programming of '{}' because it is not allowed", this.ecuName);
      return FlashProcedure.FlashState.FLASH_ERROR;

  public FlashProcedureResult execute(boolean forceFlash) {
    LOGGER.info("Executing flash procedure for '{}' with forceFlash '{}'", this.ecuName, Boolean.valueOf(forceFlash));
    FlashProcedure.FlashState flashState = checkContainerBikeCompatibility(forceFlash);
    if (flashState == FlashProcedure.FlashState.CHECKED)
      for (int retry = 1; retry <= 3 && (flashState == FlashProcedure.FlashState.FLASH_ERROR || flashState == FlashProcedure.FlashState.CHECKED); retry++) {
        LOGGER.info("Flash try number '{}' for '{}'", Integer.valueOf(retry), this.ecuName);
        if (!doesConnectedEcuMatchBackUppedIdentificationData()) {
          sleepWithLogging(1000);
        } else {
          if (isHidDeviceFlashed())
            this.diagServer.setHidMessageInterval(2500000L * (long)retry); 
          flashState = flashEcu();
        } 
      }  
    if (isHidDeviceFlashed())
      this.diagServer.setHidMessageIntervalDefault(); 
    if (flashState == FlashProcedure.FlashState.FLASH_ERROR)
      LOGGER.error("Failed to program '{}'", this.ecuName); 
    this.flashStatistic.setFlashState(flashState);
    return new FlashProcedureResult(flashState, new FlashStatistic[] { this.flashStatistic });
  }

  protected FlashProcedure.FlashState flashEcu() {

  private void executeHardResetAndExtendedSession() {
    CommunicationWrapper.Response response = this.comWrapper.executeHardReset(this.ecuName, false);
    for (int attempts = 2; attempts > 0 && response != CommunicationWrapper.Response.POSITIVE; attempts--) {
      LOGGER.debug("FlashEcu: Hard reset (left attempts='{}')", Integer.valueOf(attempts));
      waitControlled(1000);
      Constants.EcuMode mode = this.comWrapper.getContext(this.ecuName);
      if (mode == Constants.EcuMode.APP) {
        executeExtendedSessionAfterHardReset();
        return;
      } 
      prepareHardResetRetry(this.ecuName, (response == CommunicationWrapper.Response.TIMEOUT));
      response = this.comWrapper.executeHardReset(this.ecuName, true);
    } 
    if (response == CommunicationWrapper.Response.POSITIVE) {
      waitAfterHardReset();
      executeExtendedSessionAfterHardReset();
    } 
  }

    LOGGER.debug("FlashEcu: Wait {} ms for ecu {} to be ready", Integer.valueOf(1000), this.componentType);

      this.comWrapper.startExtendedSession(this.ecuName, 5);
    } 
  }

  protected List<String> executeFlashJob() {
    List<String> result = new ArrayList<>();
    Thread t = new Thread(() -> {
          Thread.currentThread().setName("FlashJob-Thread-" + Thread.currentThread().getId());
          result.addAll(startFlashRoutine(this.ecuName));
        });
    t.start();
    while (t.isAlive()) {
      int progress = this.diagServer.getFlashProgress(this.ecuName);
      fireProgressChanged(progress);
      sleep(250);
    } 
    if (result.isEmpty()) {
      result.add("NEGATIVE");
      result.add("Programming " + this.ecuName + " failed.");
      LOGGER.warn("Programming {} failed.", this.ecuName);
    } 
    LOGGER.info("Result from Flash {}: {}", this.ecuName, result.toString());
    return result;
  }

  }
  
  private List<String> startFlashRoutine(String ecu) throws NumberFormatException {
    BoschContainerDataElementWrapper boschContainerDuParaData;
    try {
      boschContainerDuParaData = this.cffController.getBoschContainerCommonInterface().getDataElement(BoschConstants.SwType.SW_TYPE_CFF2_APP, this.componentType);
      LOGGER.debug("Flashing file: '{}' on ECU '{}'", boschContainerDuParaData.getDataContent(), ecu);
    } catch (FlashDataNotAvailableException ex) {
      List<String> result = new ArrayList<>();
      result.add("NEGATIVE");
      result.add("FLASH DATA CORRUPT");
      LOGGER.error("Flash data not found in container", (Throwable)ex);
      return result;
    } 
    FlashData flashData = this.cffController.getBoschContainerCommonInterface().getBoschContainerFlashData(this.componentType, boschContainerDuParaData);
    LOGGER.debug("Flashing file: '{}' on ECU '{}'", boschContainerDuParaData.getDataContent(), ecu);
    if (flashData.getResult().contains("NEGATIVE"))
      return flashData.getResult(); 
    if (StringUtils.equals(flashData.getAppDataContentFormat(), "hex")) {
      FlashDataSet fdsBoot = createFlashDataSetMultipleBlocks(flashData, true);
      FlashDataSet fdsApp = createFlashDataSetMultipleBlocks(flashData, false);
      return this.diagServer.flashStartDownload(ecu, fdsBoot, fdsApp);
    } 
    if (StringUtils.equals(flashData.getAppDataContentFormat(), "bin")) {
      FlashDataSet fdsBoot = createFlashDataSetSingleBlock(flashData, true);
      FlashDataSet fdsApp = createFlashDataSetSingleBlock(flashData, false);
      return this.diagServer.flashStartDownload(ecu, fdsBoot, fdsApp);
    } 
    LOGGER.warn("Invalid content format: {}! No programming initiated for {}", flashData.getAppDataContentFormat(), ecu);
    return flashData.getResult();
  }
  
  protected FlashDataSet createFlashDataSetMultipleBlocks(FlashData fd, boolean bootData) {
    List<byte[]> addressList, dataList;
    byte[] checksum;
    boolean encrypted;
    List<Integer> blockSizes;
    if (bootData) {
      addressList = fd.getSwilHexData().get(0);
      dataList = fd.getSwilHexData().get(1);
      checksum = fd.getSwilDrvChecksum();
      encrypted = fd.isIsSwilDrvEncrypted();
      blockSizes = fd.getSwilBlockSizes();
    } else {
      addressList = fd.getAppHexData().get(0);
      dataList = fd.getAppHexData().get(1);
      checksum = fd.getAppChecksum();
      encrypted = fd.isIsAppEncrypted();
      blockSizes = fd.getAppBlockSizes();
    } 
    FlashDataSet fds = null;
    if (dataList != null) {
      fds = new FlashDataSet(encrypted);
      for (int n = 0; n < dataList.size(); n++) {
        fds.addBlock(addressList.get(n), dataList.get(n));
        if (encrypted)
          fds.setSize(n, blockSizes.get(n).intValue()); 
      } 
      fds.setChecksum(checksum);
    } 
    return fds;
  }
  
  private FlashDataSet createFlashDataSetSingleBlock(FlashData fd, boolean bootData) {
    FlashDataSet fds = new FlashDataSet();
    fds.setChecksum(bootData ? fd.getSwilDrvChecksum() : fd.getAppChecksum());
    byte[] address = bootData ? new byte[] { 0, 0, 0, 48 } : new byte[] { 0, 0, 0, 80 };
    if (!isClassicPlusEcu())
      LOGGER.warn("BIN format used for flashing non classic plus ECU: {}", this.ecuName); 
    fds.addBlock(address, bootData ? fd.getSwilDrvData() : fd.getAppData());
    return fds;
  }

    this.subscribers.forEach(listener -> listener.onFlashProgressChanged(progType, this.componentType, progress, ""));
  }

  public BmsDualFlashProcedure(String ecuName1, String ecuName2, EcuIdentification pp1Config, EcuIdentification pp2Config, ContainerConfiguration containerConfig) {

    this.ecuNameBms2 = ecuName2;

    this.waitingTimeAfterFlashing = SettingsProxy.getInstance().getGlobalSettingsNoException().getInteger("waitingTimeAfterDualBatFlashing", 2000);
    this.waitingTimeBetweenHardResets = SettingsProxy.getInstance().getGlobalSettingsNoException().getInteger("waitingTimeBetweenHardResetsDualBatFlashing", 100);

  public FlashProcedure.CheckResult check() {
    this.bms1CheckResult = this.bms1FlashProcedure.check();
    this.bms2CheckResult = this.bms2FlashProcedure.check();
    return getCombinedCheckResult(this.bms1CheckResult, this.bms2CheckResult);
  }
  
  private FlashProcedure.CheckResult getCombinedCheckResult(FlashProcedure.CheckResult bms1, FlashProcedure.CheckResult bms2) {
    if (bms1 == FlashProcedure.CheckResult.FIRMWARE_DOWNGRADE || bms2 == FlashProcedure.CheckResult.FIRMWARE_DOWNGRADE)
      return FlashProcedure.CheckResult.FIRMWARE_DOWNGRADE; 
    if (bms1 == FlashProcedure.CheckResult.FIRMWARE_EQUAL && bms2 == FlashProcedure.CheckResult.FIRMWARE_EQUAL)
      return FlashProcedure.CheckResult.FIRMWARE_EQUAL; 
    if (isFlashingAllowed(bms1) && isFlashingAllowed(bms2))
      return FlashProcedure.CheckResult.FIRMWARE_UPDATE; 
    return bms1;
  }
  
  private boolean isFlashingAllowed(FlashProcedure.CheckResult checkResult) {
    return Arrays.<FlashProcedure.CheckResult>asList(FlashProcedure.CheckResult.FIRMWARE_EQUAL, FlashProcedure.CheckResult.FIRMWARE_UPDATE).contains(checkResult);
  }
  
  public FlashProcedureResult execute(boolean forceFlash) {
    FlashProcedure.FlashState flashSuccess = FlashProcedure.FlashState.FLASH_ERROR;
    FlashProcedure.CheckResult combinedCheckResult = check();
    combinedCheckResult = handleForceFlashState(forceFlash, combinedCheckResult);
    flashSuccess = handleSkippedState(flashSuccess);
    if (combinedCheckResult == FlashProcedure.CheckResult.FIRMWARE_UPDATE)
      flashSuccess = flashEcu(); 
    return new FlashProcedureResult(flashSuccess, new FlashStatistic[] { getBms1FlashStatistic(), getBms2FlashStatistic() });
  }
  
  private FlashProcedure.CheckResult handleForceFlashState(boolean forceFlash, FlashProcedure.CheckResult combinedCheckResult) {
    if (forceFlash && isFlashingAllowed(combinedCheckResult)) {
      combinedCheckResult = FlashProcedure.CheckResult.FIRMWARE_UPDATE;
      this.bms1CheckResult = FlashProcedure.CheckResult.FIRMWARE_UPDATE;
      this.bms2CheckResult = FlashProcedure.CheckResult.FIRMWARE_UPDATE;
    } 
    return combinedCheckResult;
  }
  
  private FlashProcedure.FlashState handleSkippedState(FlashProcedure.FlashState flashSuccess) {
    if (this.bms1CheckResult == FlashProcedure.CheckResult.FIRMWARE_EQUAL) {
      this.bms1FlashProcedure.fireProgressChanged(100);
      getBms1FlashStatistic().setFlashState(FlashProcedure.FlashState.SKIPPED);
    } 
    if (this.bms2CheckResult == FlashProcedure.CheckResult.FIRMWARE_EQUAL) {
      this.bms2FlashProcedure.fireProgressChanged(100);
      getBms2FlashStatistic().setFlashState(FlashProcedure.FlashState.SKIPPED);
    } 
    if (this.bms1CheckResult == FlashProcedure.CheckResult.FIRMWARE_EQUAL && this.bms2CheckResult == FlashProcedure.CheckResult.FIRMWARE_EQUAL)
      flashSuccess = FlashProcedure.FlashState.SKIPPED; 
    return flashSuccess;
  }
  
  FlashProcedure.FlashState flashEcu() {
    BmsDualFlashResult result = new BmsDualFlashResult(this.bms1CheckResult, this.bms2CheckResult);
    for (int retry = 1; retry <= 3 && !result.wasFlashingSuccessful(); retry++) {
      LOGGER.info("Flash try number '{}' for '{}' (state='{}')", retry, "Combined BMS", result);
      result = flashEcuTry(result);
      result.prepareNextIteration();
    } 
    saveResultForFlashSuccessCounterLog(result);
    return result.wasFlashingSuccessful() ? FlashProcedure.FlashState.FLASHED : FlashProcedure.FlashState.FLASH_ERROR;
  }
  
  private BmsDualFlashResult flashEcuTry(BmsDualFlashResult prequel) {
    BmsFlashState bms1Result = prequel.bms1FlashState;
    BmsFlashState bms2Result = prequel.bms2FlashState;
    waitNow(this.waitingTimeBeforeFlashing);
    this.comWrapper.functionalExtendedSession();
    this.comWrapper.functionalConfigureDtcSetting(false);
    this.comWrapper.functionalConfigureEcuInternalCommunicationSetting(false);
    this.comWrapper.transmitService(this.ecuNameBms1, "PrepareProgramming_Start", null, null, false);
    this.comWrapper.transmitService(this.ecuNameBms2, "PrepareProgramming_Start", null, null, false);
    if (!this.comWrapper.checkProgrammingPreConditions(this.ecuNameBms1, 10000)) {
      bms1Result = bms1Result.getErrorState();
    } else if (!this.comWrapper.checkProgrammingPreConditions(this.ecuNameBms2, 10000)) {
      bms2Result = bms2Result.getErrorState();
    } 
    if (bms1Result.proceed() && bms2Result.proceed())
      if (!this.comWrapper.startProgrammingSessionBoolean(this.ecuNameBms1)) {
        bms1Result = bms1Result.getErrorState();
      } else if (!this.comWrapper.startProgrammingSessionBoolean(this.ecuNameBms2)) {
        bms2Result = bms2Result.getErrorState();
      }  
    if (bms1Result.proceed() && bms2Result.proceed())
      if (!this.comWrapper.securityAccessFlashBoolean(this.ecuNameBms1)) {
        bms1Result = bms1Result.getErrorState();
      } else if (!this.comWrapper.securityAccessFlashBoolean(this.ecuNameBms2)) {
        bms2Result = bms2Result.getErrorState();
      }  
    if (bms1Result.proceed() && bms2Result.proceed() && 
      !bms1Result.alreadyBeenFlashed() && this.bms1CheckResult == FlashProcedure.CheckResult.FIRMWARE_UPDATE)
      bms1Result = this.bms1FlashProcedure.executeFlashJobBoolean() ? BmsFlashState.NOW_FLASHED_SUCCESS : BmsFlashState.NOW_FLASHED_ERROR; 
    if (bms1Result.proceed() && bms2Result.proceed() && 
      !bms2Result.alreadyBeenFlashed() && this.bms2CheckResult == FlashProcedure.CheckResult.FIRMWARE_UPDATE)
      bms2Result = this.bms2FlashProcedure.executeFlashJobBoolean() ? BmsFlashState.NOW_FLASHED_SUCCESS : BmsFlashState.NOW_FLASHED_ERROR; 
    this.diagServer.hardReset(this.ecuNameBms1);
    waitNow(this.waitingTimeBetweenHardResets);
    this.diagServer.hardReset(this.ecuNameBms2);
    waitNow(this.waitingTimeAfterFlashing);
    this.comWrapper.functionalExtendedSession();
    BmsDualFlashResult result = createResultObject(bms1Result, bms2Result);
    adaptProgressBar(result);
    saveResultForFlashAbortInfoLog(result);
    return result;
  }
  
  private BmsDualFlashResult createResultObject(BmsFlashState bms1Result, BmsFlashState bms2Result) {
    BmsDualFlashResult result = new BmsDualFlashResult(bms1Result, bms2Result);
    LOGGER.debug("Quit current flash try with state='{}'", result);
    return result;
  }
  
  private void adaptProgressBar(BmsDualFlashResult result) {
    if (result.hasBms1BeenFlashedNow())
      this.bms1FlashProcedure.fireProgressChanged((result.bms1FlashState == BmsFlashState.NOW_FLASHED_SUCCESS) ? 100 : 0); 
    if (result.hasBms2BeenFlashedNow())
      this.bms2FlashProcedure.fireProgressChanged((result.bms2FlashState == BmsFlashState.NOW_FLASHED_SUCCESS) ? 100 : 0); 
  }
  
  private void saveResultForFlashAbortInfoLog(BmsDualFlashResult result) {
    if (result.bms1FlashState == BmsFlashState.NOW_FLASHED_ERROR)
      this.bms1FlashProcedure.fireFlashError(FlashErrorType.ERROR_PROGRAMMING); 
    if (result.bms2FlashState == BmsFlashState.NOW_FLASHED_ERROR)
      this.bms2FlashProcedure.fireFlashError(FlashErrorType.ERROR_PROGRAMMING); 
  }

  }

  static class BmsDualFlashResult {
    private BmsDualFlashProcedure.BmsFlashState bms1FlashState;
    
    private BmsDualFlashProcedure.BmsFlashState bms2FlashState;
    
    BmsDualFlashResult(FlashProcedure.CheckResult bms1CheckResult, FlashProcedure.CheckResult bms2CheckResult) {
      this.bms1FlashState = (bms1CheckResult == FlashProcedure.CheckResult.FIRMWARE_UPDATE) ? BmsDualFlashProcedure.BmsFlashState.NOT_YET_FLASHED : BmsDualFlashProcedure.BmsFlashState.ALREADY_FLASHED_SUCCESS;
      this.bms2FlashState = (bms2CheckResult == FlashProcedure.CheckResult.FIRMWARE_UPDATE) ? BmsDualFlashProcedure.BmsFlashState.NOT_YET_FLASHED : BmsDualFlashProcedure.BmsFlashState.ALREADY_FLASHED_SUCCESS;
    }
    
    BmsDualFlashResult(BmsDualFlashProcedure.BmsFlashState bms1FlashState, BmsDualFlashProcedure.BmsFlashState bms2FlashState) {
      this.bms1FlashState = bms1FlashState;
      this.bms2FlashState = bms2FlashState;
    }
    
    private boolean hasBms1BeenFlashedNow() {
      return (this.bms1FlashState == BmsDualFlashProcedure.BmsFlashState.NOW_FLASHED_SUCCESS || this.bms1FlashState == BmsDualFlashProcedure.BmsFlashState.NOW_FLASHED_ERROR);
    }
    
    private boolean hasBms2BeenFlashedNow() {
      return (this.bms2FlashState == BmsDualFlashProcedure.BmsFlashState.NOW_FLASHED_SUCCESS || this.bms2FlashState == BmsDualFlashProcedure.BmsFlashState.NOW_FLASHED_ERROR);
    }
    
    private boolean hasBms1BeenFlashedSuccessful() {
      return (this.bms1FlashState == BmsDualFlashProcedure.BmsFlashState.ALREADY_FLASHED_SUCCESS || this.bms1FlashState == BmsDualFlashProcedure.BmsFlashState.NOW_FLASHED_SUCCESS);
    }
    
    private boolean hasBms2BeenFlashedSuccessful() {
      return (this.bms2FlashState == BmsDualFlashProcedure.BmsFlashState.ALREADY_FLASHED_SUCCESS || this.bms2FlashState == BmsDualFlashProcedure.BmsFlashState.NOW_FLASHED_SUCCESS);
    }
    
    private boolean wasFlashingSuccessful() {
      return (hasBms1BeenFlashedSuccessful() && hasBms2BeenFlashedSuccessful());
    }
    
    public String toString() {
      return "bms1FlashState=" + this.bms1FlashState + ", bms2FlashState=" + this.bms2FlashState;
    }
    
    private void prepareNextIteration() {
      this.bms1FlashState = prepareNextIteration(this.bms1FlashState);
      this.bms2FlashState = prepareNextIteration(this.bms2FlashState);
    }
    
    private BmsDualFlashProcedure.BmsFlashState prepareNextIteration(BmsDualFlashProcedure.BmsFlashState state) {
      return 
        (state == BmsDualFlashProcedure.BmsFlashState.NOW_FLASHED_SUCCESS || state == BmsDualFlashProcedure.BmsFlashState.ALREADY_FLASHED_SUCCESS || state == BmsDualFlashProcedure.BmsFlashState.NOW_ERROR_BUT_ALREADY_FLASHED) ? 
        
        BmsDualFlashProcedure.BmsFlashState.ALREADY_FLASHED_SUCCESS : BmsDualFlashProcedure.BmsFlashState.NOT_YET_FLASHED;
    }
  }
  
  enum BmsFlashState {
    NOT_YET_FLASHED, ALREADY_FLASHED_SUCCESS, NOW_FLASHED_SUCCESS, NOW_FLASHED_ERROR, NOW_ERROR_BUT_ALREADY_FLASHED;
    
    private boolean proceed() {
      return (this == NOT_YET_FLASHED || this == ALREADY_FLASHED_SUCCESS || this == NOW_FLASHED_SUCCESS);
    }
    
    private boolean alreadyBeenFlashed() {
      return (this == ALREADY_FLASHED_SUCCESS || this == NOW_ERROR_BUT_ALREADY_FLASHED);
    }
    
    private BmsFlashState getErrorState() {
      return (this == ALREADY_FLASHED_SUCCESS || this == NOW_ERROR_BUT_ALREADY_FLASHED) ? 
        NOW_ERROR_BUT_ALREADY_FLASHED : NOW_FLASHED_ERROR;
    }
  }
}

  public FlashProcedureResult execute(boolean forceFlash) {

    FlashProcedure.CheckResult checkResult = check();

      checkResult = FlashProcedure.CheckResult.FIRMWARE_UPDATE; 
    try {
      if (checkResult == FlashProcedure.CheckResult.FIRMWARE_EQUAL) {
        fireProgressChanged(100);
        flashState = FlashProcedure.FlashState.SKIPPED;
      } else if (checkResult == FlashProcedure.CheckResult.FIRMWARE_UPDATE) {
        if (this.resetToFactoryDefault) {
          flashState = flashEcu("ucl2_update_erase_all.xml");
        } else {
          flashState = flashEcu("ucl2_update_full.xml");
        } 
      } 
      if (flashState != FlashProcedure.FlashState.FLASH_ERROR) {
        String parId = this.containerConfig.getConfiguredParId();
        boolean splashScreenPresent = this.cffController.getBoschContainerCommonInterface().isSplashScreenPresent(this.componentType, parId);
        if (splashScreenPresent)
          flashState = flashSplashScreen(this.splashScreenParameter); 
      } 
      if (flashState != FlashProcedure.FlashState.FLASH_ERROR) {
        fireProgressChanged(100);
        fireProgressChanged(BaseFlashProcedure.FlashProgressType.HMI_STEP, 100, "");
      } 
    } catch (ConfigurationNotFoundException e) {
      LOGGER.error("Internal settings file is missing : Error -> ", (Throwable)e);
      fireFlashError(FlashErrorType.ERROR_PROGRAMMING);
    } 
    this.flashStatistic.setFlashState(flashState);
    return new FlashProcedureResult(flashState, new FlashStatistic[] { this.flashStatistic });
  }

        fireFlashError(FlashErrorType.ERROR_PROGRAMMING);
        return FlashProcedure.FlashState.FLASH_ERROR;
      } 
      if (flashInfo.contains("TIMEOUT")) {
        fireFlashError(FlashErrorType.NYON_TIMEOUT);
        return FlashProcedure.FlashState.FLASH_ERROR;
      } 
      fireProgressChanged(100);
      LOGGER.info("Splashscreen successfully flashed!");
      return FlashProcedure.FlashState.FLASHED;
    } catch (ConfigurationNotFoundException e) {
      LOGGER.error("Internal settings file is missing : Error -> ", (Throwable)e);
      return FlashProcedure.FlashState.FLASH_ERROR;
    } 
  }
  
  private List<String> startFlashProcess(String uclFilePath, String uclListName, boolean flashingSplashcreen) throws ConfigurationNotFoundException {
    ControlProcess process = new ControlProcess();
    String autoPlay = process.disableAutoPlay();
    process.startFormatDiskPopupBlocker();
    List<String> result = flashStartWithMfgTool(uclFilePath, uclListName, flashingSplashcreen);
    process.stopFormatDiskPopupBlocker();
    process.enableAutoPlay(autoPlay);
    return result;
  }
  
  private List<String> flashStartWithMfgTool(String uclFilePath, String uclListName, boolean flashingSplashcreen) throws ConfigurationNotFoundException {
    List<String> result = new ArrayList<>();
    LOGGER.info("Loading MfgTool..");
    MfgToolLibCtrl mfgToolLibCtrl = new MfgToolLibCtrl();
    MfgLibError errorCode = mfgToolLibCtrl.initializeOperation(uclFilePath, uclListName);
    if (errorCode.isError()) {
      result.add("NEGATIVE");
      result.add(errorCode.toString());
      return result;
    } 
    errorCode = mfgToolLibCtrl.startOperation();
    if (errorCode.isError()) {
      result.add("NEGATIVE");
      result.add(errorCode.toString());
      return result;
    } 
    int timeOutMax = 60;
    int timeOutCount = 0;
    int lastProgress = 0;
    while (mfgToolLibCtrl.getOperationStatus() == 3) {
      int progress = mfgToolLibCtrl.getProgress();
      fireProgressChanged(progress);
      int subProgress = mfgToolLibCtrl.getIntermediateProgress();
      fireProgressChanged(BaseFlashProcedure.FlashProgressType.HMI_STEP, mfgToolLibCtrl.getIntermediateProgress(), mfgToolLibCtrl.getInfoText());
      int currentProgress = progress * 100 + subProgress;
      if (currentProgress == lastProgress) {
        timeOutCount++;
      } else {
        timeOutCount = 0;
        lastProgress = currentProgress;
      } 
      if (timeOutCount > timeOutMax)
        break; 
      sleep(1000);
    } 
    mfgToolLibCtrl.stopOperation(-1);
    if (mfgToolLibCtrl.getOperationStatus() == 1) {
      if ((this.mapsInfoHashMap != null && !this.mapsInfoHashMap.isEmpty()) || flashingSplashcreen) {
        LOGGER.info("Wait for BUI270 to restart..");
        fireProgressChanged(BaseFlashProcedure.FlashProgressType.HMI_STEP, 98, "Waiting for Nyon to restart ...");
        sleep(20000);
      } else {
        LOGGER.info("Wait for KSS to get flashed..");
        fireProgressChanged(BaseFlashProcedure.FlashProgressType.HMI_STEP, 95, "Updating KSS. See progress on cycle computer.");
        sleep(90000);
      } 
      timeOutCount = 0;
      timeOutMax = 360000;
      boolean nyonConnected = false;
      while (!nyonConnected && timeOutCount < timeOutMax) {
        Constants.HidDevice device = this.diagServer.getDevice();
        if (device == Constants.HidDevice.BUI270_APP || device == Constants.HidDevice.BUI275_APP)
          nyonConnected = true; 
        sleep(5000);
        timeOutCount += 5000;
      } 
      if (nyonConnected) {
        result.add("POSITIVE");
      } else {
        result.add("TIMEOUT");
      } 
    } else {
      result = new ArrayList<>();
      result.add("NEGATIVE");
      result.add(mfgToolLibCtrl.getErrorString());
    } 
    return result;
  }

  private boolean writeApplicationParameter(String userLevel) {
    fireProgressChanged(BaseFlashProcedure.FlashProgressType.PARAM, 1);
    String applicationParameterReferenceId = this.containerConfig.getConfiguredParId();
    fireProgressChanged(BaseFlashProcedure.FlashProgressType.PARAM, 5);
    this.comWrapper.functionalExtendedSession();
    sleep(500);
    this.comWrapper.securityAccess(this.ecuName, TokenPluginProxy.getInstance().getAccessLevel(), 1);
    fireProgressChanged(BaseFlashProcedure.FlashProgressType.PARAM, 10);
    if (userLevel.equals("IBD")) {
      List<String> partNumber = this.comWrapper.transmitService(this.ecuName, "DU_Part_Number_Customer_Read", "ASCII_PartNumber", null, false);
      if (partNumber.contains("NEGATIVE")) {
        LOGGER.error("Error reading application parameter id");
        fireFlashError(FlashErrorType.WRITE_APPLICATION_PARAMS_FAILED, BaseFlashProcedure.FlashProgressType.PARAM);
        return false;
      } 
      String appliParaId = BoschHelper.customizePartNumberCustomer(partNumber.get(1));
      if (!appliParaId.equals(applicationParameterReferenceId)) {
        fireFlashError(FlashErrorType.PRE_CHECK_ERROR_CFF, BaseFlashProcedure.FlashProgressType.PARAM);
        return false;
      } 
    } 
    fireProgressChanged(BaseFlashProcedure.FlashProgressType.PARAM, 50);
    if (programApplicationParameter().contains("POSITIVE")) {
      String newAppParaRef = BoschHelper.customizePartNumberCustomerWrite(applicationParameterReferenceId);
      List<String> result = this.comWrapper.transmitService(this.ecuName, "DU_Part_Number_Customer_Write", "ASCII_PartNumber", newAppParaRef, false);
      if (result.contains("NEGATIVE")) {
        LOGGER.error("Error writing application parameter id");
        fireFlashError(FlashErrorType.WRITE_APPLICATION_PARAMS_FAILED, BaseFlashProcedure.FlashProgressType.PARAM);
        return false;
      } 
      fireProgressChanged(BaseFlashProcedure.FlashProgressType.PARAM, 100);
      return true;
    } 
    fireFlashError(FlashErrorType.WRITE_APPLICATION_PARAMS_FAILED, BaseFlashProcedure.FlashProgressType.PARAM);
    return false;
  }
  
  private List<String> programApplicationParameter() {
    byte[] decodedParamFlashData = null;
    byte[] paramFlashData = null;
    byte[] paramChecksum = null;
    List<String> result = new ArrayList<>();
    SignatureHelper sigChecker = new SignatureHelper();
    BoschContainerDataElementWrapper data = null;
    try {
      data = this.cffController.getBoschContainerCommonInterface().getDuAppParamData(this.containerConfig.getConfiguredParId());
    } catch (FlashDataNotAvailableException ex) {
      LOGGER.error("Could not extract data from container", (Throwable)ex);
      result.add("NEGATIVE");
      result.add("FLASH DATA CORRUPT");
      return result;
    } 
    paramChecksum = BoschHelper.hexStringToByteArray(data.getCrc32());
    String flashDataPath = this.cffController.getZipPath() + this.cffController.getZipPath() + File.separator;
    paramFlashData = BoschHelper.readBinaryData(flashDataPath);
    if ("Uniform".compareTo(data.getKeyId()) == 0) {
      decodedParamFlashData = CryptoHelper.decrypt(paramFlashData, data.getKeyIv(), null, CryptoInfo.getSkey(), CryptoConstants.getcInstance5());
      result = sigChecker.checkSignature(decodedParamFlashData, data.getSignatureId(), data.getSignatureSalt(), data.getSignatureValue());
      if (result.contains("NEGATIVE"))
        return result; 
      paramFlashData = decodedParamFlashData;
    } else {
      result = sigChecker.checkSignature(paramFlashData, data.getSignatureId(), data.getSignatureSalt(), data.getSignatureValue());
      if (result.contains("NEGATIVE"))
        return result; 
    } 
    return this.diagServer.flashWriteData(this.ecuName, paramFlashData, paramChecksum);
  }

  private void initializeFunctionalEcuName() {
    if (this.bikeIdentification.isHmiConnected()) {
      this.buiFunctionalName = this.bikeIdentification.getHmiIdentification().getComponentType().getOdxEcuName();
    } else if (this.bikeIdentification.isBctConnected()) {
      this.buiFunctionalName = "BCT";
    } 
    if (this.bikeIdentification.isBms1Connected())
      if (this.bikeIdentification.getBms1Identification().getComponentType().isBBP1xx()) {
        this.pp1FunctionalName = "BMS";
      } else if (this.bikeIdentification.getBms1Identification().getComponentType().isBBP2xx()) {
        this.pp1FunctionalName = "BBP1";
      }  
    if (this.bikeIdentification.isBms2Connected())
      this.pp2FunctionalName = "BBP2"; 
    if (this.bikeIdentification.isDuConnected())
      if (this.bikeIdentification.getDuIdentification().getComponentType().isBDU1xx()) {
        this.duFunctionalName = "DU";
      } else if (this.bikeIdentification.getDuIdentification().getComponentType().isBDUVictor()) {
        this.duFunctionalName = "BDU";
      }  
    if (this.bikeIdentification.isAbsConnected())
      this.absFunctionalName = "ABS"; 
    if (this.bikeIdentification.isBcsConnected())
      this.bcsFunctionalName = "BCS"; 
  }

  protected boolean startExtendedSession(String ecu, int maxTry) {
    boolean response = !this.boschServer.startExtendedSession(ecu).contains("NEGATIVE");
    int attempts;
    for (attempts = maxTry - 1; attempts > 0 && !response; attempts--) {
      this.boschServer.startDefaultSession(ecu);
      sleep(500);
      response = !this.boschServer.startExtendedSession(ecu).contains("NEGATIVE");
    } 
    LOGGER.debug("{} status in extended session: {}, number of attempts: {}", ecu, response, maxTry - attempts);
    return response;
  }
  
  protected boolean startExtendedSession(String ecu) {
    return startExtendedSession(ecu, 1);
  }
  
  protected List<String> startProgrammingSession(String ecu, int maxTry) {
    int numTry = 0;
    boolean success = false;
    List<String> result = new ArrayList<>();
    while (!success && numTry < maxTry) {
      result = this.boschServer.startProgrammingSession(ecu);
      success = !result.contains("NEGATIVE");
      if (success) {
        LOGGER.debug("{} in programming session", ecu);
      } else {
        LOGGER.error("Failed to set {} in programming session.", ecu);
        if (maxTry > 1)
          sleep(500); 
      } 
      numTry++;
    } 
    return result;
  }
  
  protected boolean securityAccess(String ecu, int accessLevel, int maxTry) {
    int numTry = 0;
    boolean success = false;
    while (!success && numTry < maxTry) {
      List<String> result = this.boschServer.securityAccess(ecu, accessLevel);
      success = !result.contains("NEGATIVE");
      if (!success) {
        sleep(500);
        numTry++;
      } 
    } 
    return success;
  }
  
  protected List<String> startProgrammingSession(String ecu) {
    return startProgrammingSession(ecu, 1);
  }
  
  protected boolean startProgrammingSessionBoolean(String ecu) {
    return !startProgrammingSession(ecu, 1).contains("NEGATIVE");
  }
  
  protected List<String> securityAccessFlash(String ecu) {
    List<String> rslt = new ArrayList<>();
    if (securityAccess(ecu, 0, 3)) {
      rslt.add("POSITIVE");
    } else {
      rslt.add("NEGATIVE");
      rslt.add("Security access failed.");
    } 
    return rslt;
  }
  
  protected boolean securityAccessFlashBoolean(String ecu) {
    return !securityAccessFlash(ecu).contains("NEGATIVE");
  }
  
  protected boolean securityAccessFlashing(String ecu) {
    List<String> result = securityAccessFlash(ecu);
    return result.contains("POSITIVE");
  }
  
  protected boolean checkProgrammingPreConditions(String ecu, int maxDurationTime) {
    int numTry = 0;
    int maxTry = maxDurationTime / 500;
    boolean success = false;
    while (!success && numTry < maxTry) {
      List<String> arrayresult = transmitService(ecu, "CheckProgrammingPreconditions_Start", null, null, false);
      success = !arrayresult.contains("NEGATIVE");
      if (success)
        return true; 
      sleep(500);
      numTry++;
    } 
    return false;
  }
  
  protected Response executeHardReset(String ecu, boolean waitForReconnectionAndDoHandshake) {
    byte[] hardResetPdu = { 17, 1 };
    byte[] response = transmitPdu(ecu, hardResetPdu);
    if (waitForReconnectionAndDoHandshake) {
      byte[] testerPresentPdu = { 62, 0 };
      for (int i = 0; i < deviceReconnectionTimeout / 500; i++) {
        sleep(500);
        Constants.HidDevice handshakeResponse = this.boschServer.getComLayer().initDevice();
        if (handshakeResponse != Constants.HidDevice.NONE) {
          response = transmitPdu(ecu, testerPresentPdu);
          return handleResponse(response);
        } 
      } 
      return Response.NEGATIVE;
    } 
    return handleResponse(response);
  }
  
  private Response handleResponse(byte[] response) {
    if (Helper.isTimeoutResponse(response))
      return Response.TIMEOUT; 
    if (Helper.isNegativeResponse(response))
      return Response.NEGATIVE; 
    return Response.POSITIVE;
  }
  
  protected List<String> transmitService(String ecu, String service, String parameterName, String parameterValue, boolean unit) {
    List<String> parameterList = new ArrayList<>();
    try {
      if (parameterName != null && !parameterName.isEmpty() && parameterValue != null && !parameterValue.isEmpty())
        this.boschServer.setRequestParameterValue(ecu, service, parameterName, parameterValue); 
      List<String> result = this.boschServer.transmitService(ecu, service, unit);
      if (!result.isEmpty()) {
        if (StringUtils.equals(result.get(0), "NEGATIVE")) {
          parameterList.add("NEGATIVE");
          parameterList.add("NEGATIVE");
        } else if (result.size() > 5) {
          parameterList.addAll(result.subList(5, result.size()));
        } 
      } else {
        LOGGER.warn("transmitService {}.{} - ResultSet is empty!", ecu, service);
      } 
    } catch (DatabaseException e) {
      LOGGER.error("transmitService {}.{} - Error transmitting service", ecu, service, e);
      parameterList.add("NEGATIVE");
      parameterList.add("NEGATIVE");
    } 
    return parameterList;
  }
  
  public Constants.EcuMode getContext(String ecu) {
    List<String> result = new ArrayList<>();
    switch (ecu) {
      case "DU":
      case "BDU":
        result = this.boschServer.transmitService(ecu, "DU_getContext_Read", false);
        break;
      case "ABS":
        result = this.boschServer.transmitService(ecu, "ABS_getContext_Read", false);
        break;
      case "BCS":
        result = this.boschServer.transmitService(ecu, "BCS_getContext_Read", false);
        break;
      case "BMS":
      case "BBP1":
      case "BBP2":
        result = this.boschServer.transmitService(ecu, "BMS_getContext_Read", false);
        break;
      default:
        LOGGER.warn("The ecuName doesn't correspond to any Ecu: {}", ecu);
        break;
    } 
    return parseResponseToEcuMode(result);
  }
  
  private Constants.EcuMode parseResponseToEcuMode(List<String> response) {
    if (response.isEmpty())
      return Constants.EcuMode.NONE; 
    for (int i = 0; i < response.size(); i++) {
      if (response.get(i).equals("active_context") && response.get(i + 1) != null) {
        switch ((String)response.get(i + 1)) {
          case "Application running":
            return Constants.EcuMode.APP;
          case "Bootloader running":
            return Constants.EcuMode.FBL;
        } 
        return Constants.EcuMode.NONE;
      } 
    } 
    return Constants.EcuMode.NONE;
  }
  
  public void functionalExtendedSession() {
    functionalExtendedSession(1);
  }
  
  public void functionalExtendedSession(int retry) {
    boolean expectedResponse = false;
    for (int attempts = retry; attempts > 0 && !expectedResponse; attempts--) {
      if (StringUtils.isNotEmpty(this.buiFunctionalName))
        expectedResponse = startExtendedSession(this.buiFunctionalName); 
      expectedResponse = startBatteriesExtendedSession(expectedResponse);
      if (!expectedResponse) {
        if (attempts > 1)
          sleep(500); 
      } else {
        if (StringUtils.isNotEmpty(this.duFunctionalName))
          expectedResponse = startExtendedSession(this.duFunctionalName); 
        if (StringUtils.isNotEmpty(this.absFunctionalName))
          expectedResponse &= startExtendedSession(this.absFunctionalName); 
        if (StringUtils.isNotEmpty(this.bcsFunctionalName))
          expectedResponse &= startExtendedSession(this.bcsFunctionalName); 
        if (!expectedResponse && attempts > 1)
          sleep(500); 
      } 
    } 
  }
  
  private boolean startBatteriesExtendedSession(boolean expectedResponse) {
    if (StringUtils.isNotEmpty(this.pp1FunctionalName))
      expectedResponse &= startExtendedSession(this.pp1FunctionalName); 
    if (StringUtils.isNotEmpty(this.pp2FunctionalName))
      expectedResponse &= startExtendedSession(this.pp2FunctionalName); 
    return expectedResponse;
  }
  
  public void functionalSecurityAccess() {
    int level = TokenPluginProxy.getInstance().getAccessLevel();
    if (StringUtils.equals(this.buiFunctionalName, ComponentType.COMP_BUI330.getOdxEcuName()))
      securityAccess(this.buiFunctionalName, level, 1); 
    if (StringUtils.isNotEmpty(this.pp1FunctionalName))
      securityAccess(this.pp1FunctionalName, level, 3); 
    if (StringUtils.isNotEmpty(this.pp2FunctionalName))
      securityAccess(this.pp2FunctionalName, level, 3); 
    if (StringUtils.isNotEmpty(this.duFunctionalName))
      securityAccess(this.duFunctionalName, level, 3); 
    if (StringUtils.isNotEmpty(this.absFunctionalName))
      securityAccess(this.absFunctionalName, level, 1); 
  }
  
  void functionalConfigureDtcSetting(boolean setting) {
    byte[] pduTx;
    int i = setting ? 1 : 2;
    if (this.bikeIdentification.getSystemType() == Constants.SystemType.CLASSIC_PLUS) {
      pduTx = new byte[] { -123, i };
    } else {
      pduTx = new byte[] { -123, i, -1, -1, -1 };
    } 
    transmitFunctionalPdu(pduTx);
  }
  
  boolean controlDtcSettings(String ecu, boolean on, int maxTries) {
    for (int currentTry = 1; currentTry <= maxTries; currentTry++) {
      if (controlDtcSettings(ecu, on))
        return true; 
      if (currentTry < maxTries)
        sleep(500); 
    } 
    return false;
  }
  
  boolean controlDtcSettings(String ecu, boolean on) {
    byte[] txPdu;
    if (StringUtils.startsWith(ecu, "DU") || StringUtils.startsWith(ecu, "BMS")) {
      txPdu = new byte[] { -123, on ? 1 : 2 };
    } else {
      txPdu = new byte[] { -123, on ? 1 : 2, -1, -1, -1 };
    } 
    byte[] rxPdu = this.boschServer.transmitPdu(ecu, txPdu);
    return Helper.isPositiveResponse(txPdu, rxPdu);
  }
  
  boolean enableRxTx(String ecu, boolean on, int maxTries) {
    for (int currentTry = 1; currentTry <= maxTries; currentTry++) {
      List<String> response = this.boschServer.silentMode(ecu, !on);
      if (response.contains("POSITIVE"))
        return true; 
      if (currentTry < maxTries)
        sleep(500); 
    } 
    return false;
  }

  }

    if (StringUtils.isNotEmpty(this.buiFunctionalName))

  protected void sleep(int ms) {
    try {
      Thread.sleep((long)ms);

    }

public class Cui11xFlashProcedure extends HidBaseFlashProcedure {
  private static final Logger LOGGER = LoggerFactory.getLogger(Cui11xFlashProcedure.class);
  
  public Cui11xFlashProcedure(String ecuName, EcuIdentification ecuIdentification, ContainerConfiguration containerConfig) {
    super(ecuName, ecuIdentification, BaseFlashProcedure.FlashProgressType.HMI, containerConfig);
  }
  
  public FlashProcedure.CheckResult check() {
    if (isCobiFlashActivatedInSettings())
      return super.check(); 
    return FlashProcedure.CheckResult.FIRMWARE_EQUAL;
  }
  
  public FlashProcedureResult execute(boolean forceFlash) {
    this.flashStatistic.setFlashState(FlashProcedure.FlashState.SKIPPED);
    FlashProcedureResult result = new FlashProcedureResult(FlashProcedure.FlashState.SKIPPED, new FlashStatistic[] { this.flashStatistic });
    if (isCobiFlashActivatedInSettings()) {
      result = super.execute(forceFlash);
    } else {
      LOGGER.info("Skip programming of '{}' because it is not activated by settings flag", this.ecuName);
      fireProgressChanged(100);
    } 
    return result;
  }
  
  private boolean isCobiFlashActivatedInSettings() {
    try {
      return SettingsProxy.getInstance().getGlobalSettings().getBoolean("cobi_flash_active");
    } catch (ConfigurationNotFoundException ex) {
      LOGGER.warn("Flag for Cobi flash activation not found in settings ", ex.getMessage());
      return false;
    } 
  }
}

package com.gigatronik.beds.service.ecuflash.impl;

import com.gigatronik.beds.service.configuration.impl.ContainerConfiguration;
import com.gigatronik.beds.service.ecu.api.Du;
import com.gigatronik.ebike.common.configuration.ComponentType;
import com.gigatronik.ebike.common.configuration.Constants;
import com.gigatronik.ebike.common.configuration.EcuIdentification;
import com.gigatronik.ebike.common.configuration.ProductLine;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

public abstract class DuFlashProcedure extends BaseFlashProcedure {
  private static final Logger LOGGER = LoggerFactory.getLogger(DuFlashProcedure.class);
  
  private final Du du;
  
  private ProductLine ecuProductLine = ProductLine.NONE;
  
  private ComponentType ecuComponentType = ComponentType.NONE;
  
  public DuFlashProcedure(Du du, EcuIdentification ecuIdentification, ContainerConfiguration containerConfig) {
    super(ecuIdentification.getComponentType().getOdxEcuName(), ecuIdentification, BaseFlashProcedure.FlashProgressType.DU, containerConfig);
    this.du = du;
  }
  
  boolean doesConnectedEcuMatchBackUppedIdentificationData() {
    if (this.ecuIdentification.getComponentType().isBDUVictor()) {
      LOGGER.debug("Gather parameters from currently connected DU and compare them with given ECU identification data, to ensure ECU hasn't been exchanged since flash procedure start.");
      return checkIdentification();
    } 
    return true;
  }
  
  private boolean checkIdentification() {
    this.du.startExtendedSession();
    readIdentification();
    if ((this.ecuProductLine != ProductLine.NONE && checkProductLine()) || (this.ecuProductLine == ProductLine.NONE && 
      checkComponentType())) {
      LOGGER.info("Product line check for drive unit was successful");
      return true;
    } 
    if (this.ecuProductLine == ProductLine.NONE && this.ecuComponentType == ComponentType.NONE) {
      LOGGER.info("Reading drive unit identification data failed");
      fireFlashError(FlashErrorType.PRE_CHECK_ERROR_READ_IDENTIFICATION);
      return false;
    } 
    LOGGER.info("Product line check for drive unit was not successful");
    fireFlashError(FlashErrorType.PRE_CHECK_ERROR_PRODUCT_LINE);
    return false;
  }
  
  private boolean checkProductLine() {
    return (this.ecuProductLine == this.containerConfig.getProductLine());
  }
  
  private boolean checkComponentType() {
    return ((this.ecuComponentType.isBDU1xx() && this.containerConfig.getProductLine().isGen1()) || (
      this.ecuComponentType.isBDU2xx() && this.containerConfig.getProductLine().isGen2()) || (
      this.ecuComponentType.isBDU3xx() && this.containerConfig.getProductLine().isGen3()) || (
      this.ecuComponentType.isBDU4xx() && this.containerConfig.getProductLine().isGen4()));
  }
  
  private void readIdentification() {
    if (this instanceof ClassicDuFlashProcedure) {
      this.ecuProductLine = ProductLine.CLASSIC_PLUS;
      this.ecuComponentType = ComponentType.COMP_BDU100;
    } else {
      Constants.EcuMode mode = this.du.readEcuState();
      switch (mode) {
        case APP:
          readIdentificationFromApp();
          break;
        case FBL:
          readIdentificationFromFbl();
          break;
        default:
          clearIdentification();
          break;
      } 
    } 
  }
  
  private void clearIdentification() {
    this.ecuProductLine = ProductLine.NONE;
    this.ecuComponentType = ComponentType.NONE;
  }
  
  private void readIdentificationFromApp() {
    this.ecuProductLine = ProductLine.fromOdxName(this.du.readProductLine().toString());
    this.ecuComponentType = ComponentType.parse(this.du.readComponentType().toString());
  }
  
  private void readIdentificationFromFbl() {
    this.du.configureDtcSetting(false);
    this.du.checkProgrammingPreconditions();
    this.du.startProgrammingSession(false);
    this.du.executeSecurityAccess(Constants.AccessLevel.LEVEL_0);
    this.ecuProductLine = ProductLine.fromOdxName(this.du.readProductLineFbl().toString());
    this.ecuComponentType = ComponentType.parse(this.du.readComponentTypeFbl().toString());
    this.du.startDefaultSession(false);
  }
  
  protected abstract boolean writeApplicationParameters();
}

package com.gigatronik.beds.service.ecuflash.impl.event;

import com.gigatronik.beds.service.ecuflash.api.FlashProgressEventListener;
import java.util.ArrayList;
import java.util.List;
import java.util.concurrent.Executor;
import java.util.concurrent.Executors;
import org.apache.commons.lang3.concurrent.BasicThreadFactory;

public class FlashProgressEventPublisher {
  private final Executor executor = Executors.newSingleThreadExecutor(new BasicThreadFactory.Builder().namingPattern("FlashProgressPublisher-%d").build());
  
  private final List<FlashProgressEventListener> listeners = new ArrayList<>();
  
  public void addListener(FlashProgressEventListener listener) {
    if (!this.listeners.contains(listener))
      this.listeners.add(listener); 
  }
  
  public void removeListener(FlashProgressEventListener listener) {
    this.listeners.remove(listener);
  }
  
  public void publishEvent(FlashProgressEvent event) {
    this.listeners.forEach(listener -> this.executor.execute(() -> listener.onFlashProgress(event)));
  }
}

        break;

  }

package com.gigatronik.beds.service.ecuflash.impl;

import com.gigatronik.beds.service.ecuflash.api.FlashProcedure;
import com.gigatronik.ebike.common.configuration.ComponentType;
import java.util.ArrayList;
import java.util.HashSet;
import java.util.List;
import java.util.Set;

public class FlashResult {
  private FlashProcedure.FlashState resultState = FlashProcedure.FlashState.FLASH_ERROR;
  
  private FlashError flashError = null;
  
  private Boolean oemParamsWritten = null;
  
  private Boolean configParamsWritten = null;
  
  private Boolean appParamsWritten = null;
  
  private final Set<FlashStatistic> statistics = new HashSet<>();
  
  void addFlashStatistic(FlashStatistic statistic) {
    if (statistic == null)
      throw new NullPointerException("Flash statistic to add is null"); 
    if (getFlashStatisticOf(statistic.getEcuType()) != null)
      throw new IllegalArgumentException("A flash statistic of this component already exists"); 
    this.statistics.add(statistic);
  }
  
  void addFlashStatisticOverwrite(FlashStatistic statistic) {
    if (statistic == null)
      throw new NullPointerException("Flash statistic to add is null"); 
    FlashStatistic existingStatistic = getFlashStatisticOf(statistic.getEcuType());
    if (existingStatistic != null)
      this.statistics.remove(existingStatistic); 
    this.statistics.add(statistic);
  }
  
  public Set<FlashStatistic> getStatistics() {
    return this.statistics;
  }
  
  void setState(FlashProcedure.FlashState state) {
    this.resultState = state;
  }
  
  public FlashProcedure.FlashState getState() {
    return this.resultState;
  }
  
  List<ComponentType> getSuccessfullyFlashedComponentTypes() {
    List<ComponentType> result = new ArrayList<>();
    for (FlashStatistic stat : getStatistics()) {
      if (stat.getFlashState() == FlashProcedure.FlashState.FLASHED)
        result.add(stat.getComponentType()); 
    } 
    return result;
  }
  
  List<FlashStatistic> getAllStatisticsWithFlashErrors() {
    List<FlashStatistic> result = new ArrayList<>();
    for (FlashStatistic stat : getStatistics()) {
      if (!stat.getErrors().isEmpty())
        result.add(stat); 
    } 
    return result;
  }
  
  public FlashError getFlashError() {
    return this.flashError;
  }
  
  void setFlashError(FlashError error) {
    this.flashError = error;
  }
  
  public FlashStatistic getFlashStatisticOf(BaseFlashProcedure.FlashProgressType progressType) {
    for (FlashStatistic stat : getStatistics()) {
      if (stat.getEcuType() == progressType)
        return stat; 
    } 
    return null;
  }
  
  public boolean isSuccessful() {
    boolean flashedSuccesful = (this.resultState == FlashProcedure.FlashState.FLASHED || this.resultState == FlashProcedure.FlashState.SKIPPED);
    return (flashedSuccesful && !isWriteOemParametersFailed() && !isWriteConfigParametersFailed() && !isWriteAppParametersFailed());
  }
  
  public boolean isWriteOemParametersFailed() {
    return (this.oemParamsWritten != null && !this.oemParamsWritten);
  }
  
  public boolean isWriteConfigParametersFailed() {
    return (this.configParamsWritten != null && !this.configParamsWritten);
  }
  
  private boolean isWriteAppParametersFailed() {
    return (this.appParamsWritten != null && !this.appParamsWritten);
  }
  
  public void setOemParamsWritten(boolean success) {
    this.oemParamsWritten = success;
  }
  
  public void setConfigParamsWritten(boolean success) {
    this.configParamsWritten = success;
  }
  
  void setAppParamsWritten(boolean success) {
    this.appParamsWritten = success;
  }
  
  public String toString() {
    return String.format("FlashResult: Result-State: %s with flashError=%s, appParams-Written=%s, oemParamsWritten=%s, configParamsWritten=%s", this.resultState, this.flashError, this.appParamsWritten, this.oemParamsWritten, this.configParamsWritten);
  }
}

package com.gigatronik.beds.service.ecuflash.impl;

import com.gigatronik.ebike.common.configuration.ComponentType;
import java.io.BufferedWriter;

public class FlashResultLogger {
  private static final Logger LOGGER = LoggerFactory.getLogger(FlashResultLogger.class);
  
  private static final String DART2_HOME_PATH = System.getenv("DART2_HOME");
  
  private static final String LOG_PATH = DART2_HOME_PATH + "\\logs\\flashlog\\";
  
  private static final String FLASH_SUCCESS_COUNTER_LOG_FILENAME = "flash_success_counter.log";
  
  private static final String FLASH_ABORT_COUNTER_LOG_FILENAME = "flash_abort_counter.log";
  
  private static final String FLASH_ABORT_INFO_LOG_FILENAME = "flash_abort_info.log";
  
  private static final File FLASH_SUCCESS_COUNTER_LOGFILE = new File(LOG_PATH + "flash_success_counter.log");
  
  private static final File FLASH_ABORT_COUNTER_LOGFILE = new File(LOG_PATH + "flash_abort_counter.log");
  
  private static final File FLASH_ABORT_INFO_LOGFILE = new File(LOG_PATH + "flash_abort_info.log");
  
  private static final String KEY_RECORDING_START_DATE = "Recording-start-date";
  
  private static final String KEY_PURION = "BUI21x";
  
  private static final String KEY_INTUVIA = "BUI25x";
  
  private static final String KEY_NYON = "BUI27x";
  
  private static final String KEY_KIOX = "BUI330";
  
  private static final String KEY_NYON2 = "BUI350";
  
  private static final String KEY_BCT = "BCT";
  
  private static final String KEY_BDU_GEN1_CLASSIC_PLUS = "BDU100";
  
  private static final String KEY_BDU_GEN2 = "BDU2xx";
  
  private static final String KEY_BDU_GEN3 = "BDU3xx";
  
  private static final String KEY_BDU_GEN4 = "BDU4xx";
  
  private static final String KEY_BBP_GEN1_CLASSIC_PLUS = "BBP100";
  
  private static final String KEY_BBP_GEN2 = "BBP2xx";
  
  private static final String KEY_ABS = "BAS";
  
  private static final String KEY_BCS = "BCS250";
  
  private final ExecutorService logExecutor = Executors.newSingleThreadExecutor(new BasicThreadFactory.Builder().namingPattern("FlashResultLogger-Thread-%d").build());
  
  public void logFlashResult(FlashResult flashResult) {
    if (flashResult == null) {
      LOGGER.warn("Flash result is null. No flash statistics will be logged");
      return;
    } 
    this.logExecutor.execute(() -> {
          LOGGER.info("Logging flash result into flash logs ...");
          updateFlashCounterLog(FLASH_SUCCESS_COUNTER_LOGFILE, flashResult.getSuccessfullyFlashedComponentTypes());
          List<FlashStatistic> statisticsWithFlashErrors = flashResult.getAllStatisticsWithFlashErrors();
          if (!statisticsWithFlashErrors.isEmpty())
            logFlashAbortInformation(statisticsWithFlashErrors, flashResult.isSuccessful()); 
        });
  }
  
  private void logFlashAbortInformation(List<FlashStatistic> statisticsWithFlashErrors, boolean flashProcessSuccessful) {
    List<ComponentType> compTypes = new ArrayList<>();
    statisticsWithFlashErrors.forEach(stat -> compTypes.add(stat.getComponentType()));
    if (!flashProcessSuccessful)
      updateFlashCounterLog(FLASH_ABORT_COUNTER_LOGFILE, compTypes); 
    updateFlashAbortsLog(statisticsWithFlashErrors);
  }
  
  private void updateFlashCounterLog(File file, List<ComponentType> compTypes) {
    Properties props;
    if (compTypes == null || compTypes.isEmpty()) {
      LOGGER.error("No flashed components received to log to flash counter log");
      return;
    } 
    LOGGER.debug("Update {} with component types: {}", file.getName(), compTypes.toString());
    try {
      checkLogfile(file);
    } catch (IOException ex) {
      LOGGER.error("Failed to create logfile", (Throwable)ex);
      return;
    } 
    try {
      props = loadLogFile(file);
    } catch (IOException ex) {
      LOGGER.error("Failed to load flash counter logfile", (Throwable)ex);
      return;
    } 
    for (ComponentType compType : compTypes) {
      String key = mapToKey(compType);
      String entry = props.getProperty(key);
      props.setProperty(key, (entry != null) ? String.valueOf(Integer.valueOf(entry) + 1) : "1");
    } 
    try {
      storeLogFile(file, props);
    } catch (IOException ex) {
      LOGGER.error("Failed to store flash counter logfile", (Throwable)ex);
    } 
  }
  
  private void updateFlashAbortsLog(List<FlashStatistic> statistics) {
    try {
      checkLogfile(FLASH_ABORT_INFO_LOGFILE);
      for (FlashStatistic stat : statistics) {
        for (FlashStatistic.FlashErrorPair error : stat.getErrors())
          writeLogEntry(FLASH_ABORT_INFO_LOGFILE, generateLogEntry(stat.getComponentType(), error)); 
      } 
    } catch (IOException ex) {
      LOGGER.error("Failed to create logfile", (Throwable)ex);
    } 
  }
  
  private String generateLogEntry(ComponentType componentType, FlashStatistic.FlashErrorPair error) {
    String strDate = new SimpleDateFormat("yyyy-MM-dd HH:mm:ss").format(error.getDate());
    String compType = componentType.getName();
    String lastProgress = error.getLastProgress();
    String errorType = error.getErrorType().toString();
    String errorTypeInfo = error.getErrorType().getDescription();
    String udsError = error.getErrorCode();
    return strDate + ";" + strDate + ";" + compType + ";" + lastProgress + ";" + errorType + ";" + errorTypeInfo + "\n";
  }
  
  private void checkLogfile(File file) throws IOException {
    new File(LOG_PATH).mkdirs();
    if (file.exists())
      checkSizeOfLogFile(file); 
    if (!file.exists())
      createLogfile(file); 
  }
  
  private void createLogfile(File file) throws IOException {
    if (file.createNewFile()) {
      if (file == FLASH_ABORT_INFO_LOGFILE) {
        writeFlashAbortsLogHeader(file);
      } else {
        logInitialDate(file, new Date());
      } 
      LOGGER.debug("Created new logfile {}", file.getName());
    } 
  }
  
  private void writeFlashAbortsLogHeader(File file) {
    writeLogEntry(file, "Date;Component-Type;Last-Progress;Error-Type;Error-Type-Info;Uds-Error(s)\n");
  }
  
  private void logInitialDate(File logFile, Date date) {
    try {
      Properties props = loadLogFile(logFile);
      SimpleDateFormat sdfDate = new SimpleDateFormat("yyyy-MM-dd HH:mm");
      String strDate = sdfDate.format(date);
      props.setProperty("Recording-start-date", strDate);
      storeLogFile(logFile, props);
    } catch (IOException ex) {
      LOGGER.error("Failed to set initial date in logfile", (Throwable)ex);
    } 
  }
  
  private void checkSizeOfLogFile(File file) {
    float sizeFlashCounterLogInMb = (float)file.length() / 1048576.0F;
    if (sizeFlashCounterLogInMb >= 10.0F) {
      File newLogfile = new File(file.getParent() + file.getParent() + File.separator + "_1.log");
      if (newLogfile.exists())
        try {
          Files.delete(newLogfile.toPath());
        } catch (IOException ex) {
          LOGGER.error("Failed to delete logfile {}", newLogfile.getName(), ex);
        }  
      if (!file.renameTo(newLogfile))
        LOGGER.error("Failed to rename logfile {}", newLogfile.getName()); 
    } 
  }
  
  private String mapToKey(ComponentType compType) {
    String key;
    if (compType.isBUI21x()) {
      key = "BUI21x";
    } else if (compType.isBUI25x()) {
      key = "BUI25x";
    } else if (compType.isBUI27x()) {
      key = "BUI27x";
    } else if (compType.isBUI33x()) {
      key = "BUI330";
    } else if (compType.isBUI35x()) {
      key = "BUI350";
    } else if (compType.isBct()) {
      key = "BCT";
    } else if (compType.isBDU1xx()) {
      key = "BDU100";
    } else if (compType.isBDU2xx()) {
      key = "BDU2xx";
    } else if (compType.isBDU3xx()) {
      key = "BDU3xx";
    } else if (compType.isBDU4xx()) {
      key = "BDU4xx";
    } else if (compType.isBBP1xx()) {
      key = "BBP100";
    } else if (compType.isBBP2xx()) {
      key = "BBP2xx";
    } else if (compType.isAbs()) {
      key = "BAS";
    } else if (compType.isBcs()) {
      key = "BCS250";
    } else {
      LOGGER.error("Failed to map component type to log key");
      key = "unknown";
    } 
    LOGGER.debug("ComponentType {} map to key: {}", compType, key);
    return key;
  }
  
  private void writeLogEntry(File file, String logEntry) {
    try (BufferedWriter bw = new BufferedWriter(new FileWriter(file, true))) {
      bw.append(logEntry);
    } catch (IOException ex) {
      LOGGER.error("Failed to write log entry to {}", file.getName(), ex);
    } 
  }
  
  private Properties loadLogFile(File file) throws IOException {
    Properties props;
    try (FileInputStream in = new FileInputStream(file.getAbsolutePath())) {
      props = new Properties();
      props.load(in);
    } 
    return props;
  }
  
  private void storeLogFile(File file, Properties props) throws IOException {
    StringWriter writer = new StringWriter();
    props.list(new PrintWriter(writer));
    clearFile(file);
    try (BufferedWriter bw = new BufferedWriter(new FileWriter(file, true))) {
      bw.append("Recording-start-date=" + props.getProperty("Recording-start-date") + "\n");
      for (Enumeration<?> e = props.propertyNames(); e.hasMoreElements(); ) {
        String curKey = e.nextElement().toString();
        if (!StringUtils.equals(curKey, "Recording-start-date"))
          bw.append(curKey + "=" + curKey + "\n"); 
      } 
    } catch (IOException ex) {
      LOGGER.error("Failed to write logfile entries", (Throwable)ex);
    } 
  }
  
  private void clearFile(File file) throws FileNotFoundException {
    try (PrintWriter writer = new PrintWriter(file)) {
      writer.print("");
    } 
  }
}

  private FlashProcedure.FlashState flashEcus() {

    FlashProcedure.FlashState flashState = FlashProcedure.FlashState.FLASH_ERROR;
    try {
      if (this.bcsFlashProcedure != null) {
        FlashProcedureResult flashProcedureResult = this.bcsFlashProcedure.execute(true);
        this.flashResult.addFlashStatistic(flashProcedureResult.getFlashStatistic());
        return flashProcedureResult.getFlashState();
      } 
      if (this.hidFlashProcedure != null) {
        FlashProcedureResult flashProcedureResult = this.hidFlashProcedure.execute((this.bikeIdentification.isBctConnected() || this.forceFlashing));
        this.flashResult.addFlashStatistic(flashProcedureResult.getFlashStatistic());
        flashState = flashProcedureResult.getFlashState();
        if (flashState == FlashProcedure.FlashState.FLASHED && this.bikeIdentification.getHmiIdentification().getComponentType().isBUI25x() && !this.bikeIdentification.isHmiStandalone() && !configureBaudrate()) {
          this.flashResult.addFlashStatisticOverwrite(createBaudrateConfigurationFailsFlashStatistic());
          flashState = FlashProcedure.FlashState.FLASH_ERROR;
        } 
      } 
      if (flashState != FlashProcedure.FlashState.FLASH_ERROR && this.bmsFlashProcedure != null) {
        FlashProcedureResult flashProcedureResult = this.bmsFlashProcedure.execute(this.forceFlashing);
        if (this.bmsFlashProcedure instanceof BmsDualFlashProcedure) {
          this.flashResult.addFlashStatistic(flashProcedureResult.getFlashStatistics().get(0));
          this.flashResult.addFlashStatistic(flashProcedureResult.getFlashStatistics().get(1));
        } else {
          this.flashResult.addFlashStatistic(flashProcedureResult.getFlashStatistic());
        } 
        flashState = flashProcedureResult.getFlashState();
      } 
      if (flashState != FlashProcedure.FlashState.FLASH_ERROR && this.absFlashProcedure != null) {
        FlashProcedureResult flashProcedureResult = this.absFlashProcedure.execute((this.forceFlashing && this.userLevel == UserLevel.OEM));
        this.flashResult.addFlashStatistic(flashProcedureResult.getFlashStatistic());
        flashState = flashProcedureResult.getFlashState();
      } 
      if (flashState != FlashProcedure.FlashState.FLASH_ERROR && this.duFlashProcedure != null) {
        FlashProcedureResult flashProcedureResult = this.duFlashProcedure.execute(this.forceFlashing);
        this.flashResult.addFlashStatistic(flashProcedureResult.getFlashStatistic());
        flashState = flashProcedureResult.getFlashState();
        if (flashState != FlashProcedure.FlashState.FLASH_ERROR && this.duFlashProcedure instanceof DuFlashProcedure)
          if (!this.bikeIdentification.getDuIdentification().isBootLoader()) {
            this.flashResult.setAppParamsWritten(((DuFlashProcedure)this.duFlashProcedure).writeApplicationParameters());

          }  
      }

      flashState = FlashProcedure.FlashState.FLASH_ERROR;
    } 
    return flashState;
  }

  private List<String> writeOemParam(String ecu, OemParamDto param) {
    String responseString;
    List<String> result = new ArrayList<>();
    LOGGER.debug("Write OEM parameter '{}' to ECU '{}'", param, ecu);
    int currentTry = 0;
    result.add(param.getRequest());
    byte[] request = ByteArrayConverterUtil.hexStringToByteArray(param.getRequest());
    do {
      byte[] response = this.comWrapper.transmitPdu(ecu, request);
      responseString = ByteArrayConverterUtil.byteArrayToHexString(response);
      if (!responseString.startsWith(param.getNegativResponseId()))
        break; 
      currentTry++;
      if (param.isIgnoreFailure())
        continue; 
      sleep(500);
    } while (currentTry < 3 && !param.isIgnoreFailure());
    if (!param.isIgnoreFailure()) {
      LOGGER.info("Response '{}' not ignored for request '{}'", responseString, param.getRequest());
      if (responseString.startsWith(param.getPositiveResponseId())) {
        result.add("POSITIVE");
      } else {
        LOGGER.error("Failed to write oem parameters to {}. Negative response '{}' to request '{}'", ecu, responseString, param.getRequest());
        result.add("NEGATIVE");
      } 
    } else {
      LOGGER.info("Response '{}' ignored for request '{}'", responseString, param.getRequest());
      result.add("POSITIVE");
    } 
    return result;
  }

  private boolean configureBaudrate() {
    String ecuName;
    String baudrate = (this.bikeIdentification.getSystemType() == Constants.SystemType.CLASSIC_PLUS) ? "CAN baudrate 125kBaud" : "CAN baudrate 500 kBaud";
    if (this.bikeIdentification.isBms1Connected()) {
      if (this.bikeIdentification.getSystemType() == Constants.SystemType.CLASSIC_PLUS) {
        ecuName = ComponentType.COMP_BBP100.getOdxEcuName();
      } else {
        ecuName = "BBP1";
      } 
    } else if (this.bikeIdentification.isBms2Connected()) {
      ecuName = "BBP2";
    } else {
      return false;
    } 
    byte[] testerPresentPdu = { 62, 0 };
    for (int i = 0; i < 3; i++) {
      LOGGER.debug("Intuvia baudrate configuration try no. '{}' of '{}'", Integer.valueOf(i + 1), Integer.valueOf(3));
      List<String> hmiResponse = this.comWrapper.transmitService(ComponentType.COMP_BUI250.getOdxEcuName(), "CanInterfaceControl_Control", "CAN_Baudrate", baudrate, false);
      if (hmiResponse.contains("NEGATIVE"))
        LOGGER.error("Setting baudrate to '{}' failed", baudrate); 
      sleep(500 * (i + 1));
      byte[] bmsResponse = this.comWrapper.transmitPdu(ecuName, testerPresentPdu);
      if (Helper.isPositiveResponse(testerPresentPdu, bmsResponse)) {
        LOGGER.debug("Configuring baudrate to '{}' worked!", baudrate);
        return true;
      } 
      if (i < 2) {
        LOGGER.warn("Failed to interact with other ECUs after configuring baudrate, so repeat procedure with hard reset");

  private static void saveXmlToFile(Document xml, File file) throws TransformerException {
    LOGGER.info("Saving XML file {}", file.getAbsolutePath());
    Transformer tf = TransformerFactory.newInstance().newTransformer();
    tf.setOutputProperty("omit-xml-declaration", "yes");
    tf.setOutputProperty("encoding", "UTF-8");
    tf.setOutputProperty("indent", "yes");
    tf.transform(new DOMSource(xml), new StreamResult(file));
  }
  
  static String parseUclForMapsOnly(String filePath, Map<String, String> mapsInfoHashMap) {
    try {
      LOGGER.info("Parsing {}", filePath);
      File fXmlFile = new File(filePath);
      DocumentBuilderFactory dbFactory = DocumentBuilderFactory.newInstance();
      DocumentBuilder dBuilder = dbFactory.newDocumentBuilder();
      Document parsedDoc = dBuilder.parse(fXmlFile);
      LOGGER.info("Parsing {} done... normalizing...", filePath);
      parsedDoc.getDocumentElement().normalize();
      LOGGER.info("Root element: {}", parsedDoc.getDocumentElement().getNodeName());
      Document newDoc = dBuilder.newDocument();
      Element parsedRoot = parsedDoc.getDocumentElement();
      Node importRootNode = newDoc.importNode(parsedRoot, true);
      newDoc.appendChild(importRootNode);
      newDoc.getDocumentElement().setAttribute("desc", "AKKA custom UCL for maps flashing only");
      NodeList listTags = newDoc.getElementsByTagName("LIST");
      List<Node> mapNodes = createBuiMapInfoElements(mapsInfoHashMap, newDoc, filePath.contains("FS_275"));
      NodeList cmds = newDoc.getElementsByTagName("CMD");
      for (int i = 0; i < cmds.getLength(); i++) {
        if (StringUtils.equals(cmds.item(i).getTextContent(), "Flashing maps ...")) {
          Node standardMapNode = cmds.item(i);
          for (Node map : mapNodes)
            listTags.item(0).insertBefore(map, standardMapNode); 
          Node standardMapFlush = standardMapNode.getNextSibling().getNextSibling();
          Node standardMapMemRelease = standardMapFlush.getNextSibling().getNextSibling();
          LOGGER.info("Deleting node with content: {}", standardMapNode.getTextContent());
          listTags.item(0).removeChild(standardMapNode);
          LOGGER.info("Deleting node with content: {}", standardMapFlush.getTextContent());
          listTags.item(0).removeChild(standardMapFlush);
          LOGGER.info("Deleting node with content: {}", standardMapMemRelease.getTextContent());
          listTags.item(0).removeChild(standardMapMemRelease);
          break;
        } 
      } 
      File newFile = new File(fXmlFile.getParent(), "ucl2_update_custom_maps_only.xml");
      saveXmlToFile(newDoc, newFile);
      return newFile.getAbsolutePath();
    } catch (ParserConfigurationException|org.xml.sax.SAXException|java.io.IOException|TransformerException e) {
      LOGGER.error("Error parsing XML " + filePath, (Throwable)e);
      return "";
    } 
  }
  
  private static List<Node> createBuiMapInfoElements(Map<String, String> mapsInfoHashMap, Document newDoc, boolean is275) {
    List<Node> mapElements = new ArrayList<>(mapsInfoHashMap.size() * 4);
    for (Map.Entry<String, String> entry : mapsInfoHashMap.entrySet()) {
      Comment comment = newDoc.createComment("extract map " + (String)entry.getKey() + " add-on archive to mounted UBI partition");
      mapElements.add(comment);
      Element cmd1 = newDoc.createElement("CMD");
      cmd1.setAttribute("state", "Updater");
      cmd1.setAttribute("type", "push");
      if (is275) {
        cmd1.setAttribute("body", "pipe tar -xjv -C /mnt/rootfs");
      } else {
        cmd1.setAttribute("body", "pipe tar -xjv -C /mnt/ubi0");
      } 
      String mapfile = entry.getValue();
      cmd1.setAttribute("file", mapfile);
      cmd1.setTextContent("Flashing map id " + (String)entry.getKey() + " ...");
      mapElements.add(cmd1);
      Element cmd2 = newDoc.createElement("CMD");
      cmd2.setAttribute("state", "Updater");
      cmd2.setAttribute("type", "push");
      cmd2.setAttribute("body", "frf");
      cmd2.setTextContent("flush");
      mapElements.add(cmd2);
      Element cmd3 = newDoc.createElement("CMD");
      cmd3.setAttribute("state", "Updater");
      cmd3.setAttribute("type", "push");
      cmd3.setAttribute("body", "$ echo 3 > /proc/sys/vm/drop_caches");
      cmd3.setTextContent("Release Memory");
      mapElements.add(cmd3);
    } 
    return mapElements;
  }

  public boolean writeApplicationParameters() {
    boolean successful = false;
    LOGGER.info("Write application parameters to '{}'", this.ecuName);
    for (int retry = 1; retry <= 3 && !successful; retry++) {
      LOGGER.info("Flash try number '{}' for application parameters", Integer.valueOf(retry));
      successful = writeApplicationParameter((String)TokenPluginProxy.getInstance().getTokenProperties().get("User level"), retry);
      if (!successful && retry == 3) {
        LOGGER.error("Failed to write application parameters after 3 unsuccessful attempts");
      } else if (successful) {
        LOGGER.info("Application parameters have been written successfully");
      } 
    } 
    return successful;
  }
  
  boolean writeApplicationParameter(String userLevel, int tryNum) {
    fireProgressChanged(BaseFlashProcedure.FlashProgressType.PARAM, 1);
    List<String> result = new ArrayList<>();
    result.add("NEGATIVE");
    this.comWrapper.functionalExtendedSession();
    boolean securityAccessSuccess = this.comWrapper.securityAccess(this.ecuName, TokenPluginProxy.getInstance().getAccessLevel(), 1);
    fireProgressChanged(BaseFlashProcedure.FlashProgressType.PARAM, 10);
    try {
      writeApplicationParameterReferenceId(userLevel);
    } catch (WriteApplicationParameterReferenceIdException|IOException|NumberFormatException ex) {
      LOGGER.error("Error writing application parameter reference ID in attempt {} of 3", Integer.valueOf(tryNum), ex);
      fireFlashError(securityAccessSuccess);
      return false;
    } 
    fireProgressChanged(BaseFlashProcedure.FlashProgressType.PARAM, 15);
    this.comWrapper.functionalConfigureDtcSetting(false);
    fireProgressChanged(BaseFlashProcedure.FlashProgressType.PARAM, 17);
    this.comWrapper.functionalConfigureEcuInternalCommunicationSetting(false);
    fireProgressChanged(BaseFlashProcedure.FlashProgressType.PARAM, 19);
    this.comWrapper.transmitService(this.ecuName, "PrepareProgramming_Start", null, null, false);
    fireProgressChanged(BaseFlashProcedure.FlashProgressType.PARAM, 20);
    boolean success = this.comWrapper.checkProgrammingPreConditions(this.ecuName, 10000);
    if (success)
      success = !this.comWrapper.startProgrammingSession(this.ecuName).contains("NEGATIVE"); 
    if (success) {
      success = this.comWrapper.securityAccess(this.ecuName, 0, 1);
      securityAccessSuccess = (securityAccessSuccess && success);
    } 
    fireProgressChanged(BaseFlashProcedure.FlashProgressType.PARAM, 25);
    if (success)
      result = programApplicationParameterGen2(); 
    if (result == null) {
      result = new ArrayList<>();
      result.add("NEGATIVE");
      LOGGER.warn("Failed to write application parameters in attempt {} of 3", Integer.valueOf(tryNum));
    } 
    LOGGER.info("Result from Flash: {}", result);
    fireProgressChanged(BaseFlashProcedure.FlashProgressType.PARAM, 60);
    this.diagServer.hardReset(this.ecuName);
    sleep(1000);
    fireProgressChanged(BaseFlashProcedure.FlashProgressType.PARAM, 75);
    if (result.contains("POSITIVE")) {
      fireProgressChanged(BaseFlashProcedure.FlashProgressType.PARAM, 80);
      return true;
    } 
    fireFlashError(securityAccessSuccess);
    return false;
  }
  
  private void fireFlashError(boolean wasSecurityAccessSuccessful) {
    if (wasSecurityAccessSuccessful) {
      fireFlashError(FlashErrorType.WRITE_APPLICATION_PARAMS_FAILED, BaseFlashProcedure.FlashProgressType.PARAM);

      fireFlashError(FlashErrorType.SECURITY_ACCESS_FAILED, BaseFlashProcedure.FlashProgressType.PARAM);
    } 
  }
  
  private List<String> programApplicationParameterGen2() {

  private void writeApplicationParameterReferenceId(String userLevel) throws IOException, WriteApplicationParameterReferenceIdException {
    String newAppParId = this.containerConfig.getConfiguredParId();
    List<String> writeApplParIdResult = new ArrayList<>();
    if (userLevel.contains("OEM")) {
      writeApplParIdResult = this.comWrapper.transmitService(this.ecuName, "DU_Conf_OemBikeApplId_Write_1", "Bike_Appl_ID", newAppParId, false);
    } else if (isIbdApplicationParameterIdSupported()) {
      writeApplParIdResult = this.comWrapper.transmitService(this.ecuName, "DU_Conf_IbdBikeApplId_Write", "Bike_Appl_ID", newAppParId, false);
    } 
    if (writeApplParIdResult.contains("NEGATIVE"))
      throw new WriteApplicationParameterReferenceIdException("Writing application parameter reference id to DU failed"); 
  }
  
  boolean isIbdApplicationParameterIdSupported() throws IOException, WriteApplicationParameterReferenceIdException {
    List<String> compTypeResponse = this.comWrapper.transmitService(this.ecuName, "DU_Conf_ComponentType_Read_1", "DuComponentType", null, false);
    if (compTypeResponse == null || compTypeResponse.contains("NEGATIVE") || compTypeResponse.size() != 2)
      throw new WriteApplicationParameterReferenceIdException("Determining of DU component type failed"); 
    ComponentType componentType = ComponentType.parse(compTypeResponse.get(1));
    if (componentType.isBDU4xx())
      return true; 
    List<String> swVerResponse = this.comWrapper.transmitService(this.ecuName, "DU_SW_Version_Read", "SW_Version", null, false);
    if (swVerResponse == null || swVerResponse.contains("NEGATIVE") || swVerResponse.size() != 8)
      throw new WriteApplicationParameterReferenceIdException("Reading software version from DU failed"); 
    Version duSwVer = new Version(new Integer[] { Integer.parseInt(swVerResponse.get(1)), Integer.parseInt(swVerResponse.get(3)), Integer.parseInt(swVerResponse.get(5)), Integer.parseInt(swVerResponse.get(7)) });
    Version compareDuSwVer = Version.parse(componentType.isBDU2xx() ? "1.7.0.0" : "1.0.1.0");
    if (compareDuSwVer.compareTo(duSwVer) < 0) {
      LOGGER.debug("Writing application parameter reference ID will be written, because DU firmware version's new enough. Is '{}', must be greater than '{}'", duSwVer, compareDuSwVer);
      return true;
    } 
    LOGGER.debug("Writing application parameter reference ID has been skipped, because DU firmware version's too low. Is '{}', must be greater than '{}'", duSwVer, compareDuSwVer);
    return false;
  }

    }
  }
