  }

import com.gigatronik.ebike.common.configuration.ComplexResponse;
import java.io.IOException;
import java.util.Date;
import java.util.TreeMap;
import java.util.concurrent.TimeUnit;

public class Bct1DataMiningSnapshot extends AbstractDataMiningSnapshot {
  private static final Version MIN_SW_VERSION_CELL_TYPE = new Version(new Integer[] { 1, 1, 0, 0 });
  
  private static final String DATA_TYPE = "bct1-v2";
  
  private final Bct bct;
  
  public Bct1DataMiningSnapshot(CommunicationManager commManager, ConfigurationManager confManager) {
    super(commManager, confManager);
    this.bct = commManager.getBctCom();
  }
  
  public void readData() {
    if (this.bikeIdentification.isBctConnected()) {
      this.data = new TreeMap<>();
      LOGGER.debug("Reading Bct1-DataMining Data...");
      this.data.put(DataMiningKey.BIKE_DATA_TYPE.getCffKey(), "bct1-v2");
      ComplexResponse swVersion = this.bct.readSoftwareVersionInSourceCode();
      this.data.put(DataMiningKey.BCT1_SW_VERSION.getCffKey(), swVersion.isPositive() ? swVersion.toString() : "");
      this.data.put(DataMiningKey.BCT1_HW_VERSION.getCffKey(), this.bikeIdentification.getBctIdentification().getHwVersion());
      ComplexResponse hwVersion = this.bct.readBatteryNominalCapacity();
      this.data.put(DataMiningKey.BCT1_NOMINAL_ENERGY.getCffKey(), hwVersion.isPositive() ? hwVersion.toString() : "");
      ComplexResponse measuredEnergy = this.bct.readMeasuredBatteryEnergy();
      this.data.put(DataMiningKey.BCT1_MEASURED_ENERGY.getCffKey(), measuredEnergy.isPositive() ? measuredEnergy.toString() : "");
      ComplexResponse partNumber = this.bct.readBatteryPartNumber();
      this.data.put(DataMiningKey.BCT1_BMS_PART_NUMBER.getCffKey(), partNumber.isPositive() ? partNumber.toString() : "");
      ComplexResponse batteryNumOfCharge = this.bct.readBatteryNumberOfChargeCycles();
      this.data.put(DataMiningKey.BCT1_BMS_CHARGE_CYCLES.getCffKey(), batteryNumOfCharge.isPositive() ? batteryNumOfCharge.toString() : "");
      ComplexResponse manufactureDate = this.bct.readBatteryManufactureDate();
      this.data.put(DataMiningKey.BCT1_BMS_AGE.getCffKey(), getBmsAge(manufactureDate));
      if (hasServiceCellType()) {
        ComplexResponse cellType = this.bct.readBatteryCellType();
        this.data.put(DataMiningKey.BCT1_BMS_CELL_TYPE.getCffKey(), cellType);
      } 
      LOGGER.debug("Bct1-DataMining Data successfully read");
    } 
  }
  
  private String getBmsAge(ComplexResponse response) {
    try {
      Date measurementDate = new Date();
      Date manufactureDate = response.getDate();
      return String.valueOf(getDifferenceDays(manufactureDate, measurementDate));
    } catch (ComplexResponse.DataAccessException ex) {
      LOGGER.error("Error reading battery manufacture date", (Throwable)ex);
      return "";
    } 
  }
  
  private long getDifferenceDays(Date d1, Date d2) {
    long diff = d2.getTime() - d1.getTime();
    return TimeUnit.DAYS.convert(diff, TimeUnit.MILLISECONDS);
  }
  
  private boolean hasServiceCellType() {
    boolean result = false;
    try {
      Version swVersion = Version.parse(this.bikeIdentification.getBctIdentification().getSwVersion());
      return (swVersion != null) ? ((swVersion.compareTo(MIN_SW_VERSION_CELL_TYPE) >= 0)) : false;
    } catch (IOException ex) {
      LOGGER.error("Failed to parse sw version of bct", (Throwable)ex);
      return result;
    } 
  }
}

abstract class AbstractFlashUpdateSnapshot {
  protected static final Logger LOGGER = LoggerFactory.getLogger(AbstractFlashUpdateSnapshot.class);
  
  protected CommunicationManager communicationManager;
  
  protected BikeIdentification bikeIdentification;
  
  protected EcuIdentification identification;
  
  protected BaseFlashProcedure.FlashProgressType ecuType;
  
  protected Map<String, Object> data;
  
  protected int flashCounter = 0;
  
  protected String previousSwVersion = "";
  
  protected List<String> lastProgress = new ArrayList<>();
  
  protected List<String> errorCodes = new ArrayList<>();
  
  AbstractFlashUpdateSnapshot(CommunicationManager commManager, ConfigurationManager confManager) {
    this.communicationManager = commManager;
    this.bikeIdentification = confManager.getBikeIdentification();
    initIdentification();
  }
  
  abstract void initIdentification();
  
  Map<String, Object> getData() {
    return this.data;
  }
  
  BaseFlashProcedure.FlashProgressType getEcuType() {
    return this.ecuType;
  }
  
  boolean hasData() {
    return (this.data != null && !this.data.isEmpty());
  }
  
  void resetData() {
    this.data = null;
    this.flashCounter = 0;
    this.previousSwVersion = "";
    this.lastProgress = new ArrayList<>();
    this.errorCodes = new ArrayList<>();
  }
  
  void updateStatistics(FlashStatistic statistic) {
    if (statistic.getFlashState() != FlashProcedure.FlashState.SKIPPED) {
      this.flashCounter++;
      this.lastProgress.add(statistic.getLastProgressAsString());
      this.errorCodes.addAll(statistic.getErrorCodes());
    } 
  }
  
  abstract void collectData();
  
  protected void setPreviousSwVersion() {
    if (StringUtils.isBlank(this.previousSwVersion))
      this.previousSwVersion = this.identification.getSwVersion(); 
  }
  
  boolean isFlashed() {
    return (this.flashCounter > 0);
  }
}

package com.gigatronik.beds.service.datamining.snapshots.flashupdate;

import com.gigatronik.beds.service.communication.api.CommunicationManager;
import com.gigatronik.beds.service.configuration.ConfigurationManager;
import com.gigatronik.beds.service.datamining.DataMiningKey;
import com.gigatronik.beds.service.datamining.snapshots.AbstractDataMiningSnapshot;
import com.gigatronik.beds.service.ecuflash.impl.FlashStatistic;
import java.util.ArrayList;
import java.util.List;
import java.util.Map;
import java.util.Set;
import java.util.TreeMap;

public class FlashUpdateSnapshot extends AbstractDataMiningSnapshot {
  private static final String DATA_TYPE = "swupdate-v2";
  
  private final List<AbstractFlashUpdateSnapshot> flashUpdateSnapshots = new ArrayList<>();
  
  private boolean anyCompFlashed = false;
  
  public FlashUpdateSnapshot(CommunicationManager commManager, ConfigurationManager confManager) {
    super(commManager, confManager);
    initFlashUpdateSnapshots(confManager);
  }
  
  private void initFlashUpdateSnapshots(ConfigurationManager confManager) {
    this.flashUpdateSnapshots.add(new HmiFlashUpdateSnapshot(this.communicationManager, confManager));
    this.flashUpdateSnapshots.add(new DuFlashUpdateSnapshot(this.communicationManager, confManager));
    this.flashUpdateSnapshots.add(new Bms1FlashUpdateSnapshot(this.communicationManager, confManager));
    this.flashUpdateSnapshots.add(new Bms2FlashUpdateSnapshot(this.communicationManager, confManager));
    this.flashUpdateSnapshots.add(new AbsFlashUpdateSnapshot(this.communicationManager, confManager));
    this.flashUpdateSnapshots.add(new BcsFlashUpdateSnapshot(this.communicationManager, confManager));
    this.flashUpdateSnapshots.add(new BctFlashUpdateSnapshot(this.communicationManager, confManager));
  }
  
  public void setPreviousSwVersion() {
    this.flashUpdateSnapshots.forEach(snapshot -> snapshot.setPreviousSwVersion());
  }
  
  public void readData() {
    this.flashUpdateSnapshots.forEach(snapshot -> snapshot.collectData());
  }
  
  public Map<String, Object> getData() {
    this.data = new TreeMap<>();
    if (hasData())
      this.data.put(DataMiningKey.BIKE_DATA_TYPE.getCffKey(), "swupdate-v2"); 
    this.flashUpdateSnapshots.forEach(snapshot -> {
          Map<String, Object> snapshotData = snapshot.getData();
          if (snapshotData != null && !snapshotData.isEmpty())
            this.data.putAll(snapshotData); 
        });
    return this.data;
  }
  
  public void clearData() {
    super.clearData();
    this.flashUpdateSnapshots.forEach(snaphot -> snaphot.resetData());
    this.anyCompFlashed = false;
  }
  
  public boolean hasData() {
    if (this.anyCompFlashed)
      return true; 
    for (AbstractFlashUpdateSnapshot snapshot : this.flashUpdateSnapshots) {
      if (snapshot.hasData() && snapshot.isFlashed()) {
        this.anyCompFlashed = true;
        return true;
      } 
    } 
    return false;
  }
  
  public void setFlashStatistics(Set<FlashStatistic> statistics) {
    statistics.forEach(statistic -> {
          for (AbstractFlashUpdateSnapshot snapshot : this.flashUpdateSnapshots) {
            if (snapshot.getEcuType() == statistic.getEcuType())
              snapshot.updateStatistics(statistic); 
          } 
        });
  }
}
