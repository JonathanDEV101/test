  public BctResponse writeBatteryDataToCapacityTester(BctBatteryData bctBatteryData) {
    List<ComplexResponse> responses = writeBatteryCommonData(bctBatteryData);
    BctBatteryType batteryType = this.bct.getSelectedBattery();
    switch (batteryType) {
      case CLASSIC:
        responses.addAll(writeBatteryGen1Data(bctBatteryData));
        break;
      case VICTOR:
        responses.addAll(writeBatteryGen2Data(bctBatteryData));
        break;
      default:
        LOGGER.error("Read out unknown battery type '{}'", batteryType);
        break;
    } 
    if (!responses.isEmpty())
      for (ComplexResponse response : responses) {
        if (response.isNegative())
          return BctResponse.REJECTED; 
      }  
    return BctResponse.OK;
  }

  }

    LOGGER.info("Display security access failed dialog; blocking='{}', ecu='{}'", Boolean.valueOf(blocking), ecu);
    displaySecurityAccessFailedDialog(blocking, (ecu != null) ? ecu.getProductName() : "unknown");
  }
  
  public void displaySecurityAccessFailedDialog(boolean blocking, ComponentType ecu) {
    LOGGER.info("Display security access failed dialog; blocking='{}', ecu='{}'", Boolean.valueOf(blocking), ecu);
    displaySecurityAccessFailedDialog(blocking, (ecu != null) ? ecu.getProductName() : "unknown");
  }
  
  private void displaySecurityAccessFailedDialog(boolean blocking, String ecu) {
    String textRaw = this.rightsManager.isUserInLevel(UserLevel.OEM) ? BUNDLE.getString("Popup.failure.securityAccess.text.oem") : BUNDLE.getString("Popup.failure.securityAccess.text.ibd");
    String text = String.format(textRaw, ecu, ecu);
    if (SwingUtilities.isEventDispatchThread()) {
      displayPopup(text);
    } else {
      delegateDisplayingToEventDispatchThread(blocking, text);
    } 
  }
  
  private void displayPopup(String text) {
    TestableJOptionPane.showMessageDialog(text, BUNDLE.getString("Bosch.diagnostic.tool"), 0);
  }
  
  private void delegateDisplayingToEventDispatchThread(boolean blocking, String text) {
    if (blocking) {
      try {
        EventQueue.invokeAndWait(() -> displayPopup(text));
      } catch (InterruptedException ex) {
        LOGGER.warn("Waiting interrupted!", (Throwable)ex);
        Thread.currentThread().interrupt();
      } catch (InvocationTargetException ex) {
        LOGGER.error("invokeAndWait() stopped with InvocationTargetException", (Throwable)ex);
      } 
    } else {
      EventQueue.invokeLater(() -> displayPopup(text));
    } 
  }
}
