import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

public class ListSortMachine<T extends OrderableListener> {
  private static final Logger LOGGER = LoggerFactory.getLogger(ListSortMachine.class);
  
  private final List<String> orderGuideListHead;
  
  private final List<String> orderGuideListTail;
  
  public ListSortMachine(List<String> orderGuideListHead, List<String> orderGuideListTail) {
    this.orderGuideListHead = orderGuideListHead;
    this.orderGuideListTail = orderGuideListTail;
  }
  
  public void sortList(List<T> list) {
    List<T> sortedList = createSortedList(list);
    transferOrdering(list, sortedList);
  }
  
  private List<T> createSortedList(List<T> list) {
    List<List<T>> lists = splitListInGroups(list);
    orderLists(lists);
    return mergeLists(lists);
  }
  
  private List<List<T>> splitListInGroups(List<T> list) {
    List<T> head = new ArrayList<>();
    List<T> body = new ArrayList<>();
    List<T> tail = new ArrayList<>();
    list.forEach(listener -> {
          if (isInHeadList(listener)) {
            head.add(listener);
          } else if (isInTailList(listener)) {
            tail.add(listener);
          } else {
            body.add(listener);
          } 
        });
    List<List<T>> lists = new ArrayList<>();
    lists.add(head);
    lists.add(body);
    lists.add(tail);
    return lists;
  }
  
  private boolean isInHeadList(T element) {
    return this.orderGuideListHead.contains(element.getOrderingName());
  }
  
  private boolean isInTailList(T element) {
    return this.orderGuideListTail.contains(element.getOrderingName());
  }
  
  private void orderLists(List<List<T>> lists) {
    orderListPart(lists.get(0), this.orderGuideListHead);
    orderListPart(lists.get(2), this.orderGuideListTail);
  }
  
  private void orderListPart(List<T> list, List<String> orderGuideList) {
    list.sort(Comparator.comparingInt(element -> computeShouldIndex(element, orderGuideList)));
  }
  
  private int computeShouldIndex(T element, List<String> orderGuideList) {
    try {
      return computeShouldIndexTry(element, orderGuideList);
    } catch (ListSortingException ex) {
      LOGGER.warn("Ordering list element '" + element + "' failed", (Throwable)ex);
      return 0;
    } 
  }
  
  private int computeShouldIndexTry(T listener, List<String> orderGuideList) throws ListSortingException {
    Optional<String> match = orderGuideList.stream().filter(listElement -> listElement.equalsIgnoreCase(listener.getOrderingName())).findFirst();
    if (match.isPresent())
      return orderGuideList.indexOf(match.get()); 
    throw new ListSortingException("Computing index for listener '" + listener + "' not possible, 'cause element's not in list");
  }
  
  private List<T> mergeLists(List<List<T>> lists) {
    List<T> mergedList = new ArrayList<>();
    Objects.requireNonNull(mergedList);
    lists.forEach(mergedList::addAll);
    return mergedList;
  }
  
  private void transferOrdering(List<T> unsortedList, List<T> sortedList) {
    unsortedList.clear();
    unsortedList.addAll(sortedList);
  }
}

      return getDisplayName(); 
    if (isGen2()) {
      compType = "BDU2xx";
    } else if (isGen3()) {
      compType = "BDU3xx";
    } else if (isGen4()) {
      compType = "BDU4xx";
    } 
    return StringUtils.isBlank(compType) ? "" : String.format("%s %s", getDisplayName(), compType);

    if (StringUtils.isNotBlank(odxName))

          return pLine; 
      }

  public static ProductLine fromDisplayNameWithCompType(String displayName) {
    if (StringUtils.isBlank(displayName)) {
      LOGGER.error("Cannot parse empty string or null to product line");
      return NONE;
    } 
    int indexWhitespace = displayName.lastIndexOf(' ');
    if (indexWhitespace < 0)
      return CLASSIC_PLUS; 
    String compTypeName = displayName.substring(indexWhitespace + 1);
    String prodLineName = displayName.substring(0, indexWhitespace);
    if (StringUtils.isNotBlank(prodLineName) && StringUtils.isNotBlank(compTypeName))
      for (ProductLine pLine : values()) {
        if (pLine != NONE && prodLineName.equals(pLine.getDisplayName()) && isSameGeneration(pLine, compTypeName))
          return pLine; 
      }  
    return NONE;
  }

  }

  public static String customizePartNumberCustomer(String partNumberCustomer) {

    } 
  }
  
  public static String customizePartNumberCustomerWrite(String partNumberCustomer) {
    try {
      String partNumber = String.format("%012x", new BigInteger(partNumberCustomer.getBytes()));
      for (int i = partNumber.length(); i < 24; i++)
        partNumber = partNumber.concat("0"); 
      return "0x" + partNumber;
    } catch (NumberFormatException ex) {
      return "";
    } 
  }
  
  public static byte[] concatByteArrays(byte[] by1, byte[] by2) {
    byte[] by = new byte[by1.length + by2.length];
    System.arraycopy(by1, 0, by, 0, by1.length);
    System.arraycopy(by2, 0, by, by1.length, by2.length);
    return by;
  }
  
  public static String customizeBikeType(String bikeTypeValue) {
    String result = bikeTypeValue;
    BikeType bikeType = BikeType.getValueByXmlAttribute(bikeTypeValue);
    if (bikeType != BikeType.UNDEFINED)
      result = bikeType.getDisplayName(); 
    return result;
  }
  
  public static byte[] hexStringToByteArray(String hex) {
    if (hex.startsWith("0x"))
      hex = hex.substring(2); 
    try {
      return Hex.decodeHex(hex.toCharArray());
    } catch (DecoderException e) {
      throw new NumberFormatException(e.getMessage());
    } 
  }
  
  static String hex2str(String hex, int offset) {
    hex = hex.replace("0x", "");
    byte[] swVerECU = new byte[offset / 2];
    for (int i = 0; i < offset; i += 2) {
      String in = hex.substring(i, i + 2);
      swVerECU[i / 2] = (byte)Integer.parseInt(in, 16);
    } 
    return new String(swVerECU);
  }
  
  public static int hex2int(String hex) {
    return Integer.parseInt(hex.replace("0x", ""), 16);
  }
  
  public static int byteArrayToInteger(byte[] by) {
    if (by == null)
      return 0; 
    int iSize = 4;
    int iVal = 0;
    if (by.length < iSize)
      iSize = by.length; 
    for (int i = 0; i < iSize; i++)
      iVal = (iVal << 8) + (by[i] & 0xFF); 
    return iVal;
  }
  
  public static byte[] integerToByteArray(int value) {
    byte[] by = new byte[4];
    for (int i = 0; i < 4; i++) {
      int shift = i << 3;
      by[3 - i] = (byte)((value & 255 << shift) >>> shift);
    } 
    return by;
  }
  
  public static byte[] readBinaryData(String fileName) {
    try {
      return FileUtils.readFileToByteArray(new File(fileName));
    } catch (FileNotFoundException e) {
      LOGGER.error("File not found: {}", fileName, e);
    } catch (IOException e) {
      LOGGER.error("Error reading BinaryData from {}", fileName, e);
    } 
    return new byte[0];
  }
}

      return Integer.toHexString(flipBits(value, 16)).toUpperCase();
    } catch (NumberFormatException e) {
      LOGGER.error("Error calculating wheel size checksum for {}", wheelSize, e);
      return null;
    } 
  }
  
  private static int flipBits(int n, int k) {
    int mask = (1 << k) - 1;
    return n ^ mask;
  }

package com.gigatronik.ebike.common.util;

public class ConverterUtils {
  public static boolean isDouble(String str) {
    if (str == null)
      return false; 
    int length = str.length();
    if (length == 0)
      return false; 
    int i = 0;
    if (str.charAt(0) == '-') {
      if (length == 1)
        return false; 
      i++;
    } 
    int integerPartSize = 0;
    int exponentPartSize = -1;
    while (i < length) {
      char c = str.charAt(i);
      if (c < '0' || c > '9') {
        if (c == '.' && integerPartSize > 0 && exponentPartSize == -1) {
          exponentPartSize = 0;
        } else {
          return false;
        } 
      } else if (exponentPartSize > -1) {
        exponentPartSize++;
      } else {
        integerPartSize++;
      } 
      i++;
    } 
    if ((str.charAt(0) == '0' && i > 1 && exponentPartSize < 1) || exponentPartSize == 0 || 
      str.charAt(length - 1) == '.')
      return false; 
    return true;
  }
  
  public static boolean isInteger(String str) {
    if (str == null)
      return false; 
    int length = str.length();
    if (length == 0)
      return false; 
    int i = 0;
    if (str.charAt(0) == '-') {
      if (length == 1)
        return false; 
      i++;
    } 
    while (i < length) {
      char c = str.charAt(i);
      if (c < '0' || c > '9')
        return false; 
      i++;
    } 
    return true;
  }
}

package com.gigatronik.ebike.common.util;

import java.io.File;
import java.io.FileInputStream;
import java.io.FileOutputStream;
import java.io.IOException;
import java.io.InputStream;
import java.io.OutputStream;
import java.util.Enumeration;
import java.util.Objects;
import java.util.zip.ZipEntry;
import java.util.zip.ZipFile;
import org.apache.commons.compress.archivers.ArchiveException;
import org.apache.commons.compress.archivers.ArchiveStreamFactory;
import org.apache.commons.compress.archivers.tar.TarArchiveEntry;
import org.apache.commons.compress.archivers.tar.TarArchiveInputStream;
import org.apache.commons.compress.utils.IOUtils;
import org.apache.commons.io.FileUtils;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

public class FileIOUtil {
  private static final Logger LOGGER = LoggerFactory.getLogger(FileIOUtil.class);
  
  private static final int MAX_BUFFER = 4096;
  
  private static final File TEMPORARY_DIRECTORY = new File(System.getProperty("java.io.tmpdir"));
  
  public static String unzipWithValidation(File fSourceZip) throws IOException {
    try (ZipFile zipFile = new ZipFile(fSourceZip)) {
      File tempdirectory = getJavaTmpDirectory();
      String fileDirectory = tempdirectory.getPath();
      String upzippedPath = fileDirectory + fileDirectory + File.separator + "Unzipped";
      File tempDir = new File(upzippedPath);
      tempDir.mkdir();
      Enumeration<? extends ZipEntry> enumZipEntries = zipFile.entries();
      while (enumZipEntries.hasMoreElements()) {
        ZipEntry entry = enumZipEntries.nextElement();
        File destinationFilePath = new File(upzippedPath, entry.getName());
        destinationFilePath.getParentFile().mkdirs();
        if (!entry.isDirectory() || !destinationFilePath.exists() || !destinationFilePath.getName().equals(entry.getName()) || destinationFilePath.length() != entry.getSize() || 
          destinationFilePath.lastModified() != entry.getTime()) {
          try(InputStream is = zipFile.getInputStream(entry); FileOutputStream fos = new FileOutputStream(destinationFilePath)) {
            IOUtils.copy(is, fos, 4096);
            fos.flush();
          } 
          if (!destinationFilePath.setLastModified(entry.getTime()))
            LOGGER.warn("The modified date in {} has not been changed!", destinationFilePath); 
        } 
      } 
      return upzippedPath;
    } 
  }
  
  private static void unTar(File inputFile, File outputDir) throws IOException, ArchiveException {
    try (TarArchiveInputStream tarInputStream = (TarArchiveInputStream)new ArchiveStreamFactory().createArchiveInputStream("tar", new FileInputStream(inputFile))) {
      TarArchiveEntry entry;
      while ((entry = (TarArchiveEntry)tarInputStream.getNextEntry()) != null) {
        File outputFile = new File(outputDir, entry.getName());
        if (entry.isDirectory()) {
          if (!outputFile.exists() && !outputFile.mkdirs())
            throw new IllegalStateException(String.format("Couldn't create directory %s.", outputFile.getAbsolutePath())); 
          continue;
        } 
        try (OutputStream outputFileStream = new FileOutputStream(outputFile)) {
          IOUtils.copy((InputStream)tarInputStream, outputFileStream);
          outputFileStream.flush();
        } 
      } 
    } 
  }
  
  public static void unTar(String inputFileName) throws IOException, ArchiveException {
    File tarFile = new File(inputFileName);
    File outputDir = tarFile.getParentFile();
    unTar(tarFile, outputDir);
  }
  
  public static void deleteDirectoryOnExit(String path) {
    deleteDirectoryOnExit(new File(path));
  }
  
  private static void deleteDirectoryOnExit(File file) {
    try {
      FileUtils.forceDeleteOnExit(file);
    } catch (IOException e) {
      LOGGER.warn("Error scheduling directory to be deleted: {}", file.getPath());
    } 
  }
  
  public static void markTmpFilesForDeletion() {
    try {
      File tmpDirectory = getJavaTmpDirectory();
      for (File file : Objects.<File[]>requireNonNull(tmpDirectory.listFiles())) {
        if (file.getName().endsWith("Unzipped") || file.getName().startsWith("NanoHTTPD") || file.getName().startsWith("LargeContainer") || 
          file.getName().startsWith("Ebike_tmp"))
          deleteDirectoryOnExit(file); 
      } 
    } catch (RuntimeException e) {
      LOGGER.error("Violation in the access to the Temporal Files. No file will be marked to be deleted.", (Throwable)e);
    } 
  }
  
  public static File getJavaTmpDirectory() {
    return TEMPORARY_DIRECTORY;
  }
}

  private static String cryptDateYmmdd(String dateDdmmyy) {
    if (StringUtils.length(dateDdmmyy) >= 6) {
      int date;
      int year = Integer.parseInt(dateDdmmyy.substring(4, 6), 10);
      int month = Integer.parseInt(dateDdmmyy.substring(2, 4), 10);
      if (year < 10) {
        date = year * 100 + month + 80;
      } else if (year < 14) {
        date = (year - 10) * 100 + month;
      } else if (year < 20) {
        date = (year - 10) * 100;
        if (month < 5) {
          date = date + month + 16;
        } else {
          date = date + month + 48;
        } 
      } else {
        date = (year - 20) * 100;
        if (month < 5) {
          date = date + month + 36;
        } else {
          date = date + month + 68;
        } 
      } 
      String code = String.format("%03d", date);
      return code + code;
    } 
    return null;
  }
  
  private static String cryptDateDUymmdd(String dateDdmmyy) {
    int year = Integer.parseInt(dateDdmmyy.substring(4, 6), 10);
    int month = Integer.parseInt(dateDdmmyy.substring(2, 4), 10);
    int date = 0;
    if (year >= 7 && year < 10) {
      date = year * 100 + 80 + month;
    } else if (year >= 10 && year < 14) {
      date = (year - 10) * 100 + month;
    } else if (year >= 14 && year < 20) {
      date = (year - 10) * 100;
      date += ((month < 5) ? 16 : 48) + month;
    } else if (year >= 20 && year < 30) {
      date = (year - 20) * 100;
      date += ((month < 5) ? 36 : 68) + month;
    } else if (year >= 30 && year < 90) {
      char[] yearCharArray = Integer.toString(year).toCharArray();
      int secondDigit = Character.getNumericValue(yearCharArray[0]) + 7;
      String yearMonth = "" + yearCharArray[1] + yearCharArray[1] + Integer.toHexString(secondDigit);
      return yearMonth.toUpperCase() + yearMonth.toUpperCase();
    } 
    String code = String.format("%03d", date);
    return code + code;
  }
