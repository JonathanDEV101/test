public class BlacklistDownloader {
  private static final String BLACKLIST_FILE_NAME = "eb.bl";
  
  private static final Charset CHARSET = Charset.forName("UTF-8");
  
  private final Logger logger = LoggerFactory.getLogger(BlacklistDownloader.class);
  
  private final FileBasedBlacklist blacklist;
  
  private final int retryInterval;
  
  private ScheduledFuture<?> future;
  
  private ScheduledExecutorService scheduler;
  
  private final BlacklistDownloadStateChangedPublisher publisher;
  
  private final UpdateServerConnection updateServerConnection;
  
  private final ApplicationProcessManager applicationProcessManager;
  
  BlacklistDownloader(FileBasedBlacklist blacklist, int retryInterval) {
    this.blacklist = blacklist;
    this.retryInterval = retryInterval;
    this.publisher = new BlacklistDownloadStateChangedPublisher();
    this.updateServerConnection = new UpdateServerConnection();
    this.applicationProcessManager = ApplicationProcessManager.getInstance();
  }
  
  public synchronized void start(BlacklistUpdateListener listener) {
    if (this.future == null) {
      this.scheduler = Executors.newScheduledThreadPool(2);
      this.future = this.scheduler.scheduleAtFixedRate(() -> download(listener), 0L, (long)this.retryInterval, TimeUnit.MILLISECONDS);
    } else {
      throw new IllegalStateException("Download process is already running");
    } 
  }
  
  private String readFile(String filePath) throws IOException {
    try (FileInputStream stream = new FileInputStream(new File(filePath))) {
      FileChannel fc = stream.getChannel();
      MappedByteBuffer bb = fc.map(FileChannel.MapMode.READ_ONLY, 0L, fc.size());
      return CHARSET.decode(bb).toString();
    } 
  }
  
  private void download(BlacklistUpdateListener listener) {
    if (!isBlacklistDownloadAllowed()) {
      this.logger.warn("Prevented downloading blacklist");
      return;
    } 
    listener.onUpdateCycleStart();
    if (!this.updateServerConnection.isServerAvailable()) {
      this.logger.warn("Failed to connect to server on downloading blacklist");
      this.publisher.fireChangedState(BlacklistDownloadStateChanged.CONNECTION_ERROR);
      listener.onUpdateCycleEnd();
      return;
    } 
    this.logger.info("Downloading blacklist...");
    String downloadPath = this.updateServerConnection.downloadFileSync(UPDATE_SERVER_URL.get(UPDATE_SERVER_URL.BLACKLIST_DOWNLOAD).toString() + "eb.bl");
    if (StringUtils.isNotBlank(downloadPath) && new File(downloadPath).exists()) {
      this.logger.info("Successfully downloaded blacklist");
      try {
        String responseBlackl = readFile(downloadPath);
        FileUtils.writeStringToFile(this.blacklist.getFile(), responseBlackl, CHARSET);
        this.blacklist.checkIntegrity();
        listener.onListAvailable(this.blacklist);
        stop();
      } catch (BlacklistIntegrityException e) {
        this.logger.warn("Downloaded blacklist is corrupt", (Throwable)e);
        this.publisher.fireChangedState(BlacklistDownloadStateChanged.DOWNLOAD_ERROR);
      } catch (IOException e) {
        this.logger.warn("Unable to save downloaded blacklist", (Throwable)e);
        this.publisher.fireChangedState(BlacklistDownloadStateChanged.DOWNLOAD_ERROR);
      } 
    } else {
      this.logger.warn("Error downloading blacklist.");
      this.publisher.fireChangedState(BlacklistDownloadStateChanged.DOWNLOAD_ERROR);
    } 
    listener.onUpdateCycleEnd();
  }
  
  public synchronized void stop() {
    if (this.future != null) {
      this.future.cancel(false);
      this.future = null;
    } 
    if (this.scheduler != null) {
      this.scheduler.shutdown();
      this.scheduler = null;
    } 
  }

    return this.publisher;
  }
  
  private boolean isBlacklistDownloadAllowed() {
    return !this.applicationProcessManager.isBlockingProcessActive();
  }

package com.gigatronik.beds.service.dongleblacklist.impl;

import com.gigatronik.beds.service.dongleblacklist.api.BlacklistIntegrityException;
import java.io.File;
import java.io.IOException;
import java.io.Reader;
import java.io.StringReader;
import java.nio.charset.StandardCharsets;
import java.nio.file.Files;
import java.util.LinkedList;
import java.util.List;
import java.util.concurrent.atomic.AtomicReference;
import org.apache.commons.csv.CSVFormat;
import org.apache.commons.csv.CSVParser;
import org.apache.commons.csv.CSVRecord;
import org.apache.commons.io.FileUtils;
import org.jasypt.exceptions.EncryptionOperationNotPossibleException;

class FileBasedBlacklist implements DongleBlacklist {
  private final File blacklistFile;
  
  private final BlacklistCryptor cryptor;
  
  private final AtomicReference<List<String>> blacklist = new AtomicReference<>();
  
  FileBasedBlacklist(File blacklistFile, String encryptionKey) {
    this.cryptor = new BlacklistCryptor(encryptionKey);
    this.blacklistFile = blacklistFile;
  }
  
  private String readEncryptedBlacklistContents() throws IOException {
    return FileUtils.readFileToString(this.blacklistFile, StandardCharsets.UTF_8);
  }
  
  private List<String> parseBlacklist() throws BlacklistIntegrityException {
    if (!Files.isWritable(this.blacklistFile.toPath()))
      throw new BlacklistIntegrityException("File " + this.blacklistFile + " is not writable"); 
    try {
      String plainData = this.cryptor.decrypt(readEncryptedBlacklistContents());
      Reader in = new StringReader(plainData);
      CSVParser cSVParser = CSVFormat.EXCEL.parse(in);
      List<String> result = new LinkedList<>();
      for (CSVRecord record : (Iterable<CSVRecord>)cSVParser) {
        String value = record.get(0);
        result.add(value);
      } 
      return result;
    } catch (IOException e) {
      throw new BlacklistIntegrityException("Error reading blacklist", e);
    } catch (EncryptionOperationNotPossibleException e) {
      throw new BlacklistIntegrityException("Error decrypting blacklist", e);
    } 
  }
  
  public boolean isBlacklisted(String dongleId) throws BlacklistIntegrityException {
    parseBlacklist();
    if (this.blacklist.get() == null) {
      List<String> parsed = parseBlacklist();
      this.blacklist.compareAndSet(null, parsed);
    } 
    return this.blacklist.get().contains(dongleId);
  }
  
  public void checkIntegrity() throws BlacklistIntegrityException {
    parseBlacklist();
  }
  
  public File getFile() {
    return this.blacklistFile;
  }
}
