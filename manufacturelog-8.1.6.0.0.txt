    if (response.isPositive())

        serialNo = SerialNumberHelper.getSerialnumberBui21x(response).toString();
      } else if (hmiType.isBUI25x()) {
        serialNo = SerialNumberHelper.getSerialnumberBui25x(response).toString();
      } else if (hmiType.isNyon()) {
        serialNo = SerialNumberHelper.getSerialnumberBui27x(response).toString();
      } else if (hmiType.isBUI33x()) {
        serialNo = SerialNumberHelper.getSerialnumberBui33x(response).toString();
      } else if (hmiType.isCUI1xx()) {
        serialNo = SerialNumberHelper.getSerialnumberBui21x(response).toString();
      }  
    return serialNo;
  }

    } catch (ConfigurationNotFoundException|java.awt.HeadlessException ex) {
      LOGGER.error("Error saving manufacture data. Please check path settings.", (Throwable)ex);
      TestableJOptionPane.showMessageDialog(BUNDLE.getString("EBikeOnExit.ManufacturerError"), "", 1);
      return false;
    } 
    return false;
  }
  
  public boolean isManufacturingLogTooBig() {
    return (size() > 2000);
  }
  
  private boolean isManufactureLogConsistent() {
    boolean check = false;
    if (isManufacturerlogPathValid()) {
      String fileName;
      try {
        fileName = this.settingsProxy.getGlobalSettings().getProperty("manufacture_log_directory") + this.settingsProxy.getGlobalSettings().getProperty("manufacture_log_directory") + "manufacture.log";
      } catch (ConfigurationNotFoundException e) {
        LOGGER.error("Error checking manufacture log file. Check path settings.", (Throwable)e);
        return true;
      } 
      try (BufferedReader in = new BufferedReader(new FileReader(fileName))) {
        String header = in.readLine();
        String[] headings = header.split(",");
        if (headings.length == (ManufacturingEntry.values()).length)
          check = true; 
      } catch (IOException e) {
        LOGGER.error("Error checking manufacture log file. Check path settings.", (Throwable)e);
        return true;
      } 
    } 
    return check;
  }
  
  private void addFlashResultValue(Map<ManufacturingEntry, String> manufacturingMap, ManufacturingLog.FlashSubject flashSubject, boolean wasSuccessful) {
    String flashResultEntry = wasSuccessful ? (flashSubject.toString() + "_SUCCESS") : (flashSubject.toString() + "_FAILED");
    manufacturingMap.put(ManufacturingEntry.FLASH_RESULT, flashResultEntry);
  }
  
  private void addPositiveResponseValue(Map<ManufacturingEntry, String> manufacturingMap, ManufacturingEntry key, ComplexResponse response) {
    if (response.isPositive())
      manufacturingMap.put(key, response.toString()); 
  }
  
  private int size() {
    String filename;
    int count = 0;
    boolean empty = true;
    try {
      filename = SettingsProxy.getInstance().getGlobalSettings().getProperty("manufacture_log_directory") + SettingsProxy.getInstance().getGlobalSettings().getProperty("manufacture_log_directory") + "manufacture.log";
    } catch (ConfigurationNotFoundException e) {
      LOGGER.error("error calculating size", (Throwable)e);
      return 0;
    } 
    try (InputStream is = new BufferedInputStream(new FileInputStream(filename))) {
      byte[] c = new byte[1024];
      int readChars;
      while ((readChars = is.read(c)) != -1) {
        empty = false;
        for (int i = 0; i < readChars; i++) {
          if (c[i] == 10)
            count++; 
        } 
      } 
    } catch (IOException ex) {
      LOGGER.error("error calculating size", (Throwable)ex);
      return 0;
    } 
    return (count == 0 && !empty) ? 1 : count;
  }
