    ServiceLoader<T> loader = ServiceLoader.load(interfaceType);
    Iterator<T> it = loader.iterator();
    List<T> result = new LinkedList<>();
    while (it.hasNext()) {

      current = substitudeWithSingletonInstance(current);
      result.add(current);
    }

    UUID uuid = plugin.getUuid();
    if (this.INSTANTIATED_SINGLETON_PLUGINS.containsKey(uuid)) {
      DartPlugin reference = this.INSTANTIATED_SINGLETON_PLUGINS.get(uuid);
      T tmp = (T)reference;
      if (tmp != null)
        return tmp; 
    } else if (plugin.isSingleton()) {
      this.INSTANTIATED_SINGLETON_PLUGINS.put(uuid, plugin);
    } 
    return plugin;
  }

  }

  }

  public static int byteArrayToInteger(byte[] data, int inBytePos, int bitPos, int bitLen) {
    if (bitPos < 0 || bitPos > 7)
      throw new IllegalArgumentException("ODX XSD defines a valid range from 0-7 for bitPos"); 
    if (bitLen < 1)
      throw new IllegalArgumentException("ODX XSD defines a minimum value of 1 for bitLen"); 
    int bitIndex = bitPos + inBytePos * 8;
    int dataBitsLen = bitLen - 1;
    BitUtils bit = new BitUtils(data);
    bit.setCurrentBitIndex(bitIndex);
    boolean isNegative = bit.getNextBoolean();
    int result = bit.getNextInteger(dataBitsLen);
    if (isNegative)
      result |= Integer.MAX_VALUE << dataBitsLen; 
    return result;
  }
  
  public static long byteArrayToUnsignedInteger(byte[] dump, int bytePos, int bitPos, int bitLen) {
    long value = 0L;
    int byteLength = (bitPos + bitLen) / 8;
    if ((bitPos + bitLen) % 8 > 0)
      byteLength++; 
    try {
      for (int i = 0; i < byteLength; i++)
        value = (value << 8L) + (long)(dump[bytePos + i] & 0xFF); 
      if (bitPos != 0 || bitLen < 8)
        value = value >> bitPos & (long)(255 >> 8 - bitLen); 
    } catch (ArrayIndexOutOfBoundsException ex) {
      logger.error("Invalid Array index", (Throwable)ex);
      value = 0L;
    } 
    return value;
  }
  
  public static int byteArrayToInteger(byte[] by) {
    if (by == null)
      return 0; 
    int iSize = 4;
    int iVal = 0;
    if (by.length < iSize)
      iSize = by.length; 
    for (int i = 0; i < iSize; i++)
      iVal = (iVal << 8) + (by[i] & 0xFF); 
    return iVal;
  }

  }

      return "";

  }

    String hex = inHex;

    hex = hex.replaceAll("\\s+", "").trim();
    int len = hex.length() / 2;
    byte[] by = new byte[len];
    for (int n = 0; n < len; n++) {

  public FlashContainer(InputStream stream) throws IOException, ParseException {
    ZipInputStream zipStream = new ZipInputStream(stream);
    MetaInfoEntity metainfo = null;
    Map<String, byte[]> datafiles = new HashMap<>();
    ZipEntry entry;
    while ((entry = zipStream.getNextEntry()) != null) {
      if (isMetaInfoFile(entry.getName())) {
        ByteArrayOutputStream rawMetaFile = new ByteArrayOutputStream();
        IOUtils.copy(zipStream, rawMetaFile);
        ObjectMapper mapper = new ObjectMapper();
        metainfo = (MetaInfoEntity)mapper.readValue(rawMetaFile.toByteArray(), MetaInfoEntity.class);
        continue;
      } 
      ByteArrayOutputStream rawBinFile = new ByteArrayOutputStream();
      IOUtils.copy(zipStream, rawBinFile);
      datafiles.put(entry.getName(), rawBinFile.toByteArray());
    } 
    if (metainfo == null)
      throw new IOException("No metainfo found"); 
    this.NAME = metainfo.packageName;
    this.ISSUE_DATE = DATE_FORMAT.parse(metainfo.issueDate);
    for (FlashFile file : metainfo.dataList.entries) {
      String expectedFileName = file.getShortName() + ".bin";
      byte[] data = datafiles.get(expectedFileName);
      if (data == null)
        throw new IOException("Meta specified a flash file that does not exist"); 
      file.setContents(data);
      this.flashfiles.add(file);
    } 
    for (String startupScreenIdentifier : metainfo.startupScreenLinks.keySet()) {
      String filename = metainfo.startupScreenLinks.get(startupScreenIdentifier);
      byte[] data = datafiles.get(filename);
      if (data == null)
        throw new IOException("Meta specified a startup screen '" + filename + "' that does not exist"); 
      this.startupScreens.put(startupScreenIdentifier, data);
    } 
  }

  }

  public GlobalSingleton(String lockId) {

      this.lockFilesDirectoryPath = ".";
    }

      this.channel = new RandomAccessFile(this.file, "rw").getChannel();
      this.lock = this.channel.tryLock();

        throw new OverlappingFileLockException();

      closeLock();

      this.logger.error("Error while accuireing lock", (Throwable)e);
    } 
  }
  
  public void unlock() {
    try {
      if (this.lock != null)
        this.lock.release(); 
    } catch (IOException e) {
      this.logger.error("Error while releasing lock", (Throwable)e);
    } finally {
      closeLock();
      deleteLockFile();
    } 
  }
  
  private void closeLock() {
    try {
      this.lock.release();
      this.channel.close();
    } catch (Exception e) {
      this.logger.error("Error closing ressources", (Throwable)e);
    } 
  }
  
  private void deleteLockFile() {
    try {
      this.file.delete();
    } catch (Exception e) {
      this.logger.error("Error deleting ressources", (Throwable)e);
    } 
  }

import java.lang.reflect.Method;
import java.lang.reflect.Proxy;
import javax.measure.Measure;
import javax.measure.quantity.Quantity;
import javax.measure.unit.Unit;

public class AutoConvertingMeasureProxy implements InvocationHandler {
  private final Object obj;
  
  private final UnitSystemProvider provider;
  
  private AutoConvertingMeasureProxy(Object obj2, UnitSystemProvider provider) {
    this.obj = obj2;
    this.provider = provider;
  }
  
  public static Object newInstance(Object obj, UnitSystemProvider provider) {
    return Proxy.newProxyInstance(obj.getClass().getClassLoader(), obj.getClass().getInterfaces(), new AutoConvertingMeasureProxy(obj, provider));
  }
  
  public Object invoke(Object proxy, Method method, Object[] args) throws Throwable {
    if (method.getReturnType() == Measure.class) {
      Measure<?, ?> original = (Measure<?, ?>)method.invoke(this.obj, args);
      return convertUnit((Measure)original, this.provider);
    } 
    return method.invoke(this.obj, args);
  }
  
  public static <V, Q extends Quantity> Measure<V, Q> convertUnit(Measure<V, Q> original, UnitSystemProvider provider) {
    if (original != null) {
      Unit<Q> targetUnit = (Unit<Q>)getTargetUnit(provider.getUnitSystem(), original.getUnit(), provider);
      return original.to(targetUnit);
    } 
    return null;
  }
  
  public static Unit<? extends Quantity> getTargetUnit(UnitLocale.UnitSystem unitSystem, Unit<? extends Quantity> originalUnit, UnitSystemProvider provider) {
    Unit<?> result = null;
    switch (unitSystem) {
      case IMPERIAL:
        result = provider.getImperialMapping().get(originalUnit);
        break;
      default:
        result = provider.getMetricMapping().get(originalUnit);
        break;
    } 
    if (result == null)
      result = originalUnit; 
    return (Unit)result;
  }
}

class JsonFilter {
