package com.gigatronik.beds.diagnosticdatapool.impl.reader;

import com.gigatronik.beds.diagnosticdatapool.api.exception.ServiceNotImplementedException;
import com.gigatronik.beds.diagnosticdatapool.api.serviceparameter.AbsServiceParameter;
import com.gigatronik.beds.diagnosticdatapool.api.serviceparameter.ServiceParameter;
import com.gigatronik.beds.service.communication.api.CommunicationManager;
import com.gigatronik.beds.service.ecu.api.Abs;
import com.gigatronik.beds.service.ecu.api.Ecu;
import com.gigatronik.beds.service.ecu.data.DtcResponse;
import com.gigatronik.ebike.common.configuration.BikeIdentification;
import com.gigatronik.ebike.common.configuration.ComplexResponse;
import com.gigatronik.ebike.common.configuration.NegativeResponse;
import java.util.Collections;
import java.util.HashMap;
import java.util.Map;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

public class AbsServiceParameterReader extends EcuServiceParameterReader {
  private static final Logger LOGGER = LoggerFactory.getLogger(AbsServiceParameterReader.class);
  
  private double wheelSpeedFactor = 0.132D;
  
  public AbsServiceParameterReader(CommunicationManager communicationManager, BikeIdentification bikeIdentification) {
    super(communicationManager, bikeIdentification);
    this.ecu = (Ecu)communicationManager.getAbsCom();
    this.ecuIdentification = bikeIdentification.getAbsIdentification();
  }
  
  public void updateWheelSizeFromDriveUnit() {
    int wheelSize = 2200;
    if (this.bikeIdentification.isDuConnected())
      try {
        wheelSize = Integer.parseInt(this.communicationManager.getDuCom().readWheelSize().getString());
      } catch (ComplexResponse.DataAccessException|NumberFormatException ex) {
        LOGGER.error("Parsing wheel size response failed", (Throwable)ex);
      }  
    this.wheelSpeedFactor = (double)wheelSize * 60.0D / 1000000.0D;
  }
  
  public Map<ServiceParameter, ComplexResponse> readServiceParameter(ServiceParameter serviceParameter) {
    return mappingParameterToMethod(serviceParameter);
  }
  
  public DtcResponse readDtcs() {
    return getEcuDtcs();
  }
  
  protected Map<ServiceParameter, ComplexResponse> mappingParameterToMethod(ServiceParameter serviceParameter) {
    switch ((AbsServiceParameter)serviceParameter) {
      case MANUFACTURE_DATE:
        return readManufactureDate(serviceParameter);
      case PARTNUMBER_KIT:
        return readPartNumberKit(serviceParameter);
      case NUMBER_OF_CONTROLS:
        return readNumberOfControls(serviceParameter);
      case SAFETY_LAMP_DEACTIVATION_EVENT:
        return readSafetyLampDeactivationEvent(serviceParameter);
      case FLASH_COUNTER:
        return readFlashCounter(serviceParameter);
      case EXTENDED_WSS_MOUNTING_TEST_RESULT:
        return readMountingTestResult(serviceParameter);
      case VARIANT_CHECKSUM:
      case VARIANT_CODE:
        return readVariantCode();
      case PART_NUMBER:
        return readPartNumber(serviceParameter);
      case PRODUCT_CODE:
        return readProductCode(serviceParameter);
      case COMPONENT_TYPE:
        return readComponentType(serviceParameter);
      case SERIAL_NUMBER:
        return readSerialNumber(serviceParameter);
      case HW_VERSION:
        return readHardwareVersion(serviceParameter);
      case SW_VERSION:
        return readSoftwareVersion(serviceParameter);
      case WSS_MOUNTING_TEST_BACK_STATUS:
      case WSS_MOUNTING_TEST_FRONT_STATUS:
        return readExtendedWssMountingTest();
      case MOUNTING_TEST_RESULT:
        return readMountingTestResult();
      case WHEEL_SPEED_FRONT:
      case WHEEL_SPEED_REAR:
      case BRAKE_PRESSURE:
        return readWheelSpeedAndBrakePressure();
      case SLIP_EVENTS:
      case LIFT_EVENTS:
      case BREAK_EVENTS:
        return readEventCounterSpeed();
    } 
    throw new ServiceNotImplementedException(serviceParameter);
  }
  
  private Map<ServiceParameter, ComplexResponse> readWheelSpeedAndBrakePressure() {
    Map<ServiceParameter, ComplexResponse> responseBundle = new HashMap<>();
    Map<String, String> expectedResponse = ((Abs)this.ecu).readWheelSpeedsAndPressures();
    if (!expectedResponse.isEmpty()) {
      responseBundle.put(AbsServiceParameter.WHEEL_SPEED_FRONT, customizeWheelSpeed(expectedResponse.get("WheelSpeedFront")));
      responseBundle.put(AbsServiceParameter.WHEEL_SPEED_REAR, customizeWheelSpeed(expectedResponse.get("WheelSpeedRear")));
      responseBundle.put(AbsServiceParameter.BRAKE_PRESSURE, new ComplexResponse(expectedResponse.get("EffectiveBrakePressure")));
    } else {
      this.log.warn("The response from the ABS is empty!");
      responseBundle.put(AbsServiceParameter.WHEEL_SPEED_FRONT, new ComplexResponse(NegativeResponse.UNSPECIFIED));
      responseBundle.put(AbsServiceParameter.WHEEL_SPEED_REAR, new ComplexResponse(NegativeResponse.UNSPECIFIED));
      responseBundle.put(AbsServiceParameter.BRAKE_PRESSURE, new ComplexResponse(NegativeResponse.UNSPECIFIED));
    } 
    return responseBundle;
  }
  
  private ComplexResponse customizeWheelSpeed(String value) {
    try {
      int speed = Integer.parseInt(value);
      return new ComplexResponse(Long.toString(Math.round((double)speed * this.wheelSpeedFactor)));
    } catch (NumberFormatException ex) {
      LOGGER.error("Parsing wheel speed response failed", (Throwable)ex);
      return new ComplexResponse(value);
    } 
  }
  
  private Map<ServiceParameter, ComplexResponse> readMountingTestResult(ServiceParameter serviceParameter) {
    return Collections.singletonMap(serviceParameter, ((Abs)this.ecu).readExtWSSMountingTestResult());
  }
  
  private Map<ServiceParameter, ComplexResponse> readExtendedWssMountingTest() {
    Map<ServiceParameter, ComplexResponse> responseBundle = new HashMap<>();
    Map<String, String> expectedResponse = ((Abs)this.ecu).readExtWSSMountingTest();
    if (!expectedResponse.isEmpty()) {
      responseBundle.put(AbsServiceParameter.WSS_MOUNTING_TEST_FRONT_STATUS, new ComplexResponse(expectedResponse.get("WSS_front_status")));
      responseBundle.put(AbsServiceParameter.WSS_MOUNTING_TEST_BACK_STATUS, new ComplexResponse(expectedResponse.get("WSS_back_status")));
    } else {
      this.log.warn("The response from the ABS is empty!");
      responseBundle.put(AbsServiceParameter.WSS_MOUNTING_TEST_FRONT_STATUS, new ComplexResponse(NegativeResponse.UNSPECIFIED));
      responseBundle.put(AbsServiceParameter.WSS_MOUNTING_TEST_BACK_STATUS, new ComplexResponse(NegativeResponse.UNSPECIFIED));
    } 
    return responseBundle;
  }
  
  private Map<ServiceParameter, ComplexResponse> readVariantCode() {
    Map<ServiceParameter, ComplexResponse> responseBundle = new HashMap<>();
    ComplexResponse expectedResponse = ((Abs)this.ecu).readVariantCode();
    try {
      responseBundle.put(AbsServiceParameter.VARIANT_CODE, new ComplexResponse(expectedResponse.getParameterByName("AbsVariantCode").getValue()));
      responseBundle.put(AbsServiceParameter.VARIANT_CHECKSUM, new ComplexResponse(expectedResponse.getParameterByName("AbsVariantCode_CheckPattern").getValue()));
    } catch (ComplexResponse.DataAccessException e) {
      this.log.warn("Problem retrieving data from the ABS response", (Throwable)e);
      responseBundle.put(AbsServiceParameter.VARIANT_CODE, expectedResponse);
      responseBundle.put(AbsServiceParameter.VARIANT_CHECKSUM, expectedResponse);
    } 
    return responseBundle;
  }
  
  private Map<ServiceParameter, ComplexResponse> readMountingTestResult() {
    Map<ServiceParameter, ComplexResponse> responseBundle = new HashMap<>();
    responseBundle.put(AbsServiceParameter.SAFETY_LAMP_DEACTIVATION_EVENT, ((Abs)this.ecu).readSafetyLampDeactivatonEvent());
    responseBundle.put(AbsServiceParameter.EXTENDED_WSS_MOUNTING_TEST_RESULT, ((Abs)this.ecu).readExtWSSMountingTestResult());
    return responseBundle;

    if (value.isPositive())
      try {
        Calendar c = new GregorianCalendar();
        c.setTime(value.getDate());
        return new ComplexResponse(String.format("%02d:%02d", c.get(11), c.get(12)));
      } catch (ComplexResponse.DataAccessException ex) {
        LOGGER.error("Error processing kiox present time", (Throwable)ex);
      }  
    return value;
  }

    if (value.isPositive())
      try {
        Calendar c = new GregorianCalendar();
        c.setTime(value.getDate());
        return new ComplexResponse(String.format("%04d-%02d-%02d", c.get(1), c.get(2) + 1, c.get(5)));
      } catch (ComplexResponse.DataAccessException ex) {
        LOGGER.error("Error processing kiox present date", (Throwable)ex);
      }  
    return value;
  }

  private static final Logger LOGGER = LoggerFactory.getLogger(ServiceParameterDatapool.class);
  
  private Map<ServiceParameter, ComplexResponse> parameterDatapool;
  
  private Map<ServiceParameter.Ecu, DtcResponse> dtcDatapool;

  ServiceParameterDatapoolImpl(CommunicationManager communicationManager, ConfigurationManager configurationManager) {
    this.parameterDatapool = new HashMap<>();
    this.dtcDatapool = new EnumMap<>(ServiceParameter.Ecu.class);
    this.listeners = new LinkedList<>();
    this.configurationManager = configurationManager;
    this.bikeIdentification = configurationManager.getBikeIdentification();
    this.serviceParameterReaderProvider = new ServiceParameterReaderProvider(communicationManager, this.bikeIdentification);
    listenPreliminaryContainerDownloader(PreliminaryContainerDownloaderFactory.getPreliminaryContainerDownloader());
  }
  
  public ComplexResponse getServiceParameter(ServiceParameter serviceParameter) {
    ComplexResponse expectedResponse = this.parameterDatapool.get(serviceParameter);
    try {
      if (expectedResponse == null)
        expectedResponse = handleServiceParameterRequest(serviceParameter); 
    } catch (ServiceNotImplementedException e) {
      LOGGER.error(e.getMessage());
      expectedResponse = handleServiceParameterNotImplementedError(serviceParameter);
    } 
    return expectedResponse;
  }
  
  private ComplexResponse handleServiceParameterNotImplementedError(ServiceParameter serviceParameter) {
    ComplexResponse expectedResponse = new ComplexResponse(NegativeResponse.REQUEST_OUT_OF_RANGE);
    this.parameterDatapool.put(serviceParameter, expectedResponse);
    return expectedResponse;
  }
  
  private ComplexResponse handleServiceParameterRequest(ServiceParameter serviceParameter) {
    EcuServiceParameterReader ecuServiceParameterReader = this.serviceParameterReaderProvider.getParameterReader(serviceParameter);
    Map<ServiceParameter, ComplexResponse> responseBundle = ecuServiceParameterReader.readServiceParameter(serviceParameter);
    ComplexResponse expectedResponse = responseBundle.get(serviceParameter);
    this.parameterDatapool.putAll(responseBundle);
    return expectedResponse;
  }
  
  public Map<ServiceParameter, ComplexResponse> getServiceParameterBundle(List<ServiceParameter> serviceParameterList) {
    Map<ServiceParameter, ComplexResponse> returnMap = new HashMap<>();
    for (ServiceParameter parameter : serviceParameterList)
      returnMap.put(parameter, getServiceParameter(parameter)); 
    return returnMap;
  }
  
  public Map<ServiceParameter, ComplexResponse> getCleanServiceParameterBundle(List<ServiceParameter> serviceParameterList) {
    Map<ServiceParameter, ComplexResponse> returnMap = new HashMap<>();
    for (ServiceParameter parameter : serviceParameterList)
      resetServiceParameterCache(parameter); 
    for (ServiceParameter parameter : serviceParameterList)
      returnMap.put(parameter, getServiceParameter(parameter)); 
    return returnMap;
  }
  
  public DtcResponse getErrorMemoryContent(ServiceParameter.Ecu ecu) {
    DtcResponse expectedResponse = this.dtcDatapool.get(ecu);
    if (expectedResponse == null) {
      EcuServiceParameterReader ecuServiceParameterReader = this.serviceParameterReaderProvider.getParameterReader(ecu);
      expectedResponse = (ecu == ServiceParameter.Ecu.BMS2) ? ((BmsServiceParameterReader)ecuServiceParameterReader).readDtcsBms2() : ecuServiceParameterReader.readDtcs();
      this.dtcDatapool.put(ecu, expectedResponse);
    } 
    return expectedResponse;
  }
  
  private void resetCache() {
    resetServiceParameterCache();
    resetDtcCache();
  }
  
  private void resetServiceParameterCache() {
    this.parameterDatapool = new HashMap<>();
  }
  
  public void resetServiceParameterCache(ServiceParameter.Ecu ecu) {
    this.parameterDatapool.keySet().removeIf(parameter -> (parameter.getEcuType() == ecu));
    if (ecu.equals(ServiceParameter.Ecu.DU))
      resetServiceParameterCache(ServiceParameter.Ecu.ESHIFT); 
  }
  
  public void resetServiceParameterCache(ServiceParameter serviceParameter) {
    try {
      this.parameterDatapool.remove(serviceParameter);
    } catch (NullPointerException e) {
      LOGGER.info("The service parameter {} has not been yet retrieved from the ECU", serviceParameter.getText());
    } 
  }
  
  private void resetDtcCache() {
    this.dtcDatapool = new EnumMap<>(ServiceParameter.Ecu.class);
  }
  
  public void resetDtcCache(ServiceParameter.Ecu ecu) {
    this.dtcDatapool.remove(ecu);
    if (ecu.equals(ServiceParameter.Ecu.DU))
      resetDtcCache(ServiceParameter.Ecu.ESHIFT); 
  }
  
  public void onHardwareConnectionChanged(HardwareConnectionChangedEvent event) {
    LOGGER.debug("HardwareConnectionChangedEvent received: '{}'", event);
    if (event.getAction() == HardwareConnectionChangedEvent.HardwareConnectionStateChange.ECU_CONNECTED && event.getComponent().isDriveUnit()) {
      updateAbsReaderOnDriveUnitDataChange();
      if (this.configurationManager.isContainerLoaded()) {
        extractModelNameFromLoadedContainer();
        if (shouldServiceTabBeRepainted())
          notifyDatapoolUpdated(); 
      } 
    } 
    if (event.getAction() == HardwareConnectionChangedEvent.HardwareConnectionStateChange.ECU_DISCONNECTED) {
      ServiceParameter.Ecu ecu;
      switch (event.getComponent()) {
        case BUI21X:
        case BUI25X:
        case BUI270:
        case BUI275:
        case BUI330:
        case BUI350:
        case COBI:
          resetCache();
          return;
        case CLASSIC_DU:
        case VICTOR_DU:
          ecu = ServiceParameter.Ecu.DU;
          updateAbsReaderOnDriveUnitDataChange();
          break;
        case CLASSIC_PP:
        case VICTOR_PP1:
          ecu = ServiceParameter.Ecu.BMS1;
          break;
        case VICTOR_PP2:
          ecu = ServiceParameter.Ecu.BMS2;
          break;
        case BAS100:
          ecu = ServiceParameter.Ecu.ABS;
          break;
        case BCS250:
          ecu = ServiceParameter.Ecu.BCS;
          break;
        case BCT:
          ecu = ServiceParameter.Ecu.BCT;
          break;
        default:
          LOGGER.warn("The disconnected ECU: {} is not recognized by the Datapool, aborting cleaning operation.", event.getComponent());
          return;
      } 
      resetServiceParameterCache(ecu);
      resetDtcCache(ecu);
    } 
  }
  
  private void updateAbsReaderOnDriveUnitDataChange() {
    AbsServiceParameterReader absReader = (AbsServiceParameterReader)this.serviceParameterReaderProvider.getParameterReader(ServiceParameter.Ecu.ABS);
    absReader.updateWheelSizeFromDriveUnit();
  }
  
  public void onPreliminaryContainerDownloaded(PreliminaryContainerEvent event) {
    ComplexResponse applicationIdResponse = this.bikeIdentification.getApplicationIdResponse();
    if (StringUtils.isNotBlank(applicationIdResponse.toString())) {
      ComplexResponse modelName = new ComplexResponse(event.getTranslatedBikeModel());
      if (StringUtils.isNotBlank(modelName.toString()))
        this.parameterDatapool.put(DuServiceParameter.MODEL_NAME, modelName); 
    } 
    notifyDatapoolUpdated();
  }
  
  public void registerListener(DatapoolUpdatesListener listener) {
    this.listeners.add(listener);
  }
  
  private void notifyDatapoolUpdated() {
    LOGGER.info("Fire datapool updated event.");
    for (DatapoolUpdatesListener listener : this.listeners)
      listener.onDatapoolUpdated(); 
  }
  
  public void onBikeConfigurationChanged(BikeConfigurationChangedEvent event) {
    LOGGER.debug("BikeConfigurationChangedEvent received: '{}'", event);
    if (event.getAction() == BikeConfigurationChangedEvent.Action.WRITE_TO_BIKE) {
      resetServiceParameterCache(ServiceParameter.Ecu.HMI);
      resetServiceParameterCache(ServiceParameter.Ecu.DU);
      updateAbsReaderOnDriveUnitDataChange();
    } 
  }
  
  public void onContainerConfigurationChanged(ContainerConfigurationChangedEvent event) {
    LOGGER.debug("ContainerConfigurationChangedEvent received: '{}'", event);
    if (event.getAction() == ContainerConfigurationChangedEvent.Action.LOADED && 
      extractModelNameFromLoadedContainer() && 
      shouldServiceTabBeRepainted())
      notifyDatapoolUpdated(); 
  }
  
  private boolean shouldServiceTabBeRepainted() {
    ComplexResponse response = this.parameterDatapool.get(DuServiceParameter.APPLICATION_ID);
    return (response != null && response.isPositive());
  }
  
  private boolean extractModelNameFromLoadedContainer() {
    ComplexResponse modelNameResponse = this.parameterDatapool.getOrDefault(DuServiceParameter.MODEL_NAME, new ComplexResponse(""));
    ComplexResponse applicationIdResponse = this.bikeIdentification.getApplicationIdResponse();
    if (StringUtils.isBlank(modelNameResponse.toString()) && StringUtils.isNotBlank(applicationIdResponse.toString())) {
      String modelNameRaw = this.configurationManager.getContainerConfiguration().getContainerController().getBoschContainerCommonInterface().getEbikeModel(applicationIdResponse.toString().trim());
      ComplexResponse modelName = new ComplexResponse(BikeModelTranslationHelper.createTranslatedBikeModel(modelNameRaw));
      if (StringUtils.isNotBlank(modelName.toString())) {
        this.parameterDatapool.put(DuServiceParameter.MODEL_NAME, modelName);
        return true;
      } 
    } 
    return false;
  }
}

  private EcuServiceParameterReader absParameterReader;
  
  private EcuServiceParameterReader bcsParameterReader;

  ServiceParameterReaderProvider(CommunicationManager communicationManager, BikeIdentification bikeIdentification) {
    this.communicationManager = communicationManager;
    this.bikeIdentification = bikeIdentification;
  }
  
  EcuServiceParameterReader getParameterReader(ServiceParameter ecuServiceParameter) {
    ServiceParameter.Ecu ecu = ecuServiceParameter.getEcuType();
    return getParameterReader(ecu);
  }
  
  EcuServiceParameterReader getParameterReader(ServiceParameter.Ecu ecu) {
    switch (ecu) {
      case HMI:
        if (this.hmiParameterReader == null)
          this.hmiParameterReader = new HmiServiceParameterReader(this.communicationManager, this.bikeIdentification); 
        return this.hmiParameterReader;
      case BMS1:
      case BMS2:
        if (this.bmsParameterReader == null)
          this.bmsParameterReader = new BmsServiceParameterReader(this.communicationManager, this.bikeIdentification); 
        return this.bmsParameterReader;
      case DU:
        if (this.bduParameterReader == null)
          this.bduParameterReader = new DuServiceParameterReader(this.communicationManager, this.bikeIdentification); 
        return this.bduParameterReader;
      case ABS:
        if (this.absParameterReader == null)
          this.absParameterReader = new AbsServiceParameterReader(this.communicationManager, this.bikeIdentification); 
        return this.absParameterReader;
      case BCS:
        if (this.bcsParameterReader == null)
          this.bcsParameterReader = new BcsServiceParameterReader(this.communicationManager, this.bikeIdentification); 
        return this.bcsParameterReader;
      case BCT:
        if (this.bctParameterReader == null)
          this.bctParameterReader = new BctServiceParameterReader(this.communicationManager, this.bikeIdentification); 
        return this.bctParameterReader;
      case ESHIFT:
        if (this.eShiftParameterReader == null)
          this.eShiftParameterReader = new EShiftServiceParameterReader(this.communicationManager, this.bikeIdentification); 
        return this.eShiftParameterReader;
    } 
    return null;
  }
