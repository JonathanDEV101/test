class AbsWrapper {
  private static final Logger LOGGER = LoggerFactory.getLogger(AbsWrapper.class);
  
  private static final int MAX_TESTER_PRESENT = 3;
  
  private Constants.EcuMode mode = Constants.EcuMode.NONE;
  
  private final Abs absCom;
  
  private final RightsManager rightsManager;
  
  private final SecurityAccessFailurePopupHandler securityAccessFailurePopupHandler;
  
  AbsWrapper(Abs abs, RightsManager rightsManager, SecurityAccessFailurePopupHandler securityAccessFailurePopupHandler) {
    this.absCom = abs;
    this.rightsManager = rightsManager;
    this.securityAccessFailurePopupHandler = securityAccessFailurePopupHandler;
  }
  
  Constants.EcuMode checkState() {
    if (ping()) {
      if (this.mode == Constants.EcuMode.NONE)
        if (this.rightsManager.isUserInLevel(UserLevel.UNKNOWN)) {
          LOGGER.info("Prevented initialization of new detected device, 'cause of missing dongle.");
        } else {
          LOGGER.info("Detected new ECU '{}'", Constants.EcuType.BAS100);
          onConnect();
        }  
    } else if (this.mode != Constants.EcuMode.NONE) {
      this.mode = Constants.EcuMode.NONE;
    } 
    return this.mode;
  }
  
  void resetState() {
    this.mode = Constants.EcuMode.NONE;
  }
  
  private boolean ping() {
    ComplexResponse response = new ComplexResponse(NegativeResponse.UNSPECIFIED);
    if (this.mode == Constants.EcuMode.NONE) {
      response = this.absCom.startExtendedSession();
    } else {
      for (int n = 0; n < 3 && !response.isPositive(); n++)
        response = this.absCom.sendTesterPresent(); 
    } 
    return response.isPositive();
  }
  
  private void onConnect() {
    this.mode = this.absCom.getEcuMode();
    if (this.mode == Constants.EcuMode.APP) {
      ComplexResponse response;
      switch (this.rightsManager.getUserLevel()) {
        case IBD:
          response = this.absCom.executeSecurityAccess(Constants.AccessLevel.IBD);
          break;
        case OEM:
          response = this.absCom.executeSecurityAccess(Constants.AccessLevel.OEM);
          break;
        default:
          response = this.absCom.executeSecurityAccess(Constants.AccessLevel.LEVEL_0);
          break;
      } 
      if (!response.isPositive()) {
        this.absCom.startDefaultSession(false);
        this.securityAccessFailurePopupHandler.displaySecurityAccessFailedDialog(true, ComponentType.COMP_BBA100);
        this.mode = Constants.EcuMode.NONE;
      } 
    } 
  }
}

package com.gigatronik.beds.service.ecuconnectionstate.imp;

import com.gigatronik.beds.service.communication.api.CommunicationManager;

import com.gigatronik.beds.service.rightsmanager.api.UserLevel;
import com.gigatronik.ebike.common.configuration.Constants;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

public class BaudrateChecker {
  private static final Logger LOGGER = LoggerFactory.getLogger(BaudrateChecker.class);
  
  private static int WAITING_TIME_AFTER_BAUDRATE_CONFIGURATION = 1000;
  
  private boolean bms1connected;
  
  private boolean bms2connected;
  
  private final CommunicationManager communicationManager;
  
  private final RightsManager rightsManager;
  
  BaudrateChecker(CommunicationManager communicationManager, RightsManager rightsManager) {
    this.communicationManager = communicationManager;
    this.rightsManager = rightsManager;
  }
  
  Constants.SystemType checkBaudrate(boolean hmiIsIntuvia) {
    resetLocalMembers();
    if (!hmiIsIntuvia)
      return onNotIntuvia(); 
    Hmi.ConfiguredCanBaudrate baudrate = readBaudrateSetting();
    if (baudrate.isIntuviaConfiguredCorrectly()) {
      Constants.SystemType systemType = tryCurrentSetting(baudrate);
      if (systemType != Constants.SystemType.NONE)
        return systemType; 
      return tryOtherSetting(baudrate);
    } 
    return pingAnyEcu();
  }
  
  private Constants.SystemType onNotIntuvia() {
    LOGGER.debug("No check necessary, 'caus display does only support Victor ECUs.");
    pingVictorEcus(false);
    return Constants.SystemType.VICTOR;
  }
  
  private Constants.SystemType tryCurrentSetting(Hmi.ConfiguredCanBaudrate configuredBaudrate) {
    if (configuredBaudrate == Hmi.ConfiguredCanBaudrate.CLASSIC_125k) {
      LOGGER.debug("Intuvia's configured Classic+ / 125k, check connection");
      if (pingClassicPlusEcus()) {
        LOGGER.debug("SystemType detected! --> It's a CLASSIC+ system");
        return Constants.SystemType.CLASSIC_PLUS;
      } 
    } else {
      LOGGER.debug("Intuvia's configured Victor / 500k, check connection");
      if (pingVictorEcus(true)) {
        LOGGER.debug("SystemType detected! --> It's a VICTOR system");
        return Constants.SystemType.VICTOR;
      } 
    } 
    LOGGER.debug("Not able to build up a communication with current setting");
    return Constants.SystemType.NONE;
  }
  
  private Constants.SystemType tryOtherSetting(Hmi.ConfiguredCanBaudrate configuredBaudrate) {
    if (configuredBaudrate == Hmi.ConfiguredCanBaudrate.VICTOR_500k) {
      LOGGER.debug("Try Classic+ / 125k baudrate");
      this.communicationManager.getHmiCom().writeCanBaudrate(Hmi.ConfiguredCanBaudrate.CLASSIC_125k);
      waitForReconfigurationIsProcessedByEcu();
      if (pingClassicPlusEcus()) {
        LOGGER.debug("CAN baudrate configuration done --> It's a CLASSIC+ system");
        return Constants.SystemType.CLASSIC_PLUS;
      } 
    } else {
      LOGGER.debug("Try Victor / 500k baudrate");
      this.communicationManager.getHmiCom().writeCanBaudrate(Hmi.ConfiguredCanBaudrate.VICTOR_500k);
      waitForReconfigurationIsProcessedByEcu();
      if (pingVictorEcus(true)) {
        LOGGER.debug("CAN baudrate configuration done --> It's a VICTOR system");
        return Constants.SystemType.VICTOR;
      } 
    } 
    LOGGER.warn("Not able to build up a communication with any setup");
    return Constants.SystemType.NONE;
  }
  
  private Constants.SystemType pingAnyEcu() {
    LOGGER.debug("Read out CAN baudrate is still unvalid, try to reach any ECU");
    if (pingVictorEcus(true)) {
      LOGGER.debug("SystemType detected! --> It's a VICTOR system");
      return Constants.SystemType.VICTOR;
    } 
    if (pingClassicPlusEcus()) {
      LOGGER.debug("SystemType detected! --> It's a CLASSIC+ system");
      return Constants.SystemType.CLASSIC_PLUS;
    } 
    LOGGER.warn("SystemType could not be detected!");
    return Constants.SystemType.NONE;
  }
  
  private Hmi.ConfiguredCanBaudrate readBaudrateSetting() {
    Hmi hmiCom = this.communicationManager.getHmiCom();
    Hmi.ConfiguredCanBaudrate baudrate = hmiCom.readCanBaudrate();
    if (!baudrate.isIntuviaConfiguredCorrectly()) {
      LOGGER.debug("Read out CAN baudrate is unvalid ('{}'), write Victor / 500k", baudrate);
      hmiCom.writeCanBaudrate(Hmi.ConfiguredCanBaudrate.VICTOR_500k);
      baudrate = hmiCom.readCanBaudrate();
    } 
    return baudrate;
  }
  
  private boolean pingVictorEcus(boolean pingBcs) {
    boolean bcsConnected;
    this.bms1connected = this.communicationManager.getBms1Com().getBmsVictor().sendTesterPresent().isPositive();
    this.bms2connected = this.communicationManager.getBms2Com().getBmsVictor().sendTesterPresent().isPositive();
    if (pingBcs && this.rightsManager.isUserInLevel(UserLevel.IBD)) {
      bcsConnected = this.communicationManager.getBcsCom().sendTesterPresent().isPositive();
    } else {
      bcsConnected = false;
    } 
    return (this.bms1connected || this.bms2connected || bcsConnected);
  }
  
  private boolean pingClassicPlusEcus() {
    this.bms1connected = this.communicationManager.getBms1Com().getBmsClassicPlus().sendTesterPresent().isPositive();
    this.bms2connected = false;
    return this.bms1connected;
  }
  
  private void resetLocalMembers() {
    this.bms1connected = false;
    this.bms2connected = false;
  }
  
  boolean bms1HasBeenDetectedDuringLastBaudrateCheck() {
    return this.bms1connected;
  }
  
  boolean bms2HasBeenDetectedDuringLastBaudrateCheck() {
    return this.bms2connected;
  }
  
  private void waitForReconfigurationIsProcessedByEcu() {
    try {
      Thread.sleep((long)WAITING_TIME_AFTER_BAUDRATE_CONFIGURATION);
    } catch (InterruptedException ex) {
      LOGGER.warn("Sleep interrupted", (Throwable)ex);
      Thread.currentThread().interrupt();
    } 
  }
}

package com.gigatronik.beds.service.ecuconnectionstate.imp;

import com.gigatronik.beds.service.ecuconnectionstate.impl.synchronization.ActionSynchronizer;
import java.util.ArrayList;
import java.util.List;

class BctEventPublisher implements BctEventHandler {
  private final List<BctEventHandler.BctMeasurementProgressListener> progressListeners = new ArrayList<>();
  
  private final List<BctEventHandler.BctOperationStateListener> stateListeners = new ArrayList<>();
  
  private ApplicationEventQueue applicationEventQueue = ApplicationEventQueue.getInstance();
  
  public void subscribeBctMeasurementProgressChanges(BctEventHandler.BctMeasurementProgressListener listener) {
    if (!this.progressListeners.contains(listener))
      this.progressListeners.add(listener); 
  }
  
  public void unsubscribeBctMeasurementProgressChanges(BctEventHandler.BctMeasurementProgressListener listener) {
    this.progressListeners.remove(listener);
  }
  
  protected void publishProgressEvent(BctMeasurementProgressEvent event) {
    this.applicationEventQueue.queueJob(() -> this.progressListeners.forEach(listener -> listener.onBctMeasurementProgress(event)), "Publish BCT measurement progress event");
  }
  
  public void subscribeBctOperationStateChanges(BctEventHandler.BctOperationStateListener listener) {
    if (!this.stateListeners.contains(listener))
      if (listener.getClass().toString().endsWith("ConfigurationManagerImpl")) {
        this.stateListeners.add(0, listener);
      } else if (listener.getClass().toString().endsWith("StartMeasurementPanelController") && this.stateListeners.size() > 1) {
        this.stateListeners.add(1, listener);
      } else {
        this.stateListeners.add(listener);
      }  
  }
  
  public void unsubscribeBctOperationStateChanges(BctEventHandler.BctOperationStateListener listener) {
    this.stateListeners.remove(listener);
  }
  
  protected void publishOperationStateEvent(BctOperationStateChangeEvent event) {
    this.applicationEventQueue.queueJob(() -> this.stateListeners.forEach(listener -> listener.onBctOperationStateChanged(event)), "Publish BCT operation state change event");
    queueReleaseJobForServiceTabThread(event);
  }
  
  private void queueReleaseJobForServiceTabThread(BctOperationStateChangeEvent event) {
    if (event.getOperatingState() == BctState.IDLE_BATTERY_SET)
      this.applicationEventQueue.queueJob(() -> ActionSynchronizer.getInstance().notifyComplete(), "Queue notify job for waiting service tab thread at the end of pipe"); 
  }
}

package com.gigatronik.beds.service.ecuconnectionstate.imp;

import com.gigatronik.beds.common.bct.BctBatteryType;
import com.gigatronik.beds.common.bct.BctState;
import com.gigatronik.beds.common.bct.BctTestProgress;
import com.gigatronik.beds.service.communication.api.CommunicationManager;
import com.gigatronik.beds.service.ecuconnectionstate.impl.event.BctMeasurementProgressEvent;
import com.gigatronik.beds.service.ecuconnectionstate.impl.event.BctOperationStateChangeEvent;
import com.gigatronik.ebike.common.configuration.Constants;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

class BctWrapper {
  private static final Logger LOGGER = LoggerFactory.getLogger(BctWrapper.class);
  
  private final CommunicationManager communicationManager;
  
  private final BctEventPublisher publisher;
  
  private BctState previousState = BctState.NOT_CONNECTED;
  
  BctWrapper(BctEventPublisher publisher, CommunicationManager communicationManager) {
    this.publisher = publisher;
    this.communicationManager = communicationManager;
  }
  
  BctState checkOperationState() {
    BctState currentState = this.communicationManager.getBctCom().readOperatingState();
    if (currentState == BctState.NOT_CONNECTED && startExtendedSession())
      currentState = this.communicationManager.getBctCom().readOperatingState(); 
    if (currentState != this.previousState) {
      LOGGER.info("BCT operation state changed from '{}' -> '{}'", this.previousState, currentState);
      actionOnBctOperationStateChange(currentState);
      this.previousState = currentState;
      publishStateChangeEvent(currentState);
    } 
    if (currentState == BctState.TEST_RUNNING || currentState == BctState.DISCHARGE_RUNNING) {
      BctTestProgress progress = this.communicationManager.getBctCom().getMeasurementProgress();
      publishMeasurementProgressEvent(progress);
    } 
    return currentState;
  }
  
  private void actionOnBctOperationStateChange(BctState newState) {
    BctBatteryType selectedBattery;
    switch (newState) {
      case NOT_CONNECTED:
      case CONNECTED_FBL:
      case IDLE:
        this.communicationManager.setEcuType(Constants.EcuType.VICTOR_PP1, Constants.EcuMode.NONE);
        break;
      case IDLE_BATTERY_SET:
        LOGGER.debug("Request configured battery generation from BCT and configure communication layer.");
        selectedBattery = this.communicationManager.getBctCom().getSelectedBattery();
        this.communicationManager.setEcuType((selectedBattery == BctBatteryType.CLASSIC) ? Constants.EcuType.CLASSIC_PP : Constants.EcuType.VICTOR_PP1, Constants.EcuMode.APP);
        break;
    } 
  }
  
  boolean startExtendedSession() {
    return this.communicationManager.getBctCom().startExtendedSession().isPositive();
  }
  
  void publishStateChangeEvent(BctState bctState) {
    this.publisher.publishOperationStateEvent(new BctOperationStateChangeEvent(bctState));
  }
  
  private void publishMeasurementProgressEvent(BctTestProgress progress) {
    this.publisher.publishProgressEvent(new BctMeasurementProgressEvent(progress));
  }
  
  void resetState() {
    this.previousState = BctState.NOT_CONNECTED;
  }

    if (hmiPresentMode == Constants.EcuMode.APP && !bcsConnected && !isInterrupted()) {

        return new ChargingCheckResult();
      } 
      if (!chargingState && System.currentTimeMillis() - this.lastBcsConnectCheck < 3000L)
        return new ChargingCheckResult(); 
      LOGGER.debug("Checking if BCS's connected");
      this.lastBcsConnectCheck = System.currentTimeMillis();

  private ChargingCheckResult checkBcsConnectionState(boolean chargingState, boolean duConnected, boolean bms1Connected, boolean bms2Connected, boolean absConnected) {
    if (!this.vciWrapper.sendTesterPresent(Constants.EcuType.BCS250, chargingState ? 3 : 1, 200)) {
      if (chargingState)
        return new ChargingCheckResult(true, ChargingStateEvent.ChargingMessage.IBD_HINT, Constants.EcuType.NONE, Constants.EcuMode.NONE); 
      return new ChargingCheckResult();
    } 
    if (this.rightsManager.isUserInLevel(UserLevel.UNKNOWN)) {
      LOGGER.info("Prevented initialization of new detected device, 'cause of missing dongle.");
      return new ChargingCheckResult(false, ChargingStateEvent.ChargingMessage.NO_TEXT, Constants.EcuType.NONE, Constants.EcuMode.NONE);
    } 
    LOGGER.info("Detected new ECU '{}'", Constants.EcuType.BCS250);
    if (!bms1Connected)
      bms1Connected = this.communicationManager.getBms1Com().getBmsVictor().sendTesterPresent().isPositive(); 
    if (!bms2Connected)
      bms2Connected = this.communicationManager.getBms2Com().getBmsVictor().sendTesterPresent().isPositive(); 
    if (bms1Connected || bms2Connected) {
      ChargingStateEvent.ChargingMessage preConditionsCheckResult = checkBcsPreConditions(duConnected, bms1Connected, bms2Connected, absConnected);
      if (preConditionsCheckResult == ChargingStateEvent.ChargingMessage.NO_TEXT) {
        if (this.communicationManager.getBcsCom().readEcuState() == Constants.EcuMode.APP) {
          if (this.restoreBcsSession || askUserForBcsDiag()) {
            boolean bool1 = connectBcs(false, false);
            return new ChargingCheckResult(false, ChargingStateEvent.ChargingMessage.NO_TEXT, bool1 ? Constants.EcuType.BCS250 : Constants.EcuType.NONE, Constants.EcuMode.APP);
          } 
          return new ChargingCheckResult(true, ChargingStateEvent.ChargingMessage.OEM, Constants.EcuType.NONE, Constants.EcuMode.NONE);
        } 
        boolean bool = connectBcs(false, true);
        return new ChargingCheckResult(false, ChargingStateEvent.ChargingMessage.NO_TEXT, bool ? Constants.EcuType.BCS250 : Constants.EcuType.NONE, Constants.EcuMode.FBL);
      } 
      return new ChargingCheckResult(true, preConditionsCheckResult, Constants.EcuType.NONE, Constants.EcuMode.NONE);
    } 
    Constants.EcuMode bcsMode = this.communicationManager.getBcsCom().readEcuState();
    boolean success = connectBcs(true, (bcsMode == Constants.EcuMode.FBL));
    return new ChargingCheckResult(false, ChargingStateEvent.ChargingMessage.NO_TEXT, success ? Constants.EcuType.BCS250 : Constants.EcuType.NONE, bcsMode);
  }
  
  private ChargingStateEvent.ChargingMessage checkBcsPreConditions(boolean duConnected, boolean bms1Connected, boolean bms2Connected, boolean absConnected) {
    boolean cDualBat = false;
    boolean duIsSleeping = false;
    Version duSwVer = null;
    Version bmsSwVer = null;
    boolean cEbikeOk = eBikeEcusFblCheck(duConnected, bms1Connected, bms2Connected, absConnected);
    if (cEbikeOk) {
      if (!this.vciWrapper.startExtendedSession(Constants.EcuType.VICTOR_DU, 2, 200))
        LOGGER.debug("DU extended session failed! Try gathering SW version anyway ..."); 
      ComplexResponse duSwVerResp = this.communicationManager.getDuCom().getDuVictor().readSoftwareVersion();
      if (duSwVerResp.isPositive()) {
        duSwVer = conv2Version(duSwVerResp.toString());
      } else {
        try {
          if (duSwVerResp.getNegativeResponse() == NegativeResponse.TIMEOUT) {
            LOGGER.debug("DU seems to be sleeping, 'cause BMS is fully charged, so skip DU firmware check!");
            duIsSleeping = true;
          } else {
            LOGGER.warn("DU refused to share SW version, BCS preconditions are not fulfilled!");
            cEbikeOk = false;
          } 
        } catch (ComplexResponse.DataAccessException ex) {
          LOGGER.error("Error accesssing negative firmware version request response from DU '{}'", duSwVerResp, ex);
          cEbikeOk = false;
        } 
      } 
    } 
    ComplexResponse response = this.communicationManager.getDuCom().getDuVictor().readComponentType();
    ComponentType compType = response.isPositive() ? ComponentType.parse(response.toString()) : ComponentType.NONE;
    if (compType.isBDU2xx() || compType.isBDU3xx()) {
      Version minReqDuSwVer = conv2Version(compType.isBDU2xx() ? "1.7.0.1" : "1.0.1.1");
      if (!duIsSleeping && (!cEbikeOk || duSwVer == null || duSwVer.compareTo(minReqDuSwVer) < 0)) {
        LOGGER.warn("BCS preconditions not fulfilled: DU SW Version too low: '{}', should be at least: '{}'", duSwVer, minReqDuSwVer);
        cEbikeOk = false;
      } 
    } 
    if (cEbikeOk) {
      this.vciWrapper.startExtendedSession(Constants.EcuType.VICTOR_PP1, 2, 200);
      bmsSwVer = conv2Version(this.communicationManager.getBms1Com().getBmsVictor().readSoftwareVersion().toString());
    } 
    Version minReqBmsSwVer = conv2Version("1.7.1.1");
    if (!cEbikeOk || bmsSwVer == null || bmsSwVer.compareTo(minReqBmsSwVer) < 0) {
      LOGGER.warn("BCS preconditions not fulfilled: BMS SW Version too low: '{}', should be at least: '{}'", bmsSwVer, minReqBmsSwVer);
      cEbikeOk = false;
    } 
    if (bms2Connected) {
      LOGGER.warn("BCS preconditions not fulfilled: Dual battery system detected!");
      cDualBat = true;
    } 
    return interpretPreConditionCheckResult(cEbikeOk, cDualBat);
  }
  
  private boolean eBikeEcusFblCheck(boolean duConnected, boolean bms1Connected, boolean bms2Connected, boolean absConnected) {

      this.communicationManager.getDuCom().getDuVictor().readEcuState() != Constants.EcuMode.APP) {

    if (bms1Connected && 
      this.communicationManager.getBms1Com().getBmsVictor().readEcuState() != Constants.EcuMode.APP) {
      LOGGER.warn("BMS1's in FBL, BCS diagnose session conditions not given!");
      return false;
    } 
    if (bms2Connected && 
      this.communicationManager.getBms2Com().getBmsVictor().readEcuState() != Constants.EcuMode.APP) {
      LOGGER.warn("BMS2's in FBL, BCS diagnose session conditions not given!");
      return false;
    } 
    if ((absConnected || this.communicationManager.getAbsCom().sendTesterPresent().isPositive()) && 
      this.communicationManager.getAbsCom().readEcuState() != Constants.EcuMode.APP) {
      LOGGER.warn("ABS's in FBL, BCS diagnose session conditions not given!");
      return false;
    } 
    LOGGER.debug("All connected eBike ECUs are in APP");
    return true;
  }

    private final Constants.EcuType connectedHmi;
    
    ChargingStateThread(Constants.EcuType connectedHmi) {
      this.connectedHmi = connectedHmi;
    }
    
    public void run() {
      try {
        synchronized (this) {
          while (true) {
            wait((long)ChargingStateHandler.ITERATION_WAITING_TIME);
            if (!ChargingStateHandler.this.vciWrapper.sendTesterPresent(this.connectedHmi))
              return; 
          } 
        } 
      } catch (InterruptedException e) {
        ChargingStateHandler.LOGGER.error("ChargingStateThread: Interrupted -> {}", e.getMessage(), e);

      } 
    }
  }

    this.vciWrapper = vciWrapper;
  }
  
  synchronized void startChargingStatePolling(ChargingStateEvent.ChargingMessage message, Constants.EcuType hmiPresentType) {
    ChargingStateEvent event = new ChargingStateEvent(message);
    event.startCharging();
    publishChargingStateEvent(event);
    waitForHmiDisconnect(hmiPresentType);
    event.stopCharging();
    publishChargingStateEvent(event);
  }
  
  void waitForHmiDisconnect(Constants.EcuType hmiPresentType) {
    Thread hmiPollingThread = new Thread(new ChargingStateThread(hmiPresentType));
    hmiPollingThread.start();
    try {
      hmiPollingThread.join();
    } catch (InterruptedException e) {
      LOGGER.error("waitForHmiDisconnect(): Error while calling hmiPollingThread.join() -> ", (Throwable)e);
      Thread.currentThread().interrupt();
    } 
  }
  
  synchronized void subscribeChargingStateChanges(EcuConnectionStateHandler.ChargingStateListener listener) {
    if (!this.chargingStateCallbacks.contains(listener))
      this.chargingStateCallbacks.add(listener); 
  }
  
  synchronized void unsubscribeChargingStateChanges(EcuConnectionStateHandler.ChargingStateListener listener) {
    this.chargingStateCallbacks.remove(listener);
  }
  
  private void publishChargingStateEvent(ChargingStateEvent event) {
    this.chargingStateCallbacks.forEach(listener -> listener.onConnectedBikeChargingStateChanges(event));
  }
}

  EcuStateThread(CommunicationManager communicationManager, RightsManager rightsManager) {
    this.communicationManager = communicationManager;
    this.rightsManager = rightsManager;
    this.securityAccessFailurePopupHandler = SecurityAccessFailurePopupHandlerFactory.createInstance(rightsManager);
    this.vciWrapper = new VciWrapper(rightsManager, communicationManager);
    this.bctEventPublisher = new BctEventPublisher();
    this.bctWrapper = new BctWrapper(this.bctEventPublisher, communicationManager);
    this.absWrapper = new AbsWrapper(communicationManager.getAbsCom(), rightsManager, this.securityAccessFailurePopupHandler);
    this.hidDeviceConnectionController = new HidDeviceConnectionController(this);
    this.pollingStarter = new PollingStarter(this, this.hidDeviceConnectionController);
    this
      .suspensionManager = new SuspensionManager(Executors.newSingleThreadExecutor(new BasicThreadFactory.Builder().namingPattern("ResumeEcuConnectionStateDetectionExecutor").build()), this);
    this.initializationDialogController = new InitializationDialogController();
    this.baudrateChecker = new BaudrateChecker(communicationManager, rightsManager);
    this.batteryBootloaderChecker = new BatteryBootloaderChecker(communicationManager);
    this.chargingStateBcsChecker = new ChargingStateBcsChecker(communicationManager, this.vciWrapper, rightsManager);
    this.chargingStateHandler = new ChargingStateHandler(this.vciWrapper);
    this.systemTypeDetectionFailureHandler = new SystemTypeDetectionFailureHandler(communicationManager, this.suspensionManager);
    this.hidDeviceDisconnectionDetector = new HidDeviceDisconnectionDetector(this.hidDeviceConnectionController, this);
    this.publisher = new EcuConnectionStatePublisher();
    rightsManager.addListener(this);
  }
  
  public void startConnectionStateDetection() {
    this.pollingStarter.startEcuPolling();
  }
  
  public void suspendConnectionStateDetection(boolean ignoreUsbEvents) {
    this.suspensionManager.controlSuspension(ignoreUsbEvents);
  }
  
  public void resumeConnectionStateDetection(boolean reinitializeHardware, boolean restoreBcsSession) {
    this.suspensionManager.controlResumption(reinitializeHardware, restoreBcsSession);
  }
  
  public boolean isConnectionStateDetectionSuspended() {
    return this.suspensionManager.isSuspensionExecuted();
  }
  
  public synchronized void onHidDeviceConnect(Constants.HidDevice device) {
    LOGGER.debug("Start connecting HID-Device: '{}', connected device is: '{}'", device, this.connectedDevice);
    if (this.rightsManager.isUserInLevel(UserLevel.UNKNOWN)) {
      LOGGER.error("Unknown user level: System initialization is prevented");
      return;
    } 
    if (this.connectedDevice == Constants.HidDevice.NONE && device != Constants.HidDevice.NONE && this.suspensionManager.isHidEventProcessingAllowed()) {
      initGatewayConnectionStateFlags(device);
      switch (this.hmiPresentType) {
        case BUI21X:
        case BUI25X:
        case BUI270:
        case BUI275:
        case BUI330:
        case BUI350:
        case COBI:
          if (initializeHmiAndFireEcuConnectCallback(device)) {
            checkBikeState(true);
            break;
          } 
          resetGatewayConnectionStateFlags();
          return;
        case BCT:
          if (initializeBctAndFireEcuConnectCallback(device)) {
            updateBctConnectionState();
            fireBctSystemConnectCallback();
            break;
          } 
          resetGatewayConnectionStateFlags();
          return;
      } 
      this.initializationDialogController.closeDialog();
      processInitializationResult(device);
    } 
    LOGGER.debug("Finish connecting HID-Device: '{}', connected device is: '{}', interruption state is: '{}'", device, this.connectedDevice, isInterrupted());
  }
  
  private void processInitializationResult(Constants.HidDevice device) {
    if (this.systemTypeDetectionFailureHandler.isDetectionFailed() && !this.systemTypeDetectionFailureHandler.isContinuationInHmiStandAloneModeOk()) {
      LOGGER.warn("Detected HMI disconnection during system type detection procedure!");
      resetConnectionStatesAndFireEcuDisconnectCallbacks(true, false);
    } else {
      this.connectedDevice = device;
    } 
  }
  
  public boolean shouldHidEventsBeIgnored() {
    return !this.suspensionManager.isHidEventProcessingAllowed();
  }
  
  private void initGatewayConnectionStateFlags(Constants.HidDevice device) {
    this.hmiPresentType = ConfigurationHelper.getEcuTypeFromDevice(device);
    this.hmiPresentMode = ConfigurationHelper.getEcuModeFromDevice(device);
    ((HmiImpl)this.communicationManager.getHmiCom()).configureInstance(this.hmiPresentType);
  }
  
  private void resetGatewayConnectionStateFlags() {
    LOGGER.warn("Stopped initialization of HMI / BCT! Device connection event aboarded!");
    this.hmiPresentType = Constants.EcuType.NONE;
    this.hmiPresentMode = Constants.EcuMode.NONE;
  }
  
  public synchronized void onHidDeviceDisconnect(Constants.HidDevice device) {
    LOGGER.debug("Start disconnecting HID-Device: '{}', connected device is: '{}'", device, this.connectedDevice);
    if (this.connectedDevice == device && this.suspensionManager.isHidEventProcessingAllowed()) {
      determineAndFireSystemDisconnectionEvent();
      if (this.hmiPresentType == Constants.EcuType.BCT)
        this.bctWrapper.publishStateChangeEvent(BctState.NOT_CONNECTED); 
      resetConnectionStatesAndFireEcuDisconnectCallbacks(true, false);
    } 
    LOGGER.debug("Finish disconnecting HID-Device: '{}', connected device is: '{}'", device, this.connectedDevice);
  }
  
  private void determineAndFireSystemDisconnectionEvent() {
    if (this.systemConnectionCallbackFired) {
      if (this.hmiPresentType == Constants.EcuType.BCT) {
        fireSystemConnectionStateChangeCallback(SystemConnectionStateChangeEvent.InitAction.BCT_DISCONNECT);
      } else {
        fireSystemConnectionStateChangeCallback(SystemConnectionStateChangeEvent.InitAction.BIKE_DISCONNECT);
      } 
      this.systemConnectionCallbackFired = false;
    } 
  }
  
  private void resetLocalEcuConnectionStateFlags(boolean kickHmiAsWell, boolean stayInBcsMode) {
    if (kickHmiAsWell) {
      this.hmiPresentType = Constants.EcuType.NONE;
      this.hmiPresentMode = Constants.EcuMode.NONE;
      this.connectedDevice = Constants.HidDevice.NONE;
      this.hidDeviceDisconnectionDetector.setHidDevice(Constants.HidDevice.NONE);
    } 
    this.duPresentType = Constants.EcuType.NONE;
    this.duPreviousType = Constants.EcuType.NONE;
    this.duPresentMode = Constants.EcuMode.NONE;
    this.bms1PresentType = Constants.EcuType.NONE;
    this.bms1PreviousType = Constants.EcuType.NONE;
    this.bms1PresentMode = Constants.EcuMode.NONE;
    this.bms2PresentType = Constants.EcuType.NONE;
    this.bms2PreviousType = Constants.EcuType.NONE;
    this.bms2PresentMode = Constants.EcuMode.NONE;
    this.absPresentType = Constants.EcuType.NONE;
    this.absWrapper.resetState();
    this.bcsPresentType = Constants.EcuType.NONE;
    this.bctWrapper.resetState();
    this.systemType = Constants.SystemType.NONE;
    this.systemTypeDetectionFailureHandler.onDetectionFinished();
    this.bikeConnected = false;
    this.chargingStateBcsChecker.onResetCommunicationStates(stayInBcsMode);
  }
  
  private void fireEcuDisconnectCallbacks(boolean kickHmiAsWell) {
    if (this.hmiPresentType != Constants.EcuType.NONE && kickHmiAsWell)
      fireEcuDisconnectCallback(this.hmiPresentType); 
    if (this.duPresentType != Constants.EcuType.NONE)
      fireEcuDisconnectCallback(this.duPresentType); 
    if (this.bms1PresentType != Constants.EcuType.NONE)
      fireEcuDisconnectCallback(this.bms1PresentType); 
    if (this.bms2PresentType != Constants.EcuType.NONE)
      fireEcuDisconnectCallback(this.bms2PresentType); 
    if (this.absPresentType != Constants.EcuType.NONE)
      fireEcuDisconnectCallback(this.absPresentType); 
    if (this.bcsPresentType != Constants.EcuType.NONE)
      fireEcuDisconnectCallback(this.bcsPresentType); 
  }
  
  public synchronized void resetConnectionStatesAndFireEcuDisconnectCallbacks(boolean kickHmiAsWell, boolean stayInBcsMode) {
    LOGGER.debug("Reset (kickHmiAsWell='{}', stayInBcsMode='{}')! Currently connected are: HMI='{}', DU='{}' , BMS1='{}', BMS2='{}', ABS='{}', BCS='{}'", kickHmiAsWell, stayInBcsMode, this.hmiPresentType, this.duPresentType, this.bms1PresentType, this.bms2PresentType, this.absPresentType, this.bcsPresentType);
    fireEcuDisconnectCallbacks(kickHmiAsWell);
    resetLocalEcuConnectionStateFlags(kickHmiAsWell, stayInBcsMode);
  }
  
  private synchronized void checkSystemState() {
    switch (this.hmiPresentType) {
      case BUI21X:
      case BUI25X:
      case BUI270:
      case BUI275:
      case BUI330:
      case BUI350:
      case COBI:
        checkBikeState(false);
        break;
      case BCT:
        updateBctConnectionState();
        break;
    } 
    this.initializationDialogController.closeDialog();
  }
  
  private void onDriveUnitInitialized() {
    if (this.hmiPresentMode != Constants.EcuMode.NONE) {
      String presentDuSerial;
      if (this.hmiPresentMode == Constants.EcuMode.APP && this.duPresentMode == Constants.EcuMode.APP) {
        presentDuSerial = this.vciWrapper.readSerialNumberDu(this.duPresentType);
        LOGGER.debug("DU serial number => '{}'", presentDuSerial);
      } else {
        presentDuSerial = "";
      } 
      this.initializationDialogController.closeDialog();
      if (StringUtils.isNotBlank(presentDuSerial) && StringUtils.equals(presentDuSerial, this.previousDuSerial)) {
        fireSystemConnectionStateChangeCallback(SystemConnectionStateChangeEvent.InitAction.BIKE_RECONNECT);
      } else {
        fireSystemConnectionStateChangeCallback(SystemConnectionStateChangeEvent.InitAction.BIKE_CONNECT);
        this.previousDuSerial = presentDuSerial;
      } 
      this.systemConnectionCallbackFired = true;
    } 
  }
  
  private void onHmiStandaloneInitialized() {
    if (this.hmiPresentType != Constants.EcuType.NONE && this.duPresentType == Constants.EcuType.NONE && this.bms1PresentType == Constants.EcuType.NONE && this.bms2PresentType == Constants.EcuType.NONE) {
      this.initializationDialogController.closeDialog();
      fireSystemConnectionStateChangeCallback(SystemConnectionStateChangeEvent.InitAction.HMI_STANDALONE_CONNECT);
      this.systemConnectionCallbackFired = true;
    } 
  }
  
  private void fireBctSystemConnectCallback() {
    fireSystemConnectionStateChangeCallback(SystemConnectionStateChangeEvent.InitAction.BCT_CONNECT);
    this.systemConnectionCallbackFired = true;
  }
  
  public void run() {
    while (true) {
      if (this.connectedDevice != Constants.HidDevice.NONE && this.suspensionManager.isPollingAllowed())
        checkSystemState(); 
      this.suspensionManager.suspendIfRequested();
      this.suspensionManager.waitForNextPollingCycle(500);
    } 
  }
  
  private void fireEcuConnectCallback(Constants.EcuType type, Constants.EcuMode mode) {
    this.communicationManager.setEcuType(type, mode);
    EcuConnectionStateChangedEvent event = EcuConnectionStateChangedEvent.createEcuConnectEvent(type, mode);
    this.publisher.publishEcuConnectionStateChangedEvent(event);
  }
  
  private void fireEcuDisconnectCallback(Constants.EcuType type) {
    this.communicationManager.setEcuType(type, Constants.EcuMode.NONE);
    EcuConnectionStateChangedEvent event = EcuConnectionStateChangedEvent.createEcuDisconnectEvent(type);
    this.publisher.publishEcuConnectionStateChangedEvent(event);
  }
  
  private void fireSystemConnectionStateChangeCallback(SystemConnectionStateChangeEvent.InitAction action) {
    this.publisher.publishSystemConnectionStateChangeEvent(new SystemConnectionStateChangeEvent(action));
  }
  
  private boolean checkOnboardState() {
    boolean onboardState = getHmiSocketState();
    if (onboardState != this.bikeConnected && !isInterrupted()) {
      if (onboardState) {
        onBuiConnectToBike();
      } else {
        onBuiDisconnectFromBike();
      } 
      this.bikeConnected = onboardState;
    } 
    return this.bikeConnected;
  }
  
  private boolean getHmiSocketState() {
    Hmi.OnboardState onboardResponse = readHmiOnboardState();
    if (onboardResponse.isNegative() && this.bikeConnected) {
      LOGGER.debug("Retry reading the HMI socket state after a negative response, but wait '{}' before retry", Integer.valueOf(500));
      sleep(500);
      onboardResponse = readHmiOnboardState();
    } 
    boolean onboardState = (onboardResponse == Hmi.OnboardState.ONBOARD);
    LOGGER.debug("HMI <-> socket connection state: '{}'", Boolean.valueOf(onboardState));
    return onboardState;
  }
  
  private Hmi.OnboardState readHmiOnboardState() {
    Hmi.OnboardState onboardResponse = this.communicationManager.getHmiCom().readOnboardCondition();
    feedHidDeviceDisconnectionDetector((onboardResponse == Hmi.OnboardState.TIMEOUT));
    return onboardResponse;
  }
  
  private void feedHidDeviceDisconnectionDetector(boolean timoutOccured) {
    if (timoutOccured) {
      this.hidDeviceDisconnectionDetector.incrementTimeoutCounter();
    } else {
      this.hidDeviceDisconnectionDetector.clearTimeoutCounter();
    } 
  }
  
  private void onBuiConnectToBike() {
    this.initializationDialogController.openDialog();
    this.systemType = detectSystemTypeWithFailureHandling();
    if (this.bcsPresentType == Constants.EcuType.NONE)
      checkBatteryBootloaderState(); 
  }
  
  private void checkBatteryBootloaderState() {
    if (this.batteryBootloaderChecker.checkBootloaderState(this.systemType, this.baudrateChecker.bms1HasBeenDetectedDuringLastBaudrateCheck(), this.baudrateChecker.bms2HasBeenDetectedDuringLastBaudrateCheck())) {
      this.initializationDialogController.closeDialog();
      this.batteryBootloaderChecker.showBatteryBootloaderWarning();
    } 
  }
  
  private Constants.SystemType detectSystemTypeWithFailureHandling() {
    Constants.SystemType detectedSystemType;
    boolean hmiIsConnected = true;
    do {
      detectedSystemType = detectSystemType();
      if (detectedSystemType != Constants.SystemType.NONE) {
        this.systemTypeDetectionFailureHandler.onDetectionFinished();
      } else if (!isInterrupted()) {
        hmiIsConnected = this.systemTypeDetectionFailureHandler.onDetectionFailure();
      } 
    } while (detectedSystemType == Constants.SystemType.NONE && hmiIsConnected && this.systemTypeDetectionFailureHandler.isFurtherRetryAllowed() && !isInterrupted());
    return detectedSystemType;
  }
  
  private Constants.SystemType detectSystemType() {
    Constants.SystemType detectedSystemType = Constants.SystemType.NONE;
    for (int n = 0; n < 3 && !isInterrupted(); n++) {
      LOGGER.debug("Configuring CAN baudrate, try no. '{}'", Integer.valueOf(n + 1));
      detectedSystemType = this.baudrateChecker.checkBaudrate((this.hmiPresentType == Constants.EcuType.BUI25X));
      if (detectedSystemType != Constants.SystemType.NONE)
        break; 
      sleep(1000);
    } 
    return detectedSystemType;
  }
  
  private void onBuiDisconnectFromBike() {
    fireEcuDisconnectCallbacks(false);
    resetLocalEcuConnectionStateFlags(false, false);
  }
  
  private void handleChargingState(ChargingStateEvent.ChargingMessage message) {
    LOGGER.debug("Display charging window with text: '{}'", message);
    this.vciWrapper.startDefaultSession(this.hmiPresentType, false);
    this.vciWrapper.startDefaultSession(this.duPresentType, false);
    this.vciWrapper.startDefaultSession(this.bms1PresentType, false);
    this.vciWrapper.startDefaultSession(this.bms2PresentType, false);
    this.vciWrapper.startDefaultSession(this.absPresentType, false);
    this.chargingStateHandler.startChargingStatePolling(message, this.hmiPresentType);
  }
  
  private void checkBikeState(boolean initialPollingRun) {
    boolean hmiHasBeenRemoved = false;
    if (this.hmiPresentMode == Constants.EcuMode.APP && checkOnboardState()) {
      ChargingCheckResult chargingCheckResult = this.chargingStateBcsChecker.checkChargingStateAndBcsExistence(this.hmiPresentType, this.hmiPresentMode, (this.duPresentType != Constants.EcuType.NONE), (this.bms1PresentType != Constants.EcuType.NONE), (this.bms2PresentType != Constants.EcuType.NONE), (this.absPresentType != Constants.EcuType.NONE), (this.bcsPresentType != Constants.EcuType.NONE));
      processChargingCheckResult(chargingCheckResult);
      if (chargingCheckResult.isEbikeCharging())
        return; 
      updateBms1ConnectionState();
      updateBms2ConnectionState();
      updateDuConnectionState();
      updateAbsConnectionState();
      hmiHasBeenRemoved = updateBcsConnectionState();
    } 
    if (isSystemConnectCallbackFiringAllowed() && !hmiHasBeenRemoved)
      determineAndFireSystemConnectionStateChangeEventIfNecessary(initialPollingRun); 
    this.duPreviousType = this.duPresentType;
    this.bms1PreviousType = this.bms1PresentType;
    this.bms2PreviousType = this.bms2PresentType;
  }
  
  private boolean isSystemConnectCallbackFiringAllowed() {
    return ((!this.systemTypeDetectionFailureHandler.isDetectionFailed() || 
      this.systemTypeDetectionFailureHandler.isContinuationInHmiStandAloneModeOk()) && 
      !isInterrupted());
  }
  
  private void determineAndFireSystemConnectionStateChangeEventIfNecessary(boolean initialPolllingRun) {
    if (this.duPresentType != Constants.EcuType.NONE && this.duPreviousType == Constants.EcuType.NONE) {
      onDriveUnitInitialized();
    } else if ((this.bms1PresentType != Constants.EcuType.NONE || this.bms2PresentType != Constants.EcuType.NONE) && this.bms1PreviousType == Constants.EcuType.NONE && this.bms2PreviousType == Constants.EcuType.NONE) {
      fireSystemConnectionStateChangeCallback(SystemConnectionStateChangeEvent.InitAction.BIKE_CONNECT);
    } else if (initialPolllingRun) {
      onHmiStandaloneInitialized();
    } 
  }
  
  private void updateBctConnectionState() {
    if (this.hmiPresentMode == Constants.EcuMode.APP && this.bctWrapper.checkOperationState() == BctState.NOT_CONNECTED) {
      LOGGER.warn("BCT's not responding on the operation state request!");
      this.bctStatusResponseFailureCounter++;
      if (this.bctStatusResponseFailureCounter == 10) {
        resetConnectionStatesAndFireEcuDisconnectCallbacks(true, false);
        this.bctStatusResponseFailureCounter = 0;
      } 
    } else {
      this.bctStatusResponseFailureCounter = 0;
    } 
  }
  
  private void updateDuConnectionState() {
    if ((this.bms1PresentMode == Constants.EcuMode.APP || this.bms2PresentMode == Constants.EcuMode.APP || this.duPreviousType != Constants.EcuType.NONE) && !isInterrupted())
      if (this.systemType == Constants.SystemType.VICTOR) {
        this.duPresentType = checkEcuState(Constants.EcuType.VICTOR_DU, (this.duPreviousType == Constants.EcuType.NONE)) ? Constants.EcuType.VICTOR_DU : Constants.EcuType.NONE;
      } else if (this.systemType == Constants.SystemType.CLASSIC_PLUS) {
        this.duPresentType = checkEcuState(Constants.EcuType.CLASSIC_DU, (this.duPreviousType == Constants.EcuType.NONE)) ? Constants.EcuType.CLASSIC_DU : Constants.EcuType.NONE;
      }  
  }
  
  private void updateBms1ConnectionState() {
    if (this.systemType == Constants.SystemType.VICTOR && !isInterrupted()) {
      this.bms1PresentType = checkEcuState(Constants.EcuType.VICTOR_PP1, (this.bms1PreviousType == Constants.EcuType.NONE)) ? Constants.EcuType.VICTOR_PP1 : Constants.EcuType.NONE;
    } else if (this.systemType == Constants.SystemType.CLASSIC_PLUS && !isInterrupted()) {
      this.bms1PresentType = checkEcuState(Constants.EcuType.CLASSIC_PP, (this.bms1PreviousType == Constants.EcuType.NONE)) ? Constants.EcuType.CLASSIC_PP : Constants.EcuType.NONE;
    } 
  }
  
  private void updateBms2ConnectionState() {
    if (this.systemType == Constants.SystemType.VICTOR && this.bcsPresentType == Constants.EcuType.NONE && !isInterrupted())
      this.bms2PresentType = checkEcuState(Constants.EcuType.VICTOR_PP2, (this.bms2PreviousType == Constants.EcuType.NONE)) ? Constants.EcuType.VICTOR_PP2 : Constants.EcuType.NONE; 
  }
  
  private void updateAbsConnectionState() {
    Constants.EcuMode mode;
    if (((this.bms1PresentType == Constants.EcuType.VICTOR_PP1 && this.bms1PresentMode == Constants.EcuMode.APP) || (this.bms2PresentType == Constants.EcuType.VICTOR_PP2 && this.bms2PresentMode == Constants.EcuMode.APP)) && 
      
      !isInterrupted()) {
      mode = this.absWrapper.checkState();
    } else {
      this.absWrapper.resetState();
      mode = Constants.EcuMode.NONE;
    } 
    if ((mode == Constants.EcuMode.APP || mode == Constants.EcuMode.FBL) && this.absPresentType == Constants.EcuType.NONE) {
      this.absPresentType = Constants.EcuType.BAS100;
      fireEcuConnectCallback(this.absPresentType, mode);
    } else if (mode == Constants.EcuMode.NONE && this.absPresentType == Constants.EcuType.BAS100) {
      fireEcuDisconnectCallback(this.absPresentType);
      this.absPresentType = Constants.EcuType.NONE;
    } 
  }
  
  private boolean updateBcsConnectionState() {
    if (this.bcsPresentType != Constants.EcuType.NONE && this.rightsManager.isUserInLevel(UserLevel.IBD) && !this.vciWrapper.pingEcu(Constants.EcuType.BCS250, false)) {
      LOGGER.debug("Initiate disconnection procedure of BCS");
      fireEcuConnectionStateRefreshCallback(new EcuConnectionStateRefreshEvent(this.bcsPresentType, Constants.EcuMode.NONE));
      JDialog dialog = openBlockingDialog();
      this.chargingStateHandler.waitForHmiDisconnect(this.hmiPresentType);
      dialog.dispose();
      return true;
    } 
    return false;
  }
  
  private boolean checkEcuState(Constants.EcuType ecu, boolean ecuNotYetConnected) {
    Constants.EcuMode mode = Constants.EcuMode.NONE;
    boolean ecuIsConnected = this.vciWrapper.pingEcu(ecu, ecuNotYetConnected);
    if (ecuIsConnected && ecuNotYetConnected) {
      if (this.rightsManager.isUserInLevel(UserLevel.UNKNOWN)) {
        LOGGER.info("Prevented initialization of new detected device, 'cause of missing dongle.");
        return false;
      } 
      LOGGER.info("Detected new ECU '{}'", ecu);
      mode = determineEcuModeAndExecuteSecurityAccess(ecu);
      if (mode == Constants.EcuMode.NONE) {
        LOGGER.warn("Drop ECU detection on initialization failure.");
        return false;
      }

class HidDeviceConnectionController implements ApplicationProcessStateChangesListener {
  private static final Logger LOGGER = LoggerFactory.getLogger(HidDeviceConnectionController.class);
  
  private final ScheduledExecutorService deviceStateChangeReporter = Executors.newSingleThreadScheduledExecutor(new BasicThreadFactory.Builder().namingPattern("deviceStateChangeReporter-%d").build());
  
  private volatile Constants.HidDevice connectedDevice = Constants.HidDevice.NONE;
  
  private volatile Constants.HidDevice newDevice = Constants.HidDevice.NONE;
  
  private ScheduledFuture connectionJobFuture;
  
  private UserLevel currentUserLevel = UserLevel.UNKNOWN;
  
  private boolean blockingTaskIsCurrentlyExecuted = false;
  
  private boolean guiIsInitialized = false;
  
  private final EcuStateFacade ecuStateFacade;
  
  HidDeviceConnectionController(EcuStateFacade ecuStateFacade) {
    this.ecuStateFacade = ecuStateFacade;
    ApplicationProcessManager.getInstance().addListener(this);
  }
  
  synchronized void onHidDeviceConnectionChanged(HidDeviceConnectionChangedEvent event) {
    if (event.getVendorId() == 4236) {
      Constants.HidDevice device = Constants.HidDevice.fromPid(event.getProductId());
      if (device != null)
        if (event.getState()) {
          onDeviceConnected(device);
        } else {
          onDeviceDisconnected(device);
        }  
    } 
  }
  
  private void onDeviceConnected(Constants.HidDevice device) {
    LOGGER.info("HID device {} [PID_{},VID_{}] connected", device, String.format("%x", device.getPid()), String.format("%x", device.getVid()));
    this.newDevice = device;
    if (connectConditions())
      connect(this.newDevice); 
  }
  
  private void connect(Constants.HidDevice device) {
    LOGGER.debug("Initialize now new connected device '{}'", device);
    this.connectedDevice = device;
    distributeConnectionEvent(device);
    this.newDevice = Constants.HidDevice.NONE;
  }
  
  private void onDeviceDisconnected(Constants.HidDevice device) {
    LOGGER.info("HID device {} [PID_{},VID_{}] disconnected", device, String.format("%x", device.getPid()), String.format("%x", device.getVid()));
    removeUninitializedDeviceIfNecessary(device);
    removeInitializedDeviceIfNecessary(device);
  }
  
  private void removeUninitializedDeviceIfNecessary(Constants.HidDevice device) {
    if (device == this.newDevice) {
      LOGGER.debug("Remove '{}' from connection queue", device);
      this.newDevice = Constants.HidDevice.NONE;
    } 
  }
  
  private void removeInitializedDeviceIfNecessary(Constants.HidDevice device) {
    if (device == this.connectedDevice) {
      LOGGER.debug("Remove connected device '{}'", device);
      distributeDisconnectionEvent(device);
      this.connectedDevice = Constants.HidDevice.NONE;
      this.connectionJobFuture = null;
    } 
  }
  
  private void distributeConnectionEvent(Constants.HidDevice device) {
    if (!this.ecuStateFacade.shouldHidEventsBeIgnored()) {
      this.connectionJobFuture = this.deviceStateChangeReporter.schedule(() -> this.ecuStateFacade.onHidDeviceConnect(device), 0L, TimeUnit.MILLISECONDS);
    } else {
      LOGGER.warn("HID connection event is not passed to the EcuStateThread, 'cause HID events are ignored at the moment");
    } 
  }
  
  private void distributeDisconnectionEvent(Constants.HidDevice device) {
    if (!this.ecuStateFacade.shouldHidEventsBeIgnored()) {
      interruptRunningConnectionJob();
      this.deviceStateChangeReporter.execute(() -> this.ecuStateFacade.onHidDeviceDisconnect(device));
    } else {
      LOGGER.warn("HID connection event is not passed to the EcuStateThread, 'cause HID events are ignored at the moment");
    } 
  }
  
  private void interruptRunningConnectionJob() {
    if (this.connectionJobFuture != null && !this.connectionJobFuture.isDone()) {
      LOGGER.info("Interrupt currently running ECU initialization / connection job!");
      this.connectionJobFuture.cancel(true);
    } 
  }
  
  synchronized void onUserLevelChanged(UserLevel newUserLevel) {
    this.currentUserLevel = newUserLevel;
    LOGGER.debug("User level changed to '{}'! Prove if there's a device that should be connected now ...", this.currentUserLevel);
    if (connectConditions())
      connect(this.newDevice); 
  }
  
  synchronized void onEcuPollingStarted() {
    this.guiIsInitialized = true;
    LOGGER.debug("Main-Thread / GUI / application are now initialized! Prove if there's a device that should be connected now ...");
    if (connectConditions())
      connect(this.newDevice); 
  }
  
  public synchronized void onApplicationProcessStateChanged(ApplicationProcessStateChangedEvent event) {
    LOGGER.debug("ApplicationProcessStateChangedEvent received: '{}'", event);
    boolean currentState = event.getApplicationProcessManager().isProcessActive(ApplicationProcessManager.ApplicationProcess.READ_SERVICE_PARAMS);
    if (currentState != this.blockingTaskIsCurrentlyExecuted) {
      this.blockingTaskIsCurrentlyExecuted = currentState;
      LOGGER.debug("Blocking application state changed to '{}'! Prove if there's a device that should be connected now ...", Boolean.valueOf(this.blockingTaskIsCurrentlyExecuted));
      if (connectConditions())
        connect(this.newDevice); 
    } 
  }
  
  private boolean connectConditions() {
    boolean noDeviceIsInitialized = (this.connectedDevice == Constants.HidDevice.NONE);
    boolean uninitializedDeviceIsAvailable = (this.newDevice != Constants.HidDevice.NONE);
    boolean userLevelIsKnown = (this.currentUserLevel != UserLevel.UNKNOWN);
    boolean applicationIsInitialized = this.guiIsInitialized;
    boolean initializationOfNewDeviceIsAllowed = !this.blockingTaskIsCurrentlyExecuted;
    LOGGER.debug("connectConditions(): noDeviceIsInitialized='{}', uninitializedDeviceIsAvailable='{}', userLevelIsKnown='{}', applicationIsInitialized='{}', initializationOfNewDeviceIsAllowed='{}'", noDeviceIsInitialized, uninitializedDeviceIsAvailable, userLevelIsKnown, applicationIsInitialized, initializationOfNewDeviceIsAllowed);
    return (noDeviceIsInitialized && uninitializedDeviceIsAvailable && userLevelIsKnown && applicationIsInitialized && initializationOfNewDeviceIsAllowed);
  }
}

    LOGGER.debug("(Re)start ECU connection state change detection with resetConnectionStates='{}', restoreBcsSession='{}'", Boolean.valueOf(resetConnectionStates), Boolean.valueOf(restoreBcsSession));
    if (resetConnectionStates)
      this.ecuStateFacade.resetConnectionStatesAndFireEcuDisconnectCallbacks(true, restoreBcsSession); 
    this.suspendState.set(SuspensionState.NOT_SUSPENDED);
    if (resetConnectionStates)
      this.resumeEcuConnectionStateDetectionExecutor.execute(() -> {
            LOGGER.trace("Asynchronous reinitialization of connected ECUs job started");
            reInitializeConnectedEcus();
          }); 
  }
  
  private void reInitializeConnectedEcus() {
    Constants.HidDevice device = BoschServer.getInstance().getDevice();
    if (device != Constants.HidDevice.NONE) {
      LOGGER.info("Simulate '{}' has been physically re-connected and restart ECU initialization", device);
      this.ecuStateFacade.onHidDeviceConnect(device);
    } else {
      LOGGER.warn("Reconnection simulation of ECUs not possible, 'cause connected HID device is '{}'", device);
    } 
  }
  
  void controlSuspension(boolean ignoreUsbEvents) {
    LOGGER.info("controlSuspension(ignoreUsbEvents='{}') called! '{}' thread(s) ordered suspension as well.", Boolean.valueOf(ignoreUsbEvents), this.ecuPollingStopCounter);
    this.ecuPollingStopCounter.incrementAndGet();
    if (isWaitingConditionFulfilled(ignoreUsbEvents)) {
      LOGGER.debug("ECU state thread has been ordered to stop polling. Wait for request execution.");
      waitForSuspensionExecution(this.suspensionExecutionTimeout);
    } 
  }
  
  private synchronized boolean isWaitingConditionFulfilled(boolean ignoreUsbEvents) {
    if (this.suspendState.compareAndSet(SuspensionState.NOT_SUSPENDED, predictSuspensionOrderState(ignoreUsbEvents)))
      return true; 
    if (isSuspensionOrdered()) {
      if (ignoreUsbEvents)
        this.suspendState.compareAndSet(SuspensionState.SUSPENSION_ORDERED_CARE_ABOUT_USB_EVENTS, SuspensionState.SUSPENSION_ORDERED_IGNORE_USB_EVENTS); 
      return true;
    } 
    if (ignoreUsbEvents)
      this.suspendState.compareAndSet(SuspensionState.SUSPENSION_EXECUTED_CARE_ABOUT_USB_EVENTS, SuspensionState.SUSPENSION_EXECUTED_IGNORE_USB_EVENTS); 
    return false;
  }
  
  private SuspensionState predictSuspensionOrderState(boolean ignoreUsbEvents) {
    return ignoreUsbEvents ? SuspensionState.SUSPENSION_ORDERED_IGNORE_USB_EVENTS : SuspensionState.SUSPENSION_ORDERED_CARE_ABOUT_USB_EVENTS;
  }
  
  private void waitForSuspensionExecution(int timeout) {
    long startTime = System.currentTimeMillis();
    synchronized (this.syncObjectSleepInterruption) {
      this.syncObjectSleepInterruption.notify();
    } 
    synchronized (this.syncObjectSuspensionStatus) {
      try {
        this.syncObjectSuspensionStatus.wait((long)timeout);
      } catch (InterruptedException ex) {
        Thread.currentThread().interrupt();
        LOGGER.warn("Waiting for suspension of the ECU state polling thread has been interrupted", (Throwable)ex);
      } 
    } 
    long passedTime = System.currentTimeMillis() - startTime;
    LOGGER.debug("Waiting for suspension execution finished after '{}' ms", Long.valueOf(passedTime));
  }
  
  void suspendIfRequested() {
    if (this.suspendState.compareAndSet(SuspensionState.SUSPENSION_ORDERED_CARE_ABOUT_USB_EVENTS, SuspensionState.SUSPENSION_EXECUTED_CARE_ABOUT_USB_EVENTS) || 
      this.suspendState.compareAndSet(SuspensionState.SUSPENSION_ORDERED_IGNORE_USB_EVENTS, SuspensionState.SUSPENSION_EXECUTED_IGNORE_USB_EVENTS))
      synchronized (this.syncObjectSuspensionStatus) {
        this.syncObjectSuspensionStatus.notifyAll();
      }  
  }
  
  void waitForNextPollingCycle(int waitingTimeBetweenPollingCycles) {
    synchronized (this.syncObjectSleepInterruption) {
      try {
        this.syncObjectSleepInterruption.wait((long)waitingTimeBetweenPollingCycles);
      } catch (InterruptedException ex) {
        Thread.currentThread().interrupt();
        LOGGER.warn("EcuStateThread sleeping has been interrupted", (Throwable)ex);
      } 
    } 
  }
}

  }

    return this.systemTypeDetectionFailed;
  }
  
  boolean isContinuationInHmiStandAloneModeOk() {
    return this.preceedInHmiStandAloneMode;
  }
  
  synchronized void onDetectionFinished() {
    this.systemTypeDetectionFailed = false;
    this.preceedInHmiStandAloneMode = false;
    this.systemTypeDetectionTries = 0;
  }
  
  synchronized boolean onDetectionFailure() {
    LOGGER.warn("Intuvia's on socket, but system type could not be determined, retry with hard reset! (Try: '{}' of '{}')", Integer.valueOf(this.systemTypeDetectionTries + 1), Integer.valueOf(3));
    this.systemTypeDetectionFailed = true;
    return executeSynchronousHardResetOnHmi();
  }
  
  private boolean executeSynchronousHardResetOnHmi() {
    this.suspensionManager.controlSuspension(true);
    this.communicationManager.getHmiCom().executeHardReset(true);
    this.suspensionManager.controlResumption(false, false);
    return this.communicationManager.startHmiDiagnosticSession();
  }
  
  synchronized boolean isFurtherRetryAllowed() {
    if (++this.systemTypeDetectionTries < 3)
      return true; 
    LOGGER.warn("Finally give up detecting system type, work with Intuvia in standalone mode");
    this.preceedInHmiStandAloneMode = true;
    return false;
  }

package com.gigatronik.beds.service.ecuconnectionstate.impl.synchronization;

public class ActionSynchronizer {
  private static final ActionSynchronizer SYNCHRONIZER = new ActionSynchronizer();
  
  private boolean completed;
  
  public static ActionSynchronizer getInstance() {
    return SYNCHRONIZER;
  }
  
  public synchronized void clear() {
    this.completed = false;
  }
  
  public synchronized void notifyComplete() {
    this.completed = true;
    notifyAll();
  }
  
  public synchronized boolean waitCompletion(long timeout) throws InterruptedException {
    long endTime = System.currentTimeMillis() + timeout;
    while (!this.completed && (timeout = endTime - System.currentTimeMillis()) > 0L)
      wait(timeout); 
    return this.completed;
  }
}
