  }
  
  private ComponentType fallbackForCorruptVictorDuComponentType() {
    this.logger.warn("Preparing the DU for accessing the requested service parameter 'component type' & 'product line'");
    if (!this.ecu.executeHardReset(true).isEcuStillConnected())
      return ComponentType.NONE; 
    boolean fblMode = (this.ecu.readEcuState() == Constants.EcuMode.FBL);
    if (fblMode) {
      ((Du)this.ecu).startFblIdentificationSession();
    } else {
      this.communicationManager.startDuDiagnosticSession();
    } 
    ComponentType duCompType = ComponentType.parse((fblMode ? ((Du)this.ecu).readComponentTypeFbl() : this.ecu.readComponentType()).toString());
    if (duCompType != null && duCompType != ComponentType.NONE && duCompType != ComponentType.COMP_UNSPECIFIED) {
      this.logger.info("Reading the component type a second time helped retrieving the actual value '{}'", duCompType);
      return duCompType;
    } 
    ProductLine productLine = ProductLine.fromOdxName((fblMode ? ((Du)this.ecu).readProductLineFbl() : ((Du)this.ecu).readProductLine()).toString());
    if (productLine != null && productLine != ProductLine.NONE) {
      this.logger.info("ComponentType of DU could be mapped via the productLine");
      switch (productLine) {
        case ACTIVE:
        case PERFORMANCE:
        case X_LINE:
          duCompType = ComponentType.COMP_BDU250;
          break;
        case ACTIVE2:
          duCompType = ComponentType.COMP_BDU310;
          break;
        case ACTIVE_PLUS:
          duCompType = ComponentType.COMP_BDU350;
          break;
        case GEN3_PERFORMANCE:
          duCompType = ComponentType.COMP_BDU365;
          break;
        case GEN4_PERFORMANCE_CX:
          duCompType = ComponentType.COMP_BDU450;
          break;
        case GEN4_PERFORMANCE:
          duCompType = ComponentType.COMP_BDU490;
          break;
        default:
          this.logger.error("Akward productLine detected, handle this as corrupt value");
          duCompType = ComponentType.COMP_UNSPECIFIED;
          break;
      } 
    } 
    return duCompType;
  }
  
  private ProductLine mapToPerformanceLine(ComponentType comType) {
    if (comType != null)
      if (comType == ComponentType.NONE) {
        this.logger.error("Mapping unknown product line by component type not possible. Also component type is unknown.");
      } else {
        if (comType.isBDU1xx())
          return ProductLine.CLASSIC_PLUS; 
        if (comType.isBDU2xx())
          return ProductLine.PERFORMANCE; 
        if (comType.isBDU3xx())
          return ProductLine.GEN3_PERFORMANCE; 
        if (comType.isBDU4xx())
          return ProductLine.GEN4_PERFORMANCE; 
      }  
    this.logger.error("Failed to map product line by component type.");
    return ProductLine.NONE;
  }

public abstract class EcuIdentificationController {
  protected static final int ERROR_HANDLING_INTERVAL = 800;
  
  protected final Logger logger = LoggerFactory.getLogger(getClass());
  
  protected Ecu ecu;
  
  protected EcuIdentification ecuIdentification;
  
  protected CommunicationManager communicationManager;
  
  public EcuIdentificationController(CommunicationManager communicationManager) {
    this.communicationManager = communicationManager;
  }
  
  public boolean populateEcuIdentification(EcuIdentification ecuIdentification, Constants.EcuType type, Constants.EcuMode mode) {
    this.ecuIdentification = ecuIdentification;
    if (!readEcuIdentification(type, mode))
      retryReadingUponFailure(type, mode); 
    return ecuIdentification.isValid();
  }
  
  protected abstract boolean setEcuInDiagnosticSession();
  
  protected boolean readEcuIdentification(Constants.EcuType type, Constants.EcuMode mode) {
    this.ecuIdentification.setMode(mode);
    readEcuIdentificationAccordingStatus(type, mode);
    EcuIdentificationUtils.mapEcuTypeToComponentTypeIfNecessary(this.ecuIdentification, type);
    return this.ecuIdentification.isValid();
  }
  
  protected void retryReadingUponFailure(Constants.EcuType type, Constants.EcuMode mode) {
    if (!this.ecuIdentification.isValid() && mode == Constants.EcuMode.APP && !isInterrupted()) {
      this.logger.info("Received negative response while reading ECU identification, so restart extended session");
      sleep(800L);
      if (!isInterrupted() && setEcuInDiagnosticSession())
        readEcuIdentification(type, mode); 
    } 
  }
  
  protected void readEcuIdentificationAccordingStatus(Constants.EcuType type, Constants.EcuMode mode) {
    switch (mode) {
      case APP:
        readEcuIdentificationApp(type);
        break;
      case FBL:
        readEcuIdentificationFbl(type);
        break;
    } 
  }
  
  protected boolean readAbsOrBcsIdentification(Constants.EcuType type, Constants.EcuMode mode) {
    this.ecuIdentification.setMode(mode);
    this.ecuIdentification.setComponentType(ComponentType.parse(this.ecu.readComponentType().toString()));
    this.ecuIdentification.setSwVariant(this.ecu.readSoftwareVariant());
    this.ecuIdentification.setHwVariant(this.ecu.readHardwareVariant());
    this.ecuIdentification.setSwVersion(this.ecu.readSoftwareVersion());
    this.ecuIdentification.setHwVersion(this.ecu.readHardwareVersion());
    if (mode == Constants.EcuMode.APP)
      readEcuIdentificationApp(type); 
    EcuIdentificationUtils.mapEcuTypeToComponentTypeIfNecessary(this.ecuIdentification, type);
    return this.ecuIdentification.isValid();
  }
  
  protected abstract void readEcuIdentificationFbl(Constants.EcuType paramEcuType);
  
  protected void readEcuIdentificationApp(Constants.EcuType type) {
    this.ecuIdentification.setSerialNumber(this.ecu.readSerialNumber());
    this.ecuIdentification.setProductCode(this.ecu.readProductCode());
    this.ecuIdentification.setPartNumber(this.ecu.readPartNumber());
    this.ecuIdentification.setSampleSoftware(this.ecu.readIsSampleSoftware());
  }
  
  protected boolean isInterrupted() {
    return Thread.currentThread().isInterrupted();
  }
  
  protected void sleep(long millis) {
    try {
      Thread.sleep(millis);
    } catch (InterruptedException ex) {
      Thread.currentThread().interrupt();
    } 
  }

  BikeConfigurationChangedEvent setValue(ConfigurationKey key, String value) {
    ConfigurationParameter parameter = this.parameters.get(key);
    if (parameter != null && isSameValue(parameter.getValue(), value))
      return null; 
    String old = null;
    if (parameter == null) {
      parameter = new ConfigurationParameter(value);
    } else {
      old = parameter.getValue();
      parameter.setValue(value);
    } 
    this.parameters.put(key, parameter);
    this.matchesFile = configurationMatchesFile();
    return new BikeConfigurationChangedEvent(key, old, value, this.matchesFile, BikeConfigurationChangedEvent.Action.VALUE_CHANGED);
  }
  
  String getValue(ConfigurationKey key) {
    ConfigurationParameter parameter = this.parameters.get(key);
    return (parameter != null) ? parameter.getValue() : null;
  }
  
  String getFileValue(ConfigurationKey key) {
    ConfigurationParameter parameter = this.parameters.get(key);
    return (parameter != null) ? parameter.getFileValue() : null;
  }
  
  void updateFromFile() {
    if (this.fileName != null) {
      for (Map.Entry<ConfigurationKey, ConfigurationParameter> entry : this.parameters.entrySet()) {
        ConfigurationParameter parameter = entry.getValue();
        parameter.setBikeValueToFileValue();
      } 
      this.matchesFile = true;
    } 
  }
  
  BikeConfigurationChangedEvent readFromFile(File file) {
    this.parameters = this.xmlUtil.readConfig(file);
    if (!this.parameters.isEmpty()) {
      this.fileName = file.getName();
      this.matchesFile = true;
    } 
    ConfigurationBackwardCompatibility.doBackwardCompatibilityChanges(this.parameters);
    ConfigurationBackwardCompatibility.checkAndFormatHexValues(this.parameters);
    BikeConfigurationChangedEvent event = new BikeConfigurationChangedEvent(null, null, null, false, BikeConfigurationChangedEvent.Action.READ_FROM_FILE);
    event.addAttribute(Attribute.Key.OPERATION_RESULT, !this.parameters.isEmpty() ? Attribute.Value.OK : Attribute.Value.READ_FROM_FILE_FAILED);
    return event;
  }
  
  BikeConfigurationChangedEvent writeToFile(File file) {
    boolean result = this.xmlUtil.writeConfig(file, this.parameters);
    if (result) {
      this.fileName = file.getName();
      this.matchesFile = true;
    } 
    BikeConfigurationChangedEvent event = new BikeConfigurationChangedEvent(null, null, null, false, BikeConfigurationChangedEvent.Action.WRITE_TO_FILE);
    event.addAttribute(Attribute.Key.OPERATION_RESULT, result ? Attribute.Value.OK : Attribute.Value.WRITE_TO_FILE_FAILED);
    return event;
  }
  
  BikeConfigurationChangedEvent clear() {
    this.fileName = null;
    this.matchesFile = true;

package com.gigatronik.beds.service.configuration.impl;

import com.gigatronik.beds.service.configuration.ConfigurationKey;
import com.gigatronik.ebike.common.configuration.ProductLine;
import com.gigatronik.ebike.common.util.ConverterUtils;
import java.util.Map;
import org.apache.commons.lang3.StringUtils;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

class ConfigurationBackwardCompatibility {
  private static final Logger LOGGER = LoggerFactory.getLogger(ConfigurationBackwardCompatibility.class);
  
  static void doBackwardCompatibilityChanges(Map<ConfigurationKey, ConfigurationParameter> parameters) {
    ConfigurationParameter readFileVersion = parameters.getOrDefault(ConfigurationKey.FILE_VERSION, null);
    if (readFileVersion == null || StringUtils.isEmpty(readFileVersion.getValue())) {
      LOGGER.info("backward compatibility check is not possible -> parameter 'fileVersion' is missing or empty in the configuration");
      return;
    } 
    try {
      int fileVersion = Integer.parseInt(readFileVersion.getValue());
      if (4 > fileVersion) {
        cffThreeProductLineCompatibiliyChanges(fileVersion, parameters.getOrDefault(ConfigurationKey.CFF3_PRODUCT_LINE, null));
        doGearRatioCompatibilityChanges(parameters.getOrDefault(ConfigurationKey.MAX_GEAR_RATIO, null));
        doGearRatioCompatibilityChanges(parameters.getOrDefault(ConfigurationKey.MIN_GEAR_RATIO, null));
        doAbsParIdCompatibilityChanges(parameters.getOrDefault(ConfigurationKey.ABS_PARAMETER_ID, null));
        doAbsParIdChecksumCompatibilityChanges(parameters.getOrDefault(ConfigurationKey.ABS_PARAMETER_ID, null), 
            parameters.getOrDefault(ConfigurationKey.ABS_PARAMETER_ID_CHECKSUM, null));
        doAutodownshiftCompatibilityChanges(parameters);
      } 
    } catch (NumberFormatException ex) {
      LOGGER.error("Could not parse the parameter 'fileVersion' to an integer value '{}'", readFileVersion.getValue(), ex);
    } 
  }
  
  private static void doAutodownshiftCompatibilityChanges(Map<ConfigurationKey, ConfigurationParameter> parameters) {
    ConfigurationParameter autoDownshiftOld = parameters.getOrDefault(ConfigurationKey.AUTO_DOWNSHIFT_COMPATIBILITY_MY18, null);
    if (autoDownshiftOld != null) {
      boolean autoDownshiftDisabled = (StringUtils.equals(autoDownshiftOld.getFileValue(), "-1") || StringUtils.equals(autoDownshiftOld.getFileValue(), "1"));
      if (autoDownshiftDisabled) {
        ConfigurationParameter defaultGear = parameters.getOrDefault(ConfigurationKey.AUTO_DOWNSHIFT_STARTGEAR, null);
        parameters.put(ConfigurationKey.AUTO_DOWNSHIFT_STARTGEAR, (defaultGear == null) ? new ConfigurationParameter("0", "0") : 
            new ConfigurationParameter(defaultGear.getValue(), "0"));
      } 
    } 
    parameters.remove(ConfigurationKey.AUTO_DOWNSHIFT_COMPATIBILITY_MY18);
  }
  
  static void checkAndFormatHexValues(Map<ConfigurationKey, ConfigurationParameter> parameters) {
    setParameterValuesToUpperCase(parameters.get(ConfigurationKey.ABS_PARAMETER_ID_CHECKSUM));
    setParameterValuesToUpperCase(parameters.get(ConfigurationKey.WHEEL_CIRCUMFERENCE_CHECKSUM));
  }
  
  private static void setParameterValuesToUpperCase(ConfigurationParameter parameter) {
    if (parameter != null) {
      if (parameter.getFileValue() != null)
        parameter.setFileValue(parameter.getFileValue().toUpperCase()); 
      if (parameter.getValue() != null)
        parameter.setValue(parameter.getValue().toUpperCase()); 
    } 
  }
  
  private static void cffThreeProductLineCompatibiliyChanges(int version, ConfigurationParameter parameter) {
    if (version <= 2 && parameter != null) {
      String value = parameter.getValue();
      if (StringUtils.equals(value, "Classic+")) {
        parameter.setFileValue(ProductLine.CLASSIC_PLUS.getOdxName());
      } else if (StringUtils.equals(value, "Active Line 1")) {
        parameter.setFileValue(ProductLine.ACTIVE.getOdxName());
      } else if (StringUtils.equals(value, "Performance Line")) {
        parameter.setFileValue(ProductLine.PERFORMANCE.getOdxName());
      } else if (StringUtils.equals(value, "Performance Line CX")) {
        parameter.setFileValue(ProductLine.X_LINE.getOdxName());
      } else if (StringUtils.equals(value, "Active Line 2")) {
        parameter.setFileValue(ProductLine.ACTIVE2.getOdxName());
      } else if (StringUtils.equals(value, "Active Line Plus")) {
        parameter.setFileValue(ProductLine.ACTIVE_PLUS.getOdxName());
      } else {
        LOGGER.error("Could not parse parameter 'cff3ProductLine' value '{}' to current format", value);
      } 
      parameter.setBikeValueToFileValue();
    } 
  }
  
  private static void doGearRatioCompatibilityChanges(ConfigurationParameter parameter) {
    if (parameter != null) {
      String value = parameter.getValue();
      if (ConverterUtils.isDouble(value)) {
        parameter.setFileValue(String.format("%.2f", Double.parseDouble(value)));
        parameter.setBikeValueToFileValue();
      } 
    } 
  }
  
  private static void doAbsParIdCompatibilityChanges(ConfigurationParameter parameter) {
    if (parameter != null) {
      String value = parameter.getValue();
      if (StringUtils.isNumeric(value)) {
        parameter.setFileValue(String.format("%03d", Integer.parseInt(value)));
        parameter.setBikeValueToFileValue();
      } 
    } 
  }

      String absParIdChecksum = (Integer.parseInt(absParId.getValue()) == 0) ? "FF" : null;
      absChecksum.setFileValue(absParIdChecksum);
      absChecksum.setBikeValueToFileValue();
    } 
  }

    ComponentType hmiType = this.bikeIdentification.getHmiIdentification().getComponentType();
    if (hmiType.isBUI25x())
      return writeBuiLanguageAndDistanceUnit(configuration); 
    if (hmiType.isBUI21x())
      return writeBui21xData(configuration); 
    if (hmiType.isBUI33x())
      return writeBui33xData(configuration, userLevel); 
    if (hmiType.isBUI35x())
      return writeBui35xData(configuration);

    Version swVersion;
    try {
      swVersion = (userLevel == UserLevel.OEM) ? getHmiSwVersion() : this.bikeIdentification.getHmiIdentification().getSwVersionResponse().getVersion();
    } catch (ComplexResponse.DataAccessException e) {
      LOGGER.error("Parsing of BUI software version failed, setting version to 0.0.0.0", (Throwable)e);
      swVersion = new Version(new Integer[] { 0, 0, 0, 0 });
    } 
    if (swVersion.compareTo(KIOX_LANGUAGE_AND_UNIT_MIN_VERSION) >= 0)
      return writeBuiLanguageAndDistanceUnit(configuration); 
    return true;
  }
  
  private boolean writeBui35xData(BikeConfigurationData configuration) {
    String language = configuration.getDisplayLanguage();
    if (this.communicationManager.getHmiCom().writeLanguageSetting(language).isNegative())
      LOGGER.warn("Setting display language '{}' for BUI350 failed.", language); 
    return this.communicationManager.getHmiCom().writeDistanceUnit(configuration.getDistanceUnit()).isPositive();
  }
  
  private boolean writeBuiLanguageAndDistanceUnit(BikeConfigurationData configuration) {
    boolean languageResult = this.communicationManager.getHmiCom().writeLanguageSetting(configuration.getDisplayLanguage()).isPositive();
    boolean distanceUnitResult = this.communicationManager.getHmiCom().writeDistanceUnit(configuration.getDistanceUnit()).isPositive();
    return (languageResult && distanceUnitResult);
  }
  
  boolean writeToBike(BikeConfigurationData configuration, UserLevel userLevel, boolean inFlashProcess) {
    try {
      LOGGER.debug("writeToBike('{}')", configuration);
      boolean duSuccess = (!this.bikeIdentification.getDuIdentification().isApplication() || 
        writeDuConfigurationWithRetryOnFailure(configuration, userLevel, inFlashProcess));
      boolean hmiSuccess = (!this.bikeIdentification.getHmiIdentification().isApplication() || 
        writeHmiConfigurationWithRetryOnFailure(configuration, userLevel));
      return (duSuccess && hmiSuccess);
    } catch (Exception e) {
      LOGGER.error("Unexpected error while writing bike configuration data", (Throwable)e);
      return false;
    } 
  }
  
  private boolean writeHmiConfigurationWithRetryOnFailure(BikeConfigurationData configuration, UserLevel userLevel) {
    int hmiTry = 1;
    for (; hmiTry <= 3 && !writeBuiConfigurationToBike(configuration, userLevel); hmiTry++) {
      LOGGER.warn("Attempt '{}' to write HMI configuration failed.", Integer.valueOf(hmiTry));
      if (hmiTry < 3)
        reinitializeHmi(); 
    } 
    return (hmiTry < 3);
  }
  
  private boolean writeDuConfigurationWithRetryOnFailure(BikeConfigurationData configuration, UserLevel userLevel, boolean inFlashProcess) {
    int duTry = 1;
    for (; duTry <= 3 && !writeDuAndAbsConfigurationToBike(configuration, userLevel, inFlashProcess); duTry++) {
      LOGGER.warn("Attempt '{}' to write DU configuration failed.", Integer.valueOf(duTry));
      if (duTry < 3)
        reinitializeDu(); 
    } 
    return (duTry < 3);
  }
  
  private boolean writeDuAndAbsConfigurationToBike(BikeConfigurationData configuration, UserLevel userLevel, boolean inFlashProcess) {
    boolean result = (userLevel == UserLevel.OEM) ? writeOemSpecificDuConfigurationToBike(configuration) : writeIbdSpecificDuConfigurationToBike(configuration);
    result &= writeCommonDuConfigurationToBike(configuration);
    return (userLevel == UserLevel.OEM) ? ((result && writeAbsConfigurationToBike(configuration, inFlashProcess))) : result;
  }
  
  private void reinitializeDu() {
    this.communicationManager.getDuCom().executeHardReset(true);
    this.communicationManager.startDuDiagnosticSession();
  }
  
  private void reinitializeHmi() {
    this.communicationManager.startHmiDiagnosticSession();
  }
}

public class ConfigurationManagerImpl implements ConfigurationManager, RightsChangedEventListener {
  private static final Logger LOGGER = LoggerFactory.getLogger(ConfigurationManagerImpl.class);
  
  private final BikeIdentificationImpl bikeIdentification;
  
  private final BikeConfigurationImpl bikeConfiguration;
  
  private final ContainerConfiguration cffConfiguration;
  
  private final EcuConnectionStateChangedEventPublisher ecuConnectionStateChangedEventPublisher;
  
  private final EcuConfigurationChangedEventPublisher ecuConfigurationChangedEventPublisher;
  
  private final ContainerConfigurationChangedEventPublisher containerConfigurationChangedEventPublisher;
  
  private final CommunicationManager communicationManager;
  
  private final RightsManager rightsManager;
  
  private final BikeIdentificationController bikeIdentificationController;
  
  private final ContainerToBikeCompatibilityValidator containerToBikeCompatibilityValidator;
  
  private final ConfigurationHandler configurationHandler;
  
  private final SecurityAccessFailurePopupHandler securityAccessFailurePopupHandler;
  
  private volatile Attribute.Value containerCompatibleWithBike = Attribute.Value.OK;
  
  private GearPropertiesObserver gearPropertiesObserver;
  
  public ConfigurationManagerImpl(CommunicationManager communicationManager, RightsManager rightsManager) {
    this.communicationManager = communicationManager;
    this.rightsManager = rightsManager;
    this.ecuConnectionStateChangedEventPublisher = new EcuConnectionStateChangedEventPublisher();
    this.ecuConfigurationChangedEventPublisher = new EcuConfigurationChangedEventPublisher();
    this.containerConfigurationChangedEventPublisher = new ContainerConfigurationChangedEventPublisher();
    this.securityAccessFailurePopupHandler = SecurityAccessFailurePopupHandlerFactory.createInstance(rightsManager);
    this.bikeIdentification = new BikeIdentificationImpl();
    this.bikeIdentificationController = new BikeIdentificationController(this.bikeIdentification, communicationManager, this);
    this.bikeConfiguration = new BikeConfigurationImpl();
    this.cffConfiguration = new ContainerConfiguration(this.containerConfigurationChangedEventPublisher);
    this.containerToBikeCompatibilityValidator = new ContainerToBikeCompatibilityValidator((BikeIdentification)this.bikeIdentification, this.cffConfiguration, rightsManager);
    this.configurationHandler = new ConfigurationHandler((BikeIdentification)this.bikeIdentification, this.cffConfiguration, communicationManager);
    createPopUpHandler();
  }
  
  private void createPopUpHandler() {
    new ContainerBikeIncompatibilityPopUpHandler(this);
    new ImportantEcuIdentificationParameterMissingPopUpHandler(this);
    new EcuInFblPopUpHandler(this);
  }
  
  public BikeIdentification getBikeIdentification() {
    return (BikeIdentification)this.bikeIdentification;
  }
  
  public ContainerConfiguration getContainerConfiguration() {
    return this.cffConfiguration;
  }
  
  public void readConfigurationFromBike() {
    LOGGER.info("readConfigurationFromBike");
    this.bikeConfiguration.clearBikeData();
    boolean resultHmi = readHmiConfigurationFromBike();
    boolean resultDu = readDuConfigurationFromBike();
    updateContainerConfigurationFromBike(false);
    BikeConfigurationChangedEvent event = new BikeConfigurationChangedEvent(null, null, null, false, BikeConfigurationChangedEvent.Action.READ_FROM_BIKE);
    event.addAttribute(Attribute.Key.OPERATION_RESULT, (resultHmi && resultDu) ? Attribute.Value.OK : Attribute.Value.READ_FROM_BIKE_FAILED);
    parametrizeAbstractConfigurationEvent(event);
    publishBikeConfigurationChangeEvent(event);
  }
  
  private void updateContainerConfigurationFromBike(boolean readAbsConfigurationFromDu) {
    if (this.cffConfiguration.getContainerController().isCff3Loaded()) {
      ProductLine line = this.bikeIdentification.getProductLine();
      String type = this.bikeIdentification.getBikeType();
      ContainerBikeDataSet parameterSet = this.cffConfiguration.getParameterSet(this.bikeIdentification.getApplicationId());
      String model = (parameterSet != null) ? parameterSet.getBikeModel() : null;
      this.cffConfiguration.setProductLine(line);
      this.cffConfiguration.setBikeType(BikeType.getValueByXmlAttribute(type));
      this.cffConfiguration.setBikeModel(model);
      this.bikeConfiguration.setValue(ConfigurationKey.CFF3_PRODUCT_LINE, line.getOdxName());
      this.bikeConfiguration.setValue(ConfigurationKey.CFF3_BIKE_TYPE, type);
      this.bikeConfiguration.setValue(ConfigurationKey.CFF3_BIKE_MODEL, model);
    } 
    if (readAbsConfigurationFromDu)
      this.configurationHandler.readAbsConfigurationFromBike(this.bikeConfiguration); 
    this.cffConfiguration.setAbsParId(getConfigurationValue(ConfigurationKey.ABS_PARAMETER_ID));
  }
  
  private void updateContainerConfigurationFromConfig() {
    this.cffConfiguration.setProductLine(ProductLine.fromOdxName(getConfigurationValue(ConfigurationKey.CFF3_PRODUCT_LINE)));
    this.cffConfiguration.setBikeType(BikeType.fromDisplayName(getConfigurationValue(ConfigurationKey.CFF3_BIKE_TYPE)));
    this.cffConfiguration.setBikeModel(getConfigurationValue(ConfigurationKey.CFF3_BIKE_MODEL));
    this.cffConfiguration.setAbsParId(getConfigurationValue(ConfigurationKey.ABS_PARAMETER_ID));
  }
  
  private boolean readHmiConfigurationFromBike() {
    boolean result = true;
    if (this.bikeIdentification.isHmiConnected()) {
      result = this.configurationHandler.readBuiConfigurationFromBike(this.bikeConfiguration);
      if (!result && this.communicationManager.startHmiDiagnosticSession())
        result = this.configurationHandler.readBuiConfigurationFromBike(this.bikeConfiguration); 
    } 
    return result;
  }
  
  private boolean readDuConfigurationFromBike() {
    boolean result = true;
    if (this.bikeIdentification.isDuConnected()) {
      result = this.configurationHandler.readDuConfigurationFromBike(this.bikeConfiguration);
      if (!result)
        if (!this.communicationManager.startDuDiagnosticSession()) {
          if (this.bikeIdentification.isDuConnected())
            this.securityAccessFailurePopupHandler.displaySecurityAccessFailedDialog(false, this.bikeIdentification.getDuIdentification().getComponentType()); 
        } else {
          result = this.configurationHandler.readDuConfigurationFromBike(this.bikeConfiguration);
        }  
    } 
    return result;
  }
  
  public boolean writeConfigurationToBike(BikeConfigurationData configurationData, UserLevel userLevel, boolean inFlashProcess) {
    boolean result = this.configurationHandler.writeToBike(configurationData, userLevel, inFlashProcess);
    if (!inFlashProcess) {
      BikeConfigurationChangedEvent event = new BikeConfigurationChangedEvent(null, null, configurationData, false, BikeConfigurationChangedEvent.Action.WRITE_TO_BIKE);
      event.addAttribute(Attribute.Key.OPERATION_RESULT, result ? Attribute.Value.OK : Attribute.Value.WRITE_TO_BIKE_FAILED);
      publishBikeConfigurationChangeEvent(event);
    } 
    return result;
  }
  
  public void readConfigurationFromFile(File file) {
    LOGGER.info("readConfigurationFromFile(file = '{}')", file);
    try {
      BikeConfigurationChangedEvent event = this.bikeConfiguration.readFromFile(file);
      ContainerConfigurationChangedEvent cffEvent = null;
      Attribute.Value cffResult = Attribute.Value.CONTAINER_INVALID;
      String cffPath = getConfigurationValue(ConfigurationKey.CFF_FILE);
      if (cffPath != null) {
        File cffFile = new File(cffPath);
        if (cffFile.exists()) {
          cffResult = Attribute.Value.OK;
          cffEvent = loadContainerFile(cffFile, false);
        } else {
          LOGGER.error("cffFile does not exist");
        } 
      } 
      Thread.sleep(10L);
      parametrizeAndPublishContainerConfigurationChangeEvent(cffEvent);
      if (cffResult == Attribute.Value.CONTAINER_INVALID || cffEvent != null) {
        if (cffResult == Attribute.Value.CONTAINER_INVALID)
          LOGGER.error("Container is invalid"); 
        Thread.sleep(10L);
        event.addAttribute(Attribute.Key.OPERATION_RESULT, cffResult);
        parametrizeAbstractConfigurationEvent(event);
        publishBikeConfigurationChangeEvent(event);
      } 
    } catch (InterruptedException ex) {
      LOGGER.warn("Configuration loading interrupted - {}", ex.getMessage());
      Thread.currentThread().interrupt();
    } 
  }
  
  public void writeConfigurationToFile(File file) {
    BikeConfigurationChangedEvent event = this.bikeConfiguration.writeToFile(file);
    publishBikeConfigurationChangeEvent(event);
  }
  
  public void clearConfiguration() {
    BikeConfigurationChangedEvent event = this.bikeConfiguration.clear();
    publishBikeConfigurationChangeEvent(event);
  }
  
  public void setConfigurationValue(ConfigurationKey key, Object value) {
    LOGGER.debug("setConfigurationValue '{}' = '{}'", key, value);
    BikeConfigurationChangedEvent event = this.bikeConfiguration.setValue(key, (value != null) ? String.valueOf(value) : null);
    publishBikeConfigurationChangeEvent(event);
  }
  
  public String getConfigurationValue(ConfigurationKey key) {
    String value = this.bikeConfiguration.getValue(key);
    return StringUtils.isEmpty(value) ? null : value;
  }
  
  public String getConfigurationFileValue(ConfigurationKey key) {
    String value = this.bikeConfiguration.getFileValue(key);
    return StringUtils.isEmpty(value) ? null : value;
  }
  
  public void updateConfigurationFromFile() {
    this.bikeConfiguration.updateFromFile();
  }
  
  public void loadContainer(File file, boolean transferDirectly) {
    LOGGER.info("loadContainer(file = '{}', transferDirectly = '{}')", file, Boolean.valueOf(transferDirectly));
    parametrizeAndPublishContainerConfigurationChangeEvent(loadContainerFile(file, transferDirectly));
  }
  
  private ContainerConfigurationChangedEvent loadContainerFile(File file, boolean transferDirectly) {
    try {
      ContainerConfigurationChangedEvent event = this.cffConfiguration.loadContainerFromFile(file, transferDirectly);
      Thread.sleep(100L);
      if (StringUtils.isEmpty(getConfigurationFileName())) {
        updateContainerConfigurationFromBike(true);
      } else {
        updateContainerConfigurationFromConfig();
      } 
      this.containerCompatibleWithBike = this.containerToBikeCompatibilityValidator.checkCompatibility();
      return event;
    } catch (InterruptedException ex) {
      LOGGER.warn("Container loading interrupted - {}", ex.getMessage());
      Thread.currentThread().interrupt();
      this.containerCompatibleWithBike = Attribute.Value.OK;
      return null;
    } 
  }
  
  public void clearContainer() {
    if (this.cffConfiguration.isLoaded()) {
      LOGGER.info("Clear container");
      this.bikeConfiguration.clearContainerData();
      setProductLine(ProductLine.NONE);
      setBikeType(BikeType.UNDEFINED);
      setBikeModel("");
      this.containerCompatibleWithBike = Attribute.Value.OK;
      parametrizeAndPublishContainerConfigurationChangeEvent(this.cffConfiguration.clearContainer());
    } 
  }
  
  public boolean isBikeConnected() {
    return this.bikeIdentification.isHmiConnected();
  }
  
  public boolean isContainerLoaded() {
    return this.cffConfiguration.isLoaded();
  }
  
  public boolean isContainerCompatibleWithBike() {
    return (this.containerCompatibleWithBike == Attribute.Value.OK);
  }
  
  public void onEcuConnectionStateChange(EcuConnectionStateChangedEvent event) {
    LOGGER.debug("EcuConnectionStateChangedEvent received: '{}'", event);
    if (event.isEcuConnected()) {
      onEcuConnection(event.getEcuType(), event.getEcuMode());
    } else {
      onEcuDisconnection(event.getEcuType());
    } 
  }
  
  private void onEcuConnection(Constants.EcuType type, Constants.EcuMode mode) {
    if (!isInterrupted()) {
      boolean correctReading = this.bikeIdentificationController.readIdentificationDataFromEcu(type, mode);
      if (type == Constants.EcuType.VICTOR_DU && mode == Constants.EcuMode.APP && correctReading)
        this.bikeIdentificationController.readIdentificationDataFromEcu(Constants.EcuType.ESHIFT, Constants.EcuMode.APP); 
    } 
    if (!isInterrupted()) {
      if (type == Constants.EcuType.VICTOR_DU && mode == Constants.EcuMode.APP && this.gearPropertiesObserver == null && this.bikeIdentification.isEshiftConnected()) {
        this.gearPropertiesObserver = new GearPropertiesObserver(this.communicationManager, (BikeIdentification)this.bikeIdentification, this.bikeIdentificationController, this.ecuConnectionStateChangedEventPublisher);
        this.gearPropertiesObserver.start();
      } 
      this.containerCompatibleWithBike = this.containerToBikeCompatibilityValidator.checkCompatibility();
      HardwareConnectionChangedEvent event = new HardwareConnectionChangedEvent(HardwareConnectionChangedEvent.HardwareConnectionStateChange.ECU_CONNECTED, getHardwareSystemType(type), type, mode);
      parametrizeAndPublishHardwareConnectionChangeEvent(event);
    } 
  }
  
  private HardwareConnectionChangedEvent.HardwareSystem getHardwareSystemType(Constants.EcuType type) {
    switch (type) {
      case BUI21X:
      case BUI25X:
      case BUI270:
      case BUI275:
      case BUI330:
      case BUI350:
      case COBI:
        return HardwareConnectionChangedEvent.HardwareSystem.BIKE_SYSTEM;
      case BCT:
        return HardwareConnectionChangedEvent.HardwareSystem.BCT_SYSTEM;
    } 
    return HardwareConnectionChangedEvent.HardwareSystem.UNKNOWN;
  }
  
  private void onEcuDisconnection(Constants.EcuType type) {
    this.bikeIdentificationController.deleteIdentificationDataFromEcu(type);
    if (type == Constants.EcuType.VICTOR_DU)
      terminateGearPropertyObserverThreadIfNecessary(); 
    this.containerCompatibleWithBike = this.containerToBikeCompatibilityValidator.checkCompatibility();
    HardwareConnectionChangedEvent event = new HardwareConnectionChangedEvent(HardwareConnectionChangedEvent.HardwareConnectionStateChange.ECU_DISCONNECTED, getHardwareSystemType(type), type, Constants.EcuMode.NONE);
    parametrizeAndPublishHardwareConnectionChangeEvent(event);
  }
  
  public void onSystemConnectionStateChange(SystemConnectionStateChangeEvent systemConnectionStateChangeEvent) {
    LOGGER.debug("SystemConnectionStateChangeEvent received: '{}'", systemConnectionStateChangeEvent);
    HardwareConnectionChangedEvent.HardwareConnectionStateChange action = HardwareConnectionChangedEvent.HardwareConnectionStateChange.NONE;
    HardwareConnectionChangedEvent.HardwareSystem systemType = HardwareConnectionChangedEvent.HardwareSystem.UNKNOWN;
    switch (systemConnectionStateChangeEvent.getAction()) {
      case HMI_STANDALONE_CONNECT:
      case BIKE_CONNECT:
        if (this.bikeIdentification.getHmiIdentification().getComponentType().isBUI25x() || this.bikeIdentification.getHmiIdentification().getComponentType().isBUI33x() || this.bikeIdentification.getHmiIdentification().getComponentType().isBUI35x()) {
          LOGGER.debug("Update time & date on connected Intuvia or Kiox");
          this.communicationManager.getHmiCom().writePresentDateAndTime();
        } 
        onBikeConnectOrReconnect();
        action = HardwareConnectionChangedEvent.HardwareConnectionStateChange.SYSTEM_CONNECTED;
        systemType = HardwareConnectionChangedEvent.HardwareSystem.BIKE_SYSTEM;
        break;
      case BIKE_DISCONNECT:
        action = HardwareConnectionChangedEvent.HardwareConnectionStateChange.SYSTEM_DISCONNECTED;
        systemType = HardwareConnectionChangedEvent.HardwareSystem.BIKE_SYSTEM;
        break;
      case BIKE_RECONNECT:
        onBikeConnectOrReconnect();
        action = HardwareConnectionChangedEvent.HardwareConnectionStateChange.SYSTEM_RECONNECTED;
        systemType = HardwareConnectionChangedEvent.HardwareSystem.BIKE_SYSTEM;
        break;
      case BCT_CONNECT:
        action = HardwareConnectionChangedEvent.HardwareConnectionStateChange.SYSTEM_CONNECTED;
        systemType = HardwareConnectionChangedEvent.HardwareSystem.BCT_SYSTEM;
        break;
      case BCT_DISCONNECT:
        action = HardwareConnectionChangedEvent.HardwareConnectionStateChange.SYSTEM_DISCONNECTED;
        systemType = HardwareConnectionChangedEvent.HardwareSystem.BCT_SYSTEM;
        break;
    } 
    HardwareConnectionChangedEvent event = new HardwareConnectionChangedEvent(action, systemType, Constants.EcuType.NONE, Constants.EcuMode.NONE);
    parametrizeAndPublishHardwareConnectionChangeEvent(event);
  }
  
  private void onBikeConnectOrReconnect() {
    if (this.rightsManager.isUserInLevel(UserLevel.IBD) && this.containerCompatibleWithBike != Attribute.Value.OK)
      clearContainer(); 
  }
  
  public void onRightsChanged(RightsChangedEvent event) {
    LOGGER.debug("Received rights changed event '{}'", event);
    if (event.getUserLevel() == UserLevel.UNKNOWN)
      terminateGearPropertyObserverThreadIfNecessary(); 
  }
  
  private void terminateGearPropertyObserverThreadIfNecessary() {
    if (this.gearPropertiesObserver != null) {
      if (this.gearPropertiesObserver.isAlive())
        this.gearPropertiesObserver.interrupt(); 
      this.gearPropertiesObserver = null;
    } 
  }
  
  public void setBuiResetEnabled(boolean reset) {
    ContainerConfigurationChangedEvent event = this.cffConfiguration.setBuiResetEnabled(reset);
    if (event != null) {
      LOGGER.info("Fire ContainerConfigurationChangedEvent: '{}'", event);
      this.containerConfigurationChangedEventPublisher.publishEvent(event);
    } 
  }
  
  public void setProductLine(ProductLine productLine) {
    if (productLine == null)
      productLine = ProductLine.NONE; 
    LOGGER.debug("setProductLine = '{}'", productLine);
    setConfigurationValue(ConfigurationKey.CFF3_PRODUCT_LINE, productLine.getOdxName());
    parametrizeAndPublishContainerConfigurationChangeEvent(this.cffConfiguration.setProductLine(productLine));
  }
  
  public void setBikeType(BikeType bikeType) {
    if (bikeType == null)
      bikeType = BikeType.UNDEFINED; 
    LOGGER.debug("setBikeType = '{}'", bikeType);
    setConfigurationValue(ConfigurationKey.CFF3_BIKE_TYPE, bikeType.getDisplayName());
    parametrizeAndPublishContainerConfigurationChangeEvent(this.cffConfiguration.setBikeType(bikeType));
  }
  
  public void setBikeModel(String bikeModel) {
    LOGGER.debug("setBikeModel = '{}'", bikeModel);
    setConfigurationValue(ConfigurationKey.CFF3_BIKE_MODEL, bikeModel);
    parametrizeAndPublishContainerConfigurationChangeEvent(this.cffConfiguration.setBikeModel(bikeModel));
  }
  
  public void setAbsParId(String absParId) {
    LOGGER.debug("setAbsParId = '{}'", absParId);
    setConfigurationValue(ConfigurationKey.ABS_PARAMETER_ID, absParId);
    if (absParId == null)
      setConfigurationValue(ConfigurationKey.ABS_PARAMETER_ID_CHECKSUM, null); 
    parametrizeAndPublishContainerConfigurationChangeEvent(this.cffConfiguration.setAbsParId(absParId));
  }
  
  private void parametrizeAndPublishContainerConfigurationChangeEvent(ContainerConfigurationChangedEvent event) {
    if (event != null) {
      parametrizeAbstractConfigurationEvent(event);
      LOGGER.info("Fire ContainerConfigurationChangedEvent: {}", event);
      this.containerConfigurationChangedEventPublisher.publishEvent(event);
    } 
  }
  
  private void parametrizeAndPublishHardwareConnectionChangeEvent(HardwareConnectionChangedEvent event) {
    if (event != null) {
      parametrizeAbstractConfigurationEvent(event);
      LOGGER.info("Fire HardwareConnectionChangedEvent: {}", event);
      this.ecuConnectionStateChangedEventPublisher.publishEvent(event);
    } 
  }
  
  private void publishBikeConfigurationChangeEvent(BikeConfigurationChangedEvent event) {
    if (event != null) {
      LOGGER.info("Fire BikeConfigurationChangedEvent: {}", event);
      this.ecuConfigurationChangedEventPublisher.publishEvent(event);
    } 
  }
  
  private void parametrizeAbstractConfigurationEvent(AbstractConfigurationEvent event) {
    if (event != null) {
      event.addAttribute(Attribute.Key.CONTAINER_COMPATIBLE_WITH_BIKE, this.containerCompatibleWithBike);
      event.addAttribute(Attribute.Key.CONTAINER_CONFIGURED_TO_BIKE, isContainerConfiguredToBike() ? Attribute.Value.OK : Attribute.Value.CONTAINER_HARDWARE_CONFIGURATION_INVALID);
      event.addAttribute(Attribute.Key.PRODUCT_LINE_VALID, isProductLineValid() ? Attribute.Value.OK : Attribute.Value.CONTAINER_PRODUCT_LINE_MISSING);
      event.addAttribute(Attribute.Key.BIKE_TYPE_VALID, isBikeTypeValid() ? Attribute.Value.OK : Attribute.Value.CONTAINER_BIKE_TYPE_MISSING);
      event.addAttribute(Attribute.Key.BIKE_MODEL_VALID, isBikeModelValid() ? Attribute.Value.OK : Attribute.Value.CONTAINER_BIKE_MODEL_MISSING);
      event.addAttribute(Attribute.Key.ABS_VALID, isAbsConfigurationValid() ? Attribute.Value.OK : Attribute.Value.ABS_NOT_CONFIGURED);
    } 
  }
  
  public boolean isContainerConfiguredToBike() {
    return (isProductLineValid() && isBikeTypeValid() && isBikeModelValid() && isAbsConfigurationValid());
  }
  
  private boolean isProductLineValid() {
    ProductLine bikeProductLine = this.bikeIdentification.getProductLine();
    ProductLine cffProductLine = this.cffConfiguration.getProductLine();
    return (cffProductLine != ProductLine.NONE && (cffProductLine == bikeProductLine || bikeProductLine == ProductLine.NONE));
  }
  
  private boolean isBikeTypeValid() {
    BikeType bikeBikeType = BikeType.getValueByXmlAttribute(this.bikeIdentification.getBikeType());
    BikeType cffBikeType = this.cffConfiguration.getBikeType();
    return (cffBikeType != BikeType.UNDEFINED && (cffBikeType == bikeBikeType || bikeBikeType == BikeType.UNDEFINED));
  }
  
  private boolean isBikeModelValid() {
    return (StringUtils.isNotBlank(this.cffConfiguration.getBikeModel()) && this.containerToBikeCompatibilityValidator.isCurrentCff3ConfigurationValid() == Attribute.Value.OK);
  }
  
  private boolean isAbsConfigurationValid() {
    boolean absVariantsAvailable = this.cffConfiguration.getContainerController().getBas100Interface().isBas100DataForParIdAvailable();
    boolean absSelectionNotBlank = StringUtils.isNotBlank(getConfigurationValue(ConfigurationKey.ABS_PARAMETER_ID));
    return absVariantsAvailable ? absSelectionNotBlank : true;
  }
  
  public String getConfigurationFileName() {
    return this.bikeConfiguration.getFileName();
  }
  
  public boolean isConfigurationFileChanged() {
    return this.bikeConfiguration.isChanged();
  }
  
  public void addListener(OrderableListener listener) {
    if (listener instanceof HardwareConnectionChangesListener)
      this.ecuConnectionStateChangedEventPublisher.addListener((HardwareConnectionChangesListener)listener); 
    if (listener instanceof BikeConfigurationChangesListener)
      this.ecuConfigurationChangedEventPublisher.addListener((BikeConfigurationChangesListener)listener); 
    if (listener instanceof ContainerConfigurationChangesListener)
      this.containerConfigurationChangedEventPublisher.addListener((ContainerConfigurationChangesListener)listener); 
  }
  
  public void removeListener(OrderableListener listener) {
    if (listener instanceof HardwareConnectionChangesListener)
      this.ecuConnectionStateChangedEventPublisher.removeListener((HardwareConnectionChangesListener)listener); 
    if (listener instanceof BikeConfigurationChangesListener)
      this.ecuConfigurationChangedEventPublisher.removeListener((BikeConfigurationChangesListener)listener); 
    if (listener instanceof ContainerConfigurationChangesListener)
      this.containerConfigurationChangedEventPublisher.removeListener((ContainerConfigurationChangesListener)listener); 
  }
  
  public void onBctOperationStateChanged(BctOperationStateChangeEvent event) {
    LOGGER.debug("Received BctOperationStateChangeEvent: '{}'", event);
    switch (event.getOperatingState()) {
      case NOT_CONNECTED:
      case CONNECTED_FBL:
      case IDLE:
        onBctBatteryTypeChanged(BctBatteryType.NONE);
        break;
      case IDLE_BATTERY_SET:
        onBctBatteryTypeChanged(this.communicationManager.getBctCom().getSelectedBattery());
        break;
    } 
  }
  
  public BikeConfigurationData createBikeConfigurationSnapshot() {
    return this.bikeConfiguration.getCurrentConfiguration();
  }
  
  private void onBctBatteryTypeChanged(BctBatteryType batteryType) {
    this.bikeIdentification.clearBms1Identification();
    if (batteryType != BctBatteryType.NONE) {
      LOGGER.debug("Establish temporary connection to battery and gather BMS identification data.");
      if (this.communicationManager.getBms1Com().executeExtendedSessionAndSecurityAccess(this.rightsManager.getUserLevel()).isPositive()) {
        this.bikeIdentificationController.readIdentificationDataFromEcu((batteryType == BctBatteryType.CLASSIC) ? Constants.EcuType.CLASSIC_PP : Constants.EcuType.VICTOR_PP1, Constants.EcuMode.APP);
      } else {
        this.securityAccessFailurePopupHandler.displaySecurityAccessFailedDialog(false, ComponentType.COMP_BBP290);
      } 
      this.communicationManager.getBms1Com().startDefaultSession(false);
    } 
  }
  
  private boolean isInterrupted() {
    return Thread.currentThread().isInterrupted();
  }
}

import com.gigatronik.ebike.common.util.ConverterUtils;

public class ConfigurationParameter {
  private boolean matchesFile = true;
  
  private String bikeValue;
  
  private String fileValue = null;
  
  ConfigurationParameter(String value) {
    this.bikeValue = value;
  }
  
  ConfigurationParameter(String bike, String file) {
    this.bikeValue = bike;
    this.fileValue = file;
  }
  
  boolean matchesFile() {
    return this.matchesFile;
  }

  public boolean writeConfig(File file, Map<ConfigurationKey, ConfigurationParameter> parameters) {
    if (checkConfigMap(parameters)) {
      try (OutputStream out = new FileOutputStream(file)) {
        XMLOutputFactory outputFactory = XMLOutputFactory.newInstance();
        XMLEventWriter eventWriter = outputFactory.createXMLEventWriter(out, "UTF-8");
        XMLEventFactory eventFactory = XMLEventFactory.newInstance();
        XMLEvent end = eventFactory.createSpace("\n");
        StartDocument startDocument = eventFactory.createStartDocument();
        eventWriter.add(startDocument);
        eventWriter.add(end);
        String date = I18NDateTimesFormats.formatDateToInternalToolFormat(new Date());
        StartElement configStartElement = eventFactory.createStartElement("", "", ConfigurationKey.BIKE_ID.getFileKey());
        eventWriter.add(configStartElement);
        eventWriter.add(end);
        createNode(eventWriter, ConfigurationKey.FILE_VERSION, new ConfigurationParameter(Integer.toString(4)));
        createNode(eventWriter, ConfigurationKey.FILE_DATE, new ConfigurationParameter(date));
        for (ConfigurationKey key : NODES_TO_SAVE) {
          ConfigurationParameter parameter = parameters.get(key);
          if (parameter != null) {
            if (key == ConfigurationKey.LIGHT_POWER && !StringUtils.startsWithAny(parameters.get(ConfigurationKey.LIGHT_SWITCH_FUNCTION_VALUE).getValue(), "Bike light permanently on", "Bike light switchable"))
              parameter.setValue("0.0"); 
            parameter.setFileValueToBikeValue();
            createNode(eventWriter, key, parameter);
          } 
        } 
        eventWriter.add(eventFactory.createEndElement("", "", ConfigurationKey.BIKE_ID.getFileKey()));
        eventWriter.add(end);
        eventWriter.add(eventFactory.createEndDocument());
        eventWriter.close();
        addChecksum(file);
      } catch (FileNotFoundException|XMLStreamException e) {
        LOGGER.error("writeConfig(): Error writing config xml file -> " + e.getMessage(), (Throwable)e);
        return false;
      } catch (IOException e) {
        LOGGER.error("writeConfig(): Error closing file -> " + e.getMessage(), (Throwable)e);
      } 
      return true;
    } 
    return false;
  }
  
  private void addChecksum(File file) {
    try {
      String fileContent = new String(Files.readAllBytes(file.toPath()));
      String ebikeContent = fileContent.split("<ebike.*>", 2)[1].split("</ebike>", 2)[0];
      String md5Hex = calculateMd5Hex(ebikeContent);
      fileContent = fileContent.replaceFirst("<ebike>", "<ebike checksum=\"" + md5Hex + "\">");
      Files.write(file.toPath(), fileContent.getBytes());
    } catch (IOException|NoSuchAlgorithmException ex) {
      LOGGER.error("Error adding checksum to configuration file", (Throwable)ex);
    } 
  }
  
  private String calculateMd5Hex(String content) throws NoSuchAlgorithmException {
    MessageDigest md = MessageDigest.getInstance("MD5");
    md.update(content.getBytes());
    byte[] digest = md.digest();
    return DatatypeConverter.printHexBinary(digest).toUpperCase();
  }
  
  private void createNode(XMLEventWriter eventWriter, ConfigurationKey key, ConfigurationParameter parameter) throws XMLStreamException {
    XMLEventFactory eventFactory = XMLEventFactory.newInstance();
    XMLEvent end = eventFactory.createSpace("\n");
    XMLEvent tab = eventFactory.createSpace("\t");
    StartElement sElement = eventFactory.createStartElement("", "", key.getFileKey());
    eventWriter.add(tab);
    eventWriter.add(sElement);
    String value = parameter.getValue();
    Characters characters = (value == null) ? eventFactory.createCharacters("") : eventFactory.createCharacters(value);
    eventWriter.add(characters);
    EndElement eElement = eventFactory.createEndElement("", "", key.getFileKey());
    eventWriter.add(eElement);
    eventWriter.add(end);
  }
  
  public Map<ConfigurationKey, ConfigurationParameter> readConfig(File file) {
    Map<ConfigurationKey, ConfigurationParameter> items = new EnumMap<>(ConfigurationKey.class);
    try {
      InputStream in = new FileInputStream(file);
      items = readConfig(in);
      in.close();
      validateChecksum(items, file);
    } catch (FileNotFoundException e) {
      LOGGER.error("readConfig(): Error reading config xml file -> " + e.getMessage(), (Throwable)e);
    } catch (IOException e) {
      LOGGER.error("readConfig(): Error reading config xml file -> " + e.getMessage(), (Throwable)e);
    } 
    return items;
  }
  
  public Map<ConfigurationKey, ConfigurationParameter> readConfig(InputStream in) {
    Map<ConfigurationKey, ConfigurationParameter> items = new EnumMap<>(ConfigurationKey.class);
    try {
      XMLInputFactory inputFactory = XMLInputFactory.newInstance();
      XMLEventReader eventReader = inputFactory.createXMLEventReader(in, "UTF-8");
      while (eventReader.hasNext()) {
        XMLEvent event = eventReader.nextEvent();
        if (event.isStartElement()) {
          if (event.asStartElement().getName().getLocalPart().equals("ebike")) {
            Attribute checksumAttribute = event.asStartElement().getAttributeByName(new QName("checksum"));
            if (checksumAttribute != null) {
              String checksum = checksumAttribute.getValue();
              items.put(ConfigurationKey.CHECKSUM, new ConfigurationParameter(checksum, checksum));
            } 
          } 
          for (ConfigurationKey key : NODES_TO_READ) {
            if (event.asStartElement().getName().getLocalPart().equals(key.getFileKey())) {
              String value = eventReader.getElementText();
              value = StringUtils.isBlank(value) ? null : value;
              items.put(key, new ConfigurationParameter(value, value));
              break;
            } 
          } 
        } 
      } 
      eventReader.close();
      in.close();
    } catch (XMLStreamException|IOException e) {
      LOGGER.error("readConfig(): Error reading config xml file -> " + e.getMessage(), (Throwable)e);
    } 
    return items;
  }
  
  private void validateChecksum(Map<ConfigurationKey, ConfigurationParameter> items, File file) {
    if (items.containsKey(ConfigurationKey.CHECKSUM)) {
      try {
        String checksumFromFile = items.get(ConfigurationKey.CHECKSUM).getValue();
        String fileContent = new String(Files.readAllBytes(file.toPath()));
        String ebikeContent = fileContent.split("<ebike.*>", 2)[1].split("</ebike>", 2)[0];
        String calculatedChecksum = calculateMd5Hex(ebikeContent);
        String logEntry = StringUtils.equals(checksumFromFile, calculatedChecksum) ? "Configuration file checksum matches" : "Configuration file checksum doesn't match";
        LOGGER.info(logEntry);
      } catch (IOException|NoSuchAlgorithmException ex) {
        LOGGER.warn("Error validating configuration file checksum", (Throwable)ex);
      } 
    } else {
      LOGGER.info("Configuration file doesn't contain a checksum");
    } 
  }

import com.gigatronik.dart2.framework.plugins.exceptions.ConfigurationNotFoundException;
import com.gigatronik.dart2.framework.util.Version;
import com.gigatronik.ebike.common.configuration.BikeIdentification;
import com.gigatronik.ebike.common.configuration.ComponentType;
import com.gigatronik.ebike.common.configuration.EcuIdentification;
import com.gigatronik.ebike.common.configuration.HmiIdentification;
import com.gigatronik.ebike.common.util.BoschConstants;
import java.io.IOException;
import java.util.ArrayList;
import java.util.List;
import org.apache.commons.lang3.StringUtils;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

public class ContainerToBikeCompatibilityValidator {
  private static final Logger LOGGER = LoggerFactory.getLogger(ContainerToBikeCompatibilityValidator.class);
  
  private final BikeIdentification bikeIdentification;
  
  private final ContainerConfiguration containerConfiguration;
  
  private final RightsManager rightsManager;
  
  public ContainerToBikeCompatibilityValidator(BikeIdentification bikeIdentification, ContainerConfiguration containerConfiguration, RightsManager rightsManager) {
    this.bikeIdentification = bikeIdentification;
    this.containerConfiguration = containerConfiguration;
    this.rightsManager = rightsManager;
  }
  
  public Attribute.Value checkCompatibility() {
    Attribute.Value result = Attribute.Value.OK;
    try {
      checkCompatibilityTry();
      LOGGER.info("Loaded container is compatible with connected bike!");
    } catch (ContainerBikeIncompatibleException ex) {
      result = handleNegativeCheckTermination(ex);
    } 
    return result;
  }
  
  private void checkCompatibilityTry() throws ContainerBikeIncompatibleException {
    if (this.containerConfiguration.isLoaded() && this.bikeIdentification.isHmiConnected()) {
      checkContainer();
      checkHmi();
      checkDu();
      checkBms1();
      checkBms2();
      checkAbs();
    } 
  }
  
  private Attribute.Value handleNegativeCheckTermination(ContainerBikeIncompatibleException ex) {
    LOGGER.warn("Loaded container is not compatible with connected bike!", (Throwable)ex);
    Attribute.Value failure = ex.getFailureDetail();
    String incompatibleComponent = ex.getIncompatibleComponent();
    if (StringUtils.isNoneBlank(incompatibleComponent))
      failure.setComponent(incompatibleComponent); 
    return failure;
  }
  
  private void checkContainer() throws ContainerBikeIncompatibleException {
    if (this.bikeIdentification.isDuConnected()) {
      BoschContainerController.ContainerType containerType = this.containerConfiguration.getContainerController().getContainerType();
      if (containerType == BoschContainerController.ContainerType.CFF2) {
        checkCff2Compatibility();
      } else {
        checkCff3Compatibility();
      } 
    } 
  }
  
  private void checkCff2Compatibility() throws ContainerBikeIncompatibleException {
    if (this.rightsManager.getUserLevel() == UserLevel.IBD) {
      checkCff2CompatibilityIbd();
    } else {
      checkCff2CompatibilityOem();
    } 
  }
  
  private void checkCff2CompatibilityIbd() throws ContainerBikeIncompatibleException {
    if (!this.bikeIdentification.getDuIdentification().getComponentType().isBDU1xx() || !this.bikeIdentification.getDuIdentification().isBootLoader()) {
      String applicationIdBike = this.bikeIdentification.getApplicationId();
      if (StringUtils.isBlank(applicationIdBike))
        throw new ContainerBikeIncompatibleException("No application (id) installed on DU!", Attribute.Value.BIKE_IDENTIFICATION_ERROR, this.bikeIdentification.getDuIdentification().getComponentType()); 
      if (!this.containerConfiguration.getContainerController().getCff2Interface().isApplicationInContainer(applicationIdBike))
        throw new ContainerBikeIncompatibleException("Installed application ID on DU '" + applicationIdBike + "' is not in container!", Attribute.Value.CONTAINER_APPLICATION_PARAMETER_MISSING, this.bikeIdentification.getDuIdentification().getComponentType()); 
    } 
  }
  
  private void checkCff2CompatibilityOem() throws ContainerBikeIncompatibleException {
    List<Cff2_DuParameterReply> duParameterReplies = this.containerConfiguration.getContainerController().getCff2Interface().getDuParameterReplies();
    if (duParameterReplies.isEmpty())
      throw new ContainerBikeIncompatibleException("No applications in container!", Attribute.Value.CONTAINER_APPLICATION_PARAMETER_MISSING); 
    if (StringUtils.isNotBlank(this.bikeIdentification.getApplicationId()))
      checkAllProductLinesAndBikeTypesCff2(); 
  }
  
  private void checkAllProductLinesAndBikeTypesCff2() throws ContainerBikeIncompatibleException {
    for (Cff2_DuParameterReply duParameterReply : (Iterable<Cff2_DuParameterReply>)this.containerConfiguration.getContainerController().getCff2Interface().getDuParameterReplies()) {
      checkProductLine(duParameterReply);
      checkBikeType(duParameterReply);
    } 
  }
  
  private void checkProductLine(Cff2_DuParameterReply duParameterReply) throws ContainerBikeIncompatibleException {
    String bikeProductLine = this.bikeIdentification.getProductLine().getOdxName();
    String containerProductLine = duParameterReply.getParProductLine();
    if (!StringUtils.equals(bikeProductLine, containerProductLine))
      throw new ContainerBikeIncompatibleException("Bike productLine '" + bikeProductLine + "' does not match container <du-parameter-duParameterReply> '" + containerProductLine + "'", Attribute.Value.CONTAINER_PRODUCT_LINE_MISSING); 
  }
  
  private void checkBikeType(Cff2_DuParameterReply duParameterReply) throws ContainerBikeIncompatibleException {
    String bikeBikeType = this.bikeIdentification.getBikeType();
    String containerBikeType = duParameterReply.getParBikeType();
    if (!StringUtils.equals(bikeBikeType, containerBikeType))
      throw new ContainerBikeIncompatibleException("Bike bikeBikeType '" + bikeBikeType + "' does not match container <du-parameter-duParameterReply> '" + containerBikeType + "'", Attribute.Value.CONTAINER_BIKE_TYPE_MISSING); 
  }
  
  private void checkCff3Compatibility() throws ContainerBikeIncompatibleException {
    CompatibleApplicationIdsCheckResult matches = getApplicationIdsFromContainerThatMatchBikeProductLineAndBikeType();
    if (matches.getCompatibleApplicationIds().isEmpty())
      throw new ContainerBikeIncompatibleException("No compatible application (id) in container for productLine='" + this.bikeIdentification.getProductLine() + "' & bikeType='" + this.bikeIdentification.getBikeType() + "'!", matches.getFailureReason()); 
    if (!isValidContainerConfigurationPossible(matches.getCompatibleApplicationIds()))
      throw new ContainerBikeIncompatibleException("No compatible application in container for bike", Attribute.Value.CONTAINER_HARDWARE_CONFIGURATION_INVALID, createStringWithAllEcuNamesAndBikeModel()); 
  }
  
  private CompatibleApplicationIdsCheckResult getApplicationIdsFromContainerThatMatchBikeProductLineAndBikeType() {
    CompatibleApplicationIdsCheckResult result = new CompatibleApplicationIdsCheckResult();
    List<DuParameterReplyType> duParameterReplies = this.containerConfiguration.getContainerController().getCff3Interface().getDuParameterReplies();
    for (DuParameterReplyType duParameterReply : duParameterReplies) {
      result.updatePotentialFailureReason(Attribute.Value.CONTAINER_PRODUCT_LINE_MISSING);
      if (StringUtils.equals(this.bikeIdentification.getProductLine().getOdxName(), duParameterReply.getParProductLine())) {
        result.updatePotentialFailureReason(Attribute.Value.CONTAINER_BIKE_TYPE_MISSING);
        if (StringUtils.equals(this.bikeIdentification.getBikeType(), duParameterReply.getParBikeType()))
          result.addCompatibleApplicationId(duParameterReply.getParId()); 
      } 
    } 
    return result;
  }
  
  private boolean isValidContainerConfigurationPossible(List<String> bikeCompatibleApplicationIds) {
    for (String applicationId : bikeCompatibleApplicationIds) {
      if (isApplicationCompatibleWithAllEcus(applicationId))
        return true; 
    } 
    return false;
  }
  
  private boolean isApplicationCompatibleWithAllEcus(String applicationId) {
    for (ComponentType componentType : (Iterable<ComponentType>)this.bikeIdentification.getAllConnectedComponentTypes()) {
      if (!isApplicationCompatibleWithEcu(applicationId, componentType)) {
        LOGGER.debug("Application ID '{}' is incompatible with ECU '{}'", applicationId, componentType);
        return false;
      } 
    } 
    LOGGER.debug("Application ID '{}' is compatible with all connected ECUs", applicationId);
    return true;
  }
  
  private boolean isApplicationCompatibleWithEcu(String applicationId, ComponentType componentType) {
    if (!componentType.isBUI27x())
      return isApplicationCompatibleWithNonNyonEcu(applicationId, componentType); 
    return isApplicationCompatibleWithNyon(applicationId, componentType);
  }
  
  private boolean isApplicationCompatibleWithNonNyonEcu(String applicationId, ComponentType componentType) {
    String swMinReqVersion = this.containerConfiguration.getContainerController().getCff3Interface().getMinRequiredVersionUpdateConstraintCff3(BoschConstants.SwType.SW_TYPE_CFF2_APP, componentType, applicationId);
    return isCompatible(swMinReqVersion);
  }
  
  private boolean isApplicationCompatibleWithNyon(String applicationId, ComponentType componentType) {
    String swMinReqVersionFrontEnd = this.containerConfiguration.getContainerController().getCff3Interface().getMinRequiredVersionUpdateConstraintCff3(BoschConstants.SwType.SW_TYPE_CFF2_APP_FRONTEND, componentType, applicationId);
    String swMinReqVersionBackEnd = this.containerConfiguration.getContainerController().getCff3Interface().getMinRequiredVersionUpdateConstraintCff3(BoschConstants.SwType.SW_TYPE_CFF2_APP_BACKEND, componentType, applicationId);
    return (isCompatible(swMinReqVersionFrontEnd) && isCompatible(swMinReqVersionBackEnd));
  }
  
  private boolean isCompatible(String swMinReqVersion) {
    return (StringUtils.isNoneBlank(swMinReqVersion) && !swMinReqVersion.contains("-"));
  }
  
  private String createStringWithAllEcuNamesAndBikeModel() {
    List<String> connectedComponentNames = getComponentNamesOfAllConnectedEcus();
    String modelName = BikeModelTranslationHelper.createTranslatedBikeModel(this.containerConfiguration.getBikeModel());
    if (StringUtils.isNotBlank(modelName))
      connectedComponentNames.add(modelName); 
    return StringUtils.join(connectedComponentNames, ", ");
  }
  
  private void checkHmi() throws ContainerBikeIncompatibleException {
    if (shouldHmiBeChecked()) {
      checkComponentFirmware((EcuIdentification)this.bikeIdentification.getHmiIdentification());
      checkComponentHardwareVariant((EcuIdentification)this.bikeIdentification.getHmiIdentification());
    } 
  }
  
  private void checkDu() throws ContainerBikeIncompatibleException {
    if (this.bikeIdentification.getDuIdentification().isConnected()) {
      checkComponentFirmware((EcuIdentification)this.bikeIdentification.getDuIdentification());
      checkComponentHardwareVariant((EcuIdentification)this.bikeIdentification.getDuIdentification());
    } 
  }
  
  private void checkBms1() throws ContainerBikeIncompatibleException {
    if (this.bikeIdentification.getBms1Identification().isConnected()) {
      checkComponentFirmware((EcuIdentification)this.bikeIdentification.getBms1Identification());
      checkComponentHardwareVariant((EcuIdentification)this.bikeIdentification.getBms1Identification());
    } 
  }
  
  private void checkBms2() throws ContainerBikeIncompatibleException {
    if (this.bikeIdentification.getBms2Identification().isConnected()) {
      checkComponentFirmware((EcuIdentification)this.bikeIdentification.getBms2Identification());
      checkComponentHardwareVariant((EcuIdentification)this.bikeIdentification.getBms2Identification());
    } 
  }
  
  private void checkAbs() throws ContainerBikeIncompatibleException {
    if (this.bikeIdentification.getAbsIdentification().isConnected()) {
      checkComponentFirmware(this.bikeIdentification.getAbsIdentification());
      checkComponentHardwareVariant(this.bikeIdentification.getAbsIdentification());
    } 
  }
  
  private void checkComponentFirmware(EcuIdentification ecuIdentification) throws ContainerBikeIncompatibleException {
    if (!ecuIdentification.getComponentType().isBUI27x()) {
      checkStandardComponentFirmware(ecuIdentification);
    } else if (ecuIdentification instanceof HmiIdentification) {
      checkNyonFirmware((HmiIdentification)ecuIdentification);
    } else {
      throw new ContainerBikeIncompatibleException("ECU identification's corrupt! Firmware for component '" + ecuIdentification.getComponentType() + "' could not be checked!", Attribute.Value.BIKE_IDENTIFICATION_ERROR);
    } 
  }
  
  private void checkNyonFirmware(HmiIdentification hmiIdentification) throws ContainerBikeIncompatibleException {
    try {
      checkFrontEndFirmware(hmiIdentification);
      checkBackEndFirmware(hmiIdentification);
    } catch (ContainerBikeIncompatibleException ex) {
      handleNyonFirmwareCheckFailed(hmiIdentification, ex);
    } 
  }
  
  private void checkFrontEndFirmware(HmiIdentification hmiIdentification) throws ContainerBikeIncompatibleException {
    String ecuFrontEndFirmwareVersion = hmiIdentification.getFrontendSwVersion();
    String containerFrontEndFirmwareVersion = this.containerConfiguration.getContainerController().getBoschContainerCommonInterface().getSwVersion(BoschConstants.SwType.SW_TYPE_CFF2_APP_FRONTEND, hmiIdentification.getComponentType());
    checkComponentFirmware(ecuFrontEndFirmwareVersion, containerFrontEndFirmwareVersion, hmiIdentification.getComponentType());
  }
  
  private void checkBackEndFirmware(HmiIdentification hmiIdentification) throws ContainerBikeIncompatibleException {
    String ecuBackEndFirmwareVersion = hmiIdentification.getSwVersion();
    String containerBackEndFirmwareVersion = this.containerConfiguration.getContainerController().getBoschContainerCommonInterface().getSwVersion(BoschConstants.SwType.SW_TYPE_CFF2_APP_BACKEND, hmiIdentification.getComponentType());
    checkComponentFirmware(ecuBackEndFirmwareVersion, containerBackEndFirmwareVersion, hmiIdentification.getComponentType());
  }
  
  private void handleNyonFirmwareCheckFailed(HmiIdentification hmiIdentification, ContainerBikeIncompatibleException ex) throws ContainerBikeIncompatibleException {
    Attribute.Value adaptedResult = adaptNyonFirmwareCheckResult(ex.getFailureDetail(), (EcuIdentification)hmiIdentification);
    if (adaptedResult != Attribute.Value.OK)
      throw ex; 
    LOGGER.debug("Nyon firmware check has been skipped!");
  }
  
  private Attribute.Value adaptNyonFirmwareCheckResult(Attribute.Value checkResult, EcuIdentification ecuIdentification) {
    if (checkResult == Attribute.Value.BIKE_IDENTIFICATION_ERROR && ecuIdentification.isBootLoader())
      return Attribute.Value.OK; 
    return checkResult;
  }
  
  private void checkStandardComponentFirmware(EcuIdentification ecuIdentification) throws ContainerBikeIncompatibleException {
    String ecuFirmwareVersion = ecuIdentification.getSwVersion();
    String containerFirmwareVersion = this.containerConfiguration.getContainerController().getBoschContainerCommonInterface().getSwVersion(ecuIdentification.getComponentType());
    checkComponentFirmware(ecuFirmwareVersion, containerFirmwareVersion, ecuIdentification.getComponentType());
  }
  
  private void checkComponentFirmware(String ecuFirmwareVersion, String containerFirmwareVersion, ComponentType componentType) throws ContainerBikeIncompatibleException {
    Version ecuVersion = convertStringToVersion(ecuFirmwareVersion);
    if (ecuVersion == null)
      throw new ContainerBikeIncompatibleException("ECU firmware version '" + ecuFirmwareVersion + "' for component '" + componentType + "'is invalid!", Attribute.Value.BIKE_IDENTIFICATION_ERROR, componentType); 
    Version cffVersion = convertStringToVersion(containerFirmwareVersion);
    if (cffVersion == null)
      throw new ContainerBikeIncompatibleException("Container firmware version '" + containerFirmwareVersion + "' for component '" + componentType + "'is invalid!", Attribute.Value.CONTAINER_SOFTWARE_MISSING, componentType); 
    if (ecuVersion.compareTo(cffVersion) > 0)
      throw new ContainerBikeIncompatibleException("ECU firmware version '" + ecuFirmwareVersion + "' is greater than container firmware version '" + containerFirmwareVersion + "'!", Attribute.Value.CONTAINER_SOFTWARE_OUTDATED, componentType); 
  }
  
  private Version convertStringToVersion(String versionString) {
    Version version = null;
    try {
      version = Version.parse(versionString);
    } catch (IOException|NumberFormatException ex) {
      LOGGER.warn("String '{}' conversion to version object not possible!", (Throwable)ex);
    } 
    return version;
  }
  
  private void checkComponentHardwareVariant(EcuIdentification ecuIdentification) throws ContainerBikeIncompatibleException {
    ComponentType component = ecuIdentification.getComponentType();
    String bikeHwVariant = ecuIdentification.getHwVariant();
    if (component.isBDUVictor()) {
      checkVictorDuHardwareVariant(component, bikeHwVariant);
    } else if (component.isBUI27x()) {
      checkNyonHardwareVariant(component, bikeHwVariant);
    } else {
      checkHardwareVariant(bikeHwVariant, BoschConstants.SwType.SW_TYPE_CFF2_APP, component);
    } 
  }
  
  private void checkVictorDuHardwareVariant(ComponentType component, String bikeHwVariant) throws ContainerBikeIncompatibleException {
    checkHardwareVariant(bikeHwVariant, BoschConstants.SwType.SW_TYPE_CFF2_SWIL_DRV, component);
    checkHardwareVariant(bikeHwVariant, BoschConstants.SwType.SW_TYPE_CFF2_APP, component);
  }
  
  private void checkNyonHardwareVariant(ComponentType component, String bikeHwVariant) throws ContainerBikeIncompatibleException {
    checkHardwareVariant(bikeHwVariant, BoschConstants.SwType.SW_TYPE_CFF2_APP_FRONTEND, component);
    checkHardwareVariant(bikeHwVariant, BoschConstants.SwType.SW_TYPE_CFF2_APP_BACKEND, component);
  }
  
  private void checkHardwareVariant(String bikeHwVariant, BoschConstants.SwType swType, ComponentType component) throws ContainerBikeIncompatibleException {
    if (StringUtils.isNotBlank(bikeHwVariant)) {
      String containerHwVariant = this.containerConfiguration.getContainerController().getHwVariant(component, swType);
      if (StringUtils.isBlank(containerHwVariant) || (!StringUtils.equals(bikeHwVariant, containerHwVariant) && !StringUtils.equals(bikeHwVariant.replace("HW_", ""), containerHwVariant)))
        throw new ContainerBikeIncompatibleException("Hardware variant check for component '" + component + "' failed! Bike='" + bikeHwVariant + "'; Container='" + containerHwVariant + "'", Attribute.Value.CONTAINER_HARDWARE_VARIANT_MISSING, component); 
    } 
  }
  
  Attribute.Value isCurrentCff3ConfigurationValid() {
    if (this.containerConfiguration.getContainerController().isCff3Loaded() && this.containerConfiguration.isConfigured() && !isApplicationCompatibleWithAllEcus(this.containerConfiguration.getConfiguredParId())) {
      Attribute.Value failure = Attribute.Value.CONTAINER_HARDWARE_CONFIGURATION_INVALID;
      failure.setComponent(createStringWithAllEcuNamesAndBikeModel());
      return failure;
    } 
    return Attribute.Value.OK;
  }
  
  private List<String> getComponentNamesOfAllConnectedEcus() {
    List<String> connectedComponentNames = new ArrayList<>();
    if (this.bikeIdentification.isHmiConnected())
      connectedComponentNames.add(this.bikeIdentification.getHmiIdentification().getComponentType().getProductName()); 
    if (this.bikeIdentification.isDuConnected())
      connectedComponentNames.add(this.bikeIdentification.getProductLineDisplayName()); 
    if (this.bikeIdentification.isBms1Connected())
      connectedComponentNames.add(this.bikeIdentification.getBms1Identification().getComponentType().getProductName()); 
    if (this.bikeIdentification.isBms2Connected())
      connectedComponentNames.add(this.bikeIdentification.getBms2Identification().getComponentType().getProductName()); 
    if (this.bikeIdentification.isAbsConnected())
      connectedComponentNames.add(this.bikeIdentification.getAbsIdentification().getComponentType().getProductName()); 
    return connectedComponentNames;
  }
  
  private boolean shouldHmiBeChecked() {
    if (this.bikeIdentification.getHmiIdentification().getComponentType().isCUI1xx())
      return shouldCobiBeChecked(); 
    return this.bikeIdentification.getHmiIdentification().isConnected();
  }
  
  private boolean shouldCobiBeChecked() {
    try {
      return SettingsProxy.getInstance().getGlobalSettings().getBoolean("cobi_flash_active");
    } catch (ConfigurationNotFoundException ex) {
      LOGGER.warn("Flag for Cobi flash activation not found in settings", ex.getMessage());
      return false;
    } 
  }
  
  private class CompatibleApplicationIdsCheckResult {
    private List<String> compatibleApplicationIds = new ArrayList<>();
    
    private Attribute.Value failureReason = Attribute.Value.CONTAINER_APPLICATION_PARAMETER_MISSING;
    
    private void addCompatibleApplicationId(String applicationId) {
      if (StringUtils.isNoneBlank(applicationId))
        this.compatibleApplicationIds.add(applicationId); 
    }
    
    private List<String> getCompatibleApplicationIds() {
      return this.compatibleApplicationIds;
    }
    
    private void updatePotentialFailureReason(Attribute.Value reason) {
      if (this.failureReason != Attribute.Value.CONTAINER_BIKE_TYPE_MISSING)
        this.failureReason = reason; 
    }
    
    private Attribute.Value getFailureReason() {
      return this.failureReason;
    }
  }
}

package com.gigatronik.beds.service.configuration.impl;

import com.gigatronik.beds.service.communication.api.CommunicationManager;
import com.gigatronik.beds.service.configuration.event.HardwareConnectionChangedEvent;
import com.gigatronik.beds.service.configuration.publisher.EcuConnectionStateChangedEventPublisher;
import com.gigatronik.beds.service.ecu.api.Du;
import com.gigatronik.ebike.common.configuration.BikeIdentification;
import com.gigatronik.ebike.common.configuration.ComplexResponse;
import com.gigatronik.ebike.common.configuration.Constants;
import com.gigatronik.ebike.common.configuration.EshiftIdentification;
import java.awt.EventQueue;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

class GearPropertiesObserver extends Thread {
  private static final Logger LOGGER = LoggerFactory.getLogger(BikeIdentificationController.class);
  
  private static final int MAXIMUM_POLLING_TIME = 15000;
  
  private static final int POLLING_INTERVAL = 1000;
  
  private final BikeIdentificationController bikeIdentifcationController;
  
  private final EshiftIdentification eshiftIdentification;
  
  private final EcuConnectionStateChangedEventPublisher publisher;
  
  private final Du du;
  
  public GearPropertiesObserver(CommunicationManager communicationManager, BikeIdentification bikeIdentification, BikeIdentificationController bikeIdentificationController, EcuConnectionStateChangedEventPublisher configurationEventPublisher) {
    this.du = communicationManager.getDuCom();
    this.bikeIdentifcationController = bikeIdentificationController;
    this.eshiftIdentification = bikeIdentification.getEShiftIdentification();
    this.publisher = configurationEventPublisher;
  }
  
  public void run() {
    Thread.currentThread().setName("GearPropertyObserver-" + Thread.currentThread().getId());
    LOGGER.debug("Start polling gear properties of DU ...");
    long pollingEndTime = System.currentTimeMillis() + 15000L;
    long currentTime = System.currentTimeMillis();
    while (currentTime <= pollingEndTime && !isInterrupted()) {
      ComplexResponse response = this.du.readGearPropertiesReceivingState();
      if (response.isNegative() || response.toString().equals("TRUE")) {
        LOGGER.debug("Stop polling & re-read the affected service parameters");
        this.eshiftIdentification.setGearPropertiesReceived(EshiftIdentification.GearPropertiesReceived.TRUE_AVAILABLE);
        break;
      } 
      try {
        Thread.sleep(1000L);
      } catch (InterruptedException ex) {
        LOGGER.warn("Gear proptery observer thread interrupted during sleeping", (Throwable)ex);
        Thread.currentThread().interrupt();
        break;
      } 
      currentTime = System.currentTimeMillis();
    } 
    if (!isInterrupted()) {
      if (currentTime > pollingEndTime) {
        LOGGER.info("DU's still not ready --> Stop polling and display 'out of range'!");
        this.eshiftIdentification.setGearPropertiesReceived(EshiftIdentification.GearPropertiesReceived.TRUE_NOT_AVAILABLE);
      } else {
        this.bikeIdentifcationController.readIdentificationDataFromEcu(Constants.EcuType.ESHIFT, Constants.EcuMode.NONE);
      } 
      HardwareConnectionChangedEvent hardwareConnectionChangedEvent = new HardwareConnectionChangedEvent(HardwareConnectionChangedEvent.HardwareConnectionStateChange.ECU_CONNECTED, HardwareConnectionChangedEvent.HardwareSystem.BIKE_SYSTEM, Constants.EcuType.ESHIFT, Constants.EcuMode.APP);
      EventQueue.invokeLater(() -> this.publisher.publishEvent(hardwareConnectionChangedEvent));
    } 
  }
}

package com.gigatronik.beds.service.configuration.popup;

import com.gigatronik.beds.service.configuration.ConfigurationManager;
import com.gigatronik.beds.service.configuration.api.HardwareConnectionChangesListener;
import com.gigatronik.dart2.bosch.service.internationalization.ResourceBundleFactory;
import com.gigatronik.dart2.bosch.service.internationalization.Utf8ResourceBundleFormatted;
import com.gigatronik.ebike.gui.testinterface.TestableJOptionPane;
import java.awt.EventQueue;
import java.lang.reflect.InvocationTargetException;
import javax.swing.JDialog;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

abstract class AbstractPopUpHandler implements HardwareConnectionChangesListener {
  private static final Logger LOGGER = LoggerFactory.getLogger(AbstractPopUpHandler.class);
  
  static final Utf8ResourceBundleFormatted BUNDLE = ResourceBundleFactory.newDefaultResourceBundle();
  
  private JDialog popUp;
  
  AbstractPopUpHandler(ConfigurationManager configurationManager) {
    configurationManager.addListener(this);
  }
  
  void showPopUpBlocking(String message, String title, int messageType) throws InvocationTargetException, InterruptedException {
    showPop(message, title, messageType, true);
  }
  
  void showPopUpNonBlocking(String message, String title, int messageType) {
    try {
      showPop(message, title, messageType, false);
    } catch (InvocationTargetException|InterruptedException ex) {
      LOGGER.warn("This cannot occur ...", (Throwable)ex);
      Thread.currentThread().interrupt();
    } 
  }
  
  private void showPop(String message, String title, int messageType, boolean blocking) throws InvocationTargetException, InterruptedException {
    closeDialogIfOpen();
    createAndDisplayDialog(message, title, messageType, blocking);
  }
  
  void createAndDisplayDialog(String message, String title, int messageType, boolean blocking) throws InvocationTargetException, InterruptedException {
    Runnable task = () -> createAndDisplayDialogDelegated(message, title, messageType);
    if (blocking) {
      EventQueue.invokeAndWait(task);
    } else {
      EventQueue.invokeLater(task);
    } 
  }
  
  private void createAndDisplayDialogDelegated(String message, String title, int messageType) {
    TestableJOptionPane messagePane = new TestableJOptionPane(message, messageType);
    this.popUp = messagePane.createDialog(TestableJOptionPane.getDefaultParentFrame(), title);
    LOGGER.debug("Open popup");
    this.popUp.setVisible(true);
  }
  
  void closeDialogIfOpen() {

public class ContainerBikeIncompatibilityPopUpHandler extends AbstractPopUpHandler implements ContainerConfigurationChangesListener {
  private static final Logger LOGGER = LoggerFactory.getLogger(ContainerBikeIncompatibilityPopUpHandler.class);
  
  public ContainerBikeIncompatibilityPopUpHandler(ConfigurationManager configurationManager) {
    super(configurationManager);
  }
  
  public void onContainerConfigurationChanged(ContainerConfigurationChangedEvent event) {
    LOGGER.debug("ContainerConfigurationChangedEvent received: '{}'", event);
    if (hasContainerBeenLoaded(event) && wasOperationSuccessful(event) && !isContainerCompatibleWithBike(event)) {
      showPopUp(event);
    } else if (hasContainerBeenUnloaded(event)) {
      closeDialogIfOpen();
    } 
  }
  
  public void onHardwareConnectionChanged(HardwareConnectionChangedEvent event) {
    LOGGER.debug("HardwareConnectionChangedEvent received: '{}'", event);
    if (hasSystemBeenConnected(event) && isBike(event) && !isContainerCompatibleWithBike(event)) {
      showPopUp(event);
    } else if (hasSystemBeenDisconnected(event) && isBike(event)) {
      closeDialogIfOpen();
    } 
  }
  
  private void showPopUp(AbstractConfigurationEvent event) {
    Attribute.Value failureReason = event.getAttribute(Attribute.Key.CONTAINER_COMPATIBLE_WITH_BIKE);
    showPopUpNonBlocking(failureReason.getInfo(), BUNDLE.getString("ContainerValidation.Error"), 2);
  }
  
  private boolean isBike(HardwareConnectionChangedEvent event) {
    return (event.getSystemType() == HardwareConnectionChangedEvent.HardwareSystem.BIKE_SYSTEM);
  }
  
  private boolean hasSystemBeenConnected(HardwareConnectionChangedEvent event) {
    return (event.getAction() == HardwareConnectionChangedEvent.HardwareConnectionStateChange.SYSTEM_CONNECTED || 
      event.getAction() == HardwareConnectionChangedEvent.HardwareConnectionStateChange.SYSTEM_RECONNECTED);
  }
  
  private boolean hasSystemBeenDisconnected(HardwareConnectionChangedEvent event) {
    return (event.getAction() == HardwareConnectionChangedEvent.HardwareConnectionStateChange.SYSTEM_DISCONNECTED);
  }
  
  private boolean hasContainerBeenLoaded(ContainerConfigurationChangedEvent event) {
    return (event.getAction() == ContainerConfigurationChangedEvent.Action.LOADED);
  }
  
  private boolean hasContainerBeenUnloaded(ContainerConfigurationChangedEvent event) {
    return (event.getAction() == ContainerConfigurationChangedEvent.Action.UNLOADED);
  }
  
  private boolean wasOperationSuccessful(AbstractConfigurationEvent event) {
    return (event.getAttribute(Attribute.Key.OPERATION_RESULT) == null || 
      event.getAttribute(Attribute.Key.OPERATION_RESULT) == Attribute.Value.OK);
  }
  
  private boolean isContainerCompatibleWithBike(AbstractConfigurationEvent event) {
    return (event.getAttribute(Attribute.Key.CONTAINER_COMPATIBLE_WITH_BIKE) != null && 
      event.getAttribute(Attribute.Key.CONTAINER_COMPATIBLE_WITH_BIKE) == Attribute.Value.OK);
  }
}

  private static final List<String> CALLBACK_LIST_TAIL = Arrays.asList("");
