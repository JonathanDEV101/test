            if (getRequestorType().equals(Authenticator.RequestorType.PROXY))
              return (UpdateServerConnection.this.proxyAuthentication != null) ? UpdateServerConnection.this.proxyAuthentication : new PasswordAuthentication("", "".toCharArray()); 
            return (UpdateServerConnection.this.serverAuthentication != null) ? UpdateServerConnection.this.serverAuthentication : new PasswordAuthentication("", "".toCharArray());
          }
        });
  }
  
  private void setProxyAuthentication() {
    ProxyManager proxyManager = ProxyManager.getInstance();
    String username = proxyManager.getUsername();
    String password = proxyManager.getPassword();
    if (StringUtils.isNotEmpty(username) && StringUtils.isNotEmpty(password))
      this.proxyAuthentication = new PasswordAuthentication(username, password.toCharArray()); 
  }

  public HttpURLConnection uploadRequestContainer(String address, File file) {
    synchronized (LOCK) {
      HttpURLConnection conn = getConnection(appendRandomNumberToPath(address));
      try {
        conn.setRequestMethod("POST");
      } catch (ProtocolException ex) {
        LOGGER.error("Could not setRequestMethod 'POST'", (Throwable)ex);
      } 
      conn.setRequestProperty("Content-Type", "text/plain; charset=utf-8");
      LOGGER.info("Write file {} to address {}", file, address);
      try(OutputStream os = conn.getOutputStream(); BufferedInputStream bfis = new BufferedInputStream(new FileInputStream(file))) {
        byte[] buffer = new byte[1024];
        int length;
        while ((length = bfis.read(buffer)) > 0)
          os.write(buffer, 0, length); 
        DownloadStreamProgress.setTotalDownloadSize(conn);
        os.flush();
      } catch (IOException ex) {
        LOGGER.error("Error while uploading file {} to address {}", file, address, ex);
      } 
      return conn;
    } 
  }
  
  public String downloadContainer(String path, HttpURLConnection conn) {
    synchronized (LOCK) {
      if (path != null && conn != null)
        try(InputStream in = conn.getInputStream(); OutputStream out = new BufferedOutputStream(new FileOutputStream(path))) {
          LOGGER.info("Receive response file, save: {}", path);
          byte[] buffer = new byte[1024];
          int numRead;
          while ((numRead = in.read(buffer)) != -1) {
            out.write(buffer, 0, numRead);
            DownloadStreamProgress.addDownloadedBytes(numRead);
          } 
          out.flush();
        } catch (IOException ex) {
          LOGGER.error("Error downloading File to {}", path, ex);
        }  
      return path;
    } 
  }
  
  private boolean uploadFile(String url, File file) {
    if (!StringUtils.isBlank(url) && file != null) {
      HttpURLConnection conn = getConnection(url);
      try {
        if (conn != null) {
          conn.setRequestMethod("POST");
          conn.setRequestProperty("Content-Type", "text/plain; charset=utf-8");
          return upload(conn, file);
        } 
      } catch (ProtocolException ex) {
        LOGGER.error("Failed to set RequestMethod 'POST'", (Throwable)ex);
      } 
    } 
    return false;
  }
  
  private boolean upload(HttpURLConnection conn, File file) {
    synchronized (LOCK) {
      try (OutputStream os = conn.getOutputStream()) {
        LOGGER.info("Write file to Server...");
        try (BufferedInputStream bfis = new BufferedInputStream(new FileInputStream(file))) {
          byte[] buffer = new byte[1024];
          int length;
          while ((length = bfis.read(buffer)) > 0)
            os.write(buffer, 0, length); 
          DownloadStreamProgress.setTotalDownloadSize(conn);
          os.flush();
        } 
        boolean successful = (conn.getResponseCode() == 200);
        logServerResponse(conn);
        conn.disconnect();
        return successful;
      } 
    } 
  }
  
  private void logServerResponse(HttpURLConnection conn) throws IOException {
    StringBuilder stringBuilder;
    if (conn.getResponseCode() == 200) {
      LOGGER.info("Server returned HTTP-OK status");
      LOGGER.info("Successfully uploaded file to Server");
    } else {
      LOGGER.error("Server returned non-HTTP-OK status, Status Returned: {}", Integer.valueOf(conn.getResponseCode()));
    } 
    try (BufferedReader responseReader = new BufferedReader(new InputStreamReader(new BufferedInputStream(conn.getInputStream())))) {
      stringBuilder = new StringBuilder();
      String line;
      while ((line = responseReader.readLine()) != null)
        stringBuilder.append(line).append("\n"); 
    } 
    String response = stringBuilder.toString();
    LOGGER.info("Server returned file upload response: \n{}", response);
  }

    URI baseUrl;

    } catch (URISyntaxException|java.net.MalformedURLException ex) {
      LOGGER.error("No valid update server url", (Throwable)ex);
      return 404;
    } 
    return checkConnection("" + baseUrl + "/cff2/health_status");
  }
