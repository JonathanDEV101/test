import com.gigatronik.dart2.bosch.remote.common.dto.v1.Dto;

abstract class AbstractRequest implements Request {
  private final Dto data;
  
  private long sendTime = 0L;
  
  private final int timeout;
  
  public AbstractRequest(Dto data, int timeout) {
    this.data = data;
    this.timeout = timeout;
  }
  
  public Dto getRequestDto() {
    return this.data;
  }
  
  public boolean isTimedOut() {
    return (System.currentTimeMillis() - this.sendTime > (long)this.timeout);
  }
  
  public Dto onSend() {
    this.sendTime = System.currentTimeMillis();
    return getRequestDto();
  }
  
  public final boolean handleResponse(Dto dto) {
    boolean handled = isResponseHandled(dto);
    if (handled) {
      onResponse(dto);
      synchronized (this) {
        notifyAll();
      } 
    } 
    return handled;
  }
  
  protected int getTimeout() {
    return this.timeout;
  }
  
  protected abstract boolean isResponseHandled(Dto paramDto);
  
  protected abstract void onResponse(Dto paramDto);
}

      } else {
        compressImageJpeg(baos, image, (float)imageCompression / 100.0F);
        fileName = fileName + ".jpg";
      } 
      byte[] compressedImage = baos.toByteArray();
      byte[] imageZipped = zipImage(compressedImage, fileName);
      screenChangeDtoImpl.setImage(imageZipped);
      send((Dto)screenChangeDtoImpl);
    } catch (IOException e) {
      LOGGER.error("Error converting image", (Throwable)e);
    } 
  }
  
  private void compressImageJpeg(ByteArrayOutputStream baos, BufferedImage bi, float quality) throws IOException {
    if (quality > 1.0F) {
      quality = 1.0F;
    } else if (quality < 0.0F) {
      quality = 0.0F;
    } 
    ImageWriter iw = ImageIO.getImageWritersByFormatName("jpg").next();
    try (ImageOutputStream ios = ImageIO.createImageOutputStream(baos)) {
      iw.setOutput(ios);
      ImageWriteParam iwp = new JPEGImageWriteParam(I18NUtil.getLocale());
      iwp.setCompressionMode(2);
      iwp.setCompressionQuality(quality);
      iw.write(null, new IIOImage(bi, null, null), iwp);
    } finally {
      iw.dispose();
    } 
  }
  
  private void compressImagePng(ByteArrayOutputStream baos, BufferedImage bi) throws IOException {
    ImageWriter iw = ImageIO.getImageWritersByFormatName("png").next();
    try (ImageOutputStream ios = ImageIO.createImageOutputStream(baos)) {
      iw.setOutput(ios);
      iw.write(null, new IIOImage(bi, null, null), null);
    } finally {
      iw.dispose();
    } 
  }
  
  private byte[] zipImage(byte[] jpgImage, String fileName) {
    if (StringUtils.isBlank(fileName))
      fileName = "img.jpg"; 
    try (ByteArrayOutputStream bos = new ByteArrayOutputStream()) {
      LOGGER.trace("Size before zipping image: {}byte", Integer.valueOf(jpgImage.length));
      try (ZipOutputStream zos = new ZipOutputStream(bos)) {
        ZipEntry entry = new ZipEntry(fileName);
        zos.putNextEntry(entry);
        zos.write(jpgImage);
        zos.closeEntry();
        byte[] imageZipped = bos.toByteArray();
        LOGGER.trace("Size after zipping image: {}byte", Integer.valueOf(imageZipped.length));
        return imageZipped;
      } 
    } catch (IOException e) {
      LOGGER.error("Error zipping image", (Throwable)e);
      return jpgImage;
    } 
  }

    return false;
  }
  
  private boolean startWinLockService() {
    CancelableProgressDialog startLockserviceProgressDialog = displayLockServiceStartingProgress();
    try {
      this.lockStateMonitor.stopMonitor();
      LOGGER.debug("Start Lockservice...");
      long startTime = this.timeProvider.millis();
      WinServicesUtil.ServiceState state = this.lockServiceClient.startWinService();
      LOGGER.debug("Lockservice is starting, state={}, waiting...", state);
      while (state != WinServicesUtil.ServiceState.RUNNING && 
        !startLockserviceProgressDialog.isCanceledByUser() && 
        !isTimedOut(startTime)) {
        Thread.sleep(300L);
        state = this.lockServiceClient.getWinServiceState();
      } 
      if (state == WinServicesUtil.ServiceState.RUNNING) {
        LOGGER.debug("Lockservice started successfully");
        return true;
      } 
      if (startLockserviceProgressDialog.isCanceledByUser()) {
        LOGGER.debug("Lockservice start not completed: canceled by user");
      } else if (isTimedOut(startTime)) {
        LOGGER.debug("Lockservice start not completed: timeout");
      } else {
        LOGGER.debug("Lockservice start not completed, state={}", state);
      } 
    } catch (IOException ex) {
      LOGGER.error("Failed to start Lockservice", (Throwable)ex);
    } catch (InterruptedException ex) {
      LOGGER.error("Sleep interrupted on waiting for lockservice start", (Throwable)ex);
      Thread.currentThread().interrupt();
    } finally {
      startLockserviceProgressDialog.dispose();
      this.lockStateMonitor.startMonitor();
    } 
    return false;
  }

    return (this.lockServiceClient.getWinServiceLastState() == WinServicesUtil.ServiceState.RUNNING);
  }

  }

  public void onRx(String destinationId, Dto dto) {
    if (dto instanceof ConnectResponseDto) {
      handleConnectResponse(dto, destinationId);
    } else if (dto instanceof MouseEventDto) {
      handleMouseEvent(dto);
    } else if (dto instanceof ScreenChangeDto) {
      handleScreenChange(dto);
    } else if (dto instanceof DongleDto) {
      handleDongleDto((DongleDto)dto);
    } else if (dto instanceof SettingsDto) {
      handleSettings(dto);
    } else if (dto instanceof com.gigatronik.dart2.bosch.remote.common.dto.v1.DisconnectDto) {
      handleDisconnect();
    } else {
      super.onRx(destinationId, dto);
    } 
  }

  private void handleConnectResponse(Dto dto, String destinationId) {
    this.remSession.get().stopCheckDestinationIdTimer();
    ConnectResponseDto connectResponse = (ConnectResponseDto)dto;
    LOGGER.info("Connect Response received: {}", connectResponse);
    if (connectResponse.getResponse()) {
      prepareRemoteSession(connectResponse.getOperatingSystem(), connectResponse.getModifiers(), connectResponse.getScreens());
      startRemoteSession(connectResponse.isIbdDongle(), destinationId);
    } else {
      DialogFactory.getInstance().showMessageDialog(MainFrame.BUNDLE.getString("RDC.SessionRejectedDialog.message"), MainFrame.BUNDLE.getString("RDC.SessionRejectedDialog.title"), 2);
      regularResetSession();
    } 
  }

    if (RemoteScreen.isInitialized()) {

      this.mouse.setPosition(mouseEvent.getPosition(), bounds);
    } 
  }
  
  private void handleScreenChange(Dto dto) {
    ScreenChangeDto screenChange = (ScreenChangeDto)dto;
    byte[] imageUnzipped = unzipImage(screenChange.getImage());
    try (ByteArrayInputStream bais = new ByteArrayInputStream(imageUnzipped)) {
      BufferedImage image = ImageIO.read(bais);
      RemoteScreen screen = (RemoteScreen)this.screens[screenChange.getScreenID() - 1];
      if (RemoteScreen.isInitialized())
        screen.setDisplayImage(screenChange.getID(), image, screenChange.getArea()); 
    } catch (IOException e) {
      LOGGER.error("Error converting image", (Throwable)e);
    } 
  }

      dongleDecryptResponseDtoImpl.setId(request.getId());

      send((Dto)dongleDecryptResponseDtoImpl);
    }

package com.gigatronik.rdc.control.client;

import com.gigatronik.dart2.bosch.service.internationalization.ResourceBundleFactory;

import java.awt.event.ActionEvent;
import java.awt.event.ActionListener;
import javax.swing.Timer;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

public class SessionInactivityDetector implements ActionListener, AWTEventListener {
  private static final Logger LOGGER = LoggerFactory.getLogger(SessionInactivityDetector.class);
  
  private static final Utf8ResourceBundleFormatted BUNDLE = ResourceBundleFactory.newDefaultResourceBundle();
  
  private static final int TIME_INACTIVITY_WARNING = 840000;
  
  private static final int TIME_INACTIVITY_COUNTDOWN = 60000;
  
  private static final long KEY_EVENTS = 8L;
  
  private static final long MOUSE_EVENTS = 48L;
  
  private static final long USER_EVENTS = 56L;
  
  private final ServiceClient servClient;
  
  private final MainFrame frame;
  
  private Timer inactivityTimer;
  
  private Timer countdownTimer;
  
  private final Toolkit defaultToolkit = Toolkit.getDefaultToolkit();
  
  public SessionInactivityDetector(ServiceClient client, MainFrame frm) {
    this.servClient = client;
    this.frame = frm;
  }
  
  public void start() {
    LOGGER.info("Starting remote session inactivity listener");
    if (this.inactivityTimer == null)
      this.inactivityTimer = new Timer(840000, this); 
    this.inactivityTimer.start();
    this.defaultToolkit.addAWTEventListener(this, 56L);
  }
  
  public void stop() {
    LOGGER.info("Stopping remote session inactivity listener");
    this.inactivityTimer.stop();
    this.inactivityTimer = null;
    this.defaultToolkit.removeAWTEventListener(this);
  }
  
  public void eventDispatched(AWTEvent event) {
    if (event.getSource() == this.frame.getDisplayPanel().getDisplayLabel())
      if (this.inactivityTimer.isRunning()) {
        this.inactivityTimer.restart();
      } else {
        stopCountdownTimer();
        this.inactivityTimer.start();
      }  
  }
  
  private void onInactivityDetected() {
    LOGGER.info("Inactivity detected: Starting countdown timer to automatically close session if expired");
    this.inactivityTimer.stop();
    startCountdownTimer();
    DialogFactory.getInstance().showMessageDialog(this.frame, BUNDLE.getString("remoteSessionInactivity.warnMessage.text"), BUNDLE.getString("remoteSessionInactivity.warnMessage.title"), 2);
  }
  
  private void onCountdownExpired() {
    LOGGER.info("Countdown expired: remote session closed by inactivity");
    stopCountdownTimer();
    this.frame.dispose();
    this.servClient.onClose(BUNDLE.getString("remoteSessionInactivity.logDescription.onInactivity"));
    DialogFactory.getInstance().showMessageDialog(BUNDLE.getString("remoteSessionInactivity.sessionClosed.infoMessage.text"), BUNDLE.getString("remoteSessionInactivity.sessionClosed.infoMessage.title"), 1);
  }
  
  public void actionPerformed(ActionEvent e) {
    onInactivityDetected();
  }
  
  private void startCountdownTimer() {
    LOGGER.debug("Starting inactivity countdown timer");
    if (this.countdownTimer == null)
      this.countdownTimer = new Timer(60000, e -> onCountdownExpired()); 
    this.countdownTimer.start();
  }
  
  private void stopCountdownTimer() {
    LOGGER.debug("Stopping inactivity countdown timer");
    this.countdownTimer.stop();
    this.countdownTimer = null;
  }
}

import java.util.LinkedList;
import java.util.List;
import java.util.Set;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

public class LocalKeyboard extends AbstractDevice implements Keyboard, FlashStateListener {
  private static final Logger LOGGER = LoggerFactory.getLogger(LocalKeyboard.class);
  
  private final LocalRobot robot;
  
  private final Clipboard clipboard;
  
  protected final List<KeyListener> listeners = Collections.synchronizedList(new LinkedList<>());
  
  private final KeyMapper keyMapper = new KeyMapper();
  
  private final Set<Integer> pressedKeys = new HashSet<>();
  
  public LocalKeyboard(LocalRobot robot) {
    LOGGER.info("Creating LocalKeyboard...");
    this.robot = robot;
    this.clipboard = Toolkit.getDefaultToolkit().getSystemClipboard();
  }
  
  public void forceChange(KeyListener listener) {

  }
  
  public void subscribe(KeyListener listener) {
    this.listeners.add(listener);
  }
  
  public synchronized void executeEvent(int key, int action, int modifier) {
    if (key == 0 || action == 0) {
      LOGGER.warn("Keycode or action is 0. No key event executed!");
      return;
    } 
    if (this.flashProcess.get()) {
      LOGGER.debug("Current flash process in progress. Key event dropped!");
      return;
    } 
    LOGGER.debug("executeKeyEvent called with parameters: Keycode: {}, Action: {}, Modifier: {}", key, action, modifier);
    if (key < 0) {
      executeSpecialCharacter(key, modifier, action);
    } else {
      executeKey(key, modifier, action);
    } 
  }
  
  public void unsubscribe(KeyListener listener) {
    this.listeners.remove(listener);
  }
  
  private void releasePressedKeys() {
    for (Integer event : this.pressedKeys) {
      this.robot.keyRelease(event.intValue());
      LOGGER.debug("Release unreleased key (Keycode: {}) on session reset ", event);
    } 
    this.pressedKeys.clear();
  }
  
  private void executeKey(int key, int modifier, int action) {
    switch (action) {
      case 401:
        handleKeyPressedEvent(key, modifier);
        break;
      case 402:
        handleKeyReleasedEvent(key, modifier);
        break;
    } 
  }
  
  private void handleKeyReleasedEvent(int key, int modifier) {
    if (!this.pressedKeys.contains(Integer.valueOf(key)))
      handleKeyPressedEvent(key, modifier); 
    this.robot.keyRelease(key);
    LOGGER.debug("Key released: KeyCode: {}, KeyChar: {}, ModifierText: {}", key, 
        KeyEvent.getKeyText(key), KeyEvent.getModifiersExText(modifier));
    this.pressedKeys.remove(Integer.valueOf(key));
  }
  
  private void handleKeyPressedEvent(int key, int modifier) {
    this.robot.keyPress(key);
    LOGGER.debug("Key pressed: KeyCode: {}, KeyChar: {}, ModifierText: {}", key, 
        KeyEvent.getKeyText(key), KeyEvent.getModifiersExText(modifier));
    this.pressedKeys.add(Integer.valueOf(key));
  }
  
  private void executeSpecialCharacter(int key, int modifier, int action) {
    LOGGER.debug("Execute special character: KeyCode: {}, Modifier: {}, ModifierText: {}", key, modifier, 
        KeyEvent.getModifiersExText(modifier));
    if (action == 401)
      executeSpecialChar(this.keyMapper.getCustKeyChar(key)); 
  }
  
  private void executeSpecialChar(String chr) {
    releasePressedKeys();
    StringSelection selection = new StringSelection(chr);
    this.clipboard.setContents(selection, selection);
    this.robot.keyPress(17);
    this.robot.keyPress(86);
    this.robot.keyRelease(86);
    this.robot.keyRelease(17);
    LOGGER.info("Executed char {} ", chr);
  }

    synchronized (this) {
      if (this.executor == null || this.executor.isShutdown() || this.executor.isTerminated())
        this.executor = Executors.newSingleThreadScheduledExecutor(new BasicThreadFactory.Builder().namingPattern("MousePositionCaptureThread-%d").build());

    } 
  }

        } 
        this.executor = null;
        LOGGER.info("Mouse capturing terminated");
      } 
    } 
  }

  public void keyPress(int keycode) {
    LOGGER.trace("keyPress({})", Integer.valueOf(keycode));
    this.lockServiceManager.allowKeyEvent(401);
    this.delegate.keyPress(keycode);
  }
  
  public void keyRelease(int keycode) {
    LOGGER.trace("keyRelease({})", Integer.valueOf(keycode));

    LOGGER.trace("mousePress({})", Integer.valueOf(buttons));
    Integer button = getMouseSingleButton(buttons);
    if (button != null) {
      this.lockServiceManager.allowMouseEvent(501, button.intValue());
      this.delegate.mousePress(buttons);
    } else {
      runUnlocked(() -> this.delegate.mousePress(buttons));
    } 
  }
  
  public void mouseRelease(int buttons) {
    LOGGER.trace("mouseRelease({})", Integer.valueOf(buttons));
    Integer button = getMouseSingleButton(buttons);
    if (button != null) {
      this.lockServiceManager.allowMouseEvent(502, button.intValue());
      this.delegate.mouseRelease(buttons);
    } else {
      runUnlocked(() -> this.delegate.mouseRelease(buttons));
    } 
  }
  
  public void mouseWheel(int wheelAmt) {
    LOGGER.trace("mouseWheel({})", Integer.valueOf(wheelAmt));
    this.lockServiceManager.allowMouseEvent(507, 0);
    this.delegate.mouseWheel(wheelAmt);
  }
  
  public void mouseMove(int x, int y) {
    LOGGER.trace("mouseMove({}, {})", Integer.valueOf(x), Integer.valueOf(y));
    this.lockServiceManager.allowMouseEvent(503, 0);
    this.delegate.mouseMove(x, y);
  }
  
  private Integer getMouseSingleButton(int buttons) {
    switch (buttons) {
      case 1024:
        return 1;
      case 2048:
        return 2;
      case 4096:
        return 3;
    } 
    return null;
  }
  
  private void runUnlocked(Runnable runnable) {
    boolean lockstate = this.lockServiceManager.isInputLocked();
    if (lockstate) {
      this.lockServiceManager.stopLockStateMonitor();
      LOGGER.debug("Unlock user input to execute mouse wheel...");
      this.lockServiceManager.setLockState(false, false);
    } 
    try {
      runnable.run();
    } finally {
      if (lockstate) {
        this.lockServiceManager.setLockState(true, false);
        this.lockServiceManager.startLockStateMonitor();
        LOGGER.debug("User input locked after executing mouse event");
      } 
    } 
  }

      LOGGER.error("Image to compare is null");
      return rectangle;
    } 
    for (int x = 0; x < actualImage.getWidth(); x++) {
      for (int y = 0; y < actualImage.getHeight(); y++) {
        if (actualImage.getRGB(x, y) != expectedImage.getRGB(x, y)) {
          rectangle.add(x, y);
          rectangle.add(x + 1, y + 1);
        } 
      } 
    } 
    LOGGER.trace("Calculated result of screen area changed: x={}, y={}, width={}, height={}", rectangle.x, rectangle.y, rectangle.width, rectangle.height);

    LOGGER.debug("Capturing screen change on screen {}...", Integer.valueOf(this.screenID));
    BufferedImage currentScreenshot = takeScreenshot();
    BufferedImage changedImage = null;
    Rectangle changedBounds = null;
    if (this.screenshot == null || this.timeProvider.millis() - this.timeLastFullImage > this.fullScreenImageInterval) {
      LOGGER.debug("Sending fullimage of screen {}.", Integer.valueOf(this.screenID));
      this.timeLastFullImage = this.timeProvider.millis();
      changedImage = currentScreenshot;
      changedBounds = this.bounds;
    } else if (currentDisplayed == this.screenID) {
      LOGGER.debug("Check for screen changes on screen {}...", Integer.valueOf(this.screenID));
      Rectangle screenAreaChanged = getScreenAreaChanged(this.screenshot, currentScreenshot);

        changedImage = currentScreenshot.getSubimage(screenAreaChanged.x, screenAreaChanged.y, screenAreaChanged.width, screenAreaChanged.height);
        changedBounds = screenAreaChanged;
      } else {
        LOGGER.trace("No screen changes detected");
      } 
    } 
    if (changedImage != null) {
      this.screenshot = currentScreenshot;
      fireOnScreenAreaChanged(this.imageCount++, this.screenID, changedImage, changedBounds);
    } 
  }

  public synchronized void setDisplayImage(int id, BufferedImage image, Rectangle changedArea) {
    LOGGER.debug("Received image(id: {}) for screen{}...", Integer.valueOf(id), Integer.valueOf(this.screenID));
    if (id < this.currentID) {
      LOGGER.debug("Drop received image(id: {}) for screen{}. ID < currentID: {} ", id, this.screenID, this.currentID);
      return;
    } 
    if (this.bounds.height == image.getHeight() && this.bounds.width == image.getWidth()) {
      LOGGER.debug("Processing received fullimage(id: {}) for screen{}...", Integer.valueOf(id), Integer.valueOf(this.screenID));
      updateCurrentImage(id, image);
    } else {
      LOGGER.debug("Processing received subimage(id: {}) for screen{}...", Integer.valueOf(id), Integer.valueOf(this.screenID));
      processSubImage(id, image, changedArea);
    } 
    processUnprocessedSubimages();
    cleanUpSubImages();
    LOGGER.debug("Display image (currentID: {}) on screen{}", Integer.valueOf(this.currentID), Integer.valueOf(this.screenID));
    if (this.imageUpdated)
      displayCurrentImage(); 
  }
  
  private void processSubImage(int id, BufferedImage image, Rectangle changedArea) {
    if (id == this.currentID + 1) {
      LOGGER.debug("Update current fullimage (currentID: {}) by subimage(id:{})", Integer.valueOf(this.currentID), Integer.valueOf(id));
      updateCurrentImage(id, mergeImage(image, changedArea));
    } else if (id > this.currentID) {
      LOGGER.debug("Cache subimage (id: {}) to process later", Integer.valueOf(id));
      this.subimages.put(Integer.valueOf(id), image);
      this.subimageAreas.put(Integer.valueOf(id), changedArea);
    } 
  }
  
  private void updateCurrentImage(int id, BufferedImage image) {
    this.currentID = id;
    this.currentImage = image;
    this.imageUpdated = true;
  }

    this.displayManager.updateImage(this.screenID, this.currentImage, this.bounds);
    this.imageUpdated = false;
  }

  }
  
  private void processUnprocessedSubimages() {
    for (Map.Entry<Integer, BufferedImage> entry : this.subimages.entrySet()) {
      int key = entry.getKey();
      if (key == this.currentID + 1) {
        processSubImage(key, entry.getValue(), this.subimageAreas.get(Integer.valueOf(key)));
        processUnprocessedSubimages();
      } 
    } 
  }
  
  private void cleanUpSubImages() {
    List<Integer> trash = new ArrayList<>();
    for (Map.Entry<Integer, BufferedImage> entry : this.subimages.entrySet()) {
      int key = entry.getKey();
      if (key <= this.currentID)
        trash.add(Integer.valueOf(key)); 
    } 
    for (Integer id : trash) {
      this.subimages.remove(id);
      this.subimageAreas.remove(id);
    } 
  }
  
  private BufferedImage mergeImage(BufferedImage subimage, Rectangle changedArea) {
    LOGGER.debug("Merge subimage of screen{}", Integer.valueOf(this.screenID));
    BufferedImage mergedImage = this.currentImage;
    if (mergedImage != null && subimage != null) {
      Graphics g = mergedImage.getGraphics();
      g.drawImage(subimage, changedArea.x, changedArea.y, changedArea.width, changedArea.height, null);
    } 
    return mergedImage;
  }

    initialized = false;
    numOfScreens = 0;
    currentDisplayed = 1;
  }

    String mappedValues = this.mapping.getProperty(identifier, identifier);
    Matcher m = Pattern.compile("([0-9])-([0-9])-([0-9])").matcher(mappedValues);
    int resultKey = key;
    int resultModifier = modifier;
    int resultAction = action;
    if (m.matches()) {
      resultKey = Integer.decode(m.group(1));
      resultModifier = Integer.decode(m.group(2));
      resultAction = Integer.decode(m.group(3));
    } 
    KeyEventDtoImpl keyEventDtoImpl = new KeyEventDtoImpl();

    keyEventDtoImpl.setModifier(resultModifier);

    return (rawUserInfo == null) ? new String[0] : rawUserInfo.split(":", 2);

  }

    String[] userInfo = getUserInfo();
    return (userInfo.length > 0) ? URLDecoder.decode(userInfo[1], StandardCharsets.UTF_8) : null;
  }
  
  public int getPort() {
    return this.uri.getPort();
  }
  
  public String getHost() {
    return this.uri.getHost();
  }
  
  public String toString() {
    return "Host: " + getHost() + ", Port: " + getPort() + ", User: " + getUsername() + ", Password: " + ((getPassword() == null) ? "none" : "***");
  }

  public synchronized void connect(ConnectionProperties connectionProperties) throws RemoteServerConnectException {
    if (connectionProperties.getBaseUrl() == null)
      throw new IllegalArgumentException("Baseurl is a mandatory parameter"); 
    if (connectionProperties.getConnectTimeout() < 0 || connectionProperties.getReadTimeout() < 0)
      throw new IllegalArgumentException("Timeout has to be >= 0"); 
    if (connectionProperties.getUsername() == null || connectionProperties.getPassword() == null)
      throw new IllegalArgumentException("Remote Server Credentials for REST interfaces not set"); 
    this.readTimeout = connectionProperties.getReadTimeout();
    this.connectTimeout = connectionProperties.getConnectTimeout();
    this.baseUrl = connectionProperties.getBaseUrl();
    LOGGER.info("Using remote server URL '{}'", this.baseUrl);
    initAuthenticationAndSecurity(connectionProperties);
    try {
      LOGGER.info("Register client on backend...");
      try {
        this.serverVersion = readServerVersion();
        LOGGER.info("Remote Server Version: {}", this.serverVersion.toString());
      } catch (IOException e) {
        LOGGER.error("Error getting remote server version, try again with fallback credentials.", (Throwable)e);
        initFallbackBasicAuthentication(connectionProperties);
        this.serverVersion = readServerVersion();
      } 
      Entity<?> serializedLoggingInfo = Entity.json(getUpdatedAuditLogInfo());
      WebTarget functionRegister = getFunction("v1/clients");
      try (Response resp = functionRegister.request(new String[] { "application/json" }).post(serializedLoggingInfo)) {
        if (resp.getStatus() != Response.Status.OK.getStatusCode())
          throw new RemoteServerConnectException("Error getting client ID (Code " + resp.getStatus() + ", Body '" + (String)resp.readEntity(String.class) + "')"); 
        ConnectResponseEntity parsedResp = (ConnectResponseEntity)resp.readEntity(ConnectResponseEntity.class);
        this.clientProperties = parsedResp.properties;
        LOGGER.info("Client ID received: {} with properties {}", parsedResp.destinationId, this.clientProperties);
        setClientId(parsedResp.destinationId);
      } 
      this.remotePlugin.fireConnected(this.baseUrl.toString());
      openEventSource();
      if (this.conCheckTimer == null) {
        this.conCheckTimer = new Timer("CheckConnectionTimer");
        this.conCheckTimerTask = new CheckClientIdAvailableTimerTask(this, this.ourId);
        this.conCheckTimer.scheduleAtFixedRate(this.conCheckTimerTask, 1000L, 5000L);
        LOGGER.debug("Started CheckConnectionTimer");
      } 
    } catch (MalformedURLException|URISyntaxException e) {
      LOGGER.error("Malformed URL, unable to connect to server", (Throwable)e);
      throw new RemoteServerConnectException("Malformed URL, unable to connect to server");
    } catch (IOException|ProcessingException e) {
      LOGGER.error("No network connection on register backend", (Throwable)e);
      throw new RemoteServerConnectException("No network connection on register backend");
    } 
  }

  private Version readServerVersion() throws IOException {

  public void send(Dto dto, String destinationId) throws IOException {
    try {
      WebTarget serviceFunction = getFunction("v1/clients/" + destinationId + "/connections/" + this.ourId);
      Stopwatch sw = new Stopwatch();
      sw.start();
      try (Response resp = serviceFunction.request(new String[] { "application/json" }).post(Entity.json(this.mapper.writeValueAsString(dto)))) {
        sw.stop();
        LOGGER.trace("Sending DTO '{}' took {} ms", dto.getClass().getSimpleName(), Long.valueOf(sw.getMillisecondsElapsed()));
        if (resp.getStatus() != Response.Status.NO_CONTENT.getStatusCode()) {
          String respBody = (String)resp.readEntity(String.class);
          LOGGER.error("Error sending message. Server status: {}, Body: {}", Integer.valueOf(resp.getStatus()), respBody);
          onSendError();
          throw new IOException("Error sending message. Server returned status " + resp.getStatus());
        } 
      } 
    } catch (SocketException e) {
      LOGGER.error("No network connection", (Throwable)e);
      onSendError();
    } catch (ProcessingException|UnknownHostException e) {
      LOGGER.error("Error sending {} to '{}'", dto.getClass().getSimpleName(), destinationId, e);
      onSendError();
    } catch (MalformedURLException|URISyntaxException ex) {
      LOGGER.error("Unable to perform call to url", (Throwable)ex);
      resetConnection();
    } 
  }
  
  private synchronized void onSendError() {
    if (this.remSession.get() != null) {
      if (this.evaluatingConn.compareAndSet(false, true))
        return; 
      evaluateConnection();
    } 
    resetConnection();
  }

  private boolean evaluateConnection() {
    LOGGER.trace("Evaluate connection after connection break...");
    boolean inetAvailable = false;
    boolean serverAvailable = false;
    try {
      LOGGER.trace("Check internet connection ...");
      inetAvailable = new InternetAvailabilityChecker().isInternetAvailable();
    } catch (IOException ex) {
      LOGGER.error("No internet available", (Throwable)ex);
    } 
    if (inetAvailable) {
      LOGGER.trace("Check connection to remote server ...");
      try {
        Version servVersion = readServerVersion();
        if (servVersion != null)
          serverAvailable = true; 
      } catch (Exception ex) {
        LOGGER.error("Remote Server not available", (Throwable)ex);
      } 
    } 
    if (!inetAvailable) {
      this.remSession.get().getClient().setConnectionErrorInfo(CONN_ERROR_INFO_NOINTERNET);
    } else if (!serverAvailable) {
      this.remSession.get().getClient().setConnectionErrorInfo(CONN_ERROR_INFO_REMOTESERVER_NOT_AVAILABLE);
    } 
    LOGGER.info("Internet available: {}, Server available: {}", Boolean.valueOf(inetAvailable), Boolean.valueOf(serverAvailable));
    return (inetAvailable && serverAvailable);
  }

  public void onEvent(InboundEvent inboundEvent) {

      LOGGER.debug("Client: Ping received");
      this.pingExecutor.execute(() -> sendPong());
    } else {
      try {
        Dto dto = (Dto)this.mapper.readValue(inboundEvent.readData(), Dto.class);
        if (dto instanceof com.gigatronik.dart2.bosch.remote.common.dto.v1.ConnectRequestDto)
          createSession(AbstractClient.ClientType.CUSTOMER, inboundEvent.getId()); 
        if (this.remSession.get() != null) {
          this.remSession.get().onRx(inboundEvent.getId(), dto);
        } else {
          LOGGER.warn("Received data while no session is active. Data {} dropped", dto.getClass().getSimpleName());
        } 
      } catch (IOException e) {
        LOGGER.error("Error when trying to deserialize incoming dto", (Throwable)e);
      } catch (SessionAlreadyExistException ex) {
        LOGGER.warn("Received connect request dto, while remote session already exists. Connect request dropped", (Throwable)ex);

    }

  public URI createProxySettings(URL baseUrl, ProxyManager proxyManager) throws URISyntaxException {
    ProxyManager.ProxyMode proxyMode = proxyManager.getMode();
    boolean isProxyConfigValid = (proxyMode == ProxyManager.ProxyMode.SYSTEM || (proxyManager.getHost() != null && proxyManager.getPort() != null));
    boolean isProxyActive = (proxyMode == ProxyManager.ProxyMode.CUSTOM || proxyMode == ProxyManager.ProxyMode.SYSTEM);
    URI newProxyUri = null;
    if (isProxyActive) {
      LOGGER.info("Proxy active!");
      if (isProxyConfigValid) {
        LOGGER.info("Proxy conf valid!");
        List<Proxy> proxies = proxyManager.select(baseUrl.toURI());
        if (proxies == null || proxies.isEmpty()) {
          LOGGER.warn("Proxy is enabled but no proxy was available for {}. Disabling proxy usage", baseUrl);
        } else {
          Proxy proxy = proxies.get(0);
          newProxyUri = createProxyURI(proxy, proxyManager);
          LOGGER.info("Proxy build!");
        } 
      } else {
        LOGGER.error("Proxy settings are not configured correctly: Host: {} Port: {}", proxyManager.getHost(), proxyManager.getPort());
      } 
    } 
    return newProxyUri;
  }
  
  private URI createProxyURI(Proxy proxy, ProxyManager proxyManager) {
    InetSocketAddress proxyAddr = (InetSocketAddress)proxy.address();
    if (proxyAddr == null) {
      LOGGER.error("Proxy Address received is null");
      return null;
    } 
    String proxyUser = proxyManager.getUsername();
    String proxyPassword = proxyManager.getPassword();
    boolean proxyAuthEnabled = StringUtils.isNotEmpty(proxyUser);
    try {
      URI sUri;
      if (proxyAuthEnabled) {
        sUri = new URI("http", URLEncoder.encode(proxyUser, StandardCharsets.UTF_8) + ":" + URLEncoder.encode(proxyUser, StandardCharsets.UTF_8), proxyAddr.getHostName(), proxyAddr.getPort(), null, null, null);
      } else {
        sUri = new URI("http", null, proxyAddr.getHostName(), proxyAddr.getPort(), null, null, null);
      } 
      return sUri;
    } catch (URISyntaxException ex) {
      LOGGER.error("Unable to create proxy URL", (Throwable)ex);
      return null;
    } 
  }
