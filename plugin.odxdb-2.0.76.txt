        if ("DTC".equals(elementName)) {
          String sVal = this.reader.getAttributeValue(0);
          myOdxObject = new OdxDtc(sVal);
          myOdxObjects.add(myOdxObject);
          continue;
        } 
        if (myOdxObject == null)
          continue; 
        if ("SHORT-NAME".equals(elementName)) {

          myOdxObject.setShortName(sVal);
          myOdxObject.setDisplayTroubleCode(sVal);

          myOdxObject.setText(this.reader.getElementText());
          continue;

        if ("SDGS".equals(elementName))
          myOdxObject.setDescriptions(getDtcDescription()); 
        continue;

      if (event == 2 && 
        this.reader.getLocalName().equals(sEndTag))
        break;

      OdxSdg[] mySdgs = myDtc.getDescriptions();
      if (mySdgs != null)
        for (OdxSdg mySdg : myDtc.getDescriptions()) {

              OdxSdg myRefSdg = myRefDtc.getDescriptionByID(mySdg.getId().substring(4));

                mySdg.setShortName(myRefSdg.getShortName());
                mySdg.setLongName(myRefSdg.getLongName());
                mySdg.setDescription(myRefSdg.getDescription());
                break;
              } 
            }  
        }  
    } 
    return myOdxObjects.<OdxDtc>toArray(new OdxDtc[myOdxObjects.size()]);

    while (this.reader.hasNext()) {
      int event = this.reader.next();

          try {
            CompuMethod compuMethod = new CompuMethod(this.reader);
            if (!StringUtils.equals(compuMethod.getCategory(), "IDENTICAL"))
              LOGGER.warn("Unexpected COMPU-METHOD {} found while parsing DTC-DOP", compuMethod.getCategory()); 
          } catch (IOException e) {
            LOGGER.error("Parsing element COMPU-METHOD of DTC-DOP failed.", (Throwable)e);
          }

            DiagCodedType diagCodedType = new DiagCodedType(this.reader);
            myOdxObject.setEncoding(OdxParser.Encoding.parseFromXml(diagCodedType.getEncoding()));
            if (StringUtils.isNotBlank(diagCodedType.getDataType()))
              myOdxObject.setBaseDataType(OdxDatabaseDartPlugin.Datatype.fromXml(diagCodedType.getDataType())); 
            if (StringUtils.isNotBlank(diagCodedType.getBitLength()))
              myOdxObject.setBitLength(Integer.parseInt(diagCodedType.getBitLength()));

            LOGGER.error("Parsing element COMPU-METHOD of DTC-DOP failed.", (Throwable)e);

          } 
          continue;
        } 
        if ("DTCS".equals(elementName))
          break; 
        continue;
      } 
      if (event == 2 && 
        this.reader.getLocalName().equals(sEndTag))
        break; 
    }

    while (this.reader.hasNext()) {

      if (event == 1) {

        if ("BYTE-SIZE".equals(elementName)) {

          myOdxObject.setByteSize(iByteLen);
          continue;
        } 
        if ("PARAMS".equals(elementName)) {

                myParams[n - 1].setBitLength(myParams[n].getBitPosition() - myParams[n - 1].getBitPosition()); 
            } 
            continue;
          } 
          myParams[0].setByteLength(iByteLen);
        } 
        continue;
      } 
      if (event == 2 && 
        this.reader.getLocalName().equals(sEndTag))
        break; 
    } 
    return myOdxObjects.<OdxStructure>toArray(new OdxStructure[myOdxObjects.size()]);

public class OdxDtc extends OdxObject {
  private static final long serialVersionUID = 1L;
  
  private String troubleCode = "";
  
  private String displayCode = "";
  
  private String text = "";
  
  private OdxSdg[] arrOdxSdg = null;
  
  public OdxDtc(String id) {
    super(id);
  }
  
  public String toString() {
    return getHexTroubleCode();
  }
  
  public String getTroubleCode() {
    return this.troubleCode;
  }
  
  public String getHexTroubleCode() {
    long code = Long.parseLong(this.troubleCode);
    return ByteArrayConverterUtil.longToHexString(code, 3);
  }
  
  public void setTroubleCode(String troubleCode) {
    this.troubleCode = troubleCode;
  }
  
  public String getDisplayTroubleCode() {
    return this.displayCode;
  }
  
  public void setDisplayTroubleCode(String displayTroubleCode) {
    this.displayCode = displayTroubleCode;
  }
  
  public String getText() {
    return this.text;
  }
  
  public void setText(String text) {
    this.text = text;
  }
  
  public void setDescriptions(OdxSdg[] sdgs) {
    this.arrOdxSdg = sdgs;
  }
  
  public OdxSdg[] getDescriptions() {
    return this.arrOdxSdg;
  }
  
  public OdxSdg getDescriptionByID(String id) {
    if (this.arrOdxSdg != null)
      for (OdxSdg element : this.arrOdxSdg) {
        if (element.getId().equals(id))
          return element; 
      }  
    return null;
  }
  
  public String getDescriptionText() {
    StringBuilder description = new StringBuilder();
    for (OdxSdg mySdg : this.arrOdxSdg) {
      description.append(!mySdg.getLongName().isEmpty() ? mySdg.getLongName() : mySdg.getShortName());
      description.append(System.lineSeparator());
      if (!mySdg.getDescription().isEmpty()) {
        description.append("(");
        description.append(mySdg.getDescription());
        description.append(")");
        description.append(System.lineSeparator());
      } 
      if (!mySdg.getText().isEmpty()) {
        description.append(mySdg.getText());
        description.append(System.lineSeparator());
      } 
    } 
    return description.toString();
  }
}

  public static void calculateParameterValuesFromPdu(OdxParam[] odxParams, byte[] bPdu) throws DatabaseException {
    for (OdxParam myPar : odxParams) {
      int bytePos = myPar.getBytePosition();
      int byteLen = myPar.getByteLength();
      int bitLen = myPar.getBitLength();
      int bitPos = myPar.getBitPosition();
      OdxValue odxVal = myPar.getOdxValue();
      String sVal = "";
      if (odxVal != null) {
        int iVal;
        long lVal;
        double dVal;
        switch (odxVal.getDataType()) {
          case INT32:
            if (odxVal.getEncoding() == OdxParser.Encoding.BCD) {
              for (int m = bytePos; m < bytePos + byteLen; m++)
                sVal = sVal.concat(Integer.toString(bPdu[m] & 0xFF, 16)); 
              iVal = Integer.parseInt(sVal);
            } else {
              iVal = ByteArrayConverterUtil.byteArrayToInteger(bPdu, bytePos, bitPos, bitLen);
            } 
            iVal = (int)(((double)iVal + odxVal.getOffset()) * odxVal.getMultiplier());
            odxVal.set((long)iVal);
            myPar.setCodedValue((long)iVal);
            break;
          case UINT32:
            if (odxVal.getEncoding() == OdxParser.Encoding.BCD) {
              for (int m = bytePos; m < bytePos + byteLen; m++)
                sVal = sVal.concat(Integer.toString(bPdu[m] & 0xFF, 16)); 
              lVal = (long)Integer.parseInt(sVal);
            } else {
              lVal = ByteArrayConverterUtil.byteArrayToUnsignedInteger(bPdu, bytePos, bitPos, bitLen);
            } 
            lVal = (long)(((double)lVal + odxVal.getOffset()) * odxVal.getMultiplier());
            odxVal.set(lVal);
            myPar.setCodedValue(lVal);
            break;
          case FLOAT32:
            iVal = ByteArrayConverterUtil.byteArrayToInteger(bPdu, bytePos, bitPos, bitLen);
            dVal = ((double)iVal + odxVal.getOffset()) * odxVal.getMultiplier();
            odxVal.set(dVal);
            break;
          case FLOAT64:
            lVal = ByteArrayConverterUtil.byteArrayToUnsignedInteger(bPdu, bytePos, bitPos, bitLen);
            dVal = ((double)lVal + odxVal.getOffset()) * odxVal.getMultiplier();
            odxVal.set(dVal);
            break;
          case STRING:
            sVal = ByteArrayConverterUtil.byteArrayToASCII(ByteArrayConverterUtil.partOfByteArray(bPdu, bytePos, byteLen - odxVal.getTermination()));
            odxVal.set(sVal);
            break;
          case TEXTTABLE:
            switch (odxVal.getBaseDataType()) {
              case INT32:
                iVal = ByteArrayConverterUtil.byteArrayToInteger(bPdu, bytePos, bitPos, bitLen);
                break;
              case UINT32:
                iVal = (int)ByteArrayConverterUtil.byteArrayToUnsignedInteger(bPdu, bytePos, bitPos, bitLen);
                break;
              default:
                iVal = (int)ByteArrayConverterUtil.byteArrayToUnsignedInteger(bPdu, bytePos, bitPos, bitLen);
                break;
            } 
            odxVal.setTextTableValue(iVal);
            break;
          case BYTEFIELD:
            odxVal.set(ByteArrayConverterUtil.partOfByteArray(bPdu, bytePos, byteLen));
            break;
        } 
      } else if (myPar.hasStruct()) {
        myPar.getOdxStructure().setPDU(ByteArrayConverterUtil.partOfByteArray(bPdu, bytePos, byteLen));
        myPar.getOdxStructure().updateParameterValuesFromPdu();
      } else if (myPar.hasEndOfPduField()) {

        if (lVal != myPar.getCodedValue())

      } 
    }

  public static byte[] calculatePduFromParameterValues(OdxParam[] odxParams) {
    byte[] bPdu = null;
    if (odxParams != null) {
      int iLen = 0;
      for (OdxParam odxPar : odxParams) {
        int iSize = odxPar.getBitLength();
        if (iSize < 0)
          iSize = -iSize; 
        if (iSize == 0)
          iSize = 8; 
        iLen += iSize;
      } 
      iLen /= 8;
      bPdu = new byte[iLen];
      for (OdxParam odxPar : odxParams) {
        int bytePos = odxPar.getBytePosition();
        int byteLen = odxPar.getByteLength();
        int bitPos = odxPar.getBitPosition();
        int bitLen = odxPar.getBitLength();
        if (bitPos + bitLen > byteLen * 8)
          byteLen++; 
        byte[] parPdu = new byte[byteLen];
        OdxValue odxVal = odxPar.getOdxValue();
        if (odxPar.isConstant()) {
          parPdu = ByteArrayConverterUtil.longToByteArray(odxPar.getCodedValue() << bitPos, byteLen);
        } else if (odxVal != null) {
          String sValue;
          long iValue;
          int n;
          switch (odxVal.getDataType()) {
            case INT32:
              iValue = odxVal.getLong();
              if (odxVal.getEncoding() == OdxParser.Encoding.BCD) {
                String str = String.format("%0" + (byteLen * 2) + "d", iValue);
                for (int i = 0; i < byteLen; i++)
                  parPdu[i] = (byte)Integer.parseInt(str.substring(i * 2, i * 2 + 2), 16); 
              } else {
                iValue = (long)(int)((double)odxVal.getLong() / odxVal.getMultiplier() - odxVal.getOffset());
                parPdu = ByteArrayConverterUtil.longToByteArray(iValue << bitPos, byteLen);
              } 
              break;
            case UINT32:
              iValue = odxVal.getLong();
              if (odxVal.getEncoding() == OdxParser.Encoding.BCD) {
                String str = String.format("%0" + (byteLen * 2) + "d", iValue);
                for (int i = 0; i < byteLen; i++)
                  parPdu[i] = (byte)Integer.parseInt(str.substring(i * 2, i * 2 + 2), 16); 
              } else {
                iValue = Math.round((double)odxVal.getLong() / odxVal.getMultiplier() - odxVal.getOffset());
                parPdu = ByteArrayConverterUtil.longToByteArray(iValue << bitPos, byteLen);
              } 
              break;
            case FLOAT32:
              iValue = Math.round(odxVal.getFloat() / odxVal.getMultiplier() - odxVal.getOffset());
              parPdu = ByteArrayConverterUtil.longToByteArray(iValue << bitPos, byteLen);
              break;
            case FLOAT64:
              iValue = Math.round(odxVal.getFloat() / odxVal.getMultiplier() - odxVal.getOffset());
              parPdu = ByteArrayConverterUtil.longToByteArray(iValue << bitPos, byteLen);
              break;
            case STRING:
              sValue = odxVal.getString();
              for (n = 0; n < sValue.length(); n++)
                parPdu[n] = (byte)sValue.charAt(n); 
              break;
            case TEXTTABLE:
              sValue = odxVal.getString();
              iValue = (long)odxVal.getTextTableValueFromText(sValue);
              if (iValue < 0L) {
                odxVal.setTextTableDefault();
                sValue = odxVal.getString();
                iValue = (long)odxVal.getTextTableValueFromText(sValue);
              } 
              parPdu = ByteArrayConverterUtil.longToByteArray(iValue << bitPos, byteLen);
              break;
            case BYTEFIELD:
              parPdu = odxVal.getByteField();
              break;
          } 
        } else if (odxPar.hasStruct()) {
          odxPar.getOdxStructure().updatePduFromParameterValues();
          parPdu = odxPar.getOdxStructure().getPDU();
        } else if (odxPar.hasEndOfPduField()) {
          parPdu = odxPar.getOdxEndOfPduField().getData();
        } 
        if (parPdu != null)
          for (int n = 0; n < parPdu.length; n++) {
            if (bytePos + n < bPdu.length) {
              bPdu[bytePos + n] = (byte)(bPdu[bytePos + n] | parPdu[n]);
            } else {
              n = parPdu.length;
            } 
          }  
      } 
    } 
    return bPdu;
  }

    this.odxParams = new OdxParam[sourceParams.length];
    for (int n = 0; n < this.odxParams.length; n++)
      this.odxParams[n] = new OdxParam(sourceParams[n]); 
  }
  
  public boolean isPositive() {
    return (this.bPdu[0] != Byte.MAX_VALUE);
  }
  
  public OdxParam[] getParameter() {

    } 
    return findParam;
  }
  
  public void setPDU(byte[] pdu) throws DatabaseException {
    if (this.odxParams == null)
      throw new DatabaseException("Response has no parameters"); 
    int minLen = 0;
    int maxLen = 0;
    for (OdxParam param : this.odxParams) {
      int len = param.getBytePosition() + param.getByteLength();
      if (len > minLen)
        minLen = len; 
      if (param.hasEndOfPduField())
        maxLen = Integer.MAX_VALUE; 
    } 
    if (maxLen == 0)
      maxLen = minLen; 
    if (pdu.length >= minLen && pdu.length <= maxLen) {
      this.bPdu = pdu;
      updateParameterValuesFromPdu();
    } else {
      throw new DatabaseException(String.format("Error setting pdu - expected size (%d to %d), but was %d", minLen, maxLen, pdu.length));
    } 
  }

    return this.bPdu;
  }

    OdxHelper.calculateParameterValuesFromPdu(this.odxParams, this.bPdu);

    odxValue.setDataType(OdxDatabaseDartPlugin.Datatype.TEXTTABLE);
    for (CompuScale compuScale : this.compuMethod.getCompuScales()) {
      if (StringUtils.isNotBlank(compuScale.getLowerLimit())) {
        String text = compuScale.getConstant();
        if (OdxDatabaseDartPlugin.Datatype.BYTEFIELD == odxValue.getBaseDataType()) {
          odxValue.addTextTableEntry(ByteArrayConverterUtil.hexStringToByteArray(compuScale.getLowerLimit()), text);
          continue;
        } 
        int lowerLimit = Integer.parseInt(compuScale.getLowerLimit());
        int upperLimit = StringUtils.isNotBlank(compuScale.getUpperLimit()) ? Integer.parseInt(compuScale.getUpperLimit()) : lowerLimit;
        odxValue.addTextTableEntry(lowerLimit, upperLimit, text);
      } 
    } 
  }
  
  private void setLinearCompuMethodToOdxValue(OdxValue odxValue) {
    if (!this.compuMethod.getCompuScales().isEmpty()) {
      CompuScale compuScale = this.compuMethod.getCompuScales().get(0);
      if (compuScale.getNumerators().size() > 0)
        odxValue.setOffset(Double.parseDouble(compuScale.getNumerators().get(0))); 
      if (compuScale.getNumerators().size() > 1)
        odxValue.setMultiplier(Double.parseDouble(compuScale.getNumerators().get(1))); 
      if (compuScale.getDenominators().size() > 0) {
        double multiplier = odxValue.getMultiplier() / Double.parseDouble(compuScale.getDenominators().get(0));
        odxValue.setMultiplier(multiplier);
      } 
    } 
  }
  
  private void setConstraintsToOdxValue(OdxValue odxValue) {
    List<OdxConstraint> constraints = new ArrayList<>();
    if (this.internalConstraint != null) {
      int radix = (OdxDatabaseDartPlugin.Datatype.BYTEFIELD == odxValue.getBaseDataType()) ? 16 : 10;
      for (ScaleConstraint sc : this.internalConstraint.getScaleConstraints()) {
        if (StringUtils.isNotBlank(sc.getLowerLimit())) {
          long lowerLimit = Long.parseLong(sc.getLowerLimit(), radix);
          long upperLimit = StringUtils.isNotBlank(sc.getUpperLimit()) ? Long.parseLong(sc.getUpperLimit(), radix) : lowerLimit;
          constraints.add(new OdxConstraint(sc.getShortLabel(), lowerLimit, upperLimit));
        } 
      } 
    } 
    odxValue.setConstraints(constraints);
  }
}
