  private static void renameComponents(Component component) {
    List<Component> allComponents = searchForComponent(Component.class, component);
    int i = 0;
    for (Component element : allComponents) {
      if (JButton.class.isAssignableFrom(element.getClass())) {
        element.setName(String.format("btn_%s%d", element.getClass().getSimpleName(), i++));
        continue;
      } 
      element.setName(element.getClass().getSimpleName());
    } 
  }
  
  private static <T> List<T> searchForComponent(Class<T> clazz, Component component) {
    List<T> result = new ArrayList<>();
    if (component instanceof Container) {
      Container container = (Container)component;
      for (Component child : container.getComponents()) {
        List<T> subResult = searchForComponent(clazz, child);
        result.addAll(subResult);
      } 
    } 
    if (clazz.isAssignableFrom(component.getClass()))
      result.add((T)component); 
    return result;
  }
  
  private static class ModalPrivilegedAction implements PrivilegedAction<Method> {
    private final Class<?> clazz;
    
    private final String methodName;
    
    public ModalPrivilegedAction(Class<?> clazz, String methodName) {
      this.clazz = clazz;
      this.methodName = methodName;
    }
    
    public Method run() {
      Method method = null;
      try {
        method = this.clazz.getDeclaredMethod(this.methodName, null);
      } catch (NoSuchMethodException ex) {
        TestableJOptionPane.LOGGER.error("Error in TestableJOptionPane ", (Throwable)ex);
      } 
      if (method != null)
        method.setAccessible(true); 
      return method;
    }
  }

  private Map<String, Component> searchForComponent(ComponentPath searchpath, Component component, ComponentPath currentPath) {

        });
  }
  
  private List<KeyEvent> buildEvents(Component comp, String keyValue) {
    List<KeyEvent> resultList = new ArrayList<>();
    int keyEvent = getKeyCode(keyValue, comp);
    if (keyEvent >= 0 && StringUtils.isAllUpperCase(StringUtils.replaceAll(keyValue, "_", "")) && StringUtils.length(keyValue) > 1) {
      resultList.add(new KeyEvent(comp, 401, System.currentTimeMillis(), 0, keyEvent, Character.MAX_VALUE));
    } else {
      List<KeyEvent> singleCharList = new ArrayList<>();
      for (char c : keyValue.toCharArray())
        singleCharList.add(new KeyEvent(comp, 400, System.currentTimeMillis(), 0, 0, c)); 
      resultList.addAll(singleCharList);
    } 
    return resultList;
  }
  
  public int getKeyCode(String keyValue, Component comp) {
    int keyCode = -1;
    try {
      Object fieldValue = FieldUtils.getDeclaredField(KeyEvent.class, "VK_" + keyValue).get(new KeyEvent(comp, 0, 0L, 0, 0, Character.MAX_VALUE));
      if (fieldValue != null && fieldValue instanceof Integer)
        keyCode = (Integer)fieldValue; 
    } catch (Exception ex) {
      LOGGER.warn("Keyvalue {} could not be converted to key code.", keyValue, ex);
    } 
    return keyCode;
  }
}
