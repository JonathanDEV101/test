  public void retrieveReportAsync(String customerNumber, String caseNumber, String pcToolLanguage, CrmRetrieveReportListener crmRetrieveReportListener) {
    if (this.taskFuture != null && !this.taskFuture.isDone()) {
      LOGGER.trace("rejecting new task to receive report , an receive task is still running");
      return;
    } 
    this.taskFuture = this.reportRetrieveExecutor.submit(() -> {
          try {
            doReceiveReport(caseNumber, customerNumber, crmRetrieveReportListener);
          } catch (ApiException e) {
            LOGGER.error("Error receiving report", (Throwable)e);
            LOGGER.error("ResponseBody: {}", e.getResponseBody());
            crmRetrieveReportListener.onRetrieveError(e.getCode());
          } catch (MalformedURLException e) {
            LOGGER.error("Error receiving report", (Throwable)e);
            crmRetrieveReportListener.onRetrieveError();
          } catch (IOException e) {
            LOGGER.error("Error receiving report", (Throwable)e);
            crmRetrieveReportListener.onRetrieveError();
          } catch (JAXBException e) {
            LOGGER.error("Error parsing downloaded report", (Throwable)e);
            crmRetrieveReportListener.onRetrieveError();
          } catch (Exception e) {
            LOGGER.error("Error receiving report", (Throwable)e);
            crmRetrieveReportListener.onRetrieveError();
          } 
        });
  }
  
  void doReceiveReport(String caseNumber, String customerNumber, CrmRetrieveReportListener crmRetrieveReportListener) throws ApiException, IOException, JAXBException {
    this.findCaseCall = this.caseApi.findCaseCall(Integer.valueOf(1), null, null, "{\"caseNumber\":\"" + caseNumber + "\",\"accountId.spShipToNo\":\"" + customerNumber + "\"}", null, null, null);
    ApiResponse<CaseCollectionResponse> execute = this.caseApi.getApiClient().execute(this.findCaseCall, new TypeToken<CaseCollectionResponse>() {
        
        }.getType());
    CaseCollectionResponse response = (CaseCollectionResponse)execute.getData();
    LOGGER.info("findCaseCall return: {}", response);
    String id = "";
    if (response.isSuccess() && !response.getResponse().isEmpty()) {
      List<ModelCase> response1 = response.getResponse();
      id = response1.get(0).getId();
      if (StringUtils.isBlank(id))
        throw new ApiException("error retrieving caseId for caseNumber " + caseNumber); 
      this.findExternalFileJunctionCall = this.externalFileJunctionApi.findExternalFileJunctionCall(null, null, null, "{\"linkedEntityId\":\"" + id + "\"}", "externalFile.fileUrl,externalFile.category", null, null);
      LinkedTreeMap files = (LinkedTreeMap)this.externalFileJunctionApi.getApiClient().execute(this.findExternalFileJunctionCall, new TypeToken<LinkedTreeMap>() {
          
          }.getType()).getData();
      if ((Boolean)files.get("success")) {
        ArrayList<LinkedTreeMap> filesResponse = (ArrayList)files.get("response");
        boolean found = false;
        for (int i = filesResponse.size() - 1; i >= 0; i--) {
          LinkedTreeMap reportCandidate = filesResponse.get(i);
          LinkedTreeMap externalFile = (LinkedTreeMap)reportCandidate.get("externalFile");
          if (ExternalFile.CategoryEnum.DIAGNOSTICREPORT == 
            ExternalFile.CategoryEnum.fromValue((String)externalFile.get("category"))) {
            found = true;
            downloadReport(caseNumber, crmRetrieveReportListener, externalFile);
            break;
          } 
        } 
        if (!found)
          crmRetrieveReportListener.onRetrieveError(); 
      } else {
        crmRetrieveReportListener.onRetrieveError();
      } 
    } else {
      crmRetrieveReportListener.onRetrieveError();
    } 
  }
  
  private void downloadReport(String caseNumber, CrmRetrieveReportListener crmRetrieveReportListener, LinkedTreeMap externalFile) throws IOException, JAXBException {
    String downloadUrl = (String)externalFile.get("fileUrl");
    StringBuilder responseStringBuilder = new StringBuilder();
    BufferedReader in = new BufferedReader(new InputStreamReader(this.crmInternalFactory.openStream(new URL(downloadUrl))));
    String inputLine;
    while ((inputLine = in.readLine()) != null)
      responseStringBuilder.append(inputLine); 
    in.close();
    RetrievedReportData retrievedReportData = this.dataExtractor.extractData(responseStringBuilder.toString(), caseNumber);
    crmRetrieveReportListener.onRetrieveSuccess(retrievedReportData);
  }
  
  public void cancelReceiveRequest() {
    if (this.findCaseCall != null)
      this.findCaseCall.cancel(); 
    if (this.findExternalFileJunctionCall != null)

  }
