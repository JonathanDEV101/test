import com.gigatronik.bosch.plugin.unitoken.crypto.SignatureHelper;
import com.gigatronik.bosch.utils.UnMarshalJAXB;
import java.io.ByteArrayInputStream;
import java.io.File;
import java.io.FileNotFoundException;
import java.io.IOException;
import java.io.InputStream;
import javax.xml.bind.JAXBException;
import javax.xml.bind.UnmarshalException;
import org.apache.commons.compress.archivers.zip.ZipArchiveEntry;
import org.apache.commons.compress.archivers.zip.ZipFile;
import org.apache.commons.io.IOUtils;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

public abstract class AbstractContainer<T> {
  protected final SignatureHelper signatureHelper = new SignatureHelper();
  
  private static final Logger LOGGER = LoggerFactory.getLogger(AbstractContainer.class);
  
  protected final UnMarshalJAXB marshallingHelper = new UnMarshalJAXB();
  
  protected final File archiveFile;
  
  public static final String META_DATA_FILE_NAME = "meta.xml";
  
  public AbstractContainer(File archiveFile) {
    this.archiveFile = archiveFile;
  }
  
  protected void checkSignature(String sigSalt, String sigVal) throws IOException {
    try (InputStream metaData = getFileData("meta.xml")) {
      if (!this.signatureHelper.checkRsaSignature(CryptoInfo.getCheckString(IOUtils.toString(metaData, "utf-8")), sigSalt, sigVal, CryptoInfo.getX509()))
        throw new SecurityException("Error in signature"); 
    } 
  }
  
  protected InputStream getFileData(String fileName) throws IOException {
    InputStream result = null;
    try (ZipFile zipFile = new ZipFile(this.archiveFile)) {
      ZipArchiveEntry archiveEntry = zipFile.getEntry(fileName);
      if (archiveEntry == null)
        throw new FileNotFoundException(String.format("File '%s' not found in archive", fileName)); 
      result = IOUtils.toBufferedInputStream(zipFile.getInputStream(archiveEntry));
    } 
    return result;
  }
  
  protected T getMetaObject(String schemaName, Class<T> cls) throws IOException, JAXBException, InstantiationException, IllegalAccessException {
    InputStream metaData = getFileData("meta.xml");
    return unmarshallFrom(metaData, schemaName, cls);
  }
  
  public BasVariantInfoType getBas100VariantInfoObject(String fileName, String keyIv, String crc32, String sigSalt, String sigValue) throws IOException, JAXBException {
    BasVariantInfoType result = null;
    try (InputStream basInfoStream = getFileData(fileName)) {
      byte[] basInfoBytes = IOUtils.toByteArray(basInfoStream);
      byte[] decryptedBas100Xml = null;
      decryptedBas100Xml = CryptoHelper.decrypt(basInfoBytes, keyIv, crc32, CryptoInfo.getSkey(), CryptoConstants.getcInstance5());
      if (decryptedBas100Xml != null) {
        LOGGER.info("Checking Signature of {}", fileName);
        result = unmarshallCheckSignature(decryptedBas100Xml, sigSalt, sigValue);
      } 
    } 
    return result;
  }
  
  protected BasVariantInfoType unmarshallCheckSignature(byte[] marshalledData, String sigSalt, String sigValue) throws IOException, JAXBException {
    try (ByteArrayInputStream unmarshalStream = new ByteArrayInputStream(marshalledData)) {
      ByteArrayInputStream signatureInputStream = new ByteArrayInputStream(marshalledData);
      if (this.signatureHelper.checkRsaSignature(IOUtils.toByteArray(signatureInputStream), sigSalt, sigValue, CryptoInfo.getX509()))
        return (BasVariantInfoType)this.marshallingHelper.unmarshal(new BasVariantInfoType(), unmarshalStream, this.marshallingHelper.getSchema(getClass().getClassLoader().getResourceAsStream("schemas/Bas_Variant_Info.xsd"))); 
      LOGGER.error("Signature error");
      return null;
    } 
  }
  
  private T unmarshallFrom(InputStream metaData, String schemaName, Class<T> cls) throws JAXBException, IOException, InstantiationException, IllegalAccessException {
    T result = null;
    byte[] metaDataBytes = IOUtils.toByteArray(metaData);
    try (ByteArrayInputStream unmarshalStream = new ByteArrayInputStream(metaDataBytes)) {
      result = (T)this.marshallingHelper.unmarshal(cls.newInstance(), unmarshalStream, this.marshallingHelper.getSchema(getClass().getClassLoader().getResourceAsStream("schemas/" + schemaName)));
    } catch (UnmarshalException jaxbExc) {
      LOGGER.info("Could not parse meta.xml, now check if container is cff2 ", (Throwable)jaxbExc);
      result = loadClassicContainer(metaDataBytes, cls);
      if (result == null)
        throw jaxbExc; 
    } 
    return result;
  }
  
  private T loadClassicContainer(byte[] metaDataBytes, Class<T> cls) throws UnmarshalException, InstantiationException, IllegalAccessException, IOException, JAXBException {
    if (cls.isInstance(Cff2_Ebike.class.newInstance())) {
      T result = null;
      LOGGER.info("meta.xml is from cff2 container, reread the meta.xml without validation");
      try (ByteArrayInputStream unmarshalStream = new ByteArrayInputStream(metaDataBytes)) {
        result = (T)this.marshallingHelper.unmarshalWithoutValidation(cls.newInstance(), unmarshalStream);
        LOGGER.info("Re-reading of container successful, now checking if classic+");
        Cff2_Ebike res = (Cff2_Ebike)result;
        BoschContainerGen2Impl cff2 = new BoschContainerGen2Impl();
        cff2.setEbike(res);
        if (cff2.isClassicContainer()) {
          LOGGER.info("The loaded cff2 container is a not xsd valid classic container");
          return result;
        } 
      } 
    } 
    return null;
  }
}

  public void handleDownloadResponse(ContainerDownloadStates.ContainerDownloadResult downloadState, ContainerDownloadDataset resultData) {
    this.containerDownloadDataset = resultData;
    LOGGER.info("downloadstate: {}", downloadState);
    switch (downloadState) {
      case NO_REQUEST_CFF:
        onRequestError();
        break;
      case CONNECTION_PROBLEM:
        onConnectionProblem();
        break;
      case REPLY_CORRUPT:
        onDownloadError(resultData);
        break;
      case ERROR_MSG:
        onCffError(prepareErrorMessage(resultData));
        break;
      case DL_SUCCESSFUL:
        onDownloadSuccessful(resultData);
        break;
      case INVALID_URL:
        onInvalidUrl();
        break;
    } 
    if (resultData.getReplyCffPath() != null && resultData.getReplyCffPath().length() > 10485760)
      this.cffController.deletePreviousLargeContainers(); 
  }

  public abstract void onInvalidUrl();
  
  private List<String[]> prepareErrorMessage(ContainerDownloadDataset resultData) {
    List<String[]> errorMessages = new ArrayList<>();
    if (resultData.getEbikeDownloaded().getMessage() != null)
      for (int i = 0; i < resultData.getEbikeDownloaded().getMessage().size(); i++) {
        if (resultData.getEbikeDownloaded().getMessage().get(i).getText() != null) {
          Cff2_MessageText errorMsg = null;
          for (Cff2_MessageText text : resultData.getEbikeDownloaded().getMessage().get(i).getText()) {
            if (text.getLanguage().equals(I18NUtil.getLocale().getISO3Language()) || (errorMsg == null && text.getLanguage().equals(Locale.ENGLISH.getISO3Language())))
              errorMsg = text; 
          } 
          if (errorMsg != null) {
            errorMessages.add(new String[] { resultData.getEbikeDownloaded().getMessage().get(i).getMsgId(), errorMsg.getLanguage(), errorMsg.getValue() });
          } else if (!resultData.getEbikeDownloaded().getMessage().get(i).getText().isEmpty()) {
            errorMessages.add(new String[] { resultData.getEbikeDownloaded().getMessage().get(i).getMsgId(), resultData.getEbikeDownloaded().getMessage().get(i).getText().get(0).getLanguage(), resultData.getEbikeDownloaded().getMessage().get(i).getText().get(0).getValue() });
          } 
        } 
      }  
    return errorMessages;
  }
}

    this.showLoadingDialog = showLoadingDialog;
    this.cffDownloadState = cffDownloadState;
  }

  int percentageDownload = 0;

    this.percentageUpload = 0;
    this.percentageDownload = 0;
    if (!this.showLoadingDialog) {
      try {
        while (ContainerDownloadStates.ContainerDownloadProgress.LOADINGSTATE_FINISHED != this.cffDownloadState.getLoadingState()) {
          publish(new Void[0]);
          Thread.sleep(200L);
        } 
      } catch (InterruptedException e) {
        LOGGER.error("Thread Interrupted: " + e.getMessage(), (Throwable)e);
        Thread.currentThread().interrupt();
      } 
      return null;
    } 
    this.progressBar.showProgressBar();
    while (ContainerDownloadStates.ContainerDownloadProgress.LOADINGSTATE_FINISHED != this.cffDownloadState.getLoadingState());
  }
  
  protected void done() {
    if (this.showLoadingDialog)
      this.progressBar.disposeProgressBar(); 
  }
  
  protected void process(List<Void> chunks) {
    switch (this.cffDownloadState.getLoadingState()) {
      case LOADINGSTATE_UPLOAD:
        this.progressBar.setProgressMessage(this.bundle.getString("EBikeContainerFileDialog.lblCurrentProcess.uploading.text"));

    this.systemtype = systemtype;
    this.partNr = partNr;
    this.bctConfig = bctConfig;
    this.duConfig = duConfig;
    this.bmsConfig = bmsConfig;
    this.hmiConfig = hmiConfig;
    this.absConfig = absConfig;
    this.bcsConfig = bcsConfig;
    this.signatureHelper = new SignatureHelper();
    this.updateServerConnection = initUpdateServerConnection();
    this.marshallHelper = new UnMarshalJAXB();
  }
  
  private UpdateServerConnection initUpdateServerConnection() {
    int cffDownloadTimeout = 1000 * SettingsProxy.getInstance().getGlobalSettingsNoException().getInteger("CFF2ContainerDownloadTimeout", 90);
    LOGGER.info("Timeout for cff download set to {} ms.", Integer.valueOf(cffDownloadTimeout));
    return new UpdateServerConnection(cffDownloadTimeout);
  }
  
  private static final Logger LOGGER = LoggerFactory.getLogger(ContainerDownloader.class);
  
  private ContainerDownloadDataset download() {
    this.containerDownload = new ContainerDownloadDataset();
    if (!this.updateServerConnection.isServerAvailable()) {
      this.cffDownloadState.setDownloadResultState(ContainerDownloadStates.ContainerDownloadResult.INVALID_URL);
      return this.containerDownload;
    } 
    LOGGER.info("START CREATE REQUEST CFF2");
    Cff2_Ebike requestEbike = createRequest(this.bctConfig, this.bcsConfig, this.bmsConfig, this.hmiConfig, this.absConfig, this.duConfig, this.systemtype, this.appVersion, this.desiredReplyType, this.partNr);
    LOGGER.info("END CREATE REQUEST CFF2");
    if (requestEbike != null) {
      String localRequestCff = this.marshallHelper.marshallToString(requestEbike);
      if (localRequestCff != null) {
        Map<String, String> map = this.signatureHelper.createRsaSignature(CryptoInfo.getCheckString(localRequestCff), CryptoInfo.getPkcs8());
        LOGGER.info("create Signature...");
        requestEbike = UtilUpdateRequestCff2.addSaltValue(requestEbike, map);
        localRequestCff = this.marshallHelper.marshallToString(requestEbike, this.marshallHelper.getSchema(getClass().getClassLoader().getResourceAsStream("schemas/Container_File_Format_v2_Schema.xsd")));
        File tempRequestCff2 = UtilUpdateRequestCff2.createTempRequestCff2(localRequestCff);
        this.containerDownload.setRequestCffPath(tempRequestCff2.toString());
        this.cffDownloadState.setLoadingState(ContainerDownloadStates.ContainerDownloadProgress.LOADINGSTATE_UPLOAD);
        HttpURLConnection connection = this.updateServerConnection.uploadRequestContainer(UPDATE_SERVER_URL.get(UPDATE_SERVER_URL.CONTAINER_DOWNLOAD).toString(), tempRequestCff2);
        this.cffDownloadState.setLoadingState(ContainerDownloadStates.ContainerDownloadProgress.LOADINGSTATE_DOWNLOAD);
        File tempReplyCff2 = createTmpFile();
        if (tempReplyCff2 != null) {
          String path = this.updateServerConnection.downloadContainer(tempReplyCff2.toString(), connection);
          this.cffDownloadState.setLoadingState(ContainerDownloadStates.ContainerDownloadProgress.LOADINGSTATE_PROCESS);
          unzipContainer(path);
          readMetaData();
        } else {
          this.cffDownloadState.setDownloadResultState(ContainerDownloadStates.ContainerDownloadResult.CONNECTION_PROBLEM);
        } 
      } else {
        LOGGER.error("Couldn't create cff2!");
        this.cffDownloadState.setDownloadResultState(ContainerDownloadStates.ContainerDownloadResult.NO_REQUEST_CFF);
      } 
    } else {
      LOGGER.error("Error marshalling cff2!");
      this.cffDownloadState.setDownloadResultState(ContainerDownloadStates.ContainerDownloadResult.NO_REQUEST_CFF);
    } 
    return this.containerDownload;
  }
  
  public void downloadContainer() {
    ContainerDownloadDataset result = null;
    try {
      result = download();
    } catch (Exception ex) {
      LOGGER.error("Error downloading container", (Throwable)ex);
    } 
    this.cffDownloadState.setLoadingState(ContainerDownloadStates.ContainerDownloadProgress.LOADINGSTATE_FINISHED);
    ApplicationProcessManager.getInstance().setProcessState(ApplicationProcessManager.ApplicationProcess.DOWNLOAD_CONTAINER, false);
    this.downloadHandler.handleDownloadResponse(this.cffDownloadState.getDownloadResultState(), result);
  }
  
  private void checkSignature(String xmlString) {
    if (this.containerDownload.getEbikeDownloaded() != null && xmlString != null) {
      if (this.signatureHelper.checkRsaSignature(CryptoInfo.getCheckString(xmlString), this.containerDownload.getEbikeDownloaded().getSignatureSalt(), this.containerDownload.getEbikeDownloaded().getSignatureValue(), CryptoInfo.getX509())) {
        LOGGER.info("Signature OK");
        if (this.containerDownload.getEbikeDownloaded().getReply() == null) {
          LOGGER.info("Contains only error messages");
          this.cffDownloadState.setDownloadResultState(ContainerDownloadStates.ContainerDownloadResult.ERROR_MSG);
        } else {
          this.cffDownloadState.setDownloadResultState(ContainerDownloadStates.ContainerDownloadResult.DL_SUCCESSFUL);
        } 
      } else {
        this.cffDownloadState.setDownloadResultState(ContainerDownloadStates.ContainerDownloadResult.REPLY_CORRUPT);
      } 
    } else {
      LOGGER.info("Unmarshalling Problem!!!");
      this.cffDownloadState.setDownloadResultState(ContainerDownloadStates.ContainerDownloadResult.REPLY_CORRUPT);
    } 
  }
  
  private void readMetaData() {
    if (this.containerDownload.getZipPathDownloaded() != null) {
      File cff2ReplyXmlFile = new File(this.containerDownload.getZipPathDownloaded() + this.containerDownload.getZipPathDownloaded() + "meta.xml");
      FileIOUtil.deleteDirectoryOnExit(this.containerDownload.getZipPathDownloaded());
      if (cff2ReplyXmlFile.exists()) {
        LOGGER.info("Received cff2-file from internet");
        try {
          String xmlString = IOUtils.toString(new FileInputStream(cff2ReplyXmlFile), "utf-8");
          Cff2_Ebike unmarshalled = (Cff2_Ebike)this.marshallHelper.unmarshall(new Cff2_Ebike(), cff2ReplyXmlFile, this.marshallHelper.getSchema(getClass().getClassLoader().getResourceAsStream("schemas/Container_File_Format_v2_Schema.xsd")));
          if (unmarshalled == null) {
            this.cffDownloadState.setDownloadResultState(ContainerDownloadStates.ContainerDownloadResult.REPLY_CORRUPT);
            LOGGER.info("Unmarshalling problem!");
          } else {
            this.containerDownload.setEbikeDownloaded(unmarshalled);
            LOGGER.info("Unmarshalling OK");
            checkSignature(xmlString);
          } 
        } catch (IOException ex) {
          this.cffDownloadState.setDownloadResultState(ContainerDownloadStates.ContainerDownloadResult.REPLY_CORRUPT);
          LOGGER.info("Unmarshalling problem!!!", (Throwable)ex);
        } 
      } 
    } 
  }
  
  private void unzipContainer(String reply) {
    this.containerDownload.setReplyCff(reply);
    File cff2ReplyContainer = new File(this.containerDownload.getReplyCffPath());
    try {
      this.containerDownload.setZipPathDownloaded(FileIOUtil.unzipWithValidation(cff2ReplyContainer));
    } catch (IOException ioe) {
      LOGGER.error("Response file might not be a zip file", (Throwable)ioe);
      this.containerDownload.setZipPathDownloaded(null);
      this.cffDownloadState.setDownloadResultState(ContainerDownloadStates.ContainerDownloadResult.REPLY_CORRUPT);
    } 
    cff2ReplyContainer.deleteOnExit();
  }
  
  private File createTmpFile() {
    try {
      File tempReplyCff2 = File.createTempFile("Ebike_tmp", ".cff2");
      tempReplyCff2.deleteOnExit();
      return tempReplyCff2;
    } catch (IOException e) {
      LOGGER.error("Error creating Ebike_tmp.cff2", (Throwable)e);
      return null;
    } 
  }
  
  private Cff2_Ebike createRequest(EcuIdentification bctConfig, EcuIdentification bcsConfig, EcuIdentification bmsConfig, HmiIdentification hmiConfig, EcuIdentification absConfig, DuIdentification duConfig, Constants.SystemType systemtype, String appVersion, Cff2_DesiredReply desiredReplyType, String partNr) {
    Cff2_Ebike requestEbike;
    if (bctConfig.isConnected()) {
      requestEbike = UtilUpdateRequestCff2.createRequestCff2Bct(appVersion, desiredReplyType, partNr, bctConfig);
    } else if (bcsConfig.isConnected()) {
      requestEbike = UtilUpdateRequestCff2.createRequestCff2Bcs(appVersion, desiredReplyType, bcsConfig);
    } else if (systemtype == Constants.SystemType.CLASSIC_PLUS) {
      requestEbike = UtilUpdateRequestCff2.createRequestClassicPlusCff2(appVersion, desiredReplyType, partNr, duConfig, bmsConfig, hmiConfig);
    } else {
      requestEbike = UtilUpdateRequestCff2.createRequestCff2Ebike(appVersion, desiredReplyType, partNr, duConfig, bmsConfig, hmiConfig, absConfig);
    } 
    LOGGER.info("Create bike structure for Request CFF2");
    return requestEbike;
  }
}

  public FlashData getBoschContainerFlashData(ComponentType componentType, BoschContainerDataElementWrapper boschContainerFlashData) {
    FlashData flashData = new FlashData();
    flashData.setAppDataContentFormat(boschContainerFlashData.getContentFormat());
    try {
      if (checkFlashFileEncryption(flashData, boschContainerFlashData).getResult().contains("NEGATIVE"))
        return flashData; 
    } catch (IOException ex) {
      flashData.getResult().add("NEGATIVE");
      flashData.getResult().add("FLASH DATA CORRUPT");
      LOGGER.error("Hex file read error", (Throwable)ex);
      return flashData;
    } 
    if ("hex".equals(boschContainerFlashData.getContentFormat())) {
      String fblSignatureValue = boschContainerFlashData.getFblSignatureValue().replaceAll("0x", "").replaceAll(",", "").replaceAll(" ", "");
      if (!StringUtils.isBlank(fblSignatureValue))
        flashData.setAppChecksum(BoschHelper.hexStringToByteArray(fblSignatureValue)); 
      flashData.setIsAppEncrypted(boschContainerFlashData.isFblEncryption());
      if (flashData.isIsAppEncrypted())
        flashData.setAppBlockSizes(IntelHexReader.readBlockSizes(boschContainerFlashData.getFblBlockSize())); 
      if (flashData.getAppBlockSizes() == null || flashData.getAppBlockSizes().isEmpty()) {
        flashData.setAppBlockSizes(new ArrayList<>());
        flashData.getAppHexData().get(1).forEach(hexBlock1 -> flashData.getAppBlockSizes().add(Integer.valueOf(hexBlock1.length)));
      } 
    } else if ("bin".equals(boschContainerFlashData.getContentFormat())) {
      flashData.setAppChecksum(BoschHelper.hexStringToByteArray(boschContainerFlashData.getCrc32()));
    } 
    try {
      return calculateSwilDriver(componentType, flashData, boschContainerFlashData);
    } catch (IOException ex) {
      flashData.getResult().add("NEGATIVE");
      flashData.getResult().add("FLASH DATA CORRUPT");
      LOGGER.error("Hex file read error", (Throwable)ex);
      return flashData;
    } 
  }

  public boolean processCffParaData(String parType, String parVariant, String parId) {
    SignatureHelper sigChecker = new SignatureHelper();
    BoschContainerDataElementWrapper dataElement = null;
    try {
      dataElement = getParameterDataCff(parType, parVariant, parId);
    } catch (FlashDataNotAvailableException ex) {
      LOGGER.error("Could not read Parameter Data from cff", (Throwable)ex);
      return false;
    } 
    String flashParaDataPath = null;
    flashParaDataPath = this.containerController.getZipPath() + this.containerController.getZipPath() + File.separator;
    String decryptedParaData = null;
    if (StringUtils.equals(dataElement.getKeyId(), "Uniform")) {
      LOGGER.info("Decrypting {}", dataElement.getDataContent());
      try {
        decryptedParaData = CryptoInfo.decryptFile(this.containerController.getZipPath(), dataElement.getDataContent(), dataElement.getKeyIv(), CryptoInfo.getSkey(), CryptoConstants.getcInstance5());
      } catch (FileNotFoundException e) {
        LOGGER.error("File error", (Throwable)e);
      } catch (IOException e) {
        LOGGER.error("File error", (Throwable)e);
      } catch (InvalidKeyException|java.security.InvalidAlgorithmParameterException|java.security.NoSuchAlgorithmException|javax.crypto.NoSuchPaddingException e) {
        LOGGER.error("Decrypt error", (Throwable)e);
      } 
      if (sigChecker.checkSignature(decryptedParaData, dataElement.getSignatureId(), dataElement.getSignatureSalt(), dataElement.getSignatureValue()).contains("NEGATIVE")) {
        LOGGER.warn("Signature check failed");
        FileIOUtil.deleteDirectoryOnExit(this.containerController.getZipPath());
        return false;
      } 
    } else {
      if (sigChecker.checkSignature(flashParaDataPath, dataElement.getSignatureId(), dataElement.getSignatureSalt(), dataElement.getSignatureValue()).contains("NEGATIVE")) {
        LOGGER.warn("Signature check failed");
        FileIOUtil.deleteDirectoryOnExit(this.containerController.getZipPath());
        return false;
      } 
      decryptedParaData = flashParaDataPath;
    } 
    if (StringUtils.equals("tar", dataElement.getContentFormat())) {
      LOGGER.info("Untar {}", new File(decryptedParaData).getName());
      try {
        FileIOUtil.unTar(decryptedParaData);
      } catch (IOException|org.apache.commons.compress.archivers.ArchiveException e) {
        LOGGER.error("Cannot untar file", (Throwable)e);
        FileIOUtil.deleteDirectoryOnExit(this.containerController.getZipPath());
        return false;
      } 
    } else {
      LOGGER.error("Given content format is not 'tar'");
      FileIOUtil.deleteDirectoryOnExit(this.containerController.getZipPath());
      return false;
    } 
    FileIOUtil.deleteDirectoryOnExit(this.containerController.getZipPath());
    return true;
  }
  
  public boolean processCffFlashData(BoschConstants.SwType swType, ComponentType componentType) {
    BoschContainerDataElementWrapper appData = null;
    try {
      appData = getDataElement(swType, componentType);
    } catch (FlashDataNotAvailableException ex) {
      LOGGER.error("could not extract flash data from container", (Throwable)ex);
      return false;
    } 
    String decryptedFlashData = decryptFlashData(swType, componentType);
    if (decryptedFlashData == null)
      return false; 
    if ("tar".equals(appData.getContentFormat())) {
      LOGGER.info("Untar {}", new File(decryptedFlashData).getName());
      try {
        FileIOUtil.unTar(decryptedFlashData);
      } catch (IOException|org.apache.commons.compress.archivers.ArchiveException e) {
        LOGGER.error("Cannot untar file", (Throwable)e);
        FileIOUtil.deleteDirectoryOnExit(this.containerController.getZipPath());
        return false;
      } 
    } else {
      LOGGER.error("Given content format is not 'tar'");
      FileIOUtil.deleteDirectoryOnExit(this.containerController.getZipPath());
      return false;
    } 
    FileIOUtil.deleteDirectoryOnExit(this.containerController.getZipPath());
    return true;
  }
  
  public String decryptFlashData(BoschConstants.SwType swType, ComponentType componentType) {
    BoschContainerDataElementWrapper appData = null;
    try {
      appData = getDataElement(swType, componentType);
    } catch (FlashDataNotAvailableException ex) {
      LOGGER.error("could not extract flash data from container", (Throwable)ex);
      return null;
    } 
    SignatureHelper sigChecker = new SignatureHelper();
    LOGGER.info("Processing {}", appData.getDataContent().trim());
    String flashAppDataPath = this.containerController.getZipPath() + this.containerController.getZipPath() + File.separator;
    String decryptedFlashData = null;
    if (appData.getKeyId().compareTo("Uniform") == 0) {
      LOGGER.info("Decrypting {}", appData.getDataContent().trim());
      try {
        decryptedFlashData = CryptoInfo.decryptFile(this.containerController.getZipPath(), appData.getDataContent().trim(), appData.getKeyIv(), CryptoInfo.getSkey(), 
            CryptoConstants.getcInstance5());
      } catch (FileNotFoundException e) {
        LOGGER.error("File error", (Throwable)e);
      } catch (IOException e) {
        LOGGER.error("File error", (Throwable)e);
      } catch (InvalidKeyException|java.security.NoSuchAlgorithmException|javax.crypto.NoSuchPaddingException|java.security.InvalidAlgorithmParameterException e) {
        LOGGER.error("Decrypt error", (Throwable)e);
      } 
      if (sigChecker.checkSignature(decryptedFlashData, appData.getSignatureId(), appData.getSignatureSalt(), appData.getSignatureValue()).contains("NEGATIVE")) {
        LOGGER.warn("Signature check failed");
        FileIOUtil.deleteDirectoryOnExit(this.containerController.getZipPath());
        return null;
      } 
    } else {
      if (sigChecker.checkSignature(flashAppDataPath, appData.getSignatureId(), appData.getSignatureSalt(), appData.getSignatureValue()).contains("NEGATIVE")) {
        LOGGER.warn("Signature check failed");
        FileIOUtil.deleteDirectoryOnExit(this.containerController.getZipPath());
        return null;
      } 
      decryptedFlashData = flashAppDataPath;
    } 
    FileIOUtil.deleteDirectoryOnExit(this.containerController.getZipPath());
    return decryptedFlashData;
  }
  
  private FlashData calculateSwilDriver(ComponentType componentType, FlashData flashData, BoschContainerDataElementWrapper boschContainerFlashData) throws IOException {
    flashData.setSwilHexData(new ArrayList<>());
    flashData.getSwilHexData().add(null);
    flashData.getSwilHexData().add(null);
    flashData.setSwilDrvChecksum(null);
    flashData.setIsSwilDrvEncrypted(false);
    flashData.setSwilBlockSizes(null);
    if (componentType.isBms() || componentType.isDriveUnit() || componentType.isBcs()) {
      try {
        if (this.containerController.isCff2Loaded())
          extractCff2Data(this.cff2Interface.getFlashDataCff2(BoschConstants.SwType.SW_TYPE_CFF2_SWIL_DRV, componentType), boschContainerFlashData); 
        if (this.containerController.isCff3Loaded())
          extractCff3Data(this.cff3Interface.getFlashDataCff3(BoschConstants.SwType.SW_TYPE_CFF2_SWIL_DRV, componentType), boschContainerFlashData); 
      } catch (FlashDataNotAvailableException ex) {
        flashData.getResult().add("NEGATIVE");
        flashData.getResult().add("FLASH DATA CORRUPT");
        LOGGER.error("Flash data not found in container", (Throwable)ex);
        return flashData;
      } 
      return getSwilDriver(flashData, boschContainerFlashData);
    } 
    return flashData;
  }
  
  private FlashData checkFlashFileEncryption(FlashData flashData, BoschContainerDataElementWrapper boschContainerFlashData) throws IOException {
    String flashAppDataPath = this.containerController.getZipPath() + this.containerController.getZipPath() + File.separator;
    SignatureHelper signatureHelper = new SignatureHelper();
    byte[] binFlashData = BoschHelper.readBinaryData(flashAppDataPath);
    if (boschContainerFlashData.getKeyId().compareTo("Uniform") == 0) {
      byte[] decryptedFlashData = CryptoHelper.decrypt(binFlashData, boschContainerFlashData.getKeyIv(), null, CryptoInfo.getSkey(), CryptoConstants.getcInstance5());
      flashData.setResult(signatureHelper.checkSignature(decryptedFlashData, boschContainerFlashData.getSignatureId(), boschContainerFlashData.getSignatureSalt(), boschContainerFlashData.getSignatureValue()));
      if (flashData.getResult().contains("NEGATIVE"))
        return flashData; 
      if ("hex".equals(boschContainerFlashData.getContentFormat())) {
        flashData.setAppHexData(IntelHexReader.readHexFile(new String(decryptedFlashData)));
      } else if ("bin".equals(boschContainerFlashData.getContentFormat())) {
        flashData.setAppData(decryptedFlashData);
      } 
    } else {
      flashData.setResult(signatureHelper.checkSignature(binFlashData, boschContainerFlashData.getSignatureId(), boschContainerFlashData.getSignatureSalt(), boschContainerFlashData.getSignatureValue()));
      if (flashData.getResult().contains("NEGATIVE"))
        return flashData; 
      if ("hex".equals(boschContainerFlashData.getContentFormat())) {
        flashData.setAppHexData(IntelHexReader.readHexFile(flashAppDataPath));
      } else if ("bin".equals(boschContainerFlashData.getContentFormat())) {
        flashData.setAppData(binFlashData);
      } 
    } 
    return flashData;
  }
  
  private FlashData getSwilDriver(FlashData flashData, BoschContainerDataElementWrapper boschContainerFlashData) throws IOException {
    SignatureHelper signatureHelper = new SignatureHelper();
    String swilDrvDataPath = this.containerController.getZipPath() + this.containerController.getZipPath() + File.separator;
    byte[] drvData = BoschHelper.readBinaryData(swilDrvDataPath);
    if (boschContainerFlashData.getKeyId().compareTo("Uniform") == 0) {
      byte[] decryptedData = CryptoHelper.decrypt(drvData, boschContainerFlashData.getKeyIv(), null, CryptoInfo.getSkey(), CryptoConstants.getcInstance5());
      flashData.setResult(signatureHelper.checkSignature(decryptedData, boschContainerFlashData.getSignatureId(), boschContainerFlashData.getSignatureSalt(), boschContainerFlashData.getSignatureValue()));
      if (flashData.getResult().contains("NEGATIVE"))
        return flashData; 
      if ("hex".equals(boschContainerFlashData.getContentFormat())) {
        flashData.setSwilHexData(IntelHexReader.readHexFile(new String(decryptedData)));
      } else if ("bin".equals(boschContainerFlashData.getContentFormat())) {
        flashData.setSwilDrvData(decryptedData);
      } 
    } else {
      flashData.setResult(signatureHelper.checkSignature(drvData, boschContainerFlashData.getSignatureId(), boschContainerFlashData.getSignatureSalt(), boschContainerFlashData.getSignatureValue()));
      if (flashData.getResult().contains("NEGATIVE"))
        return flashData; 
      if ("hex".equals(boschContainerFlashData.getContentFormat())) {
        flashData.setSwilHexData(IntelHexReader.readHexFile(swilDrvDataPath));
      } else if ("bin".equals(boschContainerFlashData.getContentFormat())) {
        flashData.setSwilDrvData(drvData);
      } 
    } 
    if ("hex".equals(boschContainerFlashData.getContentFormat())) {
      String fblSignatureValue = boschContainerFlashData.getFblSignatureValue().replaceAll("0x", "").replaceAll(",", "");
      flashData.setSwilDrvChecksum(BoschHelper.hexStringToByteArray(fblSignatureValue));
      flashData.setIsSwilDrvEncrypted(boschContainerFlashData.isFblEncryption());
      if (flashData.isIsSwilDrvEncrypted())
        flashData.setSwilBlockSizes(IntelHexReader.readBlockSizes(boschContainerFlashData.getFblBlockSize())); 
      if (flashData.getSwilBlockSizes() == null || flashData.getSwilBlockSizes().isEmpty()) {
        flashData.setSwilBlockSizes(new ArrayList<>());
        flashData.getSwilHexData().get(1).forEach(hexBlock2 -> flashData.getSwilBlockSizes().add(Integer.valueOf(hexBlock2.length)));
      } 
    } else if ("bin".equals(boschContainerFlashData.getContentFormat())) {
      flashData.setSwilDrvChecksum(BoschHelper.hexStringToByteArray(boschContainerFlashData.getCrc32()));
    } 
    return flashData;
  }

  public void deletePreviousLargeContainers() {
    try {
      File javaTmpDirectory = FileIOUtil.getJavaTmpDirectory();
      for (File file : Objects.<File[]>requireNonNull(javaTmpDirectory.listFiles())) {
        String name = file.getName();
        if ((name.endsWith("Unzipped") || name.startsWith("NanoHTTPD") || name.startsWith("LargeContainer") || name.startsWith("Ebike_tmp")) && !file.getPath().equals(this.gen2Impl.getZipPath()) && file.isDirectory()) {
          long directorySize = FileUtils.sizeOfDirectory(file);
          if (directorySize > 104857600L)
            FileUtils.forceDelete(file);

    }

    return false;
  }
  
  private boolean getCffContainerFromFileSystem(File file, boolean delete, BoschContainerController.ContainerType containertype) {
    boolean result = false;
    try {
      String zipPath = unzipContainer(file, delete);
      if (containertype == BoschContainerController.ContainerType.CFF2) {
        this.gen2Impl.setZipPath(zipPath);
      } else {
        this.zipPathCff3 = zipPath;
      } 
      setCffMetaData(file, containertype);
      result = true;
    } catch (IOException ex) {
      LOGGER.error("Could not access meta.xml in container", (Throwable)ex);
    } catch (JAXBException|InstantiationException|IllegalAccessException ex) {
      LOGGER.error("Error reading meta.xml, xml malformed or not valid", (Throwable)ex);
    } catch (SecurityException ex) {
      LOGGER.error("Wrong Signature", (Throwable)ex);
    } 
    return result;
  }

    } else if (isCff2Loaded()) {
      cffVersion = getCff2Interface().getSwVersionCff2(component);

  public List<String> getReleaseNotesCff2(String compTypeEBike, BoschConstants.SwType swTypeEBike, String swVersionEBike, String swVariantEBike) {
    List<String> releaseNotes = new ArrayList<>();
    for (Cff2_ExecutableReply cExec : this.ebike.getReply().getExecutableReply()) {
      if (swTypeEBike == null) {
        releaseNotes.add(".");
        return releaseNotes;
      } 
      if (swVersionEBike == null) {
        releaseNotes.add(".");
        return releaseNotes;
      } 
      if (swVariantEBike == null) {
        releaseNotes.add(".");
        return releaseNotes;
      } 
      if (StringUtils.equals(cExec.getSwType(), swTypeEBike.getContainerSymbol()) && StringUtils.equals(cExec.getSwVariant(), swVariantEBike))
        try {
          Version cffVersion = Version.parse(cExec.getSwVersion());
          Version ecuVersion = Version.parse(swVersionEBike);
          if (cffVersion.compareTo(ecuVersion) >= 0)
            for (Cff2_ComponentCompatibility cCompat : cExec.getComponentCompatibility()) {
              if (cCompat.getHwType().compareTo(compTypeEBike) == 0) {
                Cff2_ReleaseNotes cffReleaseNotes = cExec.getReleaseNotes();
                if (cffReleaseNotes == null)
                  return releaseNotes; 
                for (Cff2_ReleaseNote reNot : cffReleaseNotes.getReleaseNote()) {
                  if (!getReleaseNoteForLang(I18NUtil.getLocale(), reNot.getReleaseNoteText(), releaseNotes))
                    getReleaseNoteForLang(Locale.ENGLISH, reNot.getReleaseNoteText(), releaseNotes); 
                } 
              } 
            }  
        } catch (IOException e) {
          LOGGER.error("Failure getting release notes because of version parse error for cff version '{}' or ecu version '{}'.", cExec.getSwVersion(), swVersionEBike, e);
        }  
    } 
    return releaseNotes;
  }

import java.io.FileReader;

import java.io.InputStream;

import org.apache.commons.lang3.StringUtils;

public class IntelHexReader {
  private static final int RECORD_TYPE_BEGIN_INDEX = 7;
  
  private static final int RECORD_TYPE_END_INDEX = 9;
  
  private static final int CHECKSUM_LENGTH = 2;
  
  public static List<List<byte[]>> readHexFile(String hexFile) throws IOException {
    BufferedReader br = null;
    byte[] blockAddress = null;
    int baseAddress = 0;
    int lastWrittenByte = 0;
    StringBuilder sb = new StringBuilder();
    List<byte[]> addresses = new ArrayList<>();
    List<byte[]> datas = new ArrayList<>();
    List<List<byte[]>> result = new ArrayList<>();
    result.add(addresses);
    result.add(datas);
    try {
      File f = new File(hexFile);
      if (f.exists()) {
        br = new BufferedReader(new FileReader(hexFile));
      } else {
        InputStream is = new ByteArrayInputStream(hexFile.getBytes());
        br = new BufferedReader(new InputStreamReader(is));
      } 
      String line = br.readLine();
      while (line != null) {
        String recordType = line.substring(7, 9);
        if ("02".equals(recordType)) {
          String sbaString = line.substring(9, 13);
          byte[] sbaBa = BoschHelper.hexStringToByteArray(sbaString);
          int sbaInt = BoschHelper.byteArrayToInteger(sbaBa);
          baseAddress = sbaInt << 4;
        } else if ("00".equals(recordType)) {
          String recordAddress = line.substring(3, 7);
          byte[] recordAddressArray = BoschHelper.hexStringToByteArray(recordAddress);
          int recordAddressInt = BoschHelper.byteArrayToInteger(recordAddressArray);
          int absoluteAddressInt = baseAddress + recordAddressInt;
          if (lastWrittenByte == 0) {
            blockAddress = BoschHelper.integerToByteArray(absoluteAddressInt);
            if (blockAddress.length < 4) {
              byte[] zero = { 0, 0 };
              blockAddress = BoschHelper.concatByteArrays(zero, blockAddress);
            } 
            sb.append(line.substring(9, line.length() - 2));
          } else if (absoluteAddressInt == lastWrittenByte) {
            sb.append(line.substring(9, line.length() - 2));
          } else {
            byte[] dataArray = BoschHelper.hexStringToByteArray(sb.toString());
            addresses.add(blockAddress);
            datas.add(dataArray);
            sb.delete(0, sb.length());
            blockAddress = BoschHelper.integerToByteArray(absoluteAddressInt);
            if (blockAddress.length < 4) {
              byte[] zero = { 0, 0 };
              blockAddress = BoschHelper.concatByteArrays(zero, blockAddress);
            } 
            sb.append(line.substring(9, line.length() - 2));
          } 
          String recordLengthString = line.subSequence(1, 3).toString();
          byte[] recordLengthArray = BoschHelper.hexStringToByteArray(recordLengthString);
          int recordLengthInt = BoschHelper.byteArrayToInteger(recordLengthArray);
          lastWrittenByte = absoluteAddressInt + recordLengthInt;
        } else if ("04".equals(recordType)) {
          String elaString = line.substring(9, 13);
          byte[] elaBa = BoschHelper.hexStringToByteArray(elaString);
          int elaInt = BoschHelper.byteArrayToInteger(elaBa);
          baseAddress = elaInt << 16;
        } else {
          if ("01".equals(recordType)) {
            byte[] dataArray = BoschHelper.hexStringToByteArray(sb.toString());
            addresses.add(blockAddress);
            datas.add(dataArray);
            result.set(0, addresses);
            result.set(1, datas);
            return result;
          } 
          if (!"05".equals(recordType))
            throw new IOException("Incorrect hex file: unknown record type." + recordType); 
        } 
        line = br.readLine();
      } 
    } finally {
      if (br != null)
        br.close(); 
    } 
    result.set(0, addresses);
    result.set(1, datas);
    return result;
  }

      if (StringUtils.isNotBlank(token))
        result.add(Integer.valueOf(BoschHelper.hex2int(token.replaceAll(" ", "")))); 
    } 
    return result;
  }

  private static final String ERROR_INVALID_SIGNATURE_ID = "Invalid Signature Id for flash data";
  
  private static final String ERROR_SIGNATURE_CORRUPT = "Signature corrupt for flash data";
  
  private static final String UNIFORM_UPDATE_SERVER = "UniformUpdateServer";
  
  private static final String SIGNATURE_WRONG = "SIGNATURE WRONG";
  
  private static final String SIGNATURE_ID_INVALID = "INVALID SIGNATURE ID";
  
  public List<String> checkSignature(byte[] flashData, String signatureId, String signatureSalt, String signatureValue) {
    List<String> info = new ArrayList<>();
    if (isArrayInitialized(flashData) && StringUtils.isNotBlank(signatureId) && StringUtils.isNotBlank(signatureSalt) && StringUtils.isNotBlank(signatureValue))
      if (StringUtils.equals(signatureId, "UniformUpdateServer")) {
        if (!checkRsaSignature(flashData, signatureSalt, signatureValue, CryptoInfo.getX509())) {
          info.add("NEGATIVE");
          info.add("SIGNATURE WRONG");
          LOGGER.error("Signature corrupt for flash data");
        } 
      } else if (!StringUtils.isBlank(signatureId)) {
        info.add("NEGATIVE");
        info.add("INVALID SIGNATURE ID");
        LOGGER.error("Invalid Signature Id for flash data");
      }  
    return info;
  }
  
  private boolean isArrayInitialized(byte[] data) {
    return (data != null && data.length > 0);
  }
  
  private boolean checkRsaSignature(byte[] byteContent, String saltString, String signatureValue, X509EncodedKeySpec x509) {
    return this.unitokenSigHelper.checkRsaSignature(byteContent, saltString, signatureValue, x509);
  }
  
  public List<String> checkSignature(String file, String signatureId, String signatureSalt, String signatureValue) {
    List<String> info = new ArrayList<>();
    if ("UniformUpdateServer".equals(signatureId)) {
      if (!checkRsaSignature(file, signatureSalt, signatureValue, CryptoInfo.getX509())) {
        info.add("NEGATIVE");
        info.add("SIGNATURE WRONG");
        LOGGER.error("Signature corrupt for flash data");
      } 
    } else if (!StringUtils.isBlank(signatureId)) {
      info.add("NEGATIVE");
      info.add("INVALID SIGNATURE ID");
      LOGGER.error("Invalid Signature Id for flash data");
    } 
    return info;
  }
  
  private boolean checkRsaSignature(String file, String saltString, String signatureValue, X509EncodedKeySpec x509) {
    try {
      PublicKey publicKey = KeyFactory.getInstance(CryptoConstants.getkInstanceRsa()).generatePublic(x509);
      byte[] salt = DatatypeConverter.parseBase64Binary(saltString);
      Signature signature = Signature.getInstance(CryptoConstants.getsInstanceSha256Rsa());
      signature.initVerify(publicKey);
      signature.update(salt);
      byte[] buf = new byte[1024];
      try (BufferedInputStream inputStream = new BufferedInputStream(new FileInputStream(new File(file)))) {
        int bytesRead;
        while ((bytesRead = inputStream.read(buf)) != -1)
          signature.update(buf, 0, bytesRead); 
      } 
      return signature.verify(DatatypeConverter.parseBase64Binary(signatureValue));
    } catch (NoSuchAlgorithmException|java.security.InvalidKeyException|java.security.SignatureException|java.io.IOException|java.security.spec.InvalidKeySpecException e) {
      LOGGER.error("RSA signature check error", (Throwable)e);
      return false;
    } 
  }
