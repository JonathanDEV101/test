public class LogfileUtils {
  private static final Logger LOGGER = LoggerFactory.getLogger(LogfileUtils.class);
  
  public static void deleteFilesOlderThanDays(String path, int days) {
    File directory = new File(path);
    if (directory.exists()) {
      IOFileFilter filter = createLogFilesOlderThanDaysFilter(days);
      deleteFilesInFolder(directory, filter);
      File[] subDirectories = directory.listFiles((FileFilter)FileFilterUtils.directoryFileFilter());
      if (subDirectories != null)
        for (File subDirectory : subDirectories) {
          if (subDirectory.getAbsolutePath().endsWith("setup")) {
            deleteOldSetupLogFiles(subDirectory.getAbsolutePath());
          } else {
            deleteFilesOlderThanDays(subDirectory.getAbsolutePath(), days);
          } 
        }  
    } 
  }
  
  private static void deleteOldSetupLogFiles(String path) {
    File directory = new File(path);
    if (directory.exists()) {
      SuffixFileFilter filter = new SuffixFileFilter("txt");
      deleteFilesInFolder(directory, filter);
    } 
  }
  
  private static void deleteFilesInFolder(File dir, IOFileFilter filter) {
    Iterator<File> filesToDelete = FileUtils.iterateFiles(dir, filter, null);
    while (filesToDelete.hasNext()) {
      File fileToDelete = filesToDelete.next();
      LOGGER.info("Deleting file: {}", fileToDelete);
      if (!fileToDelete.delete())
        LOGGER.error("Could not delete file '{}'", fileToDelete); 
    } 
  }
  
  private static IOFileFilter createLogFilesOlderThanDaysFilter(int thresholdDays) {
    Date thresholdDate = createThresholdDate(thresholdDays);
    AndFileFilter filter = new AndFileFilter();
    filter.addFileFilter(new AgeFileFilter(thresholdDate));
    filter.addFileFilter(new SuffixFileFilter("log"));
    return filter;
  }
  
  private static Date createThresholdDate(int thresholdDays) {
    LocalDate thresholdLocalDate = LocalDate.now().minusDays((long)thresholdDays);
    return Date.from(thresholdLocalDate.atStartOfDay(ZoneId.systemDefault()).toInstant());
  }
}

      pathLine = pathLine.substring(index);
      int checkMaxPos = Math.min(100, pathLine.length());
      int nextSlash = pathLine.lastIndexOf('\\', checkMaxPos);
      int nextBreakSymbol = Math.max(nextSlash, pathLine.lastIndexOf(' ', checkMaxPos));
      int maximumLength = Math.min(100, (nextBreakSymbol > -1) ? nextBreakSymbol : 100);
      int breakPoint = Math.min(maximumLength, pathLine.length());
      if (pathLine.length() <= 100)
        breakPoint = pathLine.length(); 
      String toAdd = pathLine.substring(0, (breakPoint != pathLine.length()) ? (breakPoint + 1) : breakPoint);
      if (toAdd.indexOf('(') > -1 && toAdd.indexOf(')') < 0) {
        messageTokens.add(toAdd.concat("..."));
        insidePath = true;
        isModified = true;
      } 
      if (toAdd.indexOf(')') > -1) {
        messageTokens.add("...".concat(toAdd));
        insidePath = false;
        isModified = true;
      }

        messageTokens.add("...".concat(toAdd));
      } else if (!isModified) {

      }

import java.io.IOException;
import java.io.InputStream;
import java.io.OutputStream;
import java.util.concurrent.TimeUnit;
import java.util.function.Consumer;
import org.apache.commons.io.IOUtils;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

public class ProcessUtil {
  private static final Logger LOGGER = LoggerFactory.getLogger(ProcessUtil.class);
  
  public Execution execute(long timeout, String... command) throws IOException {
    try {
      String logInfo = String.format("Execute: '%s' with timeout of '%d' ms", command, timeout);
      LOGGER.info(logInfo);
      Process process = new ProcessBuilder(command).start();
      if (process.waitFor(timeout, TimeUnit.MILLISECONDS)) {
        Execution execution = new Execution(process.exitValue(), 
            IOUtils.toString(process.getInputStream()), 
            IOUtils.toString(process.getErrorStream()));
        logInfo = String.format("Executed with result: '%s'", execution.toString());
        LOGGER.info(logInfo);
        return execution;
      } 
      LOGGER.info("Timeout in execution");
      int outAvailable = process.getInputStream().available();
      byte[] outBuffer = new byte[outAvailable];
      IOUtils.read(process.getInputStream(), outBuffer, 0, outAvailable);
      int errAvailable = process.getErrorStream().available();
      byte[] errBuffer = new byte[errAvailable];
      IOUtils.read(process.getErrorStream(), errBuffer, 0, errAvailable);
      return new Execution(-1000, 
          IOUtils.toString(outBuffer, "UTF-8"), 
          IOUtils.toString(errBuffer, "UTF-8"));
    } catch (InterruptedException e) {
      LOGGER.warn("Thread interrupted during execution.");
      Thread.currentThread().interrupt();
    } catch (IOException e) {
      LOGGER.error("Execution failed", (Throwable)e);
      throw e;
    } 
    throw new IllegalStateException("Thread was interrupted");
  }
  
  public Execution execute(Consumer<InputStream> outConsumer, String... command) throws InterruptedException, IOException {
    Process process = null;
    try {
      LOGGER.trace("Execute: '{}'", (Object[])command);
      process = new ProcessBuilder(command).start();
      outConsumer.accept(process.getErrorStream());
      process.waitFor();
      return new Execution(process.exitValue(), "", "");
    } catch (InterruptedException e) {
      LOGGER.warn("Thread interrupted during execution");
      if (process.isAlive()) {
        LOGGER.info("Thread interrupted, quit process...");
        OutputStream os = process.getOutputStream();
        os.write(113);
        os.flush();
      } 
      throw e;
    } catch (IOException e) {
      LOGGER.error("Execution failed", (Throwable)e);
      throw e;
    } 
  }
}

  private static final String SCHTASKS_EXE_PATH = "schtasks";
  
  private static final long SC_TIMEOUT = 5000L;
  
  private static final long SCHTASKS_TIMEOUT = 1000L;
  
  private static final String SCHEDULED_TASK_NAME = "Bosch - start ";
  
  private static final String SC_COMMAND_QUERY = "query";
  
  private static final String SC_COMMAND_START = "start";
  
  private static final String SC_COMMAND_STOP = "stop";
  
  private static final String SERVICE_STATE_RUNNING = "RUNNING";
  
  private static final String SERVICE_STATE_STOPPED = "STOPPED";
  
  private static final String SERVICE_STATE_START_PENDING = "START_PENDING";
  
  private static final String SERVICE_STATE_STOP_PENDING = "STOP_PENDING";
  
  public enum ServiceState {
    RUNNING, STOPPED, START_PENDING, STOP_PENDING, UNKNOWN_STATE, UNKNOWN_SERVICE;
  }
  
  public ServiceState checkService(String serviceName) throws IOException {
    LOGGER.info("Checking state of service {}...", serviceName);
    Execution execution = executeSc(serviceName, "query");
    return getState(serviceName, execution);
  }
  
  public ServiceState startService(String serviceName) throws IOException {
    LOGGER.debug("Start {} with current rights", serviceName);
    Execution execution = executeSc(serviceName, "start");
    if (execution != null && execution.exitCode != 0) {
      LOGGER.debug("Start {} by scheduled task", serviceName);
      execution = executeSchtasks(serviceName);
      if (execution != null && execution.exitCode != 0) {
        LOGGER.debug("Start {} with administrator rights", serviceName);
        execution = executeScAsAdministrator(serviceName, "start");
      } 
      if (execution != null && execution.exitCode == 0)
        execution = executeSc(serviceName, "query"); 
    } 
    ServiceState state = getState(serviceName, execution);
    if (state == ServiceState.RUNNING || state == ServiceState.START_PENDING) {
      LOGGER.info("Service {} started, state={}", serviceName, state);
    } else {
      LOGGER.debug("Service {} not started, state={}, execution={}", serviceName, state, execution);
    } 
    return state;
  }
  
  public boolean stopService(String serviceName) throws IOException {
    Execution execution = executeSc(serviceName, "stop");
    ServiceState state = getState(serviceName, execution);
    if (execution != null && execution.exitCode == 0 && (state == ServiceState.STOPPED || state == ServiceState.STOP_PENDING)) {
      LOGGER.info("Service {} stopped. Current state {}", serviceName, state);
      return true;
    } 
    return false;
  }
  
  private Execution executeSchtasks(String serviceName) throws IOException {
    return execute(1000L, "schtasks", "/run", "/TN", "\"Bosch - start " + serviceName + "\"");
  }
  
  private Execution executeSc(String serviceName, String command) throws IOException {
    return execute(5000L, "C:\\Windows\\System32\\sc.exe", command, serviceName);
  }
  
  private Execution executeScAsAdministrator(String serviceName, String command) throws IOException {
    return execute(5000L, "powershell", "Start-Process", "C:\\Windows\\System32\\sc.exe", "-ArgumentList", "'" + command + " " + serviceName + "'", "-WindowStyle", "Hidden", "-Wait", "-verb", "RunAs");
  }
  
  private Execution execute(long timeout, String... command) throws IOException {
    String commandString = Arrays.toString(command);
    try {
      LOGGER.trace("Execute \"{}\", timeout={}", commandString, Long.valueOf(timeout));
      Process process = new ProcessBuilder(command).start();
      if (process.waitFor(timeout, TimeUnit.MILLISECONDS)) {
        Execution execution = new Execution(process.exitValue(), IOUtils.toString(process.getInputStream()), 
            IOUtils.toString(process.getErrorStream()));
        LOGGER.trace("Executed \"{}\", execution result={}", commandString, execution);
        return execution;
      } 
      LOGGER.trace("Timeout \"{}\"", commandString);
      return null;
    } catch (InterruptedException e) {
      LOGGER.debug("Thread interrupted during {} execution", (Object[])command);
      Thread.currentThread().interrupt();
    } catch (IOException e) {
      LOGGER.debug("Failed to execute \"{}\"", commandString, e);
      throw e;
    } 
    throw new IllegalStateException("Thread was interrupted");
  }
  
  private ServiceState getState(String serviceName, Execution execution) {
    if (execution != null && execution.exitCode == 0) {
      if (!execution.out.contains("STATE"))
        LOGGER.info("'STATE' of Service '{}' not found in execution response, ignore and check for possible state values", serviceName); 
      if (execution.out.contains("RUNNING")) {
        LOGGER.info("Service {} is running", serviceName);
        return ServiceState.RUNNING;
      } 
      if (execution.out.contains("STOPPED")) {
        LOGGER.info("Service {} is stopped", serviceName);
        return ServiceState.STOPPED;
      } 
      if (execution.out.contains("START_PENDING")) {
        LOGGER.info("Service {} is pending start", serviceName);
        return ServiceState.START_PENDING;
      } 
      if (execution.out.contains("STOP_PENDING")) {
        LOGGER.info("Service {} is pending stop", serviceName);
        return ServiceState.STOP_PENDING;
      } 
      return ServiceState.RUNNING;
    } 
    return ServiceState.UNKNOWN_SERVICE;
  }
  
  private static final class Execution {
    final int exitCode;
    
    final String out;
    
    final String err;
    
    Execution(int exitCode, String out, String err) {
      this.exitCode = exitCode;
      this.out = out;
      this.err = err;
    }
    
    public String toString() {
      return "exitCode=" + this.exitCode + ", out='" + this.out + "', err='" + this.err + "'";
    }
  }
}
