  private static final int GENERAL_INITIALIZATION_DELAY_DEFAULT = 1000;
  
  private static final int BUI27X_INITIALIZATION_DELAY_DEFAULT = 2000;
  
  private static final int BUI350_INITIALIZATION_DELAY_DEFAULT = 2000;
  
  private static final int HANDSHAKE_RETRY_DELAY_DEFAULT = 500;
  
  private static final int MAX_RETRY_HANDSHAKE = 3;
  
  private final int bui27xInitDelay;
  
  private final int bui350InitDelay;
  
  private final int hidInitDelay;
  
  private final int handshakeRetryDelay;
  
  private Constants.HidDevice deviceInProgress = Constants.HidDevice.NONE;
  
  private Constants.HidDevice connectedDevice = Constants.HidDevice.NONE;
  
  private final CanIsoTpComLayer canIsoTpComLayer;
  
  private final DeviceConnectionChangeEventPublisher deviceConnectionChangeEventPublisher;
  
  private Timer delayedConnectTimer = new Timer("DelayedConnectTimer-" + Thread.currentThread().getId());
  
  BoschDeviceConnectionHandler(CanIsoTpComLayer comLayer, DeviceConnectionChangeEventPublisher publisher) {
    this.canIsoTpComLayer = comLayer;
    this.deviceConnectionChangeEventPublisher = publisher;
    this.hidInitDelay = SettingsProxy.getInstance().getGlobalSettingsNoException().getInteger("hidInitDelay", 1000);
    LOGGER.debug("Set general device initialization delay to {} ms", Integer.valueOf(this.hidInitDelay));
    this.bui27xInitDelay = SettingsProxy.getInstance().getGlobalSettingsNoException().getInteger("bui27xInitDelay", 2000);
    LOGGER.debug("Set BUI27x initialization delay to {} ms", Integer.valueOf(this.bui27xInitDelay));
    this.bui350InitDelay = SettingsProxy.getInstance().getGlobalSettingsNoException().getInteger("bui350InitDelay", 2000);
    LOGGER.debug("Set BUI350 initialization delay to {} ms", Integer.valueOf(this.bui350InitDelay));
    this.handshakeRetryDelay = SettingsProxy.getInstance().getGlobalSettingsNoException().getInteger("handshakeRetryDelay", 500);
    LOGGER.debug("Set handshake retry delay to {} ms", Integer.valueOf(this.handshakeRetryDelay));
  }
  
  synchronized void handleDeviceConnectionChangedEvent(HidDeviceConnectionChangedEvent event) {
    Constants.HidDevice device = Constants.HidDevice.fromPid(event.getProductId());
    if (device != null)
      if (event.getState()) {
        handleDeviceConnect(event, device);
      } else {
        handleDeviceDisconnect(event, device);
      }  
  }
  
  private void handleDeviceConnect(final HidDeviceConnectionChangedEvent event, final Constants.HidDevice device) {
    TimerTask initializationTask = new TimerTask() {
        public void run() {
          BoschDeviceConnectionHandler.LOGGER.debug("Delay's over, so check connection conditions and initialize device '{}' if necessary", device);
          if (BoschDeviceConnectionHandler.this.connectedDevice == Constants.HidDevice.NONE && BoschDeviceConnectionHandler.this.initDevice(device)) {
            BoschDeviceConnectionHandler.this.connectedDevice = device;
            BoschDeviceConnectionHandler.this.deviceConnectionChangeEventPublisher.publishEvent(event);
          } 
          BoschDeviceConnectionHandler.this.deviceInProgress = Constants.HidDevice.NONE;
        }
      };
    LOGGER.info("HID device '{}' connected", device);
    if (this.deviceInProgress != Constants.HidDevice.NONE || this.connectedDevice != Constants.HidDevice.NONE) {
      LOGGER.warn("HID device connect already in progress, so ignore connect of '{}'.", device);
      return;
    } 
    this.deviceInProgress = device;
    this.delayedConnectTimer = new Timer("DelayedConnectTimer-" + Thread.currentThread().getId());
    this.delayedConnectTimer.schedule(initializationTask, (long)getInitializationDelayTime(device));
  }
  
  private void handleDeviceDisconnect(HidDeviceConnectionChangedEvent event, Constants.HidDevice device) {
    if (this.deviceInProgress == device) {
      LOGGER.info("Not yet initialized device '{}' disconnected, cancel further initialization.", device);
      this.delayedConnectTimer.cancel();
      this.canIsoTpComLayer.closeDevice();
      this.deviceInProgress = Constants.HidDevice.NONE;
    } 
    if (this.connectedDevice == device) {
      LOGGER.info("Initialized device '{}' disconnected, inform listeners.", device);
      this.canIsoTpComLayer.closeDevice();
      this.connectedDevice = Constants.HidDevice.NONE;
      this.deviceConnectionChangeEventPublisher.publishEvent(event);
    } 
  }
  
  private boolean initDevice(Constants.HidDevice device) {
    for (int n = 1; n <= 3; n++) {
      LOGGER.debug("Handshake try no. '{}' with '{}'", Integer.valueOf(n), device);
      if (this.canIsoTpComLayer.initHidDevice(device) == device) {
        LOGGER.debug("Success handshaking with '{}'", device);
        return true;
      } 
      try {
        Thread.sleep((long)this.handshakeRetryDelay);
      } catch (InterruptedException ex) {
        Thread.currentThread().interrupt();
        LOGGER.error("Sleep after failed handshake interrupted", (Throwable)ex);
      } 
    } 
    LOGGER.error("Handshaking with '{}' failed", device);
    return false;
  }
  
  private int getInitializationDelayTime(Constants.HidDevice device) {
    int usedDelayTime;
    if (device == Constants.HidDevice.BUI270_APP || device == Constants.HidDevice.BUI270_FBL || device == Constants.HidDevice.BUI270_STO || device == Constants.HidDevice.BUI275_APP || device == Constants.HidDevice.BUI275_FBL || device == Constants.HidDevice.BUI275_STO) {
      usedDelayTime = this.bui27xInitDelay;
    } else if (device == Constants.HidDevice.BUI350_APP || device == Constants.HidDevice.BUI350_FBL) {
      usedDelayTime = this.bui350InitDelay;
    } else {
      usedDelayTime = this.hidInitDelay;
    } 
    LOGGER.debug("Wait '{}' ms before starting initialization of device '{}'", Integer.valueOf(usedDelayTime), device);
    return usedDelayTime;
  }
}

      try(InputStream in = getClass().getResourceAsStream(file); 
          CipherInputStream cin = new CipherInputStream(in, c); 
          ObjectInputStream oin = new ObjectInputStream(cin)) {
        SealedObject sealedObject = (SealedObject)oin.readObject();

      } catch (IOException|ClassNotFoundException|javax.crypto.BadPaddingException ex) {

      }

public class SessionKeepAliveHandler {

  private static final long DELAY_TIME_BEFORE_SENDING_FIRST_FUNCTIONAL_TESTER_PRESENT_AFTER_START = 1000L;

  public void adaptKeepAliveUdsSessionHandlingState(byte[] pduRx, ComCtrl communicationPartner) {
    if (!this.isAutomaticAdaptionFunctionTesterPresentTaskSuspended && isValidResponse(pduRx)) {
      checkCommunicationPartnerForUpdateTargetAddressNecessity(pduRx, communicationPartner);
      checkPduForUpdateFailureHandlingNecessity(pduRx, communicationPartner);
      checkPduForTaskStartNecessity(pduRx);
    } 
  }
  
  private void checkPduForUpdateFailureHandlingNecessity(byte[] pduRx, ComCtrl communicationPartner) {
    if (isHidDevice(communicationPartner) && requestTriggersEcuReboot(pduRx))
      updateTransmissionFailureToleranceTimestamp(); 
  }
  
  private boolean isHidDevice(ComCtrl communicationPartner) {
    String ecuName = communicationPartner.getEcuName();
    return (ecuName.contains("BUI") || ecuName.contains("CUI") || ecuName.contains("BCT"));
  }
  
  private boolean requestTriggersEcuReboot(byte[] pduRx) {
    return (isHardResetResponse(pduRx) || isProgrammingSessionResponse(pduRx) || isBctWriteBatteryTypeResponse(pduRx));
  }
  
  private boolean isBctWriteBatteryTypeResponse(byte[] pduRx) {
    return (pduRx.length > 2 && pduRx[0] == 110 && pduRx[1] == Byte.MIN_VALUE && pduRx[2] == 1);
  }
  
  private boolean isProgrammingSessionResponse(byte[] pduRx) {
    return (pduRx[0] == 80 && pduRx[1] == 2);
  }
  
  private boolean isHardResetResponse(byte[] pduRx) {
    return ((pduRx[0] == Byte.MAX_VALUE && pduRx[1] == 17) || (pduRx[0] == 81 && pduRx[1] == 1));
  }
  
  private synchronized void updateTransmissionFailureToleranceTimestamp() {
    LOGGER.debug("Functional tester present transmission failures are tolerated for the next '{}' sec", Long.valueOf(10L));
    this.lastEcuRebootTimestamp = System.currentTimeMillis();
  }
  
  private void checkPduForTaskStartNecessity(byte[] pduRx) {
    if (!this.isFunctionalTesterPresentTaskRunning && doesPduTriggerModeChangeInEcuToNonDefaultSession(pduRx))
      startCyclicFunctionalTesterPresentMessageSendingIfNotAlreadyRunning(); 
  }
  
  private void checkCommunicationPartnerForUpdateTargetAddressNecessity(byte[] pduRx, ComCtrl communicationPartner) {

  }

    if (!this.isAutomaticAdaptionFunctionTesterPresentTaskSuspended) {

    }

    if (this.isFunctionalTesterPresentTaskRunning) {

    } else {
      stopCyclicFunctionalTesterPresentMessageSendingIfActive();
    } 
  }

    return (System.currentTimeMillis() - this.lastEcuRebootTimestamp < 10000L);

package com.gigatronik.diagserver.bosch.reboot;

import com.gigatronik.beds.service.hiddriverwrapper.api.HidDeviceConnectionChangedEvent;
import com.gigatronik.diagserver.bosch.impl.ComCtrl;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

public class CanEcuRebootHandler implements EcuRebootHandler {
  private static final Logger LOGGER = LoggerFactory.getLogger(CanEcuRebootHandler.class);
  
  private static final int TIMEOUT_CAN_ECU_RECONNECT = 3000;
  
  private static final int WAITING_TIME_BETWEEN_POLLING_CYCLES_CAN_ECU_RECONNECT = 250;
  
  protected final ComCtrl communicationController;
  
  protected final String ecuName;
  
  CanEcuRebootHandler(ComCtrl communicationController) {
    this.communicationController = communicationController;
    this.ecuName = communicationController.getEcuName();
  }
  
  public EcuRebootServiceResponse executeResetService(byte[] pduTx, boolean waitForReconnect) {
    byte[] pduRx = this.communicationController.transmitPdu(pduTx);
    boolean isStillConnected = !isTimeout(pduRx);
    if (waitForReconnect)
      isStillConnected = waitForReconnect(); 
    return new EcuRebootServiceResponse(pduRx).setStillConnectedState(isStillConnected);
  }
  
  boolean waitForReconnect() {
    for (int i = 0; i < 12 && !isInterrupted(); i++) {
      sleep(250);
      if (!isInterrupted() && pingEcu()) {
        LOGGER.debug("CAN ECU '{}' is reachable again!", this.ecuName);
        return true;
      } 
    } 
    LOGGER.warn("Timeout while waiting for CAN ECU '{}' after hard reset!", this.ecuName);
    return false;
  }
  
  private boolean pingEcu() {
    byte[] pduRx = this.communicationController.transmitPdu(new byte[] { (byte)62, (byte)0 });
    return (pduRx != null && pduRx.length > 0 && pduRx[0] != Byte.MAX_VALUE);
  }
  
  private boolean isTimeout(byte[] pduRx) {
    return (pduRx == null || (pduRx.length == 2 && pduRx[0] == Byte.MAX_VALUE));
  }
  
  public EcuRebootServiceResponse executeSessionService(byte[] pduTx, boolean waitForReconnect) {
    byte[] pduRx = this.communicationController.transmitPdu(pduTx);
    boolean isStillConnected = !isTimeout(pduRx);
    if (waitForReconnect && wasServiceRequestSuccessful(pduTx, pduRx) && doesSessionRequestTriggerReboot(pduTx))
      isStillConnected = waitForReconnect(); 
    return new EcuRebootServiceResponse(pduRx).setStillConnectedState(isStillConnected);
  }
  
  private boolean wasServiceRequestSuccessful(byte[] pduTx, byte[] pduRx) {
    return (pduTx[0] + 64 == pduRx[0] && pduTx[1] == pduRx[1]);
  }
  
  boolean doesSessionRequestTriggerReboot(byte[] pduTx) {
    return (pduTx[1] == 2);
  }
  
  void sleep(int ms) {
    try {
      Thread.sleep((long)ms);
    } catch (InterruptedException ex) {
      LOGGER.warn("Error in sleepMilliseconds ", (Throwable)ex);
      Thread.currentThread().interrupt();
    } 
  }
  
  boolean isInterrupted() {
    return Thread.currentThread().isInterrupted();
  }
  
  public void onHidDeviceConnectionChanged(HidDeviceConnectionChangedEvent event) {}
}

    return didHidConnectionEventOccur();
  }
  
  private synchronized boolean didHidConnectionEventOccur() {
    long endTime = getCurrentTime() + 10000L;
    try {
      waitNow();
    } catch (InterruptedException ex) {
      LOGGER.warn("Waiting interrupted!", (Throwable)ex);
      Thread.currentThread().interrupt();
    } 
    return (endTime > getCurrentTime() && !isInterrupted());
  }
  
  long getCurrentTime() {
    return System.currentTimeMillis();
  }
  
  synchronized void waitNow() throws InterruptedException {
    LOGGER.debug("Start waiting for HID device reconnection");

    LOGGER.debug("Terminate waiting for HID device reconnection");
  }

public class VictorFlashJob extends AbstractFlashJob {
  public VictorFlashJob(ComCtrl comController) {
    super(comController);
  }
  
  public byte[] loadFlashDriver(FlashDataSet fdsBoot) {
    byte encryption = fdsBoot.getEncryption() ? 1 : 0;
    byte[] pduRX = null;
    for (int n = 0; !Helper.isNegativeResponse(pduRX) && n < fdsBoot.getNumberOfBlocks(); n++)
      pduRX = downloadDataSegment(encryption, fdsBoot.getAddress(n), fdsBoot.getData(n), fdsBoot.getSize(n), 0.0D, 2.0D); 
    this.comCtrl.startTesterPresentTaskIfNotAlreadyRunning();
    if (!Helper.isNegativeResponse(pduRX)) {
      pduRX = checkMemory(fdsBoot.getChecksum());
      this.comCtrl.setProgress(5);
    } 
    this.comCtrl.stopTesterPresentTask();
    return pduRX;
  }
  
  public byte[] programApplicationSoftware(FlashDataSet fdsApp) {
    int numberOfBlocks = fdsApp.getNumberOfBlocks();
    int dataSizeTotal = fdsApp.getDataSize();
    byte encryption = fdsApp.getEncryption() ? 1 : 0;
    int startAdd = ByteArrayConverterUtil.byteArrayToInteger(fdsApp.getAddress(0));
    int endAdd = ByteArrayConverterUtil.byteArrayToInteger(fdsApp.getAddress(numberOfBlocks - 1)) + (fdsApp.getData(numberOfBlocks - 1)).length;
    int memSize = endAdd - startAdd;
    this.comCtrl.startTesterPresentTaskIfNotAlreadyRunning();
    byte[] pduRX = eraseMemory(fdsApp.getAddress(0), ByteArrayConverterUtil.integerToByteArray(memSize));
    this.comCtrl.setProgress(8);
    this.comCtrl.stopTesterPresentTask();
    double stepStart = 8.0D;
    for (int n = 0; !Helper.isNegativeResponse(pduRX) && n < fdsApp.getNumberOfBlocks(); n++) {
      double stepSize = 91.0D * (double)(fdsApp.getData(n)).length / (double)dataSizeTotal;
      pduRX = downloadDataSegment(encryption, fdsApp.getAddress(n), fdsApp.getData(n), fdsApp.getSize(n), stepStart, stepSize);
      stepStart += stepSize;
    } 
    this.comCtrl.startTesterPresentTaskIfNotAlreadyRunning();
    if (!Helper.isNegativeResponse(pduRX)) {
      pduRX = checkMemory(fdsApp.getChecksum());
      this.comCtrl.setProgress(100);
    } 
    this.comCtrl.stopTesterPresentTask();
    return pduRX;
  }
  
  private byte[] downloadDataSegment(byte encrypted, byte[] address, byte[] data, int size, double stepStart, double stepSize) {
    byte[] pduRX = requestDownload(address, ByteArrayConverterUtil.integerToByteArray(size), encrypted);
    if (!Helper.isNegativeResponse(pduRX)) {
      int iBlockLength = getBlockLengthFromRequestDownloadResponse(pduRX, true);
      pduRX = transferData(data, iBlockLength, 0, stepStart, stepSize, true);
    } 
    if (!Helper.isNegativeResponse(pduRX))
      pduRX = transferExit(); 
    return pduRX;
  }
}

import com.gigatronik.dart2.framework.util.ByteArrayConverterUtil;

import java.io.IOException;

public class PduSimulation {
  private static final Logger LOGGER = LoggerFactory.getLogger(PduSimulation.class);
  
  static byte[] getPositiveResponsePdu(byte[] requestPdu) {
    requestPdu[0] = (byte)(requestPdu[0] + 64);
    return requestPdu;
  }
  
  public static byte[] getResponsePdu(String simFile, String ecu, byte[] requestPdu) {
    String request = "0x" + ByteArrayConverterUtil.byteArrayToHexString(requestPdu);
    byte[] response = { Byte.MAX_VALUE, requestPdu[0] };
    try (BufferedReader brSim = new BufferedReader(new FileReader(simFile))) {
      String line = findFirstRequest(brSim, ecu);
      while (!StringUtils.isEmpty(line)) {
        response = findResponse(line, request);
        if (response != null)
          break; 
        line = brSim.readLine();
      } 
    } catch (IOException ex) {
      LOGGER.error("Error in getResponsePdu " + ex);
    } 
    return response;
  }
  
  private static byte[] findResponse(String line, String request) {
    byte[] response = null;
    String[] entries = line.split(",");
    if (entries.length >= 3)
      if (request.length() > entries[1].length()) {
        if (request.startsWith(entries[1].replace("*", "")))
          response = calculateResponsePDUFromRequest(request, entries[2]); 
      } else if (entries[1].startsWith(request)) {
        response = calculateResponsePDUFromRequest(request, entries[2]);
      }  
    return response;
  }
  
  private static String findFirstRequest(BufferedReader br, String ecu) throws IOException {
    String ecuEntry = "[" + ecu + "]";
    String line = br.readLine();
    while (line != null && !StringUtils.startsWith(line, ecuEntry))
      line = br.readLine(); 
    while (StringUtils.startsWith(line, "["))
      line = br.readLine(); 
    return line;
  }
  
  private static byte[] calculateResponsePDUFromRequest(String request, String response) {
    int pos = response.indexOf("*");
    String sResult = "";
    if (pos >= 0) {
      sResult = response.substring(0, pos) + response.substring(0, pos);
    } else {
      sResult = response;
    } 
    return ByteArrayConverterUtil.hexStringToByteArray(sResult);
  }
}
