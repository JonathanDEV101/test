class TaskSchedule implements TaskExecutionObserver {
  private static final Logger LOGGER = LoggerFactory.getLogger(TaskSchedule.class);
  
  private List<Job> openJobs;
  
  private boolean isRejectionMode = false;
  
  TaskSchedule() {
    this.openJobs = new ArrayList<>();
  }
  
  synchronized void push(Job job) {
    if (this.isRejectionMode) {
      rejectJob(job);
    } else {
      this.openJobs.add(job);
    } 
  }
  
  private void rejectJob(Job job) {
    LOGGER.warn("Rejecting incoming task '{}'", job.getType());
    job.setState(ExecutionState.TERMINATED_ERROR);
    TaskCompletionEvent taskRejectEvent = new TaskCompletionEvent(false);
    job.getCallbacks().forEach(callback -> callback.onExecutionFinished(taskRejectEvent));
  }
  
  synchronized Job pop() {
    Job highestPriorityJob = determineHighestPriorityJob();
    if (highestPriorityJob != null)
      adaptQueueContent(highestPriorityJob); 
    return highestPriorityJob;
  }
  
  private Job determineHighestPriorityJob() {
    Optional<Job> element = this.openJobs.stream().max(Comparator.comparingInt(job -> job.getType().getPriority()));
    if (element.isPresent())
      return element.get(); 
    return null;
  }
  
  private void adaptQueueContent(Job scheduledJob) {
    if (scheduledJob.getType().shouldQueueBeClearedAfterExecution()) {
      clearSchedule();
      startRejectionMode(scheduledJob);
    } else {
      this.openJobs.remove(scheduledJob);
    } 
  }
  
  private void clearSchedule() {
    LOGGER.debug("Reset task queue");
    this.openJobs.clear();
  }
  
  private void startRejectionMode(Job scheduledJob) {
    LOGGER.debug("Start job rejection mode");
    this.isRejectionMode = true;
    scheduledJob.addCallback(this);
  }

    LOGGER.debug("Stop job rejection mode");
    this.isRejectionMode = false;
  }

    return this.openJobs.stream().anyMatch(job -> (job.getType() == type));
  }
