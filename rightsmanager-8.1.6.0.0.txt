public enum Mode {
  NONE, OEM_WORKER, OEM_ADMIN;
  
  public static Mode getDefaultModeFor(UserLevel level) {
    switch (level) {
      case IBD:
        return NONE;
      case OEM:
        return OEM_WORKER;
    } 
    return NONE;
  }
  
  public boolean isValidForLevel(UserLevel level) {
    switch (this) {
      case OEM_ADMIN:
      case OEM_WORKER:
        return (level == UserLevel.OEM);
      case NONE:
        return (level == UserLevel.IBD || level == UserLevel.UNKNOWN);
    } 
    return false;
  }
}
