public class SystemDumpThread extends TimerTask {

  private static final EnvironmentSnapshotDartPlugin ENVIRONMENT_SNAPSHOT_DART_PLUGIN = new EnvironmentSnapshotPluginImpl();
  
  private static final String LINE_SEPARATOR = System.getProperty("line.separator");
  
  private static final String ZIPPED_ENCRYPTION_PASSWORD = "YWFiZjNkNDU0YmU2NDViYzAyYTNkNmQ4";
  
  private static final String ZIPPED_TEXT_FILE_NAME = "dmp_data.txt";
  
  private static final String SEARCH_ACRONYM = "**dmp";
  
  public void run() {
    Thread.currentThread().setName("SystemDump-Thread-" + Thread.currentThread().getId());
    EnvironmentSnapshotDartPlugin.Snapshot snapshot = ENVIRONMENT_SNAPSHOT_DART_PLUGIN.createSnapshot();
    List<EnvironmentSnapshotDartPlugin.CheckResult> results = snapshot.getResults();
    StringBuilder stringBuilder = new StringBuilder();
    results.stream().forEach(result -> {
          String classSimpleName = result.getCheck().getClass().getSimpleName();
          stringBuilder.append(classSimpleName).append(LINE_SEPARATOR).append(result.getResult().toString()).append(LINE_SEPARATOR);
        });
    try {
      byte[] zipContent = zipStatementsPWEntries(stringBuilder.toString().getBytes());
      String base64Content = new String(Base64.encodeBase64(zipContent));
      LOGGER.info(String.format("%s %s", "**dmp", base64Content));
    } catch (IOException|ZipException ex) {
      LOGGER.error("Error converting dump", (Throwable)ex);
    } 
  }
  
  private byte[] zipStatementsPWEntries(byte[] contentToZip) throws IOException, ZipException {
    byte[] result;
    try(ByteArrayOutputStream bos = new ByteArrayOutputStream(); 
        ZipOutputStream outputStream = new ZipOutputStream(bos)) {
      ZipParameters zipParameters = getParameters();
      outputStream.putNextEntry(null, zipParameters);
      outputStream.write(contentToZip);
      outputStream.closeEntry();
      outputStream.finish();
      result = bos.toByteArray();
    } 
    return result;
  }
  
  private ZipParameters getParameters() {
    ZipParameters zipParameters = new ZipParameters();
    zipParameters.setCompressionMethod(8);
    zipParameters.setCompressionLevel(5);
    zipParameters.setEncryptFiles(true);
    zipParameters.setEncryptionMethod(0);
    zipParameters.setPassword("YWFiZjNkNDU0YmU2NDViYzAyYTNkNmQ4");
    zipParameters.setFileNameInZip("dmp_data.txt");
    zipParameters.setSourceExternalStream(true);
    return zipParameters;
  }
